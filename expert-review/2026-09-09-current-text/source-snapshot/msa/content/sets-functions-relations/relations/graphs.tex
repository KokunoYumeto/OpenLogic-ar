\olsection{الرسوم البيانية}

\emph{الرسم البياني} مخطط ترتبط فيه نقاط---تسمى «عقدًا» أو «رؤوسًا»
(جمع «رأس»)---بحواف. والرسوم البيانية أداة واسعة الانتشار في
الرياضيات المتقطعة وعلوم الحاسوب. وهي نافعة للغاية في تمثيل العلاقات
والبنى وتصورها، بدءًا من أشياء عيانية كالشبكات بمختلف أنواعها، وانتهاءً
ببنى مجردة كالنتائج الممكنة للقرارات. وترد في المراجع أنواع كثيرة مختلفة
من الرسوم البيانية تتباين، مثلًا، بحسب ما إذا كانت الحواف موجهة أم غير
موجهة، وموسومة أم غير موسومة، وما إذا كان يمكن أن توجد حواف من عقدة إلى
العقدة نفسها أو حواف متعددة بين العقدتين نفسيهما، وغير ذلك. وللرسوم
البيانية \emph{الموجهة} صلة خاصة بالعلاقات.

\begin{defn}[الرسم البياني الموجه]
\emph{الرسم البياني الموجه} $G = \tuple{V, E}$ يتألف من مجموعة من
\emph{الرؤوس}~$V$ ومجموعة من \emph{الحواف}~$E \subseteq V^2$.
\end{defn}

\begin{explain}
بحسب تعريفنا، ليس الرسم البياني إلا مجموعة ومعها علاقة على تلك المجموعة.
وبالطبع، من الطبيعي عند الحديث عن الرسوم البيانية أن نتوقع تمثيلها
رسوميًا: فيمكننا رسمه برسم سهم من الرأس~$v_1$ إلى الرأس~$v_2$ إذا وفقط إذا
كان $\tuple{v_1, v_2} \in E$. والفرق الوحيد بين العلاقة بمفردها والرسم
البياني هو أن الرسم البياني يحدد مجموعة الرؤوس؛ أي إنه قد تكون له رؤوس
معزولة. غير أن النقطة المهمة هي أن كل علاقة~$R$ على مجموعة~$X$ يمكن النظر
إليها بوصفها رسمًا بيانيًا موجهًا $\tuple{X, R}$؛ وبالعكس، يمكن النظر إلى
الرسم البياني الموجه~$\tuple{V, E}$ بوصفه علاقة $E \subseteq V^2$ مع
تحديد المجموعة $V$ صراحة.
\end{explain}

\begin{ex}
يبدو الرسم البياني $\tuple{V, E}$ الذي فيه $V = \{1, 2, 3, 4\}$ و$E =
\{\tuple{1,1}, \allowbreak \tuple{1, 2}, \allowbreak \tuple{1, 3},
\allowbreak \tuple{2, 3}\}$ كما يأتي:
\begin{align*}
& \begin{tikzpicture}[->,node distance=2cm]
  \node[draw,circle] (A) {$1$};
  \node[draw,circle] (B) [right of=A] {$2$};
  \node[draw,circle] (C) [below of=B] {$3$};
  \node[draw,circle] (D) [right of=B] {$4$};
  \draw (A) to [loop above]  (A);
  \draw (A) to  (B);
  \draw (A) to  (C);
  \draw (B) to  (C);
  \end{tikzpicture}
\intertext{وهذا رسم بياني مختلف عن $\tuple{V', E}$ الذي فيه $V' =
  \{1, 2, 3\}$، والذي يبدو كما يأتي:}
& \begin{tikzpicture}[->,node distance=2cm]
  \node[draw,circle] (A) {$1$};
  \node[draw,circle] (B) [right of=A] {$2$};
  \node[draw,circle] (C) [below of=B] {$3$};
  \draw (A) to [loop above]  (A);
  \draw (A) to  (B);
  \draw (A) to  (C);
  \draw (B) to  (C);
  \end{tikzpicture}
\end{align*}
\end{ex}

\begin{prob}
  اعتبر علاقة أصغر من أو يساوي~$\le$ على المجموعة $\{1,
  2, 3, 4\}$ بوصفها رسمًا بيانيًا، وارسم المخطط المناظر.
\end{prob}
