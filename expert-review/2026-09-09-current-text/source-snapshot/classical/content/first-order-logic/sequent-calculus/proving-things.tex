\olsection{أمثلة على ال\usetoken{P}{derivation}}

\begin{ex}
أعط $\Log{LK}$-!!{derivation} للتتابعية~$!A \land !B \Sequent !A$.

نبدأ بكتابة التتابعية النهائية المطلوبة في أسفل ال!!{derivation}.
\begin{prooftree}
\AxiomC{}
\UnaryInf$!A\land !B \fCenter !A$
\end{prooftree}
ثم ننظر: أي استدلال يصح أن تكون هذه تتابعيته السفلى؟ قد تصلح قاعدة بنيوية، لكن الأولى أن نلتمس قاعدة منطقية. وليس في هذه التتابعية من الروابط إلا~$\land$؛ فنطلب قاعدة لـ~$\land$.
ولما كان~$\land$ في المقدَّم، تعيّن النظر في~\LeftR{\land}.
\begin{prooftree}
\AxiomC{}
\RightLabel{\LeftR{\land}}
\UnaryInf$!A\land !B \fCenter !A$
\end{prooftree}
للتتابعية العليا في~\LeftR{\land} احتمالان:
$!A \Sequent !A$، أو~$!B \Sequent !A$.
الأولى، $!A \Sequent !A$، ابتدائية، وذلك ما نطلبه؛ وأما
$!B \Sequent !A$ فلا تُشتق في العموم. فنضع الأولى في أعلى الشجرة:
\begin{prooftree}
\Axiom$!A \fCenter !A$
\RightLabel{\LeftR{\land}}
\UnaryInf$!A\land !B \fCenter !A$
\end{prooftree}
أصبح لدينا الآن $\Log{LK}$-!!{derivation} صحيح للتتابعية
$!A \land !B \Sequent !A$.
\end{ex}

\begin{ex}
أعط $\Log{LK}$-!!{derivation} للتتابعية
$\lnot !A \lor !B \Sequent !A \lif !B$.

ابدأ بكتابة التتابعية النهائية المطلوبة في أسفل ال!!{derivation}.
\begin{prooftree}
\AxiomC{}
\UnaryInf$\lnot !A \lor !B \fCenter !A \lif !B$
\end{prooftree}
ننطلق من روابط التتابعية النهائية: $\lnot$ و$\lor$ و$\lif$.
لكن النظر الآن في~$\lor$ و$\lif$ وحدهما، إذ هما
!!{main operator}s لل!!{sentence}s فيها. وأما~$\lnot$ فداخل نطاق رابط آخر، ويتأخر النظر فيه.
فللاستدلال الأخير إذن قاعدتان محتملتان: \LeftR{\lor} و~\RightR{\lif}.
كلتاهما تصلح؛ ونختار~\RightR{\lif} لأنها تؤخر تشعب البرهان إلى فرعين.
وبمطابقة صورة~\RightR{\lif} على التتابعية المطلوبة نجد:
\begin{prooftree}
\AxiomC{}
\UnaryInf$ !A, \lnot !A \lor !B \fCenter !B $
\RightLabel{\RightR{\lif}} \UnaryInf$ \lnot !A \lor !B \fCenter !A \lif !B $
\end{prooftree}

انقل~$\lnot !A \lor !B$ إلى طرف المقدَّم؛ فحينئذ تعمل قاعدة~\LeftR{\lor}.
ومخططها يقتضي تتابعيتين علويتين، على هذه الصورة:
\begin{prooftree}
\AxiomC{}
\UnaryInf$\lnot !A, !A \fCenter !B$
\AxiomC{}
\UnaryInf$!B, !A \fCenter !B$
\RightLabel{\LeftR{\lor}}
\BinaryInf$ \lnot !A \lor !B, !A \fCenter !B $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ !A, \lnot !A \lor !B \fCenter !B $
\RightLabel{\RightR{\lif}}
\UnaryInf$ \lnot !A \lor !B \fCenter !A \lif !B $
\end{prooftree}
والغاية من هذا الصعود بلوغ التتابعيات الابتدائية، وقد قربنا منها!{}
فأما الفرع الأيمن فيتم بإضعاف واحد وتبادل واحد:
\begin{prooftree}
\AxiomC{}
\UnaryInf$\lnot !A, !A \fCenter !B$
\Axiom$!B \fCenter !B$
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$!A, !B \fCenter !B$
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!B, !A \fCenter !B$
\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor !B, !A \fCenter !B $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ !A, \lnot !A \lor !B \fCenter !B $
\RightLabel{\RightR{\lif}}
\UnaryInf$ \lnot !A \lor !B \fCenter !A \lif !B $
\end{prooftree}

وأما الفرع الأيسر، فلا رابط في أي
!!{sentence} منه إلا~$\lnot$ في !!{sentence}s المقدَّم؛ فنأخذ مثيلًا من~\LeftR{\lnot}.
\begin{prooftree}
\AxiomC{}
\UnaryInf$ !A \fCenter !B, !A$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$\lnot !A, !A \fCenter !B$
\Axiom$!B \fCenter !B$
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$!A, !B \fCenter !B$
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!B, !A \fCenter !B$
\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor !B, !A \fCenter !B $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ !A, \lnot !A \lor !B \fCenter !B $
\RightLabel{\RightR{\lif}}
\UnaryInf$ \lnot !A \lor !B \fCenter !A \lif !B $
\end{prooftree}
ويتم الفرع الأيسر بمثل ما تم به الأيمن: إضعاف واحد، ثم تبادل واحد.
\begin{prooftree}
\Axiom$!A \fCenter !A$
\RightLabel{\RightR{\Weakening}}
\UnaryInf$ !A \fCenter !A, !B$
\RightLabel{\RightR{\Exchange}}
\UnaryInf$ !A \fCenter !B, !A$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$\lnot !A, !A \fCenter !B$
\Axiom$!B \fCenter !B$
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$!A, !B \fCenter !B$
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!B, !A \fCenter !B$
\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor !B, !A \fCenter !B $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ !A, \lnot !A \lor !B \fCenter !B $
\RightLabel{\RightR{\lif}}
\UnaryInf$ \lnot !A \lor !B \fCenter !A \lif !B $
\end{prooftree}
\end{ex}

\begin{ex}
أعط $\Log{LK}$-!!{derivation} للتتابعية
$\lnot !A \lor \lnot !B \Sequent \lnot (!A \land !B)$.

باستعمال الأساليب السابقة، نبدأ بكتابة التتابعية النهائية المطلوبة في
الأسفل.
\begin{prooftree}
\AxiomC{}
\UnaryInf$ \lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B) $
\end{prooftree}
ليس لل!!{sentence}s في التتابعية النهائية من المؤثرات الرئيسة إلا~$\lor$ و$\lnot$.
فتصلح~\LeftR{\lor} كما تصلح
\RightR{\lnot}. ونبدأ بـ~\RightR{\lnot} ليبقى البرهان قليلًا بلا تشعب:
\begin{prooftree}
\AxiomC{}
\UnaryInf$!A \land !B, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$\lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B)$
\end{prooftree}
ثم يتجه النظر إلى~\LeftR{\land} أو~\LeftR{\lor}.
فلنجرب~\LeftR{\land}. يمكن أن تكون البداية
$!A, \lnot !A \lor \lnot !B \Sequent \quad$ أو
$!B, \lnot !A \lor \lnot !B \Sequent \quad$.
\textbf{تنبيه تصحيحي على الأصل (C075-01).} أُعيد النفي الساقط قبل $!B$ في
التتابعيتين؛ فقاعدة إدخال العطف في اليسار لا تغيّر السياق
$\lnot !A\lor\lnot !B$ الظاهر في الشجرة التالية.
وال!!{derivation} متناظر في~$!A$ و$!B$، فنأخذ الأولى:
\begin{prooftree}
\AxiomC{}
\UnaryInf$!A, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\LeftR{\land}}
\UnaryInf$!A \land !B, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$\lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B)$
\end{prooftree}
فإذا تابعنا إنشاء ال!!{derivation} ظهر موضع الإشكال:
\begin{prooftree}
\Axiom$!A \fCenter !A$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$ \lnot !A, !A \fCenter$
\AxiomC{}
\RightLabel{?}
\UnaryInf$!A \fCenter !B$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$ \lnot !B, !A \fCenter$
\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor \lnot !B, !A \fCenter $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!A, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\LeftR{\land}}
\UnaryInf$!A \land !B, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$\lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B)$
\end{prooftree}
أعلى الفرع الأيمن لا يقبل مزيدًا من الاختزال، ولا تنقله القواعد البنيوية إلى تتابعية ابتدائية. فتبين أن هذا المسلك لا يتم، وأن تقديم~\LeftR{\land} لم يكن اختيارًا موفقًا.
فلنرجع إلى الموضع السابق ونأخذ~\LeftR{\lor}، فيحصل:
\begin{prooftree}
\AxiomC{}
\UnaryInf$\lnot !A, !A \land !B \fCenter $

\AxiomC{}
\UnaryInf$\lnot !B, !A \land !B \fCenter $

\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor \lnot !B, !A \land !B \fCenter $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!A \land !B, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$\lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B)$
\end{prooftree}
وبإكمال كل فرع كما فعلنا من قبل نحصل على
\begin{prooftree}
\Axiom$ !A \fCenter!A$
\RightLabel{\LeftR{\land}}
\UnaryInf$!A \land !B \fCenter !A$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$\lnot !A, !A \land !B \fCenter $

\Axiom$ !B \fCenter !B$
\RightLabel{\LeftR{\land}}
\UnaryInf$!A \land !B \fCenter !B$
\RightLabel{\LeftR{\lnot}}
\UnaryInf$\lnot !B, !A \land !B \fCenter $

\RightLabel{\LeftR{\lor}}
\BinaryInf$\lnot !A \lor \lnot !B, !A \land !B \fCenter $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$!A \land !B, \lnot !A \lor \lnot !B \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$\lnot !A \lor \lnot !B \fCenter \lnot (!A \land !B)$
\end{prooftree}
(وكان يمكننا تنفيذ قواعد~$\land$ في موضع أدنى من قواعد~$\lnot$ في
هذه الخطوات، مع بقائنا نحصل على !!{derivation} صحيح).
\end{ex}

\begin{ex}
وبقيت قاعدة الانكماش، ولم نحتج إليها فيما تقدم؛ لكنها لازمة في بعض المواضع.
لنطلب مثلًا~$\quad \Sequent !A \lor \lnot !A$.
فإن عكسنا~$\RightR{\lor}$ حصل أحد هذين ال!!{derivation}s:
\begin{prooftree}
\AxiomC{}
\UnaryInf$ \fCenter !A$
\RightLabel{\RightR{\lor}}
\UnaryInf$ \fCenter !A \lor \lnot !A$
\DisplayProof\qquad\bottomAlignProof
\AxiomC{}
\UnaryInf$!A \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$ \fCenter \lnot !A$
\RightLabel{\RightR{\lor}}
\UnaryInf$ \fCenter !A \lor \lnot !A$
\end{prooftree}
ولا يبلغ أيٌّ منهما تتابعية ابتدائية.
ووجه الخروج أن الانكماش يجمع نسختين من
!!a{sentence} في واحدة. وعند طلب البرهان من أسفل إلى أعلى يفيدنا ذلك إبقاء نسخة من~$!A \lor \lnot !A$ في المقدمة، هكذا:
\begin{prooftree}
\AxiomC{}
\UnaryInf$ \fCenter !A \lor \lnot !A, !A$
\RightLabel{\RightR{\lor}}
\UnaryInf$ \fCenter !A \lor \lnot !A, !A \lor \lnot !A$
\RightLabel{\RightR{\Contraction}}
\UnaryInf$ \fCenter !A \lor \lnot !A$
\end{prooftree}
ثم نعمل~$\RightR{\lor}$ مرة ثانية لنحصل على~$\lnot !A$ أيضًا، فيتم ال!!{derivation}:
\begin{prooftree}
\Axiom$!A \fCenter !A$
\RightLabel{\RightR{\lnot}}
\UnaryInf$\fCenter !A, \lnot !A$
\RightLabel{\RightR{\lor}}
\UnaryInf$\fCenter !A, !A \lor \lnot !A$
\RightLabel{\RightR{\Exchange}}
\UnaryInf$ \fCenter !A \lor \lnot !A, !A$
\RightLabel{\RightR{\lor}}
\UnaryInf$ \fCenter !A \lor \lnot !A, !A \lor \lnot !A$
\RightLabel{\RightR{\Contraction}}
\UnaryInf$ \fCenter !A \lor \lnot !A$
\end{prooftree}
\end{ex}

\begin{prob}
أعط !!{derivation}s للتتابعيات الآتية:
\begin{enumerate}
\item $!A \land (!B \land !C) \Sequent (!A \land !B) \land !C$.
\item $!A \lor (!B \lor !C) \Sequent (!A \lor !B) \lor !C$.
\item $!A \lif (!B \lif !C) \Sequent !B \lif (!A \lif !C)$.
\item $!A \Sequent \lnot\lnot !A$.
\end{enumerate}
\end{prob}

\begin{prob}
أعط !!{derivation}s للتتابعيات الآتية:
\begin{enumerate}
\item $(!A \lor !B) \lif !C \Sequent !A \lif !C$.
\item $(!A \lif !C) \land (!B \lif !C) \Sequent (!A \lor !B) \lif !C$.
\item $\Sequent \lnot(!A \land \lnot !A)$.
\item $!B \lif !A \Sequent \lnot !A \lif \lnot !B$.
\item $\Sequent (!A \lif \lnot !A) \lif \lnot !A$.
\item $\Sequent \lnot(!A \lif !B) \lif \lnot !B$.
\item $!A \lif !C \Sequent \lnot (!A \land \lnot !C)$.
\item $!A \land \lnot !C \Sequent \lnot (!A \lif !C)$.
\item $!A \lor !B, \lnot !B \Sequent !A$.
\item $\lnot !A \lor \lnot !B \Sequent \lnot(!A \land !B)$.
\item $\Sequent (\lnot !A \land \lnot !B) \lif\lnot(!A \lor !B)$.
\item $\Sequent \lnot(!A \lor !B) \lif (\lnot !A \land \lnot !B)$.
\end{enumerate}
\end{prob}

\begin{prob}
أعط !!{derivation}s للتتابعيات الآتية:
\begin{enumerate}
\item $\lnot(!A \lif !B) \Sequent !A$.
\item $\lnot(!A \land !B) \Sequent \lnot !A \lor \lnot !B$.
\item $!A \lif !B \Sequent \lnot !A \lor !B$.
\item $\Sequent \lnot \lnot !A \lif !A$.
\item $!A \lif !B, \lnot !A \lif !B \Sequent !B$.
\item $(!A \land !B) \lif !C \Sequent (!A \lif !C) \lor (!B \lif !C)$.
\item $(!A \lif !B) \lif !A \Sequent !A$.
\item $\Sequent (!A \lif !B) \lor (!B \lif !C)$.
\end{enumerate}
(تتطلب هذه كلها القاعدة~$\RightR{\Contraction}$.)
\end{prob}
