\olsection{أمثلة على الترشيحات}

بقي أن نبين وجود الترشيحات. فللنموذج~$\mModel{M}$، على ما سيظهر، ترشيحات
كثيرة لـ~$\mModel{M}$ عبر~$\Gamma$. وأخصها بالنظر ترشيحان: الأدق والأخشن.
وإنما يقع اختلاف ترشيحات النموذج الواحد في علاقة النفاذ؛ إذ تعيّن
\olref[fil]{defn:filtration} كلاًّ من~$W^*$ و$V^*$ تعيينًا مباشرًا.
فالأدق يقتصر على أقل ما يمكن من أزواج العوالم المرتبطة، والأخشن يضم
أكثر ما يمكن منها.

\begin{defn}
  إذا كانت~$\Gamma$ مغلقة تحت أخذ الصيغ الجزئية، عُرّف الترشيح
  \emph{الأدق}~$\mModel{M^*}$ لنموذج~$\mModel{M}$ بوضع:
  \[
  R^*[u][v] \quad \text{إذا وفقط إذا} \quad \exists u'\in [u] \;
  \exists v' \in [v] : Ru'v'.
  \]
\end{defn}

\begin{prop}\ollabel{prop:finest}
  الترشيح الأدق~$\mModel{M^*}$ ترشيح بالفعل.
\end{prop}

\begin{proof}
  موضع البرهان أن~$R^*$ المحددة آنفًا تحقق
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R}.
  فلننظر في الشروط الثلاثة واحدًا بعد واحد.

  متى كان~$Ruv$، كان~$u \in [u]$ و$v \in [v]$ شاهدين على
  $R^*[u][v]$؛ فثبت~\olref[fil]{defn:filtration-R1}.

  \iftag{prvBox}{\iftag{probBox}{نترك التحقق من
      \olref[fil]{defn:filtration-R2} تمرينًا.}{للتحقق من
        \olref[fil]{defn:filtration-R2}، افترض~$\Box!A \in \Gamma$
        و$R^*[u][v]$ و$\mSat{M}{\Box!A}[u]$. بحسب تعريف~$R^*$، يوجد
        $u' \equiv u$ و$v' \equiv v$ بحيث~$Ru'v'$. ولأن~$u$ و$u'$
        يتفقان على~$\Gamma$، يكون أيضًا~$\mSat{M}{\Box!A}[u']$، ومن
        ثم~$\mSat{M}{!A}[v']$. وبإغلاق~$\Gamma$ تحت أخذ
        ال!!{formula}s الجزئية، يتفق~$v$ و$v'$ على~$!A$، ولذلك
        $\mSat{M}{!A}[v]$، كما هو مطلوب.}}{}

  \iftag{prvDiamond}{\iftag{probDiamond}{نترك التحقق من
      \olref[fil]{defn:filtration-R3} تمرينًا.}{للتحقق من
        \olref[fil]{defn:filtration-R3}، افترض~$\Diamond!A \in
        \Gamma$ و$R^*[u][v]$ و$\mSat{M}{!A}[v]$. بحسب تعريف~$R^*$،
        يوجد~$u' \equiv u$ و$v' \equiv v$ بحيث~$Ru'v'$. ولأن~$v$
        و$v'$ يتفقان على~$\Gamma$، ولأن~$\Gamma$ مغلقة تحت أخذ
        ال!!{formula}s الجزئية، يكون أيضًا~$\mSat{M}{!A}[v']$، ومن
        ثم~$\mSat{M}{\Diamond !A}[u']$. ولأن~$u$ و$u'$ يتفقان أيضًا
        على~$\Gamma$، يكون~$\mSat{M}{\Diamond !A}[u]$.}}{}
\end{proof}

\begin{probtag}{probBox,probDiamond}
  أكمل برهان~\olref[nml][fil][exf]{prop:finest}.
\end{probtag}

\begin{defn}
  إذا كانت~$\Gamma$ مغلقة تحت أخذ الصيغ الجزئية، عُرّف الترشيح
  \emph{الأخشن}~$\mModel{M^*}$ لنموذج~$\mModel{M}$ بوضع
  $R^*[u][v]$ إذا وفقط إذا
  \iftag{notprvBox,notprvDiamond}{تحقق الشرط الآتي:}{تحقق
    \emph{كلا} الشرطين الآتيين:}
  \begin{tagenumerate}{prvBox,prvDiamond}
  \tagitem{prvBox}{\ollabel{defn:coarsest-Box}إذا كان~$\Box!A \in
    \Gamma$ و$\mSat{M}{\Box!A}[u]$، كان
    $\mSat{M}{!A}[v]$\iftag{prvDiamond}{؛}{.}}{}
  \tagitem{prvDiamond}{\ollabel{defn:coarsest-Diamond}إذا كان
    $\Diamond!A \in \Gamma$ و$\mSat{M}{!A}[v]$، كان
    $\mSat{M}{\Diamond!A}[u]$.}{}
  \end{tagenumerate}
\end{defn}

\begin{prop}
  الترشيح الأخشن~$\mModel{M^*}$ ترشيح بالفعل.
\end{prop}

\begin{proof}
  أما~$R^*$ فقد ضُمِّن تعريفها سائر الشروط، فلا يبقى إلا أن يستلزم~$Ruv$
  العلاقة~$R^*[u][v]$. ولنفرض لذلك~$Ruv$. \iftag{prvBox}{افترض~$\Box!A \in
    \Gamma$ و$\mSat{M}{\Box!A}[u]$؛ عندئذ من الواضح أن
    $\mSat{M}{!A}[v]$\iftag{prvDiamond}{، ومن ثم يستوفى
      \olref{defn:coarsest-Box}.}{.}}{}\iftag{prvDiamond}{افترض
    $\Diamond!A \in \Gamma$ و$\mSat{M}{!A}[v]$. عندئذ
    $\mSat{M}{\Diamond!A}[u]$ لأن~$Ruv$\iftag{prvBox}{، ومن ثم
      يستوفى~\olref{defn:coarsest-Diamond}}{}.}{}
\end{proof}

\begin{ex}
  نأخذ~$W=\PosInt$، ونضع~$Rnm$ إذا وفقط إذا كان~$m=n+1$، ونعيّن
  $V(p)=\Setabs{2n}{n \in \PosInt}$. فهذا النموذج
  $\mModel{M}=\tuple{W,R,V}$ هو المرسوم في~\olref{fig:ex-filtration}.
  عوالمه~$1$ و$2$ وما يليهما، وليس لكل عالم إلا عالم واحد ينفذ إليه،
  هو لاحقه؛ وأما~$p$ فيصدق عند الأعداد الزوجية الموجبة وحدها.

  \begin{figure}
    \centering
    \begin{tikzpicture}[modal]
      \node[world] (1) [label=below:\mFalse{p}] {$1$};
      \node[world] (2) [label=below:\mTrue{p}, right=of 1] {$2$};
      \node[world] (3) [label=below:\mFalse{p}, right=of 2] {$3$};
      \node[world] (4) [label=below:\mTrue{p}, right=of 3] {$4$};
      \node[phantom] (5) [right=of 4] {};
      \draw[->] (1) to (2);
      \draw[->] (2) to (3);
      \draw[->] (3) to (4);
      \draw[dotted] (4) to (5);
    \end{tikzpicture}

    \begin{tikzpicture}[modal]
      \node[world] (1) [label=below:\mFalse{p}] {$[1]$};
      \node[world] (2) [label=below:\mTrue{p}, right=of 1] {$[2]$};
      \draw[->,bend left] (1) to (2);
      \draw[->,bend left] (2) to (1);

      \node[world] (3) [label=below:\mFalse{p},right=of 2] {$[1]$};
      \node[world] (4) [label=below:\mTrue{p}, right=of 3] {$[2]$};
      \draw[->,bend left] (3) to (4);
      \draw[->,bend left] (4) to (3);
      \draw[->,reflexive above] (4) to (4);
    \end{tikzpicture}
    \caption{نموذج لا نهائي وترشيحاه.}
    \ollabel{fig:ex-filtration}
  \end{figure}

  ولنأخذ~$\Gamma$ مجموعة ال!!{formula}s الجزئية لـ~$\Box p \lif p$،
  وهي~$\{p,\Box p,\Box p \lif p\}$. أما~$p$ فيصدق عند الأعداد الزوجية
  وحدها، وأما~$\Box p$ فعند الأعداد الفردية وحدها؛ ولذلك تصدق
  $\Box p \lif p$ عند الزوجية دون الفردية. وتفصيل ذلك أن العالم الفردي
  يصدّق~$\Box p$ ويكذّب~$p$ و$\Box p \lif p$؛ والعالم الزوجي
  يصدّق~$p$ و$\Box p \lif p$ ويكذّب~$\Box p$. فليس إذن إلا الفئتان
  $W^*=\{[1],[2]\}$، وهما~$[1]=\{1,3,5,\dots\}$
  و$[2]=\{2,4,6,\dots\}$. ومن~$2 \in V(p)$ يلزم
  $[2] \in V^*(p)$، ومن~$1 \notin V(p)$ يلزم~$[1] \notin V^*(p)$؛
  فحصل~$V^*(p)=\{[2]\}$.

  وأما علاقة النفاذ لترشيح على~$W^*$ فلا بد أن تشتمل على
  $\tuple{[1],[2]}$ و$\tuple{[2],[1]}$. فإن~$R12$ يقتضي
  $R^*[1][2]$ بحكم
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R1}؛ ثم إن
  $R23$ يقتضي~$R^*[2][3]$، وقد علمنا أن~$[3]=[1]$.
  وأما~$\tuple{[1],[1]}$ فممتنع: لو اشتملت عليه لكان~$R^*[1][1]$،
  مع أن~$\mSat{M}{\Box p}[1]$ و$\mSat/{M}{p}[1]$،
  وهو خلاف~\olref[fil]{defn:filtration-R2}. ويبقى~$R^*[2][2]$ جائز
  الإثبات والنفي، فلا موجب له ولا مانع منه. فترشيحا~$\mModel{M}$
  الممكنان اثنان، وعلاقتا نفاذهما:
  \[
  \{\tuple{[1],[2]},\tuple{[2],[1]}\} \text{ و }
  \{\tuple{[1],[2]},\tuple{[2],[1]},\tuple{[2],[2]}\}.
  \]
  وفي كلتا الحالين يكذب~$p$ و$\Box p \lif p$ عند~$[1]$،
  ويصدق~$\Box p$؛ ويصدق~$p$ و$\Box p \lif p$ عند~$[2]$،
  ويكذب~$\Box p$.
\end{ex}


\begin{prob}
  ليكن النموذج~$\mModel{M}=\tuple{W,R,V}$ على الوجه الآتي:
  $W=\Setabs{0\sigma}{\sigma \in \Bin^*}$، أي متتاليات الأصفار والآحاد
  المبدوءة بـ~$0$؛ وتكون~$R\sigma\sigma'$ إذا وفقط إذا كان
  $\sigma'=\sigma 0$ أو~$\sigma'=\sigma 1$؛ ونضع
  $V(p)=\Setabs{\sigma 0}{\sigma \in \Bin^*}\cap W$،
  و$V(q)=\Setabs{\sigma 1}{\sigma \in
    \Bin^* \setminus \{1\}}\cap W$. وصورته:

  وفي هذا النموذج~$\mSat/{M}{\Box(p \lor q) \lif (\Box p \lor \Box q)}[w]$
  لكل~$w$.

  خذ~$\Gamma$ مجموعة ال!!{formula}s الجزئية لـ~$\Box(p \lor q) \lif
  (\Box p \lor \Box q)$. عيّن~$W^*$ و$V^*$، ثم عيّن علاقة النفاذ
  في أدق ترشيحات~$\mModel{M}$ وفي أخشنها.
\end{prob}
