\olsection{في تطعيم \usetoken{P}{proof}}

!!^{proof}s في~\Log{N1c} و\Log{N1i} إنما هي~!!{proof}s لـ!!{formula}s~$!A$
من فروض~$!B$. فإذا كان لـ$!B$ نفسها~!!a{proof}، سألنا: كيف نصل~!!{proof}
$!B$ بـ!!{proof}~$!A$ من~$!B$، حتى نحصل على~!!a{proof} لـ$!A$ لا
يستند إلى الفرض~$!B$؟

ومن الوجوه أن نحذف الفرض~$!B$ من~!!{proof}~$!A$ بإعمال
$\Intro{\lif}$، ثم نصل~!!{proof} الناتج لـ$!B \lif !A$
بـ!!{proof}~$!B$، فينتج
\[
\AxiomC{$\Discharge{!B}{x}$}
\DeduceC{$!A$}
\DischargeRule{\Intro{\lif}}{x}
\UnaryInfC{$!B \lif !A$}
\AxiomC{}
\DeduceC{$!B$}
\RightLabel{\Elim{\lif}}
\BinaryInfC{$!A$}
\DisplayProof
\]
وهذا طريق مباشر، غير أنه مشوب بفضول: فما وجه إدخال~$\lif$ إذا كنا
نحذفه من فورنا؟ والأولى أن نستغني عن هذا «الالتفاف» المار بـ~$!B \lif !A$.
(ومؤدى مبرهنة التطبيع أن الاستغناء عن هذه الالتفافات ممكن دائمًا.)

وفي~\Log{N1c} و\Log{N1i} يسهل اجتنابه: نقيم مقام الفروض~$!B^x$
كامل~!!{proof}~$!B$. ولما كان «تطعيم»~!!{proof}s بفروض براهين أخرى كثير
النفع، أثبتناه في التعريف الآتي.

\begin{defn}
ليكن~$\delta_1$~!!{proof} في~\Log{N1c} (أو~!!{proof} في
\Log{N1i}) لـ$!A$ من الفرض المفتوح~$!B^x$، وليكن~$\delta_2$~!!{proof}
في~\Log{N1c} (أو~!!{proof} في~\Log{N1i}) لـ$!B$. فنعرف
$\Subst{\delta_1}{\delta_2}{!B^x}$ بأنه ما يحصل من إحلال~$\delta_2$ محل
كل ظهور غير مبرأ للفرض~$!B^x$ في~$\delta_1$.
\end{defn}

إذا لم يكن في~$\delta_1$ و$\delta_2$~!!{formula}s فرض متغايرة تحمل
علامة واحدة، ولم يشتمل فرض مفتوح في~$\delta_2$ على متغير مميز من
$\delta_1$، كان~$\Subst{\delta_1}{\delta_2}{!B^x}$~!!{proof} صحيحًا،
وفروضه المفتوحة مجموع فروض~$\delta_1$ وفروض~$\delta_2$. وهذان الشرطان
متحققان عند تطعيم~!!{proof}s عادة؛ فالأول حاصل متى كان~$\delta_1$ و$\delta_2$
برهانين جزئيين من~!!{proof} أكبر، والثاني إن تخلف أصلحناه بإعادة تسمية
المتغيرات المميزة في~$\delta_1$.

ولهذا التعريف نظير في~\Log{N2c} و\Log{N2i}.

\begin{defn}
إذا كان~$\delta_1$~!!{proof} في~\Log{N2c} (أو~!!{proof} في
\Log{N2i}) لـ$x:!B, \Gamma_1 \Sequent !A$، وكان~$\delta_2$~!!{proof}
في~\Log{N2c} (أو~!!{proof} في~\Log{N2i}) لـ
$\Gamma_2 \Sequent !B$، فإن~$\Subst{\delta_1}{\delta_2}{x:!B}$ هو
ناتج استبدال~$\delta_2$ بكل مسلّمة~$x:!B \Sequent !B$ في~$\delta_1$،
وإضافة~$\Gamma_2$ إلى سياق كل متتالية تقع تحت تلك المسلّمات.
\end{defn}

\begin{prop}
إذا كان~$\delta_1 \Proves x:!B, \Gamma_1 \Sequent !A$ و
$\delta_2 \Proves \Gamma_2 \Sequent !B$، فإن
$\Subst{\delta_1}{\delta_2}{x:!B} \Proves \Gamma_1, \Gamma_2 \Sequent
!A$، بشرط ألا يظهر أي متغير مميز من~$\delta_1$ في~$\Gamma_2$.
\end{prop}


\begin{proof}
  بالاستقراء على~$\pheight{\delta_1}$.
\end{proof}
