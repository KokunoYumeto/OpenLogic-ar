\olsection{الإحلال}

\begin{explain}
لما كانت المتغيرات الحرة تحيل إلى متغيرات البيئة، جاز أن نضع قيمة بعينها موضع المتغير الحر.
فقد نريد، مثلًا، أن نستبدل~$f$ في
$\lambd[x][f x]$ بحد معين، هو دالة الهوية~$\lambd[y][y]$، فنحصل على
$\lambd[x][(\lambd[y][y]) x]$. وهذا الاستبدال، أعني وضع حدود لامبدا موضع المتغيرات الحرة، هو الإحلال.
\end{explain}

\begin{defn}[الإحلال] \ollabel{defn:substitution}
  \emph{إحلال} الحد~$N$ محل المتغير~$x$ في الحد~$M$، ورمزه
  $\Subst{M}{N}{x}$، يحدّد بالقواعد الاستقرائية التالية:
  \begin{enumerate}
    \item $\Subst{x}{N}{x} = N$. \ollabel{defn:substitution-1} 
    \item $\Subst{y}{N}{x}  = y$ متى كان~$x \neq y$. \ollabel{defn:substitution-2} 
    \item $\Subst{PQ}{N}{x}  = (\Subst{P}{N}{x}) (\Subst{Q}{N}{x})$.
      \ollabel{def:substitution-3}
    \item $\Subst{(\lambd[y][P])}{N}{x} = \lambd[y][\Subst{P}{N}{x}]$،
      بشرط~$x \neq y$ و$y \notin \FV{N}$؛ وإذا تخلف الشرط فالإحلال غير معرّف.
      \ollabel{defn:substitution-4}
  \end{enumerate}
\end{defn}

\begin{explain}
وجه اشتراط~$x
\neq y$ في~\olref{defn:substitution}\olref{defn:substitution-4}
أننا لا نريد وضع~$N$ محل الوقائع \emph{المقيدة} للمتغير~$x$
في~$M$. فإذا طلبنا، مثلًا، حساب~$\Subst{\lambd[x][x]}{y}{x}$، امتنع أن نجعل نتيجته~$\lambd[x][y]$.
بل هذا الإحلال غير معرّف في التعريف الذي تعمدنا جعله جزئيًا.
وسيأتي علاج ذلك بتحويل~$\alpha$، أي بتغيير أسماء المتغيرات المقيدة.

ولإحلال~$N$ محل~$x$ في~$\lambd[y][P]$ شرط آخر، هو~$y
\notin \FV{N}$.
فلا يجوز في هذا التعريف الجزئي أن نحل~$y$ محل~$x$ في~$\lambd[y][x]$؛
إذ يكون~$\Subst{\lambd[y][x]}{y}{x}$ غير معرّف. ولو أجرينا الاستبدال بلا احتراز لحصلنا على
$\lambd[y][y]$، وهو يمثل دالة تأخذ وسيطًا ثم تردّه كما هو.
لكن~$\lambd[y][x]$ يمثل دالة تردّ دائمًا الحد~$x$، أو ما يدل عليه~$x$.
فالمطلوب في الحقيقة دالة تأخذ وسيطًا وتهمله، ثم تردّ متغير البيئة~$y$.
ولا يستقيم هذا إلا بتغيير اسم المتغير المقيد~$y$ أولًا؛ وسيأتي في تحويل ألفا حل هذه المسألة.
\end{explain}

\begin{prob}
  احسب الإحلالات الآتية، وبيّن ما ليس معرّفًا منها:
  \begin{enumerate}
  \item $\Subst{\lambd[y][x(\lambd[w][vwx])]}{(uv)}{x}$
  \item $\Subst{\lambd[y][x(\lambd[x][x])]}{(\lambd[y][xy])}{x}$
  \item $\Subst{y(\lambd[v][xv])}{(\lambd[y][vy])}{x}$
  \end{enumerate}
\end{prob}

\begin{thm} \ollabel{thm:notinfv}
  من~$x \notin \FV{M}$ يلزم~$\FV{\Subst{M}{N}{x}} = \FV{M}$،
  على شرط أن يكون الطرف الأيسر معرّفًا.
\end{thm}

\begin{proof}
  نستقرئ تكوين~$M$، فنميز:
  \begin{enumerate}
  \item حالة كون~$M$ متغيرًا؛ وإثباتها تمرين.
  \item حالة كون~$M$ على صورة~$(PQ)$؛ وإثباتها تمرين.
  \item حالة كون~$M$ على صورة~$\lambd[y][P]$. لما كان
    $\Subst{\lambd[y][P]}{N}{x}$ معرّفًا، وجب أن يساوي
    $\lambd[y][\Subst{P}{N}{x}]$. فيلزم أن يكون
    $\Subst{P}{N}{x}$ معرّفًا، وأن تتحقق أيضًا~$x \neq y$ و$y \notin \FV{N}$
    و$x \notin \FV{P}$. ومن ثم نحصل على
    \begin{multline*}
      \FV{\Subst{\lambd[y][P]}{N}{x}} = \\
    \begin{aligned}
      & = \FV{\lambd[y][\Subst{P}{N}{x}]} &&
      \text{بموجب \olref{defn:substitution-4}}\\
      & = \FV{\Subst{P}{N}{x}} \setminus \{y\} &&
      \text{بموجب \olref[fv]{def:fv}\olref[fv]{def:fv2}}\\
      & = \FV{P} \setminus \{y\} && \text{بفرض الاستقراء} \\
      & = \FV{\lambd[y][P]} && \text{بموجب \olref[fv]{def:fv}\olref[fv]{def:fv2}}
    \end{aligned}
    \end{multline*}
  \end{enumerate}
\end{proof}

\begin{prob}
  أتمّ الحالات المتروكة من برهان~\olref[lam][syn][sub]{thm:notinfv}.
\end{prob}

\begin{thm} \ollabel{thm:infv}
  من~$x \in \FV{M}$ يلزم~$\FV{\Subst{M}{N}{x}} = (\FV{M} \setminus
  \{x\}) \cup \FV{N}$، متى كان الطرف الأيسر معرّفًا.
\end{thm}

\begin{proof}
  نستقرئ تكوين~$M$:
  \begin{enumerate}
  \item إن كان~$M$ متغيرًا، ترك الإثبات تمرينًا.
  \item إن كان~$M$ على صورة~$PQ$، فإن تعريف
    $\Subst{(PQ)}{N}{x}$ يقتضي أن يساوي
    $(\Subst{P}{N}{x})(\Subst{Q}{N}{x})$، وأن يكون الإحلالان معرّفين.
    ثم إن~$x \in \FV{PQ}$ يقتضي~$x \in
    \FV{P}$ أو~$x \in \FV{Q}$، وقد يجتمعان. والباقي تمرين.
  \item إن كان~$M$ على صورة~$\lambd[y][P]$، فإن تعريف
    $\Subst{\lambd[y][P]}{N}{x}$ يقتضي أن يساوي
    $\lambd[y][\Subst{P}{N}{x}]$، مع تعريف~$\Subst{P}{N}{x}$
    وتحقق~$x \neq y$ و$y \notin \FV{N}$. ومن~$x \in
    \FV{\lambd[y][P]}$ يلزم~$x \in \FV{P}$ كذلك. فنحسب
    \begin{multline*}
      \FV{\Subst{(\lambd[y][P])}{N}{x}} = \\
      \begin{aligned}
      & = \FV{\lambd[y][\Subst{P}{N}{x}]} \\
      & = \FV{\Subst{P}{N}{x}} \setminus \{y\} \\
      & = ((\FV{P}\setminus\{x\})\cup\FV{N})\setminus\{y\}
       && \text{بفرض الاستقراء} \\
      & = (\FV{P}\setminus\{x,y\})\cup\FV{N}
       && \text{لأن~$y\notin\FV{N}$} \\
      & = (\FV{\lambd[y][P]} \setminus \{x\}) \cup \FV{N}
      \end{aligned}
      \end{multline*}
  \end{enumerate}
\end{proof}

\begin{prob}
  أتمّ ما ترك من برهان~\olref[lam][syn][sub]{thm:infv}.
\end{prob}

\begin{thm}\ollabel{thm:clr}
  تتحقق~$x \notin \FV{\Subst{M}{N}{x}}$ متى كان الطرف الأيمن معرّفًا
  وتحقق~$x \notin \FV{N}$.
\end{thm}

\begin{proof}
  إثبات ذلك تمرين.
\end{proof}

\begin{prob}
  أقم برهان~\olref[lam][syn][sub]{thm:clr}.
\end{prob}

\begin{thm}\ollabel{thm:inv}
  إذا كان~$\Subst{M}{y}{x}$ معرّفًا، وتحقق~$y \notin \FV{M}$، لزم
  $\Subst{\Subst{M}{y}{x}}{x}{y} = M$.
\end{thm}

\begin{proof}
  نستقرئ تكوين~$M$:
  \begin{enumerate}
  \item إن كان~$M$ متغيرًا~$z$، فالإثبات تمرين.
  \item إن كان~$M$ على صورة~$(PQ)$، حسبنا
    \begin{align*}
      \Subst{\Subst{(PQ)}{y}{x}}{x}{y}
      &=\Subst{((\Subst{P}{y}{x})(\Subst{Q}{y}{x}))}{x}{y} \\
      &= (\Subst{\Subst{P}{y}{x}}{x}{y})(\Subst{\Subst{Q}{y}{x}}{x}{y}) \\
      &= (PQ) \text{ بفرض الاستقراء}
    \end{align*}
  \item وإن كان~$M$ على صورة~$\lambd[z][N]$، اقتضى تعريف
    $\Subst{\lambd[z][N]}{y}{x}$ أن يكون~$z\neq x$ و$z\neq y$، وأن
    يكون~$\Subst{N}{y}{x}$ معرّفًا. ومن~$y\notin\FV{\lambd[z][N]}$
    و$z\neq y$ يلزم~$y\notin\FV{N}$. فيصح فرض الاستقراء في الإحلال
    الداخلي، وتتم كذلك شروط تعريف الإحلال العكسي تحت~$\lambd[z]$؛ وعليه
    \begin{align*}
      \Subst{\Subst{(\lambd[z][N])}{y}{x}}{x}{y}\\
      & = \Subst{(\lambd[z][\Subst{N}{y}{x}])}{x}{y} \\
      & = \lambd[z][\Subst{\Subst{N}{y}{x}}{x}{y}] \\
      &= \lambd[z][N] \text{ بفرض الاستقراء} 
    \end{align*}
    % تصحيح مصدر (0361-I1): صُرّح بشروط التعريف والنضارة اللازمة للإحلالين ولفرض الاستقراء.
  \end{enumerate}
\end{proof}

\begin{prob}
  أتمّ برهان~\olref[lam][syn][sub]{thm:inv}.
\end{prob}
