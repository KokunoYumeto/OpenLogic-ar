\olsection{الاستنباط الطبيعي}

الاستنباط الطبيعي إذا حذفت منه قواعد~$\FalseCl$ نظام معياري من أنظمة
!!{derivation} للمنطق الحدسي. وسنورد قواعده، ثم نبين وجهها على تأويل BHK:
فكل قاعدة تنقلنا من وجود بناءات لمقدماتها إلى وجود بناء لنتيجتها.

ولما كانت !!{derivation}s الاستنباط الطبيعي قد تشتمل على افتراضات غير
مصروفة، حملنا !!a{derivation} لــ$!A$ من افتراضات
!!{undischarged}~$\Gamma$ على دالة تنقل بناءات كل~$!B \in \Gamma$ إلى
بناء لــ$!A$. فإذا وجدت !!a{derivation} لــ$!A$ لا افتراض فيها
!!{undischarged}، وجد بناء لــ$!A$ بحسب BHK. وسنسقط~$\Gamma$ حيث لا
يحتاج إليها البيان.

والافتراض~$!A$ وحده !!a{derivation} لــ$!A$ من الافتراض
!!{undischarged}~$!A$؛ وهذا موافق لتأويل BHK، لأن دالة الهوية تنقل كل
بناء لــ$!A$ إلى بناء لــ$!A$.

\subsection{العطف}

\begin{defish}
\AxiomC{$!A$}
\AxiomC{$!B$}
\RightLabel{\Intro{\land}}
\BinaryInfC{$!A \land !B$}
\DisplayProof
\hfill
\begin{tabular}{l}
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!A$}
\DisplayProof
\\[3ex]
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!B$}
\DisplayProof
\end{tabular}
\end{defish}

إذا كان لدينا البناءان~$N_1$ و$N_2$ لــ$!A_1$ و$!A_2$ على الترتيب،
كان لدينا بناء لــ$!A_1 \land !A_2$، وهو الزوج~$\tuple{N_1, N_2}$.

وبناء~$!A_1 \land !A_2$ في تأويل BHK زوج~$\tuple{N_1, N_2}$. فإذا
أعطي هذا الزوج حصل بناء كل معطوف: فــ$N_1$ بناء لــ$!A_1$، و$N_2$ بناء
لــ$!A_2$.

\subsection{الشرط}

\begin{defish}
  \AxiomC{$\Discharge{!A}{u}$}
  \DeduceC{$!B$}
  \DischargeRule{$\Intro{\lif}$}{u}
  \UnaryInfC{$!A \lif !B$}
  \DisplayProof
\hfill
  \AxiomC{$!A \lif !B$}
  \AxiomC{$!A$}
  \RightLabel{$\Elim{\lif}$}
  \BinaryInfC{$!B$}
  \DisplayProof
\end{defish}

إذا قامت !!a{derivation} لــ$!B$ من الافتراض
!!{undischarged}~$!A$، قامت دالة~$f$ تنقل بناءات~$!A$ إلى بناءات~$!B$.
وهذه الدالة بعينها بناء لــ$!A \lif !B$؛ فإذا كان لمقدمة~$\Intro\lif$
بناء مشروط ببناء لــ$!A$، كان للنتيجة~$!A \lif !B$ بناء.

وأما الحذف، فإذا وجد بناء~$N$ لــ$!A$ وبناء~$f$ لــ$!A \lif !B$، كانت
الدالة التي هي بناء~$!A \lif !B$ ناقلة لبناءات~$!A$ إلى بناءات~$!B$؛
فكان~$f(N)$ بناء لــ$!B$، وهو بناء نتيجة~$\Elim\lif$.

\subsection{الفصل}

\begin{defish}
\begin{tabular}{l}
\AxiomC{$!A$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\\[3ex]
\AxiomC{$!B$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\end{tabular}
\hfill
\AxiomC{$!A \lor !B$}
\AxiomC{$\Discharge{!A}{n}$}
\DeduceC{$!C$}
\AxiomC{$\Discharge{!B}{n}$}
\DeduceC{$!C$}
\DischargeRule{\Elim{\lor}}{n}
\TrinaryInfC{$!C$}
\DisplayProof
\end{defish}

إذا كان~$N_i$ بناء لــ$!A_i$، صار الزوج~$\tuple{i, N_i}$ بناء لــ
$!A_1 \lor !A_2$. وأما إذا أعطينا بناءً لــ$!A_1 \lor !A_2$، أي زوجًا
$\tuple{i, N_i}$ يكون~$N_i$ فيه بناء لــ$!A_i$، وأعطينا دالتين~$f_1$
و$f_2$ تنقلان بناءات~$!A_1$ و$!A_2$ على الترتيب إلى بناءات~$!C$، كان
$f_i(N_i)$ بناء لــ$!C$، وهو نتيجة~$\Elim\lor$.

\subsection{المحال}

\begin{defish}
  \AxiomC{$\lfalse$}
  \RightLabel{\FalseInt}
  \UnaryInfC{$!A$}
  \DisplayProof
\end{defish}

إذا قامت !!a{derivation} لــ$\lfalse$ من الافتراضات
!!{undischarged}~$!B_1$، \dots، $!B_n$، قامت دالة
$f(M_1, \dots, M_n)$ تنقل بناءات~$!B_1$، \dots، $!B_n$ إلى بناء
لــ$\lfalse$. ولما لم يكن لــ$\lfalse$ بناء، امتنع اجتماع بناءات لكل
$!B_1$، \dots، $!B_n$. ومن ثم تصدق في~$f$ القضية الآتية أيضًا:
\emph{إذا} كانت~$M_1$، \dots، $M_n$ بناءات لــ$!B_1$،
\dots، $!B_n$ على الترتيب، \emph{فإن}~$f(M_1, \dots, M_n)$ بناء
لــ$!A$.

\subsection{قواعد $\lnot$}

لما كانت~$\lnot !A$ معرّفة بــ$!A \lif \lfalse$، استغنينا من جهة
الدقة عن قواعد~$\lnot$. وإن أفردناها كانت كما يأتي:

\begin{defish}
\AxiomC{$\Discharge{!A}{n}$}
\noLine
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{n}
\UnaryInfC{$\lnot !A$}
\DisplayProof
\hfill
\AxiomC{$\lnot !A$}
\AxiomC{$!A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\DisplayProof
\end{defish}


\subsection{أمثلة على \usetoken{P}{derivation}}

\begin{enumerate}
\item $\Proves !A \lif (\lnot !A \lif \lfalse)$، أي $\Proves !A
  \lif ((!A \lif \lfalse) \lif \lfalse)$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A}{2}$}
    \AxiomC{$\Discharge{!A \lif \lfalse}{1}$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \lif \lfalse)\lif \lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!A \lif \lfalse) \lif \lfalse$}
  \end{prooftree}

\item $\Proves ((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$
  \begin{prooftree}
    \AxiomC{$\Discharge{(!A \land !B) \lif !C}{3}$}
    \AxiomC{$\Discharge{!A}{2}$}
    \AxiomC{$\Discharge{!B}{1}$}
    \RightLabel{\Intro\land}
    \BinaryInfC{$!A \land !B$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$!C$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!B \lif !C$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!B \lif !C)$}
    \DischargeRule{\Intro\lif}{3}
    \UnaryInfC{$((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$}
  \end{prooftree}

\item $\Proves \lnot(!A \land \lnot !A)$، أي $\Proves (!A \land (!A
  \lif \lfalse))\lif \lfalse$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{1}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A \lif \lfalse$}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{1}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \land (!A \lif \lfalse))\lif \lfalse$}
  \end{prooftree}

\item $\Proves \lnot\lnot(!A \lor \lnot !A)$، أي $\Proves ((!A \lor
  (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$
  \begin{prooftree}\hskip-3em
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{2}$}
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{2}$}
    \AxiomC{$\Discharge{!A}{1}$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!A \lif \lfalse$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \insertBetweenHyps{\hskip -4em}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$((!A \lor (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$}
  \end{prooftree}
\end{enumerate}

\begin{prop}
  إذا كان~$\Gamma \Proves !A$ في الاستنباط الطبيعي الحدسي، كان
  $\Gamma \Proves !A$ في المنطق الكلاسيكي. وعلى الخصوص، كل~$!A$
  تكون مبرهنة حدسية هي مبرهنة كلاسيكية.
\end{prop}

\begin{proof}
  لأن كل قاعدة من قواعد الاستنباط الطبيعي الحدسي قاعدة في الاستنباط
  الطبيعي الكلاسيكي؛ فكل !!{derivation} حدسية !!a{derivation}
  كلاسيكية أيضًا.
\end{proof}

\begin{prob}
  أقم في المنطق الحدسي !!{derivation}s لل!!{formula}s الآتية:
  \begin{enumerate}
    \item $(\lnot !A \lor !B) \lif (!A \lif !B)$
    \item $\lnot\lnot\lnot !A \lif \lnot !A$
    \item $\lnot\lnot (!A \land !B) \liff (\lnot\lnot !A\land \lnot \lnot !B)$
    \item $\lnot (!A \lor !B) \liff (\lnot !A \land \lnot !B)$
    \item $(\lnot !A \lor \lnot !B) \lif \lnot (!A \land !B)$
    \item $\lnot\lnot ( !A \land !B) \lif  (\lnot\lnot !A \lor
    \lnot\lnot !B)$
  \end{enumerate}
\end{prob}
