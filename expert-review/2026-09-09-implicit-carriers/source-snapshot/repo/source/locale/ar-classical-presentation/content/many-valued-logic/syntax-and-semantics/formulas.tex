\olsection{\usetoken{P}{formula}}

\begin{defn}[الصيغة]
\ollabel{defn:formulas}
مجموعة~$\Frm[{\OLId{latin-looped}{L}}]$، وهي مجموعة~\emph{!!{formula}s} لغة القضايا
$\Lang L$، تعرف بالاستقراء كما يأتي:
\begin{enumerate}
\item كل~!!{propositional variable}~$\Obj p_{\OLId{latin-ordinary}{i}}$~!!{formula} ذرية.
\item كل رابط ذي~${\OLMathNumeral{0}}$ من المواضع في~$\Lang L$، أي كل ثابت قضوي،
!!{formula} ذرية.
\item إذا كان~$\star$ رابطًا ذا~${\OLId{latin-ordinary}{n}}$ من المواضع في~$\Lang L$، وكان
${\OLId{latin-ordinary}{n}}>{\OLMathNumeral{0}}$، وكانت~$!A_{\OLMathNumeral{1}}$، \dots،~$!A_{\OLId{latin-ordinary}{n}}$~!!{formula}s، كانت
$\star(!A_{\OLMathNumeral{1}}, \dots, !A_{\OLId{latin-ordinary}{n}})$~!!a{formula}.
\tagitem{limitClause}{ولا يعد~!!a{formula} شيء سوى ما بني بهذه البنود.}{}
\end{enumerate}
فإذا كان~$\star$ ذا موضع واحد، أوجزنا غالبًا~$\star(!A_{\OLMathNumeral{1}})$ إلى
$\star !A_{\OLMathNumeral{1}}$. وإذا كان~$\star$ ذا موضعين، أوجزنا
$\star(!A_{\OLMathNumeral{1}},!A_{\OLMathNumeral{2}})$ إلى~$(!A_{\OLMathNumeral{1}} \star !A_{\OLMathNumeral{2}})$.
\end{defn}

وكعادتنا نحذف زوج الأقواس الخارجي حيث لا يقع لبس.

\begin{ex}
  في اللغة القياسية~$\Lang{L_0}$، تعد
  $\Obj p_{\OLMathNumeral{1}} \lif (\Obj p_{\OLMathNumeral{1}} \land \lnot \Obj p_{\OLMathNumeral{2}})$ صيغة.
  وأما في لغة منطق الضرب فتكتب
  $\Obj p_{\OLMathNumeral{1}} \lif (\Obj p_{\OLMathNumeral{1}} \odot \lnot \Obj p_{\OLMathNumeral{2}})$.
  وإذا زدنا في اللغة الرابط الأحادي~$\triangle$، دخلت فيها أيضًا
  صيغ نحو~$\triangle (\Obj p_{\OLMathNumeral{1}} \land \Obj p_{\OLMathNumeral{2}}) \lif
  (\triangle \Obj p_{\OLMathNumeral{1}} \land \triangle \Obj p_{\OLMathNumeral{2}})$.
\end{ex}
