\olsection{ال\printtoken{P}{variable} الحرة وال\printtoken{P}{sentence}}

\begin{defn}[الورودات الحرة ل!!a{variable}]
\ollabel{defn:free-occ}
أما الورودات \emph{الحرة} ل!!a{variable} في !!a{formula}،
فيُعرف حدّها الاستقرائي من الحالات الآتية:
\begin{enumerate}
\item \indcase*{!A}{$!A$ ذرية}{جميع ورودات ال!!{variable}s في
  $\indfrm$ حرة.}

\tagitem{prvNot}{\indcase{!A}{\lnot !B}{ورودات ال!!{variable}s
  الحرة في~$\indfrm$ هي وروداتها في~$!B$ بالضبط.}}{}

\item \indcase{!A}{(!B \ast !C)}{ورودات ال!!{variable}s الحرة
  في~$\indfrm$ هي وروداتها في~$!B$ مع وروداتها في~$!C$.}

\tagitem{prvAll}{\indcase{!A}{\lforall[x][!B]}{ورودات
  ال!!{variable}s الحرة في~$\indfrm$ هي جميع وروداتها في~$!B$ عدا
  ورودات~${\OLId{latin-ordinary}{x}}$.}}{}

\tagitem{prvEx}{\indcase{!A}{\lexists[x][!B]}{ورودات
  ال!!{variable}s الحرة في~$\indfrm$ هي جميع وروداتها في~$!B$ عدا
  ورودات~${\OLId{latin-ordinary}{x}}$.}}{}
\end{enumerate}
\end{defn}

\begin{defn}[المتغيرات المقيدة]
وكل ورود ل!!a{variable} في صيغة~$!A$ ليس حرًا فهو \emph{مقيد}.
\end{defn}

\begin{prob}
ضع لورودات المتغيرات المقيدة تعريفًا استقرائيًا على مثال
\olref[fol][syn][fvs]{defn:free-occ}.
\end{prob}

\begin{defn}[المجال]
\iftag{prvAll}{إذا كانت $\lforall[{\OLId{latin-ordinary}{x}}][!B]$ ورودًا لصيغة جزئية في
  صيغة~$!A$، فإن الورود المقابل لـ~$!B$ في~$!A$ يسمى
  \emph{مجال} الورود المقابل لـ~$\lforall[{\OLId{latin-ordinary}{x}}]$.
  \iftag{prvEx}{وكذلك الأمر لـ~$\lexists[{\OLId{latin-ordinary}{x}}]$.}{}}{إذا كانت
  $\lexists[{\OLId{latin-ordinary}{x}}][!B]$ ورودًا لصيغة جزئية في صيغة~$!A$، فإن الورود
  المقابل لـ~$!B$ في~$!A$ يسمى \emph{مجال} الورود المقابل
  لـ~$\lexists[{\OLId{latin-ordinary}{x}}]$.}

إذا كانت $!B$ مجال ورود المكمم
\iftag{prvAll}{$\lforall[{\OLId{latin-ordinary}{x}}]$\iftag{prvEx}{ أو
    $\lexists[{\OLId{latin-ordinary}{x}}]$}{}}{$\lexists[{\OLId{latin-ordinary}{x}}]$} في~$!A$، فإن الورودات الحرة
لـ~${\OLId{latin-ordinary}{x}}$ في~$!B$ يقيدها ورود المكمم المذكور. وبهذا الاعتبار نسمي هذه الورودات
\emph{مقيدة بورود المكمم المذكور}.
\end{defn}

\begin{ex}
انظر في الصيغة الآتية:
\[
\lexists[\Obj v_{\OLMathNumeral{0}}][\underbrace{\Atom{\Obj A^2_0}{\Obj v_0,\Obj v_1}}_{!B}]
\]
فـ$!B$ مجال~$\lexists[\Obj v_{\OLMathNumeral{0}}]$.
والذي يقيّده هذا المكمم هو ورود~$\Obj v_{\OLMathNumeral{0}}$ في~$!B$، دون ورود~$\Obj
v_{\OLMathNumeral{1}}$؛ فيبقى~$\Obj v_{\OLMathNumeral{1}}$ هنا متغيرًا حرًا.


ويتضح التفصيل في !!{formula}~$!A$ الآتية، وهي أكثر تركيبًا:
\[
\lforall[\Obj v_{\OLMathNumeral{0}}][\underbrace{(\Atom{\Obj A^1_0}{\Obj v_0} \lif
    \Atom{\Obj A^2_0}{\Obj v_0, \Obj v_1})}_{!B}] \lif \lexists[\Obj
  v_{\OLMathNumeral{1}}][\underbrace{(\Atom{\Obj A^2_1}{\Obj v_0, \Obj v_1} \lor \lforall[\Obj v_{\OLMathNumeral{0}}][\overbrace{\lnot \Atom{\Obj A^1_1}{\Obj v_0}}^{!D}])}_{!C}]
\]
$!B$ مجال الورود الأول لـ~$\lforall[\Obj v_{\OLMathNumeral{0}}]$، و$!C$ مجال
$\lexists[\Obj v_{\OLMathNumeral{1}}]$، و$!D$ مجال الورود الثاني
لـ~$\lforall[\Obj v_{\OLMathNumeral{0}}]$. فأما الورود الأول لـ~$\lforall[\Obj
v_{\OLMathNumeral{0}}]$ فيقيّد ورودات~$\Obj v_{\OLMathNumeral{0}}$ في~$!B$، وأما $\lexists[\Obj v_{\OLMathNumeral{1}}]$
فيقيّد ورود~$\Obj v_{\OLMathNumeral{1}}$ في~$!C$، وأما الورود الثاني لـ~$\lforall[\Obj
v_{\OLMathNumeral{0}}]$ فيقيّد ورود~$\Obj v_{\OLMathNumeral{0}}$ في~$!D$. ويبقى الورود الأول لـ~$\Obj v_{\OLMathNumeral{1}}$
والورود الرابع لـ~$\Obj v_{\OLMathNumeral{0}}$ حرين في~$!A$. وأما الورود الأخير
لـ~$\Obj v_{\OLMathNumeral{0}}$، فهو حر إذا اعتُبرت~$!D$ وحدها، ومقيد إذا اعتُبرت~$!C$ أو $!A$.
\end{ex}

\begin{defn}[الجملة]
نسمي !!^a{formula}~$!A$ \article{sentence} \emph{!!{sentence}} إذا
وفقط إذا خلت من كل ورود حر ل!!{variable}s.
\end{defn}
