\olsection{التعدادات ومجموعات \usetoken{S}{enumerable}}

\begin{editorial}
التعداد هنا دالة تقابلية مع $\Nat$ أو مع مقطع ابتدائي منها، على اصطلاح نظرية المجموعات. فبين هذا التعريف وتعريفات \olref[enm]{sec} فرق يسير، وترد هنا أمثلة ذلك القسم كلها مرة أخرى. وهذا العرض أوجز منه بعض الإيجاز.
\end{editorial}

من طرق تعيين المجموعة المنتهية أن نسرد !!{element}sها؛ ومن ذلك أن نكتب
\[
  {\OLId{latin-looped}{A}} = \{{\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, {\OLId{latin-ordinary}{a}}_{\OLMathNumeral{2}}, \ldots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}\}.
\]
فإذا كانت !!{element}s~${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}$، \dots،~${\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}$ متمايزة كلها، نشأ من هذا السرد !!a{bijection} بين ${\OLId{latin-looped}{A}}$ والأعداد الطبيعية الأولى التي عددها ${\OLId{latin-ordinary}{n}}$، وهي ${\OLMathNumeral{0}}$، \dots،~${\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}$. وبالعكس، كل مجموعة منتهية يمكن أن نجعلها في مثل هذه المراسلة، لأن عدد !!{element}sها منته. فمتى كانت ${\OLId{latin-looped}{A}}$ منتهية وجدت !!a{bijection} بين ${\OLId{latin-looped}{A}}$ و$\{{\OLMathNumeral{0}}, \dots, {\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}\}$، حيث ${\OLId{latin-ordinary}{n}}$ عدد !!{element}s~${\OLId{latin-looped}{A}}$.

وإذا أجزنا ضروبًا من المجموعات اللانهائية أمكن أن نجيز لبعضها تعدادًا أيضًا. وضبط هذا أن تعداد المجموعة اللانهائية يكون بـ!!a{bijection} بينها وبين مجموعة الأعداد الطبيعية بأسرها~$\Nat$.

\begin{defn}[التعداد في اصطلاح نظرية المجموعات]
\emph{تعداد} المجموعة ${\OLId{latin-looped}{A}}$ هو !!a{bijection} مداها ${\OLId{latin-looped}{A}}$، ومجالها مقطع ابتدائي من الأعداد الطبيعية $\{{\OLMathNumeral{0}}, {\OLMathNumeral{1}}, \ldots, {\OLId{latin-ordinary}{n}}\}$ {أو} مجموعة الأعداد الطبيعية كلها~$\Nat$.
\end{defn}

\begin{explain}
واختيار اسم \emph{التعداد} موافق للتصور؛ فقولنا إننا عددنا المجموعة ${\OLId{latin-looped}{A}}$ معناه وجود !!a{bijection}~${\OLId{latin-ordinary}{f}}$ نحصي بها عناصر المجموعة~${\OLId{latin-looped}{A}}$ واحدًا بعد واحد: ذو الرتبة~${\OLMathNumeral{0}}$ هو ${\OLId{latin-ordinary}{f}}({\OLMathNumeral{0}})$، والذي يليه ${\OLId{latin-ordinary}{f}}({\OLMathNumeral{1}})$، \ldots وذو الرتبة~${\OLId{latin-ordinary}{n}}$ هو ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{n}})$\ldots.\footnote{فالعد هنا مبتدئ من ${\OLMathNumeral{0}}$؛ ولو ابتدأناه من~${\OLMathNumeral{1}}$ لم يتغير المقصود تغيرًا ذا بال، وإنما نستبدل~$\Nat$ بـ~$\PosInt$.} ويزداد هذا المعنى بيانًا بالتعريف التالي.
\end{explain}

\begin{defn}
\ollabel{defn:enumerable}
المجموعة~${\OLId{latin-looped}{A}}$ !!{enumerable} إذا وفقط إذا كان ${\OLId{latin-looped}{A}} = \emptyset$ أو وجد لـ~${\OLId{latin-looped}{A}}$ تعداد. وأما ${\OLId{latin-looped}{A}}$ فتسمى !!{nonenumerable} إذا وفقط إذا لم تكن ${\OLId{latin-looped}{A}}$ !!{enumerable}.
\end{defn}

\begin{explain}
فحاصل الأمر أن المجموعة !!{enumerable} إما خالية، وإما ذات تعداد يتيح عد !!{element}sها واحدًا واحدًا.
\end{explain}

\begin{ex}
يكفينا لتعداد الأعداد الطبيعية دالة الهوية $\Id{\Nat} \colon \Nat \to \Nat$ التي يحدها $\Id{\Nat}({\OLId{latin-ordinary}{n}}) = {\OLId{latin-ordinary}{n}}$. وأما الأعداد الطبيعية \emph{الموجبة}، أي $\Nat^+ =
\Nat \setminus \{{\OLMathNumeral{0}}\}$، فتعددها دالة اللاحق ${\OLId{latin-ordinary}{g}}({\OLId{latin-ordinary}{n}}) = {\OLId{latin-ordinary}{n}} + {\OLMathNumeral{1}}$.
\end{ex}

\begin{prob}
برهن أن ${\OLId{latin-looped}{A}}$ !!{enumerable} إذا وفقط إذا كان ${\OLId{latin-looped}{A}} = \emptyset$ أو وجدت !!a{surjection} ${\OLId{latin-ordinary}{f}}\colon \Nat \to {\OLId{latin-looped}{A}}$. ثم برهن أن ${\OLId{latin-looped}{A}}$ !!{enumerable} إذا وفقط إذا وجدت !!a{injection} ${\OLId{latin-ordinary}{g}}\colon {\OLId{latin-looped}{A}} \to \Nat$.
\end{prob}

\begin{ex}
لتكن الدالتان ${\OLId{latin-ordinary}{f}}\colon \Nat \to \Nat$ و${\OLId{latin-ordinary}{g}} \colon \Nat \to \Nat$ معينتين بالحدين
\begin{align*}
  {\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{n}}) & = {\OLMathNumeral{2}}{\OLId{latin-ordinary}{n}} \text{ و}\\
  {\OLId{latin-ordinary}{g}}({\OLId{latin-ordinary}{n}}) & = {\OLMathNumeral{2}}{\OLId{latin-ordinary}{n}}+{\OLMathNumeral{1}}
\end{align*}
فالأولى تعدد الأعداد الطبيعية الزوجية، والثانية تعدد الفردية. لكن لا واحدة منهما !!{surjective} إلى المجال المقابل المعين؛ ولذلك ليست أي منهما تعدادًا لـ$\Nat$.
\end{ex}

\begin{prob}
عين دالة تعدد الأعداد المربعة ${\OLMathNumeral{1}}$، ${\OLMathNumeral{4}}$، ${\OLMathNumeral{9}}$، ${\OLMathNumeral{16}}$، \dots
\end{prob}

\begin{ex}
نكتب $\lceil {\OLId{latin-ordinary}{x}} \rceil$ لـ\emph{دالة السقف}، وقيمتها أصغر عدد صحيح لا يقل عن ${\OLId{latin-ordinary}{x}}$. فالدالة ${\OLId{latin-ordinary}{f}} \colon \Nat \to \Int$ المعرفة بالحد
\[
  {\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{n}}) = (-{\OLMathNumeral{1}})^{{\OLId{latin-ordinary}{n}}} \left\lceil\tfrac{{\OLId{latin-ordinary}{n}}}{{\OLMathNumeral{2}}}\right\rceil
\]
تعدد مجموعة الأعداد الصحيحة~$\Int$، كما يظهر من القيم الآتية:
\[
\begin{array}{c c c c c c c c}
{\OLId{latin-ordinary}{f}}({\OLMathNumeral{0}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{1}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{2}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{3}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{4}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{5}}) & {\OLId{latin-ordinary}{f}}({\OLMathNumeral{6}}) & \dots \\ \\
\big\lceil \tfrac{{\OLMathNumeral{0}}}{{\OLMathNumeral{2}}} \big\rceil & -\big\lceil \tfrac{{\OLMathNumeral{1}}}{{\OLMathNumeral{2}}}\big\rceil &  \big\lceil \tfrac{{\OLMathNumeral{2}}}{{\OLMathNumeral{2}}} \big\rceil & -\big\lceil \tfrac{{\OLMathNumeral{3}}}{{\OLMathNumeral{2}}} \big\rceil & \big\lceil \tfrac{{\OLMathNumeral{4}}}{{\OLMathNumeral{2}}} \big\rceil  & -\big\lceil \tfrac{{\OLMathNumeral{5}}}{{\OLMathNumeral{2}}}\big\rceil & \big\lceil \tfrac{{\OLMathNumeral{6}}}{{\OLMathNumeral{2}}} \big\rceil & \dots \\ \\
{\OLMathNumeral{0}} & -{\OLMathNumeral{1}} & {\OLMathNumeral{1}} & -{\OLMathNumeral{2}} & {\OLMathNumeral{2}} & -{\OLMathNumeral{3}} & {\OLMathNumeral{3}}& \dots
\end{array}
\]
وترى أن ${\OLId{latin-ordinary}{f}}$ تستوفي قيم $\Int$ بالتنقل بين الأعداد الصحيحة الموجبة والسالبة. ويجوز أن نعين ${\OLId{latin-ordinary}{f}}$ بحالتين على الوجه الآتي:
\[
{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{n}}) = \begin{cases}
  \frac{{\OLId{latin-ordinary}{n}}}{{\OLMathNumeral{2}}} & \text{إذا كان ${\OLId{latin-ordinary}{n}}$ زوجيًا}\\
  -\frac{{\OLId{latin-ordinary}{n}}+{\OLMathNumeral{1}}}{{\OLMathNumeral{2}}} & \text{إذا كان ${\OLId{latin-ordinary}{n}}$ فرديًا}
  \end{cases}
\]
\end{ex}

\begin{prob}
إذا كانت ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$ !!{enumerable}، فبين أن ${\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}$ كذلك.
\end{prob}

\begin{prob}
استعمل الاستقراء على ${\OLId{latin-ordinary}{n}}$ لإثبات أن ${\OLId{latin-looped}{A}}_{\OLMathNumeral{1}}$، ${\OLId{latin-looped}{A}}_{\OLMathNumeral{2}}$، \dots،~${\OLId{latin-looped}{A}}_{\OLId{latin-ordinary}{n}}$ إذا كانت كلها !!{enumerable}، كان ${\OLId{latin-looped}{A}}_{\OLMathNumeral{1}} \cup \dots \cup {\OLId{latin-looped}{A}}_{\OLId{latin-ordinary}{n}}$ كذلك.
\end{prob}
