\olsection{في التشاكل}

\begin{explain}
\emph{التشاكل} تقابل بين مجموعتين يصون بنيتهما، والمراد بالبنية هنا
العلاقات القائمة بين !!{element}s كل منهما. فاعتبر المجموعتين
${\OLId{latin-looped}{X}}=\{{\OLMathNumeral{1}},{\OLMathNumeral{2}},{\OLMathNumeral{3}}\}$ و${\OLId{latin-looped}{Y}}=\{{\OLMathNumeral{4}},{\OLMathNumeral{5}},{\OLMathNumeral{6}}\}$؛ ففي كل واحدة منهما علاقات التالي، وأصغر
من، وأكبر من. والتشاكل بينهما !!a{bijection} يحفظ هذه العلاقات. فإذا
  % REVIEW-0717-N1: نحفظ اتجاه الشرط الكافي الذي تقرره النسخة الإنجليزية.
  كانت ${\OLId{latin-ordinary}{f}} \colon {\OLId{latin-looped}{X}} \to {\OLId{latin-looped}{Y}}$~!!a{bijective}، كانت تشاكلًا متى تحقق،
لكل ${\OLId{latin-ordinary}{i}},{\OLId{latin-ordinary}{j}}\in {\OLId{latin-looped}{X}}$، أن ${\OLId{latin-ordinary}{i}}<{\OLId{latin-ordinary}{j}}$ إذا وفقط إذا كان ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{i}})<{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{j}})$، وأن ${\OLId{latin-ordinary}{i}}>{\OLId{latin-ordinary}{j}}$ إذا
وفقط إذا كان ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{i}})>{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{j}})$، وأن ${\OLId{latin-ordinary}{j}}$ تالي ${\OLId{latin-ordinary}{i}}$ إذا وفقط إذا كان ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{j}})$ تالي
${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{i}})$.
\end{explain}

\begin{defn}[التشاكل]
ليكن ${\OLId{latin-looped}{U}}$ الزوج $\langle {\OLId{latin-looped}{X}}, {\OLId{latin-looped}{R}}\rangle$، و${\OLId{latin-looped}{V}}$ الزوج $\langle
{\OLId{latin-looped}{Y}}, {\OLId{latin-looped}{S}}\rangle$، على أن تكون ${\OLId{latin-looped}{X}}$ و${\OLId{latin-looped}{Y}}$ مجموعتين، و${\OLId{latin-looped}{R}}$ و${\OLId{latin-looped}{S}}$ علاقتين على
${\OLId{latin-looped}{X}}$ و${\OLId{latin-looped}{Y}}$ بالترتيب. وليكن ${\OLId{latin-ordinary}{f}}$~!!{bijection} من ${\OLId{latin-looped}{X}}$ إلى ${\OLId{latin-looped}{Y}}$. فنقول إن
${\OLId{latin-ordinary}{f}}$ \emph{تشاكل} من ${\OLId{latin-looped}{U}}$ إلى ${\OLId{latin-looped}{V}}$ إذا وفقط إذا حفظ البنية العلائقية؛
أي إذا كان، لكل ${\OLId{latin-ordinary}{x}}_{{\OLMathNumeral{1}}}$ و${\OLId{latin-ordinary}{x}}_{{\OLMathNumeral{2}}}$ في ${\OLId{latin-looped}{X}}$،
$\tuple{{\OLId{latin-ordinary}{x}}_{\OLMathNumeral{1}},{\OLId{latin-ordinary}{x}}_{\OLMathNumeral{2}}} \in {\OLId{latin-looped}{R}}$ إذا وفقط إذا كان $\tuple{{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}_{\OLMathNumeral{1}}),{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}_{\OLMathNumeral{2}})} \in {\OLId{latin-looped}{S}}$.
\end{defn}

\begin{ex}
في المجموعتين ${\OLId{latin-looped}{X}}=\{{\OLMathNumeral{1}},{\OLMathNumeral{2}},{\OLMathNumeral{3}}\}$ و${\OLId{latin-looped}{Y}}=\{{\OLMathNumeral{4}},{\OLMathNumeral{5}},{\OLMathNumeral{6}}\}$، مع علاقتي
أصغر من وأكبر من، تكون الدالة ${\OLId{latin-ordinary}{f}}\colon {\OLId{latin-looped}{X}} \to {\OLId{latin-looped}{Y}}$ المعطاة بـ
${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}) = {\OLMathNumeral{7}}-{\OLId{latin-ordinary}{x}}$ تشاكلًا بين $\tuple{{\OLId{latin-looped}{X}},<}$ و$\tuple{{\OLId{latin-looped}{Y}},>}$.
\end{ex}
