\olsection{اختزال~$\beta\eta$}

ننتقل إلى إثبات خاصية تشيرش--روسر لاختزال~$\beta\eta$ ($\bered$).

\begin{lem}\ollabel{lem:one-par}
  من~${\OLId{latin-looped}{M}} \beredone {\OLId{latin-looped}{M}}'$ يلزم~${\OLId{latin-looped}{M}} \beredpar {\OLId{latin-looped}{M}}'$.
\end{lem}

\begin{proof}
  نستقرئ~!!{derivation} العبارة~${\OLId{latin-looped}{M}} \beredone {\OLId{latin-looped}{M}}'$.
  فإن كان~${\OLId{latin-looped}{M}} \beredone {\OLId{latin-looped}{M}}'$ حاصلًا بتحويل~$\eta$، أي بقاعدة
  \olref[syn][eta]{defn:beredone}، جمعنا
  \olref[pbe]{defn:beredpar5} إلى~\olref[pbe]{thm:refl}.
  وأما بقية الحالات فنجري فيها على برهان~\olref[b]{lem:one-par}.
\end{proof}


\begin{lem}\ollabel{lem:par-red}
  من~${\OLId{latin-looped}{M}} \beredpar {\OLId{latin-looped}{M}}'$ يلزم~${\OLId{latin-looped}{M}} \bered {\OLId{latin-looped}{M}}'$.
\end{lem}

\begin{proof}
  نستقرئ~!!{derivation} العبارة~${\OLId{latin-looped}{M}} \beredpar {\OLId{latin-looped}{M}}'$.

  أما الحالات الأربع الأولى فبرهانها كما في~\olref[b]{lem:par-red}.
  وأما إذا كانت القاعدة الأخيرة~\olref[pbe]{defn:beredpar5}، فإن~${\OLId{latin-looped}{M}}$
  هو~$\lambd[{\OLId{latin-ordinary}{x}}][Nx]$ و${\OLId{latin-looped}{M}}'$ هو~${\OLId{latin-looped}{N}}'$، لبعض~${\OLId{latin-ordinary}{x}}$ و${\OLId{latin-looped}{N}}$ و${\OLId{latin-looped}{N}}'$،
  مع~${\OLId{latin-ordinary}{x}} \notin \FV{{\OLId{latin-looped}{N}}}$ و${\OLId{latin-looped}{N}} \beredpar {\OLId{latin-looped}{N}}'$.
  نختزل حينئذ~$\lambd[{\OLId{latin-ordinary}{x}}][Nx]$ إلى~${\OLId{latin-looped}{N}}$ بتحويل~$\eta$،
  ثم نجري خطوات~$\beredone$ التي تحقق~${\OLId{latin-looped}{N}} \bered {\OLId{latin-looped}{N}}'$؛
  ووجود هذه المتتالية ثابت بفرض الاستقراء.
\end{proof}


\begin{lem}\ollabel{lem:str}
  العلاقة~$\bered$ أصغر علاقة متعدية تشتمل على~$\beredpar$.
\end{lem}

\begin{proof}
  يجرى البرهان على مثال~\olref[b]{lem:str}.
\end{proof}

\begin{thm}\ollabel{thm:cr}
  العلاقة~$\bered$ ذات خاصية تشيرش--روسر.
\end{thm}

\begin{proof}
  نضم~\olref[dap]{thm:str} و\olref[pbe]{thm:cr} إلى~\olref{lem:str}، فيتم المطلوب.
\end{proof}
