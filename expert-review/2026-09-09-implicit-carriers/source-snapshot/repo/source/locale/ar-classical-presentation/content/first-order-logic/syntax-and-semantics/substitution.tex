\olsection{الإحلال}

\begin{defn}[الإحلال في حد]
الرمز $\Subst{{\OLId{latin-ordinary}{s}}}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ يدل على ما ينتج من \emph{إحلال}~${\OLId{latin-ordinary}{t}}$
محل كل ورود لـ~${\OLId{latin-ordinary}{x}}$ في~${\OLId{latin-ordinary}{s}}$؛ وتعريفه العودي كما يأتي:
\begin{enumerate}
\item \indcase{s}{c}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي ${\OLId{latin-ordinary}{s}}$ نفسها.}

\item \indcase{s}{y}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي أيضًا ${\OLId{latin-ordinary}{s}}$ نفسها،
  شريطة أن يكون ${\OLId{latin-ordinary}{y}}$ متغيرًا وأن ${\OLId{latin-ordinary}{y}} \not\ident {\OLId{latin-ordinary}{x}}$.}

\item \indcase{s}{x}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي~${\OLId{latin-ordinary}{t}}$.}

\item\indcase{s}{\Atom{f}{t_1, \dots, t_n}}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي
  $\Atom{f}{\Subst{t_1}{t}{x}, \dots, \Subst{t_n}{t}{x}}$.}
\end{enumerate}
\end{defn}

\begin{defn}
نقول إن الحد~${\OLId{latin-ordinary}{t}}$ \emph{!!{free for}}~${\OLId{latin-ordinary}{x}}$ في~$!A$ متى خلا كل ورود
حر لـ~${\OLId{latin-ordinary}{x}}$ في~$!A$ من الوقوع في مجال مكمم يقيّد متغيرًا من~${\OLId{latin-ordinary}{t}}$.
\end{defn}

\begin{ex} ~
\begin{enumerate}
\item $\Obj v_{\OLMathNumeral{8}}$ حرة للإحلال محل~$\Obj v_{\OLMathNumeral{1}}$ في
  $\lexists[\Obj v_{\OLMathNumeral{3}}]\Atom{\Obj A^2_4}{\Obj v_3,\Obj v_1}$

\item $\Obj f^{\OLMathNumeral{2}}_{\OLMathNumeral{1}}(\Obj v_{\OLMathNumeral{1}}, \Obj v_{\OLMathNumeral{2}})$ \emph{ليست} حرة للإحلال
  محل~$\Obj v_{\OLMathNumeral{0}}$ في
  $\lforall[\Obj v_{\OLMathNumeral{2}}]\Atom{\Obj A^2_4}{\Obj v_0,\Obj v_2}$
\end{enumerate}
\end{ex}

\begin{defn}[الإحلال في !!a{formula}]
إذا كانت $!A$ !!a{formula}، وكان ${\OLId{latin-ordinary}{x}}$~!!a{variable}، وكان ${\OLId{latin-ordinary}{t}}$~حدًا
!!{free for}~${\OLId{latin-ordinary}{x}}$ في~$!A$، فإن $\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي نتيجة إحلال
${\OLId{latin-ordinary}{t}}$ محل جميع الورودات الحرة لـ~${\OLId{latin-ordinary}{x}}$ في~$!A$.
\begin{enumerate}
\tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي
    $\lfalse$.}}{}

\tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي
    $\ltrue$.}}{}

\item \indcase{!A}{\Atom{P}{t_1,\dots,
    t_n}}{$\Subst{\indfrm}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $\Atom{P}{\Subst{t_1}{t}{x},
    \dots, \Subst{t_n}{t}{x}}$.}

\item \indcase{!A}{\eq[t_1][t_2]}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي
  $\Subst{{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}} = \Subst{{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$.}

\tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي
    $\lnot \Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$.}}{}

\tagitem{prvAnd}{\indcase{!A}{(!B \land
    !C)}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $(\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}} \land
    \Subst{!C}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}})$.}}{}

\tagitem{prvOr}{\indcase{!A}{(!B \lor
    !C)}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $(\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}} \lor
    \Subst{!C}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}})$.}}{}

\tagitem{prvIf}{\indcase{!A}{(!B \lif
    !C)}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $(\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}} \lif
    \Subst{!C}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}})$.}}{}

\tagitem{prvIff}{\indcase{!A}{(!B \liff
    !C)}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $(\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}} \liff
    \Subst{!C}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}})$.}}{}

\tagitem{prvAll}{
\indcase{!A}{\lforall[y][!B]}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$
    هي $\lforall[{\OLId{latin-ordinary}{y}}][\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}]$، شريطة أن يكون ${\OLId{latin-ordinary}{y}}$ متغيرًا
    غير~${\OLId{latin-ordinary}{x}}$؛ وإلا كانت $\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $\indfrm$
    نفسها.}}{}

\tagitem{prvEx}{
\indcase{!A}{\lexists[y][!B]}{$\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$
    هي $\lexists[{\OLId{latin-ordinary}{y}}][\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}]$، شريطة أن يكون ${\OLId{latin-ordinary}{y}}$ متغيرًا
    غير~${\OLId{latin-ordinary}{x}}$؛ وإلا كانت $\Subst{\indfrmp}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $\indfrm$
    نفسها.}}{}
\end{enumerate}
\end{defn}

\begin{explain}
وقد لا يغيّر الإحلال شيئًا؛ فإن لم يكن لـ~${\OLId{latin-ordinary}{x}}$ في~$!A$ ورود أصلًا،
بقيت $\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي~$!A$ بعينها.

وإنما اشترطنا أن يكون~${\OLId{latin-ordinary}{t}}$ !!{free for}~${\OLId{latin-ordinary}{x}}$ في~$!A$ لئلا يقع
مثل هذا الخلل: لو كان $!A \ident \lexists[{\OLId{latin-ordinary}{y}}][{\OLId{latin-ordinary}{x}} < {\OLId{latin-ordinary}{y}}]$ وكان ${\OLId{latin-ordinary}{t}} \ident {\OLId{latin-ordinary}{y}}$،
لصارت $\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ هي $\lexists[{\OLId{latin-ordinary}{y}}][{\OLId{latin-ordinary}{y}} < {\OLId{latin-ordinary}{y}}]$؛ فيقع المتغير
الحر~${\OLId{latin-ordinary}{y}}$ عند الإحلال تحت قيد المكمم~$\lexists[{\OLId{latin-ordinary}{y}}]$، وذلك ما يسمى
«أسر» المتغير. ووجه فساده أن صدق $\lforall[{\OLId{latin-ordinary}{x}}][!B]$ ينبغي أن يستتبع
صدق $\Subst{!B}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$. فانظر إلى
$\lforall[{\OLId{latin-ordinary}{x}}][\lexists[{\OLId{latin-ordinary}{y}}][{\OLId{latin-ordinary}{x}} < {\OLId{latin-ordinary}{y}}]]$، حيث $!B$ هي $\lexists[{\OLId{latin-ordinary}{y}}][{\OLId{latin-ordinary}{x}} <
  {\OLId{latin-ordinary}{y}}]$. هذه جملة صادقة في الأعداد الطبيعية مثلًا؛ إذ لكل عدد~${\OLId{latin-ordinary}{x}}$
عدد~${\OLId{latin-ordinary}{y}}$ يفوقه. فإن أجزنا إحلال~${\OLId{latin-ordinary}{y}}$ محل~${\OLId{latin-ordinary}{x}}$، حصل
$\Subst{!B}{{\OLId{latin-ordinary}{y}}}{{\OLId{latin-ordinary}{x}}} \ident \lexists[{\OLId{latin-ordinary}{y}}][{\OLId{latin-ordinary}{y}} < {\OLId{latin-ordinary}{y}}]$، وهو كاذب.
فدفعًا لهذا الخلل نشترط ألا يصير شيء من المتغيرات الحرة في~${\OLId{latin-ordinary}{t}}$
مقيدًا بمكمم في~$!A$ عند الإحلال.
\end{explain}

وللتخفيف من ثقل الترميز نجري كثيرًا على اصطلاح آخر: متى كانت
$!A$ !!a{formula} يجوز أن يرد فيها ال!!{variable}~${\OLId{latin-ordinary}{x}}$ حرًا،
كتبنا~$!A({\OLId{latin-ordinary}{x}})$ إشارة إلى ذلك. فإذا تعيّنت~$!A$ وتعيّن~${\OLId{latin-ordinary}{x}}$ من السياق،
وكان ${\OLId{latin-ordinary}{t}}$ حدًا نفترض حريته للإحلال محل~${\OLId{latin-ordinary}{x}}$ في~$!A({\OLId{latin-ordinary}{x}})$، جعلنا
$!A({\OLId{latin-ordinary}{t}})$ اختصارًا لـ~$\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$. فقولنا مثلًا إن
$!A({\OLId{latin-ordinary}{t}})$ حالة إحلالية من~$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$ مراده هذا:
إذا كانت $!A$~أي !!{formula}، وكان ${\OLId{latin-ordinary}{x}}$~!!a{variable}، وكان ${\OLId{latin-ordinary}{t}}$~حدًا
حرًا للإحلال محل~${\OLId{latin-ordinary}{x}}$ في~$!A$، فـ$\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{x}}}$ حالة إحلالية
من~$\lforall[{\OLId{latin-ordinary}{x}}][!A]$.
