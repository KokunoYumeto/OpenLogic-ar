\olsection{الحدود وال\printtoken{P}{formula}}

إذا تعيّنت لغة من الرتبة الأولى~$\Lang L$، تعيّن المعجم الذي تُبنى
منه تعبيرات~$\Lang L$. ونخصّ من هذه التعبيرات هنا
\emph{الحدود} و\emph{!!{formula}s} بالتعريف.

\begin{defn}[الحدود]
\ollabel{defn:terms}
أما مجموعة \emph{الحدود}~$\Trm[{\OLId{latin-looped}{L}}]$ في~$\Lang L$، فحدّها الاستقرائي
هذه الشروط:
\begin{enumerate}
\item كل !!{variable} حد.
\item كل !!{constant} في~$\Lang L$ حد.
\item إذا كانت ${\OLId{latin-ordinary}{f}}$ !!{function} ذات ${\OLId{latin-ordinary}{n}}$ من المحال، وكانت
  ${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$، و\dots، و${\OLId{latin-ordinary}{t}}_{\OLId{latin-ordinary}{n}}$ حدودًا، فإن
  $\Atom{f}{t_1, \ldots, t_n}$ حد.
\tagitem{limitClause}{
  لا شيء غير ذلك حد.}{}
\end{enumerate}
والحد الذي لا يحتوي على !!{variable}s \emph{حد مغلق}.
\end{defn}

\begin{explain}
قد أفردنا ال!!{constant}s بصنف من الرموز حين حدّدنا اللغة والحدود.
ولنا وجه آخر، وهو عدّها !!{function}s صفرية المحل؛ وعندئذ يستغنى
عن البند الثاني من تعريف الحدود. وليس علينا إلا أن نحمل $\Atom{f}{t_1, \ldots, t_n}$ على أنه ${\OLId{latin-ordinary}{f}}$ وحده إذا
كان ${\OLId{latin-ordinary}{n}} = {\OLMathNumeral{0}}$.
\end{explain}

\begin{defn}[الصيغ]
\ollabel{defn:formulas}
وأما مجموعة \emph{!!{formula}s}~$\Frm[{\OLId{latin-looped}{L}}]$ في اللغة~$\Lang L$،
فتتعيّن استقرائيًا بالشروط الآتية:
\begin{enumerate}
\tagitem{prvFalse}{$\lfalse$ هي !!{formula} ذرية.}{}

\tagitem{prvTrue}{$\ltrue$ هي !!{formula} ذرية.}{}

\item إذا كان ${\OLId{latin-looped}{R}}$ !!{predicate} ذا ${\OLId{latin-ordinary}{n}}$ من المحال في~$\Lang L$، وكانت
  ${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$، و\dots، و${\OLId{latin-ordinary}{t}}_{\OLId{latin-ordinary}{n}}$ حدودًا في~$\Lang L$، فإن
  $\Atom{R}{t_1,\ldots, t_n}$ !!{formula} ذرية.

\item إذا كان ${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$ و${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}$ حدين في~$\Lang L$، فإن
  $\Atom{\eq}{t_1, t_2}$ !!{formula} ذرية.

\tagitem{prvNot}{إذا كانت $!A$ !!a{formula}، فإن $\lnot !A$
  !!a{formula}.}{}

\tagitem{prvAnd}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \land
  !B)$ !!a{formula}.}{}

\tagitem{prvOr}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \lor !B)$
  !!a{formula}.}{}

\tagitem{prvIf}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \lif !B)$
  !!a{formula}.}{}

\tagitem{prvIff}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن
  $(!A \liff !B)$ !!a{formula}.}{}

\tagitem{prvAll}{إذا كانت $!A$ !!a{formula} وكان ${\OLId{latin-ordinary}{x}}$ !!a{variable}،
  فإن $\lforall[{\OLId{latin-ordinary}{x}}][!A]$ !!a{formula}.}{}

\tagitem{prvEx}{إذا كانت $!A$ !!a{formula} وكان ${\OLId{latin-ordinary}{x}}$ !!a{variable}،
  فإن $\lexists[{\OLId{latin-ordinary}{x}}][!A]$ !!a{formula}.}{}

\tagitem{limitClause}{لا شيء غير ذلك !!a{formula}.}{}
\end{enumerate}
\end{defn}

\begin{explain}
تعريف الحدود وتعريف ال!!{formula}s كلاهما من
\emph{التعريفات الاستقرائية}. ومعنى ذلك أن بناء مجموعة
ال!!{formula}s يجري على مراحل لا تتناهى. نبدأ بجعل الصيغ الذرية
كلها صيغًا، وهو ما تقرره الحالات الأولى من التعريف، أعني حالات
\iftag{prvTrue}{$\ltrue$، }{}%
\iftag{prvFalse}{$\lfalse$، }{}%
$\Atom{R}{t_1,\dots,t_n}$ و$\Atom{\eq}{t_1,t_2}$. فكل
!!{formula} بهذه الصورة هي المرادة باسم «!!{formula} ذرية».

وأما سائر حالات التعريف، فقواعد يُنشأ بها الجديد من ال!!{formula}s
من ال!!{formula}s التي سبق إنشاؤها. ففي المرحلة الثانية نطبّقها
فننشئ !!{formula}s من !!{formula}s ذرية؛ ثم نجمع في الثالثة ما
كان ذريًا وما نشأ في الثانية، ونبني منه صيغًا أخرى، ونجري بعد ذلك
على هذا النهج. فما حصل في مرحلة من هذه المراحل فهو !!{formula}،
وما لم يحصل في شيء منها فليس كذلك.
\end{explain}

ومن اصطلاح الكتابة أن نجعل $\eq$ بين حجتيها، مع إسقاط القوسين:
فـ~$\eq[{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}]$ اختصار لـ~$\Atom{\eq}{t_1,t_2}$. ويُختصر أيضًا
$\lnot \Atom{\eq}{t_1,t_2}$ إلى~$\eq/[{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}]$. وإذا كتبنا الصيغة
$(!B \ast !C)$ المؤلّفة من $!B$ و$!C$ بالرابط الثنائي
المحل~$\ast$، فالغالب أن نسقط قوسيها الخارجيين ونقتصر
على~$!B \ast !C$.

\begin{intro}
وفي بعض كتب المنطق لا بد من ورود ال!!{variable}~${\OLId{latin-ordinary}{x}}$ في~$!A$ حتى تُعد
\iftag{prvEx}{$\lexists[{\OLId{latin-ordinary}{x}}][!A]$ }{}%
\iftag{notprvEx,notprvAll}{}{و }%
\iftag{prvAll}{$\lforall[{\OLId{latin-ordinary}{x}}][!A]$ }{}%
في عداد
\iftag{notprvEx,notprvAll}{!!a{formula}}{!!{formula}s}.
ولسنا نشترط ذلك؛ فترك هذا الشرط لا يخلّ بالبحث، وفيه تيسير له.
\end{intro}

\begin{tagblock}{defNot,defOr,defAnd,defIf,defIff,defTrue,defFalse,defEx,defAll}
\begin{defn}
والصيغ التي تدخلها المؤثرات المعرّفة تُردّ إلى أصولها على الوجه الآتي:

\begin{tagenumerate}{defTrue,defFalse,defNot,defOr,defAnd,defIf,defIff,defEx,defAll}

\tagitem{defTrue}{$\ltrue$ اختصار لـ
  \iftag{prvFalse}{$\lnot\lfalse$}{$(!A \lor \lnot !A)$، حيث تكون
    !!{formula}~$!A$ ذرية ثابتة ما}.}{}

\tagitem{defFalse}{$\lfalse$ اختصار لـ
  \iftag{prvTrue}{$\lnot\ltrue$}{$(!A \land \lnot !A)$، حيث تكون
    !!{formula}~$!A$ ذرية ثابتة ما}.}{}

\tagitem{defNot}{$\lnot !A$ اختصار لـ~$!A \lif \lfalse$.}{}

\tagitem{defOr}{$!A \lor !B$ اختصار لـ
  \iftag{prvAnd}{$\lnot(\lnot !A \land \lnot !B)$}{$\lnot !A \lif
    !B$}.}{}

\tagitem{defAnd}{$!A \land !B$ اختصار لـ
  \iftag{prvOr}{$\lnot(\lnot !A \lor \lnot !B)$}{$\lnot (!A \lif
    \lnot !B)$}.}{}

\tagitem{defIf}{$!A \lif !B$ اختصار لـ
  \iftag{prvOr}{$(\lnot !A \lor !B)$}{$\lnot (!A \land \lnot !B)$}.}{}

\tagitem{defIff}{$!A \liff !B$ اختصار لـ~$(!A \lif !B) \land (!B
  \lif !A)$.}{}

\tagitem{defAll}{$\lforall[{\OLId{latin-ordinary}{x}}][!A]$ اختصار لـ~$\lnot\lexists[{\OLId{latin-ordinary}{x}}][\lnot !A]$.}{}

\tagitem{defEx}{$\lexists[{\OLId{latin-ordinary}{x}}][!A]$ اختصار لـ~$\lnot\lforall[{\OLId{latin-ordinary}{x}}][\lnot !A]$.}{}
\end{tagenumerate}
\end{defn}
\end{tagblock}

وقد تختص اللغة بموضع استعمال، فيجري الاصطلاح على وضع ال!!{predicate}s
وال!!{function}s الثنائية المحل بين الحدين اللذين تتعلق بهما؛ مثل
${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}} < {\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}$ و$({\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}} + {\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}})$ في لغة الحساب، و${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}} \in {\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}$ في لغة
نظرية المجموعات. وفي لغة الحساب توضع دالة الخلف، على الاصطلاح الجاري،
\emph{بعد} حجتها:~${\OLId{latin-ordinary}{t}}'$. وهذه الرسوم جميعًا اختصارات لا غير؛ وأصولها
الصورية، بالترتيب، هي~$\Atom{\Obj A^2_0}{t_1, t_2}$ و
$\Obj f^{\OLMathNumeral{2}}_{\OLMathNumeral{0}}({\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}, {\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}})$ و$\Atom{\Obj A^2_0}{t_1, t_2}$ و$\Obj f^{\OLMathNumeral{1}}_{\OLMathNumeral{0}}({\OLId{latin-ordinary}{t}})$.

\begin{defn}[الهوية التركيبية]
للهوية التركيبية بين سلاسل الرموز نضع العلامة~$\ident$؛ فقولنا
$!A \ident !B$ يصدق إذا وفقط إذا كان $!A$ و$!B$ سلسلتين من الرموز لهما
الطول نفسه وتحتويان الرمز نفسه في كل موضع.
\end{defn}

وقد يقع على جانبي الرمز~$\ident$ ما تألّف من وصل سلاسل؛ فمثلًا قولنا
$!A \ident (!B \lor !C)$ معناه أن سلسلة الرموز~$!A$ بعينها تحصل من
وصل قوس فاتح بالسلسلة~$!B$، ثم بالرمز~$\lor$، ثم بالسلسلة~$!C$، ثم
بقوس غالق، على هذا الترتيب. ومن ذلك نعلم أن أول رمز في~$!A$
قوس فاتح، وأن في~$!A$ السلسلة~$!B$ جزءًا يبدأ عند الرمز الثاني،
وأن ما يعقب ذلك الجزء هو~$\lor$، وعلى هذا يُقاس الباقي.

ولما كان بناء الحدود وال!!{formula}s استقرائيًا، مبدؤه عناصر أساسية،
كان سبيل البرهان على خواصها مبدأي الاستقراء الآتيين.

\begin{lem}[\emph{مبدأ الاستقراء على الحدود}]
    \ollabel{lem:trmind}
    لتكن $\Lang L$ لغة رتبة أولى.
    إذا كانت خاصية ما~${\OLId{latin-looped}{P}}$ بحيث
    %
    \begin{enumerate}
        \item تصدق في كل !!{variable}~${\OLId{latin-ordinary}{v}}$،
        %
        \item وتصدق في كل !!{constant}~${\OLId{latin-ordinary}{a}}$ من~$\Lang L$،
        %
        \item وتصدق في ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}},\dotsc,{\OLId{latin-ordinary}{t}}_{\OLId{latin-ordinary}{n}})$ متى صدقت في
        ${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$، و\dots، و${\OLId{latin-ordinary}{t}}_{\OLId{latin-ordinary}{n}}$ وكانت ${\OLId{latin-ordinary}{f}}$ !!{function} ذات ${\OLId{latin-ordinary}{n}}$ من
            المحال في~$\Lang L$،
    \end{enumerate}
    (على افتراض أن ${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$، و\dots، و${\OLId{latin-ordinary}{t}}_{\OLId{latin-ordinary}{n}}$ حدود في~$\Lang{L}$)،
    فإن ${\OLId{latin-looped}{P}}$ تصدق في كل حد في~$\Trm[{\OLId{latin-looped}{L}}]$.
\end{lem}

\begin{prob}
    أثبت \olref[fol][syn][frm]{lem:trmind}.
\end{prob}

\begin{lem}[\emph{مبدأ الاستقراء على ال!!{formula}s}]
    \ollabel{thm:frmind}
    لتكن $\Lang L$ لغة رتبة أولى.
    إذا كانت خاصية ما~${\OLId{latin-looped}{P}}$ تصدق في جميع ال!!{formula}s الذرية
    وكانت بحيث
    %
    \begin{enumerate}
        \tagitem{prvNot}{تصدق في $\lnot !A$ متى صدقت
            في~$!A$؛}{}
        \tagitem{prvAnd}{تصدق في $(!A \land !B)$
            متى صدقت في $!A$ و~$!B$؛}{}
        \tagitem{prvOr}{تصدق في $(!A \lor !B)$
            متى صدقت في $!A$ و~$!B$؛}{}
        \tagitem{prvIf}{تصدق في $(!A \lif !B)$
            متى صدقت في $!A$ و~$!B$؛}{}
        \tagitem{prvIff}{تصدق في $(!A \liff !B)$
            متى صدقت في $!A$ و~$!B$؛}{}
        \tagitem{prvEx}{تصدق في $\lexists[{\OLId{latin-ordinary}{x}}][!A]$
            متى صدقت في~$!A$؛}{}
        \tagitem{prvAll}{تصدق في $\lforall[{\OLId{latin-ordinary}{x}}][!A]$
            متى صدقت في~$!A$؛}{}
  \end{enumerate}
  (على افتراض أن $!A$ و$!B$ !!{formula}s في~$\Lang{L}$)،
  فإن ${\OLId{latin-looped}{P}}$ تصدق في جميع الصيغ في~$\Frm[{\OLId{latin-looped}{L}}]$.
\end{lem}
