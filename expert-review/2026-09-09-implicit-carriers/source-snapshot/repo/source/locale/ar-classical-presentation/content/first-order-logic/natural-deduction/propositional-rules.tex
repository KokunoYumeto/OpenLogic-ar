\subsection{قواعد~$\land$}

\begin{defish}
\AxiomC{$!A$}
\AxiomC{$!B$}
\RightLabel{\Intro{\land}}
\BinaryInfC{$!A \land !B$}
\DisplayProof
\hfill
\begin{tabular}{r}
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!A$}
\DisplayProof
\\[3ex]
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!B$}
\DisplayProof
\end{tabular}
\end{defish}

\subsection{قواعد~$\lor$}

\begin{defish}
\begin{tabular}{r}
\AxiomC{$!A$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\\[3ex]
\AxiomC{$!B$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\end{tabular}
\hfill
\AxiomC{$!A \lor !B$}
\AxiomC{$\Discharge{!A}{{\OLId{latin-ordinary}{n}}}$}
\DeduceC{$!C$}
\AxiomC{$\Discharge{!B}{{\OLId{latin-ordinary}{n}}}$}
\DeduceC{$!C$}
\DischargeRule{\Elim{\lor}}{n}
\TrinaryInfC{$!C$}
\DisplayProof
\end{defish}

\subsection{قواعد~$\lif$}

\begin{defish}
\AxiomC{$\Discharge{!A}{{\OLId{latin-ordinary}{n}}}$}
\DeduceC{$!B$}
\DischargeRule{\Intro{\lif}}{n}
\UnaryInfC{$!A \lif !B$}
\DisplayProof
\hfill
\AxiomC{$!A \lif !B$}
\AxiomC{$!A$}
\RightLabel{\Elim{\lif}}
\BinaryInfC{$!B$}
\DisplayProof
\end{defish}

\subsection{قواعد~$\lnot$}

\begin{defish}
\AxiomC{$\Discharge{!A}{{\OLId{latin-ordinary}{n}}}$}
\noLine
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{n}
\UnaryInfC{$\lnot !A$}
\DisplayProof
\hfill
\AxiomC{$\lnot !A$}
\AxiomC{$!A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\DisplayProof
\end{defish}

\subsection{قواعد~$\lfalse$}

\begin{defish}
\AxiomC{$\lfalse$}
\RightLabel{\FalseInt}
\UnaryInfC{$!A$}
\DisplayProof
\hfill
\AxiomC{$\Discharge{\lnot !A}{{\OLId{latin-ordinary}{n}}}$}
\DeduceC{$\lfalse$}
\DischargeRule{\FalseCl}{n}
\UnaryInfC{$!A$}
\DisplayProof
\end{defish}

يشبه~$\Intro{\lnot}$ القاعدة~$\FalseCl$ شبهًا كبيرًا.
لكن~$\Intro{\lnot}$ نتيجتها !!{sentence} منفية~$\lnot !A$؛ وأما
$\FalseCl$ فنتيجتها !!{sentence} موجبة~$!A$.

والإسقاط حيث تبيحه القاعدة رخصة لا شرط لازم.
ففي~$\Intro{\lif}$، مثلًا، لنا أن نسقط من الفروض ذات الصورة~$!A$ في ال!!{derivation} المؤدي إلى~$!B$ أي عدد، ولو كان صفرًا.
