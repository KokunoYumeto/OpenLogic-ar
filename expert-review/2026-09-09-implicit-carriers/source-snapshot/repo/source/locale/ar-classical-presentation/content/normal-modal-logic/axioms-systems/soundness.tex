\olsection{السلامة}

السلامة في نسق !!^a{derivation} أن يكون كل ما يمكن !!{derive}d
صالحًا. وفي النظم الجهية التي تجيز !!{derivation}s باستعمال
أمثلة !!{formula}s~$!A_{\OLMathNumeral{1}}$، \dots،~$!A_{\OLId{latin-ordinary}{n}}$ فوق~\Ax{K}،
نطلب أن تصدق كل صيغة !!{derivable} في كل نموذج تصدق فيه
جميع أمثلة~$!A_{\OLMathNumeral{1}}$، \dots، $!A_{\OLId{latin-ordinary}{n}}$.

\begin{thm}[مبرهنة السلامة]\ollabel{thm:soundness}
  إذا كان كل مثال لـ$!A_{\OLMathNumeral{1}}$، \dots، $!A_{\OLId{latin-ordinary}{n}}$ صالحا، على الترتيب، في
  فئات النماذج $\mClass{C}_{\OLMathNumeral{1}}$، \dots، $\mClass{C}_{\OLId{latin-ordinary}{n}}$، فإن
  $\Log{K}!A_{\OLMathNumeral{1}}\dots !A_{\OLId{latin-ordinary}{n}} \Proves !B$ يستلزم أن تكون $!B$ صالحة في
  فئة النماذج $\mClass{C}_{\OLMathNumeral{1}} \cap \dots \cap \mClass{C}_{\OLId{latin-ordinary}{n}}$.
\end{thm}

\begin{proof}
  نجري الاستقراء على طول البرهان، ونجعل للاختصار
  $\mClass{C} =
  \mClass{C}_{\OLMathNumeral{1}} \cap \dots \cap \mClass{C}_{\OLId{latin-ordinary}{n}}$.
  \begin{enumerate}
  \item أساس الاستقراء: إذا كان لـ$!B$ برهان طوله~${\OLMathNumeral{1}}$، فهي إما مثال
    لقضية صادقة صدقا تحليليا، أو مثال لـ~\Ax{K}،
    \iftag{prvDiamond}{أو لـ~\Dual{}،}{} أو مثال لواحدة من $!A_{\OLMathNumeral{1}}$،
    \dots،~$!A_{\OLId{latin-ordinary}{n}}$. في الحالة الأولى تكون $!B$ صالحة في
    $\mClass{C}$، لأن أمثلة القضايا الصادقة صدقا تحليليا صالحة في
    \emph{كل} فئة من النماذج، بحسب \olref[syn][tau]{prop:valid-taut}.
    وكذلك في الحالة الثانية، بحسب
    \olref[syn][sch]{prop:Kvalid}\iftag{prvDiamond}{ و
      \olref[syn][sch]{prop:Dual-valid}}{}. وأخيرا، في الحالة الثالثة،
    لما كانت $!B$ صالحة في $\mClass{C}_{\OLId{latin-ordinary}{i}}$ وكان $\mClass{C} \subseteq
    \mClass{C}_{\OLId{latin-ordinary}{i}}$، كانت $!B$ صالحة في $\mClass{C}$ أيضا بحسب
    \olref[syn][val]{prop:subset-class}.
  \item وأما خطوة الاستقراء فليكن لـ$!B$ برهان طوله ${\OLId{latin-ordinary}{k}}>{\OLMathNumeral{1}}$.
    إن كانت $!B$ حالة تحصيل منطقي، أو حالة من~\Ax{K}،
    \iftag{prvDiamond}{أو حالة من~\Dual{}،}{} أو من إحدى
    $!A_{\OLMathNumeral{1}}$، \dots، $!A_{\OLId{latin-ordinary}{n}}$، كفى ما تقدم في الأساس.
    وإن كانت $!B$ نتيجة~\MP{} من !!{formula}s السابقتين
    $!C \lif !B$ و$!C$، فإن لـ$!C \lif !B$ و$!C$ برهانين
    أقصر من ${\OLId{latin-ordinary}{k}}$. فهما صالحتان في~$\mClass{C}$ بفرض الاستقراء،
    ومن~\olref[syn][sch]{prop:soundMP} تكون $!B$ صالحة في
    $\mClass{C}$. وبقي أن تكون $!B$ نتيجة~\Nec{} من $!C$،
    أي $!B = \Box!C$. حينئذ تصح $!C$ في $\mClass{C}$
    بفرض الاستقراء، وتصح~$!B$ بتطبيق حجة
    \olref[syn][val]{prop:Nec-rule} على كل نموذج في هذه الفئة.
  \end{enumerate}
\end{proof}
