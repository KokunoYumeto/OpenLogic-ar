\olsection{مبرهنة ${\OLId{latin-ordinary}{s}}$-${\OLId{latin-ordinary}{m}}$-${\OLId{latin-ordinary}{n}}$}

\begin{explain}
تسمى المبرهنة الآتية «مبرهنة ${\OLId{latin-ordinary}{s}}$-${\OLId{latin-ordinary}{m}}$-${\OLId{latin-ordinary}{n}}$»، وسيظهر وجه
التسمية. وإنما الصعوبة في ضبط معناها؛ فإذا استبانت عبارتها
قرب مأخذها.
\end{explain}

\begin{thm}
\ollabel{thm:s-m-n}
  مهما أخذنا عددين طبيعيين ${\OLId{latin-ordinary}{n}}$ و~${\OLId{latin-ordinary}{m}}$، وجدت دالة عودية بدائية~${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}$
  تتحقق بها، لكل أعداد طبيعية ${\OLId{latin-ordinary}{e}}$، ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}}$، …، ${\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}$، ${\OLId{latin-ordinary}{y}}_{\OLMathNumeral{0}}$،
  …،~${\OLId{latin-ordinary}{y}}_{{\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}}$، المساواة الجزئية الآتية:
  \[
  \cfind{{\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}({\OLId{latin-ordinary}{e}}, {\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}}, \dots, {\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}})}[{\OLId{latin-ordinary}{n}}]({\OLId{latin-ordinary}{y}}_{\OLMathNumeral{0}}, \dots, {\OLId{latin-ordinary}{y}}_{{\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}}) \simeq
  \cfind{{\OLId{latin-ordinary}{e}}}[{\OLId{latin-ordinary}{m}}+{\OLId{latin-ordinary}{n}}]({\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}}, \dots, {\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}, {\OLId{latin-ordinary}{y}}_{\OLMathNumeral{0}}, \dots, {\OLId{latin-ordinary}{y}}_{{\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}}).
\]
\end{thm}

% تصحيح ترجمة (C232-01): الأعيان المذكورة أعداد طبيعية، لا متتالية واحدة.
\textbf{تنبيه تصحيحي على الترجمة الأساس.} الأعيان ${\OLId{latin-ordinary}{e}}$ و${\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{i}}$ و${\OLId{latin-ordinary}{y}}_{\OLId{latin-ordinary}{i}}$
أعداد طبيعية، لا متتالية واحدة؛ فأُظهر نوعها الصحيح.

\begin{explain}
ويقرب المعنى إذا جعلنا عمل ${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}$ في \emph{البرامج}. فالدالة ${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}$
تتلقى برنامجًا~${\OLId{latin-ordinary}{e}}$ لدالة ذات ${\OLId{latin-ordinary}{m}}+{\OLId{latin-ordinary}{n}}$ حجة، وتتلقى معه المداخل المثبتة ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}}$،
…،~${\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}$؛ ثم تعطي برنامجًا
${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}({\OLId{latin-ordinary}{e}},{\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}},\dots,{\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}})$ لدالة ذات ${\OLId{latin-ordinary}{n}}$ حجة، لا يبقى من حججها إلا ما لم يُثبّت.
\iftag{TMs}{فإذا كان ${\OLId{latin-ordinary}{e}}$ وصفًا لآلة تورنغ، كانت
${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}({\OLId{latin-ordinary}{e}},{\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}},\dots,{\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}})$ هي آلة تورنغ التي تسبق سلسلة الدخل
${\OLId{latin-ordinary}{y}}_{\OLMathNumeral{0}}$، …،~${\OLId{latin-ordinary}{y}}_{{\OLId{latin-ordinary}{n}}-{\OLMathNumeral{1}}}$ بـ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{0}}$، …،~${\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}$، ثم تشغل~${\OLId{latin-ordinary}{e}}$.
فكل ${\OLId{latin-ordinary}{s}}^{\OLId{latin-ordinary}{m}}_{\OLId{latin-ordinary}{n}}$ حينئذ دالة عودية بدائية تستخرج رمز آلة تورنغ
المطلوبة.}{}
\end{explain}
