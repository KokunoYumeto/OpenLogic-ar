\olsection{المتتاليات}

قد بان اتساع ما تتناوله الدوال العودية البدائية. ويزداد ذلك اتساعًا
إذا هيأنا وسيلة لمعالجة \emph{المتتاليات}. فنرمز إلى كل متتالية
منتهية من الأعداد الطبيعية بعدد طبيعي على هذا الوجه: نجعل للمتتالية
$\langle a_0,a_1,a_2,\dots,a_k\rangle$ العدد
\[
p_0^{a_0+1} \cdot p_1^{a_1+1} \cdot p_2^{a_2+1} \cdot \dots \cdot
p_k^{a_k+1}.
\]
وإنما زدنا واحدًا في الأسس لئلا يختلط، مثلًا، رمزا المتتاليتين
$\langle2,7,3\rangle$ و$\langle2,7,3,0,0\rangle$ بعضهما ببعض.
ويجوز أن نجعل كلًا من $0$ و~$1$ رمزًا للمتتالية الخالية؛
وللتعيين، ليدل $\emptyseq$ على~$0$.

وصحة هذا الترميز مبنية على المبرهنة الأساسية في
الحساب: لكل عدد طبيعي $n\ge2$ تحليل واحد لا غير على الصورة
\[
n = p_0^{a_0} \cdot p_1^{a_1} \cdot \dots \cdot p_k^{a_k}
\]
مع $a_k\ge1$. فيكون التطبيق
$\tuple{}(a_0,\dots,a_k)=\tuple{a_0,\dots,a_k}$ أحاديًا؛ إذ تختلف
أعداد الترميز باختلاف المتتاليات، ولا يرمز العدد الواحد إلى متتاليتين
مختلفتين.

ولنثبت الآن العودية البدائية للعمليات التي نحتاج إليها: معرفة طول المتتالية وعنصرها ذي الرتبة $i$،
وإلحاق عنصر بها، ووصلها بمتتالية أخرى.

\begin{prop}
  الدالة $\len{s}$، التي تعيد طول المتتالية $s$، عودية بدائية.
\end{prop}

\begin{proof}
  لتكن $R(i,s)$ العلاقة المعرفة بـ
  \[
  R(i, s) \text{ إذا وفقط إذا }
  p_i \mid s \land p_{i+1} \nmid s.
  \]
  فالعودية البدائية لـ$R$ ظاهرة. وإذا كان $s$ يرمز إلى متتالية غير خالية،
  أي
  \[
  s = p_0^{a_0+1} \cdot \dots \cdot p_{k}^{a_{k}+1},
  \]
  صدقت $R(i,s)$ إذا كان $p_i$ أكبر عدد أولي بحيث $p_i\mid s$، أي
  إذا كان $i=k$. ومن ثم يكون طول $s$ هو $i+1$ إذا وفقط إذا كان
  $p_i$ أكبر عدد أولي يقسم~$s$، فلنا أن نضع
  \[
  \len{s} =
  \begin{cases}
    0 & \text{إذا كان $s=0$ أو $s=1$} \\
    1 + \bmin{i < s}{R(i, s)} & \text{في غير ذلك}
  \end{cases}
  \]
  وصح استعمال التصغير المحدود لأن $i$ الذي يحقق
  $R(i,s)$ وحيد متى كان $s$ رمز متتالية، ولأن هذا $i$، إن وجد، دون~$s$
  نفسها.
\end{proof}

\begin{prop}
  الدالة $\fn{append}(s,a)$، التي تعيد نتيجة إلحاق $a$ بالمتتالية
  $s$، عودية بدائية.
\end{prop}

\begin{proof}
  تعريف $\fn{append}$ على الوجه الآتي:
  \[
  \fn{append}(s,a) =
  \begin{cases}
    2^{a+1} & \text{إذا كان $s=0$ أو $s=1$} \\
    s \cdot p_{\len{s}}^{a+1} & \text{في غير ذلك.}
  \end{cases}
  \]
\end{proof}

\begin{prop}
  الدالة $\fn{element}(s,i)$، التي تعيد عنصر $s$ ذا الرتبة $i$
  (مع تسمية العنصر الأول ذا الرتبة~$0$)، أو تعيد $0$ إذا كان $i$ أكبر
  من أو مساويًا لطول $s$، عودية بدائية.
\end{prop}

\begin{proof}
نستخرج $a$، وهو عنصر~$s$ ذو الرتبة $i$، من الشرط القائل إن
$p_i^{a+1}$ أكبر قوة لـ~$p_i$ تقسم~$s$؛ أي إذا كان
$p_i^{a+1}\mid s$ ولكن $p_i^{a+2}\nmid s$. ومن ثم:
  \[
  \fn{element}(s,i) =
  \begin{cases}
    0 & \mbox{إذا كان $i \geq \len{s}$} \\
    \bmin{a < s}{(p_i^{a+2} \nmid s)} & \text{في غير ذلك.}
  \end{cases}
  \]
\end{proof}

ونختصر أسماء الدوال المتقدمة بترميز
أوجز: فنكتب $(s)_i$ مكان $\fn{element}(s,i)$، ونجعل
$\tuple{s_0,\dots,s_k}$ عبارة مختصرة عن
\[
\fn{append}(\fn{append}(\dots \fn{append}(\emptyseq,s_0)
\dots),s_k).
\]
فإذا كان طول $s$ مساويًا لـ~$k$، فعناصر $s$ هي
$(s)_0$، …،~$(s)_{k-1}$.

\begin{prop}
الدالة $\fn{concat}(s,t)$، التي تصل متتاليتين، عودية بدائية.
\end{prop}

\begin{proof}
  والمطلوب أن نعرّف $\fn{concat}$ بحيث تحقق
  \[
    \fn{concat}(\tuple{a_0, \dots, a_k}, \tuple{b_0, \dots, b_l}) =
    \tuple{a_0, \dots, a_k, b_0, \dots, b_l}.
  \]
  فنستعين بالدالة $\fn{hconcat}(s,t,n)$، وهي التي تلحق الرموز $n$ الأولى
  من $t$ بـ~$s$. وتعريفها بالعودية البدائية كما يأتي:
  \begin{align*}
    \fn{hconcat}(s,t,0) & = s\\
    \fn{hconcat}(s,t,n+1) & = \fn{append}(\fn{hconcat}(s,t,n),(t)_n)
    \intertext{ثم يمكننا تعريف $\fn{concat}$ بـ}
    \fn{concat}(s,t) & = \fn{hconcat}(s,t,\len{t}).
  \end{align*}
\end{proof}

سنكتب $s\concat t$ بدلًا من $\fn{concat}(s,t)$.

ونحتاج أيضًا إلى حد لعدد ترميز المتتالية يُعرف من طولها وأكبر
عناصرها. فلتكن $s$ متتالية ذات طول~$k$، ولتكن عناصرها جميعًا دون
عدد~$x$ أو مساوية له. فعدد عوامل $s$ الأولية لا يزيد على $k$، وأكبر
هذه العوامل لا يجاوز~$p_{k-1}$، وأقصى أس لكل عامل هو $x+1$ في التحليل
الأولي لـ~$s$. لذلك نضع $\fn{sequenceBound}(x,0)=1$، وإذا
كان $k>0$ عرفنا
\[
\fn{sequenceBound}(x,k) = p_{k-1}^{k \cdot (x+1)},
\]
فلا يجاوز عدد ترميز المتتالية~$s$ المتقدمة
~$\fn{sequenceBound}(x,k)$.

ومتى حصل حد كهذا للمتتاليات، أمكن تعريف دوال جديدة
بالبحث المحدود. فالدالة $\fn{concat}$، مثلًا، يمكن طلبها بهذا البحث؛
وإنما يلزمنا أن نضع \emph{مواصفة} عودية بدائية لما نطلبه،
وهو عدد ترميز المتتالية الموصولة، وأن نحد مقدار البحث. والصيغة المقترحة هي:
\begin{align*}
  \fn{concat}(s,t) = {} & \bmin{v < \fn{sequenceBound}(s+t,\len{s} +
    \len{t})}{} \\
  & \quad(\len{v} = \len{s} + \len{t} \land {}\\
    & \qquad \bforall{i < \len{s}}{(v)_i = (s)_i} \land {} \\
    & \qquad \bforall{j < \len{t}}{(v)_{\len{s}+j} = (t)_j})
\end{align*}

% تصحيح مصدر (C220-01): فُصل الشرطان الكليان لئلا يصير شرط t خاليًا عند خلو s.
\textbf{تنبيه تصحيحي على الأصل.} فُصل الشرطان الكليان على عنصري
$s$ و$t$؛ لأن إدخال شرط~$t$ في نطاق التكميم على~$s$ يجعله صادقًا
صدقًا خاليًا متى كانت~$s$ خالية.

\begin{prob}
أثبت وجود دالة عودية بدائية~$\fn{sconcat}(s)$ تحقق
\[
\fn{sconcat}(\tuple{s_0, \dots, s_k}) = s_0 \concat \dots \concat s_k.
\]
\end{prob}

\begin{prob}
أثبت وجود دالة عودية بدائية~$\fn{tail}(s)$ تحقق
\begin{align*}
  \fn{tail}(\emptyseq) & = 0 \text{، و}\\
  \fn{tail}(\tuple{s_0, \dots, s_{k}}) & = \tuple{s_1, \dots, s_{k}}.
\end{align*}
\end{prob}

\begin{prop}
  \ollabel{prop:subseq}
  الدالة $\fn{subseq}(s,i,n)$ التي تعيد المتتالية الجزئية من $s$ ذات
  الطول~$n$، ابتداء من العنصر ذي الرتبة $i$، عودية بدائية.
\end{prop}
