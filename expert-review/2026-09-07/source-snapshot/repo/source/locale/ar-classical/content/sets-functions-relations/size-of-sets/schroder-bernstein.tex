\olsection{مفهوم الحجم ومبرهنة شرودر–برنشتاين}

\begin{explain}
من مقتضى تصور الحجم أن $A$ إذا لم تزد حجمًا على $B$، ولم تزد $B$ حجمًا على $A$، وجب أن تتساوى $A$ و$B$ عددًا. ولو كان هذا \emph{باطلًا} لعسر علينا أن نجد لتعريف التساوي العددي صلة بمقارنة «أحجام» المجموعات!{} غير أن هذا المعنى صحيح، ومبرهنة شرودر–برنشتاين هي التي تقيم الدليل عليه.
\end{explain}

\begin{thm}[شرودر–برنشتاين]
\ollabel{thm:schroder-bernstein}
من $\cardle{A}{B}$ و$\cardle{B}{A}$ يلزم $\cardeq{A}{B}$.
\end{thm}

\begin{explain}
ومعناه أن وجود !!a{injection} من $A$ إلى~$B$، و!!a{injection} من $B$ إلى~$A$، يستلزم وجود !!a{bijection} من $A$ إلى~$B$.

على أن إقامة البرهان على هذه النتيجة ليست هينة، بل هي \emph{صعبة}. وقد صرح كانتور بالحكم، وكان إثباته على أيدي غيره.\footnote{وفي تاريخ هذه المبرهنة يراجع مثلًا \citet[ص~165--6]{Potter2004}.}
\oliflabeldef{sfr:cardinals:card-sb:sec}{ولن تتهيأ لنا وسائل \emph{إثبات} مبرهنة شرودر–برنشتاين حتى نبلغ \olref[sfr][cardinals][card-sb]{sec}.}{}%
فنستعملها الآن مسلّمين بصحتها، ونؤخر برهانها.

فقد وافقت مبرهنة شرودر–برنشتاين ذلك الحدس، وهي \emph{صحيحة}، وبصحتها يستقيم حمل العلاقتين اللتين عرفناهما، أعني $\cardeq{A}{B}$ و$\cardle{A}{B}$، على المقارنة في «الحجم». ولها، وراء هذا، \emph{نفع عظيم} في البرهان: فقد يعسر تعيين !!a{bijection} بين مجموعتين متساويتين عددًا، فتقسم المبرهنة هذا المطلب إلى مطلبين أيسر منه؛ نعين !!a{injection} من الأولى إلى الثانية، وأخرى في الجهة المقابلة.
\end{explain}
