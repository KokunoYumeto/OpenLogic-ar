\olsection{التجاوز}

\begin{thm}
\ollabel{overspill} كل مجموعة جمل~$\Gamma$ لها نماذج منتهية لا حدّ لكبرها
يكون لها نموذج لا نهائي.
\end{thm}

\begin{proof}
لنزد لغة~$\Gamma$ ثوابت جديدة معدودة $c_0$، و$c_1$،
و\dots، ولنأخذ المجموعة $\Gamma \cup \{c_i \neq c_j : i \neq
  j\}$. ومعنى أن لـ~$\Gamma$ نماذج منتهية لا حدّ لكبرها أنه،
مهما كان $m >0$، وجد $n\ge m$ وكان لـ~$\Gamma$ نموذج عدديته~$n$.
فتكون $\Gamma \cup \{c_i \neq c_j : i \neq j\}$ قابلة للإرضاء
منتهيًا. وبالتراص يوجد لـ~$\Gamma \cup \{c_i \neq c_j : i
\neq j\}$ نموذج~$\Struct M$. ولا يكون مجاله منتهيًا، لأنه يرضي
المتراجحات $c_i \neq c_j$ كلها.
\end{proof}

\begin{prop}
\ollabel{inf-not-fo}
لا توجد جملة~$!A$ من أي لغة من الرتبة الأولى تصدق في
!!a{structure}~$\Struct M$ إذا وفقط إذا كان ال!!{domain}~$\Domain{M}$
لل!!{structure} لا نهائيًا.
\end{prop}

\begin{proof}
لنفرض وجود~$!A$ على الصفة المذكورة. فيصدق نفيها~$\lnot !A$
في جميع ال!!{structure}s المنتهية، ولا يصدق في غيرها. فيكون
  له نماذج منتهية لا حدّ لكبرها، ولا يكون له نموذج لا نهائي،
خلافًا لـ \olref{overspill}.
\end{proof}
