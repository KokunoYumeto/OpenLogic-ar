\olsection{مثال لقابلية الاختزال}

ومن تطبيقات \olref[ppr]{prop:reduce} المثال الآتي.

\begin{prop}
\ollabel{prop:k1}
لتكن
\[
K_1 = \Setabs{e}{\cfind{e}(0) \fdefined}.
\]
عندئذ تكون $K_1$ قابلة للتعداد حسابيًا، ولكنها غير قابلة للحساب.
\end{prop}

\begin{proof}
لما كان $K_1=\Setabs{e}{\lexists[s][T(e,0,s)]}$، فإن $K_1$ قابلة
للتعداد حسابيًا بحسب \olref[eqc]{thm:exists-char}.

وأما عدم قابلية $K_1$ للحساب، فنثبته باختزال $K_0$
إليها.

\begin{explain}
وفي ذلك موضع دقة؛ فإن $K_1$ لا تجيب إلا عن
حسابات دخلها $0$. فتصوّر صديقًا يحسن الإجابة
عن هذا النوع من الأسئلة، ونسمي أمثاله «عُرّافين». ثم جاءك
سائل: هل $\tuple{e,x}$ من $K_0$، أي هل تقف الآلة $e$ عند
الدخل $x$؟ فلك أن تبني آلة أخرى $e_x$ لا تعتد،
عند \emph{أي} دخل، بما أُدخل فيها، بل تشغّل~$e$ عند الدخل $x$.
فيصير سؤال توقف $e$ عند $x$ مكافئًا لسؤال توقف
$e_x$ عند $0$ أو أي دخل آخر. وحين تسأل صديقك عن الآلة
الجديدة $e_x$ عند الدخل $0$، تحصل من جوابه على جواب
السؤال الأول. فهذا هو وجه اختزال $K_0$ إلى~$K_1$.
\end{explain}

ولنحرر الحجة بالدالة الجزئية الشاملة القابلة للحساب: نعرّف $f$ ذات
الحجج الثلاثة بوضع
\[
f(x,y,z) \simeq \cfind{x}(y).
\]
فهذه $f$ لا تتوقف على دخلها الثالث. ولنأخذ فهرسًا $e$ يحقق
$f=\cfind{e}[3]$؛ فيكون لدينا
\[
\cfind{e}[3](x,y,z) \simeq \cfind{x}(y).
\]
فتعطينا مبرهنة $s$-$m$-$n$ دالة $s(e,x,y)$ تحقق، لكل $z$،
\begin{align*}
\cfind{s(e,x,y)}(z) & \simeq \cfind{e}[3](x,y,z) \\
& \simeq \cfind{x}(y).
\end{align*}

\begin{explain}
وبالرجوع إلى التصوير السابق، نجد أن $s(e,x,y)$ فهرس الآلة التي تهمل
أي دخل $z$، وتحسب بدلًا منه $\cfind{x}(y)$.
\end{explain}

ويلزم على الخصوص
\[
\cfind{s(e,x,y)}(0) \fdefined \quad \text{إذا وفقط إذا} \quad
\cfind{x}(y) \fdefined.
\]
فمعنى ذلك أن $\tuple{x,y}\in K_0$ إذا وفقط إذا كان
$s(e,x,y)\in K_1$. ولنثبت عددًا~$z$ فهرسًا للدالة الجزئية غير
المعرفة عند كل دخل؛ فيكون $z\notin K_1$. وعليه نجعل الدالة~$g$
المعرّفة بـ
\[
g(w) =
\begin{cases}
s(e,(w)_0,(w)_1) & \text{إذا كان $w=\tuple{(w)_0,(w)_1}$،}\\
z & \text{في غير ذلك،}
\end{cases}
\]
اختزالًا لـ$K_0$ إلى~$K_1$.

% تصحيح مصدر (C246-01): اختُبر رمز الزوج بإعادة ترميزه، وعُرّفت قيمة g خارجه بثابت لا يقع في K_1.
\textbf{تنبيه تصحيحي على الأصل.} عُرّفت قيمة~$g$ خارج رموز الأزواج
تعريفًا صريحًا خارج~$K_1$؛ لأن الإسقاطين $(w)_0$ و$(w)_1$ وحدهما
لا يضمنان تكافؤ العضوية إذا لم يكن~$w$ رمز زوج؛ ولذلك اختُبر ذلك
بالمساواة $w=\tuple{(w)_0,(w)_1}$، لا بطول الإسقاط وحده.
\end{proof}
