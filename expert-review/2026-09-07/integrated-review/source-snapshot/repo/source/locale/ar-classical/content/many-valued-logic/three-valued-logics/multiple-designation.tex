\olsection{تمييز قيم غير~$\True$}

كانت القيم المميزة في المنطقات المتقدمة إلى الآن هي
$V^+ = \{\True\}$؛ فالحكم بالصدق كان مكافئًا لكون قيمة الصدق
$\True$. غير أنه يجوز أن نحكم بالصدق كلما لم تكن القيمة~$\False$.
وعلى هذا الوجه نضع في المصفوفة، مثلًا،
$V^+ = \{\True, \Undef\}$، فنحصل على منطق آخر.

\begin{defn}
مصفوفة \emph{منطق المفارقة}~$\LogLP$ هي:
\begin{enumerate}
  \item الجزء الخالي من الثوابت~$\Lang L_0^{-}$ من لغة القضايا القياسية، ذو الروابط
  $\lnot$ و$\land$ و$\lor$ و$\lif$.
  \item مجموعة قيم الصدق~$V = \{\True, \Undef, \False\}$.
  \item القيمتان~$\True$ و$\Undef$ مميزتان؛ أي
  $V^+ = \{\True, \Undef\}$.
  \item دوال الصدق هي دوال منطق كليني القوي نفسها.
\end{enumerate}
\end{defn}

\begin{defn}
ومصفوفة \emph{منطق اللامعنى} لهالدين~$\LogHal$ هي:
\begin{enumerate}
  \item اللغة الناتجة من الجزء الخالي من الثوابت~$\Lang L_0^{-}$
  بإضافة الرابط الأحادي~$+$.
  \item مجموعة قيم الصدق~$V = \{\True, \Undef, \False\}$.
  \item القيمتان~$\True$ و$\Undef$ مميزتان؛ أي
  $V^+ = \{\True, \Undef\}$.
  \item دوال الصدق هي دوال منطق كليني الضعيف، ويزاد عليها مؤثر
  «عديم المعنى»:
  \begin{center}
    \begin{tabular}{c|c}
    $\tf{+}$ & \\
    \hline
    $\True$ & $\False$ \\
    $\Undef$ & $\True$ \\
    $\False$ & $\False$
    \end{tabular}
  \end{center}
\end{enumerate}
\end{defn}

فهذان المنطقان، على خلاف منطقي كليني اللذين يوافقانهما في جداول
الصدق، لهما \emph{بالفعل} تحصيلات منطقية.

\begin{prop}\ollabel{prop:LP-taut-CL} تحصيلات~$\LogLP$ هي بعينها
  تحصيلات منطق القضايا الكلاسيكي.
\end{prop}

\begin{proof}
  تفيد \olref[syn][sub]{prop:mvl-cl} أن~$\Entails[\LogLP] !A$ يستلزم
  $\Entails[\LogCL] !A$. ولإثبات الجهة الأخرى يكفينا أن نبين: إذا
  وجد~!!a{valuation}~$\pAssign v\colon \PVar \to
  \{\False, \True, \Undef\}$ يحقق~$\pValue v[\LogKs](!A) = \False$،
  وجد~!!a{valuation}~$\pAssign {v'}\colon \PVar \to \{\False, \True\}$
  يحقق~$\pValue {v'}[\LogCL](!A) = \False$. وهذه الدعوى تثبت المطلوب
  لـ~$\LogLP$؛ فإن~$\LogKs$ و$\LogLP$ مشتركان في دوال الصدق،
  و$\False$ وحدها غير مميزة في~$\LogLP$، وهذا هو الفرق الوحيد بين
  $\LogLP$ و$\LogKs$. فإذا كان~$\Entails/[\LogLP] !A$، كان هناك
  !!{valuation}~$\pAssign v$ بحيث
  $\pValue v[\LogLP](!A) = \pValue v[\LogKs](!A) = \False$.
  وبالدعوى يكون~$\pValue{v'}[\LogCL](!A) = \False$؛ أي
  $\Entails/[\LogCL] !A$.
  
  بقي إثبات الدعوى. نعرّف~$\pAssign {v'}$ أولًا بالمعادلة
  \[
    \pAssign {v'}(p) =
    \begin{cases}
    \True & \text{إذا كان } \pAssign {v}(p) \in \{\True, \Undef\}\\
    \False & \text{في غير ذلك}
  \end{cases}
  \]
  ثم نثبت بالاستقراء على~$!A$ خاصتين: (أ)~إذا كان
  $\pValue v[\LogKs](!A) = \False$، كان
  $\pValue {v'}[\LogCL](!A) = \False$؛ و(ب)~إذا كان
  $\pValue v[\LogKs](!A) = \True$، كان
  $\pValue {v'}[\LogCL](!A) = \True$.
  \begin{enumerate}
    \item أساس الاستقراء هو~$!A \ident p$. فإذا كان، بحسب
    \olref[syn][val]{defn:pValue}،
    $\pValue v[\LogKs](!A) = \False$، كان
    $\pAssign v(p) = \False$؛ ومن تعريف~$\pAssign{v'}$ ينتج
    $\pValue {v'}[\LogCL](!A) = \pAssign{v'}(p) = \False$، فتثبت
    (أ). وإذا كان~$\pValue v[\LogKs](!A) = \True$، كان
    $\pAssign v(p) = \True$، ومن ثم
    $\pValue {v'}[\LogCL](!A) = \pAssign{v'}(p) = \True$، فتثبت
    (ب).
    
    وأما خطوة الاستقراء فتفصل إلى الحالات الآتية:
    \item $!A \ident \lnot !B$.
    \begin{enumerate}
      \item إذا كان~$\pValue v[\LogKs](\lnot!B) = \False$، أفاد
      تعريف~$\tf{\lnot}[\LogKs]$ أن
      $\pValue v[\LogKs](!B) = \True$. وبالحالة~(ب) من فرض
      الاستقراء يكون~$\pValue {v'}[\LogCL](!B) = \True$؛ ولذلك
      $\pValue {v'}[\LogCL](\lnot !B) = \False$.
      \item وإذا كان~$\pValue v[\LogKs](\lnot!B) = \True$، أفاد
      تعريف~$\tf{\lnot}[\LogKs]$ أن
      $\pValue v[\LogKs](!B) = \False$. وبالحالة~(أ) من فرض
      الاستقراء يكون~$\pValue {v'}[\LogCL](!B) = \False$؛ ولذلك
      $\pValue {v'}[\LogCL](\lnot !B) = \True$.
    \end{enumerate}

    \item $!A \ident (!B \land !C)$.
    \begin{enumerate}
      \item إذا كان~$\pValue v[\LogKs](!B \land !C) = \False$،
      أفاد تعريف~$\tf{\land}[\LogKs]$ أن
      $\pValue v[\LogKs](!B) = \False$ أو
      $\pValue v[\LogKs](!C) = \False$. وبالحالة~(أ) من فرض
      الاستقراء يكون~$\pValue {v'}[\LogCL](!B) = \False$ أو
      $\pValue {v'}[\LogCL](!C) = \False$؛ ومن ثم
      $\pValue {v'}[\LogCL](!B \land !C) = \False$.
      \item وإذا كان~$\pValue v[\LogKs](!B \land !C) = \True$،
      أفاد تعريف~$\tf{\land}[\LogKs]$ أن
      $\pValue v[\LogKs](!B) = \True$ و
      $\pValue v[\LogKs](!C) = \True$. وبالحالة~(ب) من فرض
      الاستقراء يكون~$\pValue {v'}[\LogCL](!B) = \True$ و
      $\pValue {v'}[\LogCL](!C) = \True$؛ ومن ثم
      $\pValue {v'}[\LogCL](!B \land !C) = \True$.
    \end{enumerate}
  \end{enumerate}

  والحالتان الباقيتان تجريان على هذا المثال، فنتركهما تمرينًا.
  ويجوز بدل ذلك أن يقال: إن البرهان السابق يثبت النتيجة لكل
  !!{formula}s التي لا يدخل فيها إلا~$\lnot$ و$\land$. ثم نستعمل،
  في~$\LogKs$ و$\LogCL$ جميعًا ولكل~$\pAssign v$، المساواتين
  $\pValue v(!B \lor !C) = \pValue v(\lnot(\lnot!B \land \lnot !C))$
  و$\pValue v(!B \lif !C) = \pValue v(\lnot(!B \land \lnot !C))$.
\end{proof}

\begin{prob}
  أكمل برهان \olref[mvl][thr][mul]{prop:LP-taut-CL}، فأثبت (أ)
  و(ب) للحالتين~$!A \ident (!B \lor !C)$ و
  $!A \ident (!B \lif !C)$.
\end{prob}

\begin{prob}
أثبت أن كل تحصيل كلاسيكي تحصيل في~$\LogHal$.
\end{prob}

ومع أن تحصيلات هذين المنطقين هي تحصيلات المنطق الكلاسيكي نفسها، فإن
علاقتي النتيجة تختلفان. فمنطق~$\LogLP$، مثلًا، \emph{متسامح مع
التناقض}؛ إذ~$\lnot p, p \Entails/ q$، فلا يصح مبدأ الانفجار
$\lnot !A, !A \Entails !B$ على العموم. وقد يصح في حالات مخصوصة من
$!A$ و$!B$، كأن تكون~$!B$ تحصيلًا منطقيًا.

\begin{prob}
  أي العلاقات الآتية تصح في (أ)~$\LogLP$، و(ب)~$\LogHal$؟ ضع لكل
  واحدة جدول صدق.
  \begin{enumerate}
    \item $p, p \lif q \Entails q$
    \item $\lnot q, p \lif q \Entails \lnot p$
    \item $p \lor q, \lnot p \Entails q$
    \item $\lnot p, p \Entails q$
    \item $p \Entails p \lor q$
    \item $p \lif q, q\lif r \Entails p \lif r$
  \end{enumerate}
\end{prob}

فماذا يكون إذا جعلنا~$\Undef$ مميزة في~$\LogLuk[3]$؟

\begin{defn}
  مصفوفة \emph{منطق R-Mingle الثلاثي القيم}~$\LogRM[3]$ هي:
  \begin{enumerate}
    \item الجزء الخالي من الثوابت~$\Lang L_0^{-}$ من لغة القضايا القياسية، ذو الروابط
    $\lnot$ و$\land$ و$\lor$ و$\lif$.
    \item مجموعة قيم الصدق~$V = \{\True, \Undef, \False\}$.
    \item القيمتان~$\True$ و$\Undef$ مميزتان؛ أي
    $V^+ = \{\True, \Undef\}$.
    \item دوال الصدق هي دوال منطق~\L ukasiewicz~$\LogLuk[3]$ نفسها.
    % تصحيح مصدر مشترك (MVL-L0-I1): وُحدت اللغات مع مخزون دوال الصدق المعرّف، وحُذف lfalse غير المفسر من R-Mingle.
  \end{enumerate}
\end{defn}

\begin{prob}
  أي العلاقات الآتية تصح في~$\LogRM[3]$؟
  \begin{enumerate}
    \item $p, p \lif q \Entails q$
    \item $p \lor q, \lnot p \Entails q$
    \item $\lnot p, p \Entails q$
    \item $p \Entails p \lor q$
  \end{enumerate}
\end{prob}

وقد تؤدي جداول صدق مختلفة إلى المنطق نفسه، أي إلى علاقة الاستلزام
نفسها، إذا غيرنا القيم المميزة. ومن ذلك أن نجعل في منطق غودل
$V^+ = \{\True, \Undef\}$ بدل~$\{\True\}$.

\begin{prop}\ollabel{prop:gl-udes}
  المصفوفة التي فيها~$V = \{\False, \Undef,\True\}$ و
  $V^+=\{\True, \Undef\}$، ودوال صدقها دوال منطق غودل الثلاثي،
  تعرّف المنطق الكلاسيكي.
\end{prop}

\begin{proof}
  يترك إثباته تمرينًا.
\end{proof}

\begin{prob}
  أثبت \olref[mvl][thr][mul]{prop:gl-udes}. فإذا كان~$\Log L$
  معرفًا كمنطق غودل، إلا أن~$V^+=\{\True,\Undef\}$، فبيّن أن
  $\Gamma \Entails/[\Log L] !B$ يستلزم~$\Gamma \Entails/[\LogCL] !B$.
  واستعمل طريقة \olref[mvl][thr][mul]{prop:LP-taut-CL}، غير أنك بدل
  إثبات (أ) و(ب) تبين أن
  $\pValue v[\LogGod](!A) = \False$ إذا وفقط إذا كان
  $\pValue {v'}[\LogCL](!A) = \False$، ومن ثم أن
  $\pValue v[\LogGod](!A) \in \{\True,\Undef\}$ إذا وفقط إذا كان
  $\pValue {v'}[\LogCL](!A) = \True$. ثم اشرح كيف تثبت هذه النتيجة
  القضية.
\end{prob}
