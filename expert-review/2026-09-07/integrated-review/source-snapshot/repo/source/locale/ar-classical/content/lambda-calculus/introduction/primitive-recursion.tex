\olsection{الدوال \usetoken{S}{lambda definable} مغلقة تحت العودية البدائية}

أما العودية البدائية فتحتاج إلى برهان نبنيه على مراحل. نفرض، كما تقدم، حدين~$G'$ و$H'$، كل منهما
!!{lambda define} الدالة المناظرة له، وهما $g$ و$h$ بالترتيب. ونطلب حدًا~$F$
!!{lambda define}s الدالة~$f$ التي تعريفها
\begin{align*}
f(0, \vec z) & = g(\vec z) \\
f(x+1, \vec z) & = h(x, f(x,\vec z), \vec z).
\end{align*}
فإذا أعطينا، على العموم، حدي لامبدا~$G'$ و$H'$، كفانا أن نجد حدًا~$F$ يحقق
\begin{align*}
F(\num{0}, \vec z) & \equiv G(\vec z) \\
F(\overline{n+1}, \vec z) & \equiv H(\num{n}, F(\num{n}, \vec z), \vec z)
\end{align*}
لكل عدد طبيعي~$n$. ذلك أن كون~$G'$ و$H'$ كل منهما !!{lambda define} دالته،
$g$ و$h$ بالترتيب، يضمن أننا إذا وضعنا الأعداد~$\num{\vec m}$ موضع~$\vec z$،
انتهى اختزال~$F(\num{n+1}, \num{\vec m})$ إلى الجواب الصحيح في صورة سوية.

ونردّ هذا المطلوب إلى إيجاد حد~$F$ يحقق
\begin{align*}
F(\num 0) & \equiv G \\
F(\overline {n+1}) & \equiv H(\num{n},F(\num{n}))
\intertext{لكل عدد طبيعي $n$، حيث}
G & = \lambd[\vec z][G'(\vec z)] \text{ و}\\
H(u,v) & = \lambd[\vec z][H'(u,v(\vec z),\vec z)].
\end{align*}
فبهذا التصرف في تجريد لامبدا نستغني عن تتبع المعلمات الزائدة
$\vec z$ على حدة؛ إذ تدخل في ترميز لامبدا نفسه.

ولتعريف الحد~$F$ لا بد أولًا من وسيلة نمثل بها الأزواج المرتبة ونتصرف فيها، وهذا ما تثبته اللمّة التالية.

\begin{lem}
يمكن اختيار حد لامبدا~$D$ بحيث يتحقق، لأي حدي لامبدا~$M$ و$N$،
$D(M,N)(\num{0}) \red M$ و$D(M,N)(\num{1}) \red N$.
\end{lem}

\begin{proof}
نبدأ بتعريف حد لامبدا~$K$ كما يلي:
\[
K(y) = \lambd[x][y].
\]
فـ$K$، وفق اصطلاحنا، هو الحد~$\lambd[y][\lambd[x][y]]$. ومعنى ذلك أن كل حد~$M$
يعطي دالة ثابتة~$K(M)$، تردّ القيمة~$M$ مهما يكن المدخل.

ثم نعرّف~$D(x,y,z)$ بالمعادلة~$D(x,y,z) = z (K(y))x$، فنحصل على
\begin{align*}
D(M,N,\num 0) & \red \num 0 (K(N)) M \red M \text{ و}\\
D(M,N,\num 1) & \red \num 1 (K(N)) M \red K(N) M \red N,
\end{align*}
وبذلك يتم المطلوب.
\end{proof}

فالمقصود أن نجعل~$D(M,N)$ تمثيلًا للزوج~$\tuple{M, N}$. ومتى كان~$P$
تمثيلًا لزوج من هذا القبيل، كان~$P(\num 0)$ و$P(\num 1)$ تمثيلين لإسقاطيه الأيسر والأيمن،
أي~$(P)_0$ و$(P)_1$. ومن الآن نعتمد هذين الرمزين الأخيرين.

\begin{lem}
العودية البدائية تحفظ كون الدوال~!!{lambda definable}.
\end{lem}

\begin{proof}
المطلوب أن نجد، لأي حدين~$G$ و$H$، حدًا~$F$ تتحقق له
\begin{align*}
F(\num{0}) & \equiv G \\
F(\num{n+1}) & \equiv H(\num{n}, F(\num{n}))
\end{align*}
عند كل عدد طبيعي~$n$. وسبيل ذلك، إجمالًا، أن نحسب متتاليات من
\emph{الأزواج}
\[
\tuple{\num 0, F(\num 0)}, \tuple{\num 1, F(\num 1)}, \dots,
\]
ونجعل الأعداد أدوات تكرار. فأول هذه الأزواج هو
$\tuple{\num 0, G}$. وإذا بلغنا الزوج~$\tuple{\num{n}, F(\num{n})}$، وجب أن يكون ما يليه، وهو~$\tuple{\num{n+1}, F(\num{n+1})}$،
مكافئًا للزوج~$\tuple{\num{n+1}, H(\num{n}, F(\num{n}))}$. ولذلك ننشئ حد لامبدا~$T$ ينقلنا من كل زوج إلى تاليه في خطوة واحدة.

وبيان هذا الإنشاء أن نعرّف~$T(u)$ بالمعادلة
\[
T(u) = \tuple{\fn{Succ}((u)_0), H((u)_0,(u)_1)}.
\]
وحينئذ يسهل التحقق، لكل عدد~$n$، من أن
\[
T(\tuple{\num{n}, M}) \red \tuple{\num{n+1}, H(\num{n}, M)}.
\]
ثم إذا أعطينا~$G$ و$H$، عرّفنا~$F(u)$ كما سبق أن اقترحنا:
\[
F(u) = (u(T,\tuple{\num 0, G}))_1.
\]
فإذا كان المدخل~$\num{n}$، كرر~$F$ تطبيق~$T$ بمقدار~$n$ من المرات
على~$\tuple{\num 0, G}$، ثم أخذ المركبة الثانية. أما حالة الابتداء فهي
\begin{enumerate}
\item $\num{0} (T, \tuple{\num 0, G}) \equiv \tuple{\num{0}, G}$
\item $F(\num{0}) \equiv G$
\end{enumerate}
ونثبت بالاستقراء على~$n$ العبارتين الآتيتين لكل عدد طبيعي:
\begin{enumerate}
\item $\num{n+1}(T, \tuple{\num{0}, G}) \equiv \tuple{\num{n+1}, F(\num{n+1})}$
\item $F(\num{n+1}) \equiv H(\num{n}, F(\num{n}))$
\end{enumerate}
أما العبارة الثانية فبيانها
\begin{align*}
F(\num{n+1}) & \red (\num{n+1}(T, \tuple{\num 0, G}))_1 \\
& \equiv (T(\num{n} (T, \tuple{\num 0, G})))_1 \\
& \equiv (T(\tuple{\num{n}, F(\num{n})}))_1 \\
& \equiv (\tuple{\num{n+1}, H(\num{n}, F(\num{n}))})_1 \\
& \equiv H(\num{n}, F(\num{n})).
\end{align*}
وفي السطر قبل الأخير استعملنا فرض الاستقراء. وأما الأولى فبيانها
\begin{align*}
\num{n+1} (T, \tuple{\num 0, G}) &
\equiv T(\num{n} (T, \tuple{\num 0, G})) \\
& \equiv T( \tuple{\num{n}, F(\num{n})}) \\
& \equiv \tuple{\num{n+1}, H(\num{n}, F(\num{n}))} \\
& \equiv \tuple{\num{n+1}, F(\num{n+1})}.
\end{align*}
وفي السطر الأخير استعملنا العبارة الثانية. فثبت
$F(\num 0) \equiv G$، وثبت أيضًا، لكل~$n$، أن~$F(\num {n+1}) \equiv H(\num n,
F(\num{n}))$؛ وذلك هو المطلوب.
\end{proof}
