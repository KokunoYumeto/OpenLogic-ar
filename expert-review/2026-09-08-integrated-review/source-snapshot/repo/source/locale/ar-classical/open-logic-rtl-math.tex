% Dynamic save/bind/ownership operations are token-exact.  No saved command is
% expanded while it is captured.
\ExplSyntaxOn
\cs_new_protected:Npn \ol_classical_registry_save_entry:n #1
  {
    \cs_if_exist:cTF {#1}
      {
        \cs_if_exist:cTF {OLClassicalSavedRegistry@#1}
          {
            \OLClassicalRTLMathError
              {Duplicate~registry~entry~#1}
              {The~typed~registry~must~own~each~command~exactly~once.}
          }
          {\cs_gset_eq:cc {OLClassicalSavedRegistry@#1} {#1}}
      }
      {
        \OLClassicalRTLMathError
          {Registry~command~#1~is~undefined}
          {Load~the~pinned~mathematical~package~stack~before~this~layer.}
      }
  }
\cs_new_protected:Npn \ol_classical_registry_bind_entry:nn #1#2
  {\cs_gset_eq:cc {#1} {#2}}
\cs_new_protected:Npn \ol_classical_registry_wrap_entry:nnn #1#2#3
  {
    \ol_classical_registry_save_entry:n {#1}
    \cs_gset_protected:cpn {OLClassicalRegistryWrapper@#1}
      {\OLClassicalRegistryDispatch{#2}{#3}}
    \ol_classical_registry_bind_entry:nn
      {#1} {OLClassicalRegistryWrapper@#1}
  }
\cs_new_protected:Npn \ol_classical_registry_restore_entry:n #1
  {
    \cs_gset_eq:cc {#1} {OLClassicalSavedRegistry@#1}
    \cs_gundefine:c {OLClassicalSavedRegistry@#1}
    \cs_gundefine:c {OLClassicalRegistryWrapper@#1}
  }
\cs_new_protected:Npn \ol_classical_registry_check_entry:n #1
  {
    \cs_if_eq:ccF {#1} {OLClassicalRegistryWrapper@#1}
      {
        \typeout{OLC-RTL-REGISTRY-DRIFT=entry/#1}%
        \OLClassicalRTLRegistryMacrosOwnedfalse
      }
  }

% Extensible arrows are ordered composites, not one-noad symbols.  In TRT a
% generic endpoint swap would expose the saved composite to a second reversal
% through its internal fill.  Typeset the selected saved peer as one TLT
% relation atom instead.  Only the saved shaft/fill composite is TLT.  Each
% optional/mandatory label is a separately boxed TRT math list in the label's
% current style, so a noncommutative label such as A\to B retains the edition's
% formula direction rather than inheriting the shaft's implementation shell.
\newcommand*{\OLClassicalBuildRTLXArrowLabel}[2]{%
  \hbox{%
    \OLClassicalSetLocalMathDirection\@ne
    \m@th$#1#2$%
  }%
}
\newcommand*{\OLClassicalRTLXArrowLabelPalette}[2]{%
  \OLClassicalBuildRTLXArrowLabel{#1}{#2}%
}
\newcommand*{\OLClassicalRTLXArrowLabel}[1]{%
  \mathpalette\OLClassicalRTLXArrowLabelPalette{#1}%
}
\newcommand*{\OLClassicalBuildBoxedXArrow}[4]{%
  \hbox{%
    \OLClassicalSetLocalMathDirection\z@
    \m@th$#1%
      \IfNoValueTF{#3}
        {\OLClassicalSavedRegistry{#2}{\OLClassicalRTLXArrowLabel{#4}}}
        {\OLClassicalSavedRegistry{#2}
          [\OLClassicalRTLXArrowLabel{#3}]
          {\OLClassicalRTLXArrowLabel{#4}}}%
    $%
  }%
}
\newcommand*{\OLClassicalBoxedXArrowChoices}[3]{%
  \mathrel{\mathchoice
    {\OLClassicalBuildBoxedXArrow\displaystyle{#1}{#2}{#3}}%
    {\OLClassicalBuildBoxedXArrow\textstyle{#1}{#2}{#3}}%
    {\OLClassicalBuildBoxedXArrow\scriptstyle{#1}{#2}{#3}}%
    {\OLClassicalBuildBoxedXArrow\scriptscriptstyle{#1}{#2}{#3}}}%
}
\NewDocumentCommand{\OLClassicalRegistryXrightarrow}{o m}{%

}

\newcommand*{\OLClassicalLiteralFences}[1]{%
  #1{40}{4}{41}#1{41}{5}{40}#1{91}{4}{93}#1{93}{5}{91}%
}
% fontspec declares these four small faces in legacymaths (not operators).
% Resolve the named family after font setup instead of assuming family zero.
\newcommand*{\OLClassicalLiteralFenceFamily}{%
  \ifdefined\symlegacymaths\number\symlegacymaths
  \else\number\symoperators\fi
}
\newcommand*{\OLClassicalSaveLiteralFence}[3]{%
  % Read the structured engine codes; never truncate a Unicode mathcode to
  % the legacy 15-bit packing. LuaTeX manual 10.3.6 specifies these arrays.
  \directlua{
    local m = tex.getmathcode(#1)
    local d = tex.getdelcode(#1)
    local family = \OLClassicalLiteralFenceFamily
    local large = ({[40]=0, [41]=1, [91]=2, [93]=3})[#1]
    texio.write_nl("term and log", "OLC-RTL-LITERAL-FENCE=#1;math="
      .. table.concat(m, ",") .. ";delimiter=" .. table.concat(d, ","))
    if not (m[1] == #2) or not (m[2] == family) or not (m[3] == #1)
       or not (d[1] == family) or not (d[2] == #1)
       or not (d[3] == \number\symlargesymbols) or not (d[4] == large) then
      tex.error("OpenLogic RTL math: unsupported literal fence code #1")
    else
      openlogic_classical_rtl_math = openlogic_classical_rtl_math or {}
      local owner = openlogic_classical_rtl_math
      owner.literal_fences = owner.literal_fences or {}
      owner.literal_fences[#1] = {math = {m[1], m[2], m[3]},
        del = {d[1], d[2], d[3], d[4]}}
    end
  }%
}
\newcommand*{\OLClassicalApplyLiteralFence}[3]{%
  % Local assignments inside the complete math group. Keep the source atom
  % class and take only the peer family/character and delimiter faces.
  \directlua{
    local owner = openlogic_classical_rtl_math
    local saved = owner and owner.literal_fences
    if not (saved and saved[#1] and saved[#3]) or tex.globaldefs > 0 then
      tex.error("OpenLogic RTL math: missing fence owner or global code assignment")
    else
      local source = saved[#1]
      local face = source
      if tex.count.OLClassicalDesiredMathDirection == 1 then
        face = saved[#3]
      end
      tex.setmathcode(#1, {source.math[1], face.math[2], face.math[3]})
      tex.setdelcode(#1, {face.del[1], face.del[2], face.del[3], face.del[4]})
    end
  }%
}
\newcommand*{\OLClassicalApplyDelimiterCodes}{%
  \ifOLClassicalDelimiterRuntimeActive
    \OLClassicalLiteralFences\OLClassicalApplyLiteralFence
  \fi
}
\newcommand*{\OLClassicalCheckLiteralFence}[3]{%
  % Outside a complete math list the original codes remain owned unchanged;
  % everymath/everydisplay apply only local assignments within that list.
  \directlua{
    local owner = openlogic_classical_rtl_math
    local saved = owner and owner.literal_fences and owner.literal_fences[#1]
    if not saved then
      tex.error("OpenLogic RTL math: missing original literal fence #1")
    else
      local m = tex.getmathcode(#1)
      local d = tex.getdelcode(#1)
      for i = 1, 3 do
        if not (m[i] == saved.math[i]) then
          tex.error("OpenLogic RTL math: literal fence mathcode owner changed #1")
        end
      end
      for i = 1, 4 do
        if not (d[i] == saved.del[i]) then
          tex.error("OpenLogic RTL math: literal fence delcode owner changed #1")
        end
      end
    end
  }%
}

% The sole retained mechanism from installed luabidi 1.1 lines 339--350 is a
% namespaced post-mlist direction-node wrapper.  It reads the one primitive
% owned here and has no second state, package patching, or list reversal.
\newcommand*{\OLClassicalInstallMathListDirectionCallback}{%
  \directlua{
    local owner = "openlogic.classical.rtl_math.ensure_math_dir"
    if not (type(luatexbase) == "table")
       or not (type(luatexbase.add_to_callback) == "function")
       or not (type(luatexbase.in_callback) == "function") then
      tex.error("OpenLogic RTL math: luatexbase callback API unavailable")
    elseif luatexbase.in_callback("post_mlist_to_hlist_filter", owner) then
      tex.error("OpenLogic RTL math: duplicate post-mlist callback owner")
    else
      openlogic_classical_rtl_math = openlogic_classical_rtl_math or {}
      function openlogic_classical_rtl_math.ensure_math_dir(head)
        if head == nil then
          return head
        end
        local direction = tex.mathdirection
        local first = node.new("dir")
        first.direction = direction
        head = node.insert_before(head, head, first)
        local last = node.new("dir", 1)
        last.direction = direction
        head = node.insert_after(head, node.tail(head), last)
        return head
      end
      openlogic_classical_rtl_math.installed_function =
        openlogic_classical_rtl_math.ensure_math_dir
      luatexbase.add_to_callback(
        "post_mlist_to_hlist_filter",
        openlogic_classical_rtl_math.ensure_math_dir,
        owner)
      texio.write_nl("term and log", "OLC-RTL-MLIST-CALLBACK=installed-once")
    end
  }%
}
\newcommand*{\OLClassicalCheckMathListDirectionCallback}{%
  \directlua{
    local owner = "openlogic.classical.rtl_math.ensure_math_dir"
    local count = 0
    if type(luatexbase) == "table"
       and type(luatexbase.callback_descriptions) == "function" then
      for _, description in ipairs(
          luatexbase.callback_descriptions("post_mlist_to_hlist_filter")) do
        if description == owner then
          count = count + 1
        end
      end
    end
    if not (count == 1) then
      tex.error("OpenLogic RTL math: post-mlist callback ownership drift")
    elseif not (type(openlogic_classical_rtl_math) == "table")
       or not (openlogic_classical_rtl_math.installed_function ==
          openlogic_classical_rtl_math.ensure_math_dir) then
      tex.error("OpenLogic RTL math: private callback identity drift")
    else
      texio.write_nl("term and log", "OLC-RTL-MLIST-CALLBACK-OWNERSHIP=PASS")
    end
  }%
}
\newcommand*{\OLClassicalRemoveMathListDirectionCallback}{%
  \directlua{
    local owner = "openlogic.classical.rtl_math.ensure_math_dir"
    if not (type(luatexbase) == "table")
       or not (type(luatexbase.in_callback) == "function")
       or not (type(luatexbase.remove_from_callback) == "function")
       or not (type(luatexbase.add_to_callback) == "function")
       or not luatexbase.in_callback("post_mlist_to_hlist_filter", owner) then
      tex.error("OpenLogic RTL math: refusing callback removal after ownership drift")
    else
      local removed, description = luatexbase.remove_from_callback(
        "post_mlist_to_hlist_filter", owner)
      local expected = type(openlogic_classical_rtl_math) == "table"
        and openlogic_classical_rtl_math.installed_function or nil
      if not (removed == expected) or not (description == owner) then
        -- A later owner reused our public description.  Put that foreign
        -- function back before failing; restoration must never delete it.
        if type(removed) == "function" and type(description) == "string" then
          luatexbase.add_to_callback(
            "post_mlist_to_hlist_filter", removed, description)
        end
        tex.error("OpenLogic RTL math: callback function identity drift")
      elseif openlogic_classical_rtl_math then
        openlogic_classical_rtl_math.ensure_math_dir = nil
        openlogic_classical_rtl_math.installed_function = nil
      end
    end
  }%
}

\newcommand*{\OLClassicalActivateRTLMathRuntime}{%
  \ifOLClassicalRTLMathRuntimeActive
    \OLClassicalRTLMathError{The RTL mathematics runtime was activated twice}
      {There must be exactly one registry and one post-mlist callback owner.}%
  \else
    \OLClassicalActivateMathListHooks
    \OLClassicalActivateRTLRegistry
    \OLClassicalActivateDelimiterRegistry
    \OLClassicalInstallMathListDirectionCallback
    % Babel and amsmath need opposite logical tag modes under TRT: reqno gives

    % exists, preserving its text-direction setup and changing only \veqno.

    \global\OLClassicalRTLMathRuntimeActivetrue
    \OLClassicalCheckRTLMath
  \fi
}

\newcommand*{\OLClassicalCheckRTLMath}{%
  \ifOLClassicalRTLMathInstalled
    \ifdefined\mathdirection
      \ifnum\mathdirection=\@ne\relax
      \else
        \OLClassicalRTLMathError
          {The installed mathematical direction is no longer TRT}
          {A later package changed mathdirection. Reinspect load order; do not silently reinstall.}%
      \fi
      \ifnum\OLClassicalDesiredMathDirection=\@ne\relax\else
        \OLClassicalRTLMathError
          {The desired mathematical direction is no longer TRT}
          {The state-aware list-boundary contract has drifted.}%
      \fi
    \else
      \OLClassicalRTLMathError
        {The LuaTeX mathdirection primitive disappeared}
        {The current engine is not the engine on which the layer was installed.}%
    \fi
    \iftagsleft@
      \OLClassicalRTLMathError
        {The physical-left display-tag policy was superseded}
        {Do not accept mixed equation and align tag sides.}%
    \fi
    \ifx\bbl@ams@equation\OLClassicalBabelPhysicalLeftEquationAdapter\else
      \OLClassicalRTLMathError
        {The Babel equation-tag adapter was superseded}
        {Restore the intervening owner before checking or restoring this layer.}%
    \fi
    \ifOLClassicalRTLMathRuntimeActive
      \OLClassicalCheckRTLRegistryOwnership
      \OLClassicalCheckRTLMathHookOwnership
      \OLClassicalCheckMathListDirectionCallback
      \OLClassicalCheckDelimiterRegistry
    \else
      \OLClassicalRTLMathError
        {The RTL mathematics runtime is not active}
        {The begin-document activation hook was removed or bypassed.}%
    \fi
    \typeout{OLC-RTL-OWNERSHIP=PASS}%
  \else
    \OLClassicalRTLMathError
      {RTL mathematics has not been installed}
      {Call OLClassicalInstallRTLMath last in the classical-edition preamble.}%
  \fi
}

% Restoration refuses to overwrite any intervening direction, macro,
% mathcode, callback, tag-policy, or hook owner.  Unlike installation it is
% intentionally available after begin-document activation: the registry and
% callback do not exist until then, so a preamble-only restore command would
% make the runtime ownership transaction impossible to reverse.
\NewDocumentCommand{\OLClassicalRestoreMathDirection}{}{%
  \ifOLClassicalRTLMathInstalled
    \ifnum\currentgrouplevel=\z@
      \ifnum\mathdirection=\@ne\relax
        \iftagsleft@
          \OLClassicalRTLMathError
            {Refusing to overwrite a later display-tag policy change}
            {Restore the intervening owner first, then call OLClassicalRestoreMathDirection again.}%
        \else
          \ifOLClassicalRTLMathRuntimeActive
            \OLClassicalRestoreRTLMathRuntime
          \fi
          \ifOLClassicalSavedTagsLeft\tagsleft@true\else\tagsleft@false\fi
          \global\OLClassicalDesiredMathDirection=\OLClassicalSavedMathDirection
          \global\mathdirection=\OLClassicalSavedMathDirection\relax
          \global\OLClassicalRTLMathInstalledfalse
          \RemoveFromHook{begindocument/end}[ol-classical-rtl-math-runtime]%
          \RemoveFromHook{enddocument}[ol-classical-rtl-math-ownership]%
          \typeout{OL-CLASSICAL-RTL-MATH=restored}%
        \fi
      \else
        \OLClassicalRTLMathError
          {Refusing to overwrite a later mathematical-direction change}
          {Restore the intervening owner first, then call OLClassicalRestoreMathDirection again.}%
      \fi
    \else
      \OLClassicalRTLMathError
        {Restore at top level outside a local group}
        {The saved edition-level owners must not be restored from a disposable group.}%
    \fi
  \else
    \OLClassicalRTLMathError
      {There is no installed RTL mathematical direction to restore}
      {No direction assignment was made by this layer.}%
  \fi
}

% These scopes must surround complete math delimiters or math environments;
% switching direction after a formula has begun is intentionally unsupported.
% A local LTR exception is supported inside the globally installed classical
% RTL edition.  A late local LTR-to-RTL switch is deliberately rejected:
% LuaTeX/hyperref can defer link-box calculation until after the group has
% restored the old direction, producing invalid annotation rectangles.  The
% edition-level installer sets TRT before document construction and does not
% have that failure mode.
\newcommand*{\OLClassicalSetLocalMathDirection}[1]{%
  \ifdefined\directlua
    \ifdefined\mathdirection
      \OLClassicalDesiredMathDirection=#1\relax
      \mathdirection=#1\relax
    \else
      \OLClassicalRTLMathError
        {The LuaTeX mathdirection primitive is unavailable}
        {The local direction scope cannot be entered.}%
    \fi
  \else
    \OLClassicalRTLMathError
      {LuaLaTeX is required}
      {The local direction scope cannot be entered.}%
  \fi
}
\newcommand*{\OLClassicalApplyDesiredMathDirection}{%
  \mathdirection=\OLClassicalDesiredMathDirection\relax
  \OLClassicalApplyDelimiterCodes
}
\newcommand*{\OLClassicalRequireInstalledRTLMath}{%
  \ifOLClassicalRTLMathInstalled
    \OLClassicalSetLocalMathDirection\@ne
  \else
    \OLClassicalRTLMathError
      {A late local switch to RTL mathematics is unsupported}
      {Install RTL mathematics at preamble top level; use the local LTR scope only for complete exceptions.}%
  \fi
}
\NewDocumentEnvironment{OLClassicalRTLMathScope}{}%
  {\begingroup\OLClassicalRequireInstalledRTLMath}%
  {\endgroup}
\NewDocumentEnvironment{OLClassicalLTRMathScope}{}%
  {\begingroup\OLClassicalSetLocalMathDirection\z@}%
  {\endgroup}

% Convenience forms create a complete inline formula.  The LTR form also
% establishes Babel's matching inline text-direction embedding; changing only
% \mathdirection inside an Arabic paragraph can otherwise give hyperref an
% off-page annotation box even when the glyphs appear near the margin.  The
% low-level environments continue to own math-list direction only, so callers
% using them directly in RTL prose must wrap the complete formula in
% \babelsublr.  In an existing math list both forms fail explicitly and leave
% the argument in its current direction.
\NewDocumentCommand{\OLClassicalRTLMath}{m}{%
  \ifmmode
    \OLClassicalRTLMathError
      {OLClassicalRTLMath was called inside math mode}
      {Place it outside math mode, or surround the complete formula with OLClassicalRTLMathScope.}%
    #1%
  \else
    \begin{OLClassicalRTLMathScope}\ensuremath{#1}\end{OLClassicalRTLMathScope}%
  \fi
}
\NewDocumentCommand{\OLClassicalLTRMath}{m}{%
  \ifmmode
    \OLClassicalRTLMathError
      {OLClassicalLTRMath was called inside math mode}
      {Place it outside math mode, or surround the complete formula with OLClassicalLTRMathScope.}%
    #1%
  \else
    \ifdefined\babelsublr
      \babelsublr{\begin{OLClassicalLTRMathScope}%
        \ensuremath{#1}%
      \end{OLClassicalLTRMathScope}}%
    \else
      \OLClassicalRTLMathError
        {Babel's inline left-to-right embedding is unavailable}
        {Load the pinned Arabic Babel stack before using OLClassicalLTRMath.}%
      \begin{OLClassicalLTRMathScope}\ensuremath{#1}\end{OLClassicalLTRMathScope}%
    \fi
  \fi
}
