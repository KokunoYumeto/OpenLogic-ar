\olsection{مثال ثان}

\begin{prop}
إذا كان $A \subseteq C$، فإن $A \cup (C \setminus A) = C$.
\end{prop}

\begin{proof}
  لنفرض أن $A \subseteq C$. والمطلوب أن نبين
  $A \cup (C \setminus A) = C$.
  \begin{quote}
    أول ما يلحظ أن العبارة شرطية، وأن تكميمها الكلي مضمر؛ فهي صادقة
    لجميع المجموعات $A$ و$C$. فـ$A$ و$C$ اسمان لمجموعتين اعتباطيتين،
    وطريق إثبات الشرطية أن نفرض مقدمها ونستخرج تاليها.

    ثم نعمل بالفرض $A \subseteq C$. وبسط~$\subseteq$ يبين أن كل
    !!{element} للمجموعة~$A$ هو !!{element} أيضاً للمجموعة~$C$.
    فلنثبت هذه الحقيقة، إذ يتكرر استعمالها في البرهان.
  \end{quote}
  فبتعريف~$\subseteq$، ولما كان $A \subseteq C$، يكون لكل $z$: إذا
  كان $z \in A$ كان $z \in C$.
  \begin{quote}
    بهذا بسطنا ما في الفرض من تعريفات، فننتقل إلى النتيجة. ولإثبات
    $A \cup (C \setminus A) = C$ نرتب الحجة كما في المثال السابق:
    نثبت أن كل !!{element} في $A \cup (C \setminus A)$ هو
    !!a{element} أيضاً في~$C$، ثم نثبت بالعكس أن كل !!{element} في
    $C$ هو !!a{element} في $A \cup (C \setminus A)$. ويختصر القولان
    في~$A \cup (C \setminus A) \subseteq C$ و
    $C \subseteq A \cup (C \setminus A)$. (وهذا رجوع من العبارة
    المبسوطة إلى الاختصار، وفيه تيسير لقراءة البرهان.) ولأنهما اقتران
    وجب إثبات جزأيه. ففي الأول، أي كون كل !!{element} في
    $A \cup (C \setminus A)$ !!a{element} أيضاً في~$C$، نأخذ
    $z \in A \cup (C \setminus A)$ لـ$z$ اعتباطي، ونثبت
    $z \in C$. ثم يرد الفرض $z \in A \cup (C \setminus A)$، بتعريف
    $\cup$، إلى $z \in A$ أو
    $z \in C \setminus A$. وقد غدا هذا المسلك مألوفاً.
    \end{quote}
  ويكون $A \cup (C \setminus A) = C$ إذا وفقط إذا كان
  $A \cup (C \setminus A) \subseteq C$ و
  $C \subseteq (A \cup (C \setminus A))$. فلنثبت أولاً
  $A \cup (C \setminus A) \subseteq C$. لتكن
  $z \in A \cup (C \setminus A)$. فإما أن $z \in A$ وإما أن
  $z \in (C \setminus A)$.
  \begin{quote}
    بلغنا فصلاً، والمطلوب استخراج $z \in C$ منه؛ فنقسم النظر إلى
    حالتين.
  \end{quote}
  الحالة 1: $z \in A$. ولما كان لكل $z$: إذا كان $z \in A$ كان
  $z \in C$، ثبت $z \in C$.
  \begin{quote}
    استعملنا الحقيقة المستخرجة آنفاً من فرض القضية
    $A \subseteq C$. فتمت الحالة الأولى، وننتقل إلى الثانية:
    $z \in (C \setminus A)$. ويدل $C \setminus A$ على \emph{فرق}
    المجموعتين، أي مجموعة !!{element} التي في~$C$ وليست
    !!{element} في~$A$. وكل !!{element} في $C$ غير منتم إلى~$A$ هو،
    من باب أولى،~!!a{element} في~$C$.
  \end{quote}
  الحالة 2: $z \in (C \setminus A)$. ومعناه أن $z \in C$ و
  $z \notin A$؛ فيثبت، بوجه خاص،~$z \in C$.
  \begin{quote}
    تم الاتجاه الأول؛ ونقصد الآن إلى الثاني، فنثبت
    $C \subseteq A \cup (C \setminus A)$. نأخذ $z \in C$ ونبين
    $z \in A \cup (C \setminus A)$.
  \end{quote}
  ولتكن الآن $z \in C$. فالمطلوب أن $z \in A$ أو
  $z \in C \setminus A$.
  \begin{quote}
    إن كل !!{element} في $A$ هو !!{element} أيضاً في $C$، وإن
    $C \setminus A$ جماعة الأشياء التي هي !!{element} في $C$ وليست
    في $A$؛ فمن شأن $z$ أن يكون إما في $A$ وإما في
    $C \setminus A$. وقد يخفى وجه اللزوم على من لم يسبق له تصور
    النتيجة، فنقيمه خطوة خطوة. نستعمل حقيقة ظاهرة لا تحتاج هنا إلى
    برهان: $z \in A$ أو $z \notin A$. وهذا هو «مبدأ الثالث المرفوع»:
    لكل عبارة~$p$، إما أن تكون $p$ صادقة وإما أن يكون نفيها صادقاً.
    (والعبارة $p$ هنا هي $z \in A$.) ولما كان هذا فصلاً عدنا إلى
    البرهان بالحالات.
  \end{quote}
  إما أن $z \in A$ أو $z \notin A$. في الحالة الأولى
  $z \in A \cup (C \setminus A)$. وفي الثانية $z \in C$ و
  $z \notin A$، ومن ثم $z \in C \setminus A$. وعندئذ
  $z \in A \cup (C \setminus A)$.
  \begin{quote}
    وبهذا تم البرهان، إذ قد ثبت $A \cup (C \setminus A) = C$.
  \end{quote}
\end{proof}
