\olsection{العلاقات والدوال}

إذا عُرّفت مجموعة من الكائنات، كالأعداد الطبيعية أو الحدود السليمة،
تعريفًا استقرائيًّا، جاز أن نعرّف كذلك \emph{علاقات على} أفرادها
بالاستقراء. فليكن الحد السليم~$t_1$ حدًّا فرعيًّا من الحد السليم~$t_2$
متى وقع جزءًا منه، ولنرمز إلى ذلك بـ$t_1 \sqsubseteq t_2$. ومن الظاهر
أن كل حد سليم فرع من نفسه: $t \sqsubseteq t$. وهذه العلاقة تقبل
التعريف الاستقرائي الآتي:

\begin{defn}
  تُعرَّف بالاستقراء على~$t_2$ علاقةُ كون الحد السليم~$t_1$ حداً فرعياً
  لـ~$t_2$، أي $t_1 \sqsubseteq t_2$، كما يأتي:
  \begin{enumerate}
  \item إذا كان $t_2$ حرفاً، فإن $t_1 \sqsubseteq t_2$ إذا وفقط إذا كان
    $t_1 = t_2$.
  \item إذا كان $t_2$ هو $[s_1 \circ s_2]$، فإن
    $t_1 \sqsubseteq t_2$ إذا وفقط إذا كان $t_1 = t_2$، أو
    $t_1 \sqsubseteq s_1$، أو $t_1 \sqsubseteq s_2$.
  \end{enumerate}
\end{defn}

ومن هذا التعريف، مثلًا، نحصل على $
\mathrm{a} \sqsubseteq [\mathrm{b} \circ \mathrm{a}]$. فإن البند~(2)
يجعل $
\mathrm{a} \sqsubseteq [\mathrm{b} \circ \mathrm{a}]$ مكافئًا لصدق
$\mathrm{a} = [\mathrm{b} \circ \mathrm{a}]$، أو
$\mathrm{a} \sqsubseteq \mathrm{b}$، أو
$\mathrm{a} \sqsubseteq \mathrm{a}$. والأولان كاذبان: فواضح أن
$\mathrm{a}$ غير مطابق لـ$[\mathrm{b} \circ \mathrm{a}]$؛ وبالبند~(1)
يكون $\mathrm{a} \sqsubseteq \mathrm{b}$ إذا وفقط إذا كان
$\mathrm{a} = \mathrm{b}$، وهو كاذب. وأما البند~(1) نفسه فيجعل
$\mathrm{a} \sqsubseteq \mathrm{a}$ إذا وفقط إذا كان
$\mathrm{a} = \mathrm{a}$، وهذا صادق.

ويتوقف إحكام هذا التعريف على حقيقة لم تثبت بعد: أن كل حد سليم~$t$
إما حرف منفرد، وإما أن له حدين سليمين $s_1$ و$s_2$
\emph{متعينين على وجه فريد} بحيث $t = [s_1 \circ s_2]$. ومعنى التفرد
أنه إذا كان $t = [s_1 \circ s_2]$ لم يصح أن يكون \emph{أيضًا}
$= [r_1 \circ r_2]$ مع $s_1 \neq r_1$ أو $s_2 \neq r_2$.
فلو جاز ذلك لناقض البند~(2) نفسه: ربما أعطت قراءة $t_2$ على صورة
$[s_1 \circ s_2]$ الحكم $t_1 \sqsubseteq t_2$، ثم أعطت قراءته على
صورة $[r_1 \circ r_2]$ نفي $t_1 \sqsubseteq t_2$. وقبل إثبات امتناع
هذا في الحدود السليمة، ننظر في مثال يقع فيه بالفعل.

\begin{defn}
  نعرّف \emph{الحدود الخالية من الأقواس} استقرائياً كما يأتي:
  \begin{enumerate}
  \item كل حرف حد خالٍ من الأقواس.
  \item إذا كان $s_1$ و$s_2$ حدين خاليين من الأقواس، فإن
    $s_1 \circ s_2$ حد خالٍ من الأقواس.
  \item لا شيء آخر حد خالٍ من الأقواس.
  \end{enumerate}
\end{defn}

ومن الحدود الخالية من الأقواس: $\mathrm{a}$، و
$\mathrm{b} \circ \mathrm{d}$، و
$\mathrm{b} \circ \mathrm{a} \circ \mathrm{b}$. ولو عرّفنا «الحد
الفرعي» لهذه الحدود على قياس التعريف السابق، لكان البند الثاني:
\begin{center}
  إذا كان $t_2 = s_1 \circ s_2$، فإن $t_1 \sqsubseteq t_2$ إذا وفقط إذا
  كان $t_1 = t_2$، أو $t_1 \sqsubseteq s_1$، أو
  $t_1 \sqsubseteq s_2$.
\end{center}

غير أن $\mathrm{b} \circ \mathrm{a} \circ \mathrm{b}$ يقع على الصورة
$s_1 \circ s_2$، حيث
\begin{align*}
  s_1 & = \mathrm{b} \text{ و} & s_2 & = \mathrm{a} \circ \mathrm{b}.
\intertext{وهو أيضاً من الصورة $r_1 \circ r_2$، حيث}
  r_1 & = \mathrm{b} \circ \mathrm{a} \text{ و} & r_2 &= \mathrm{b}.
\end{align*}
فهل $\mathrm{a} \circ \mathrm{b}$ حد فرعي من
$\mathrm{b} \circ \mathrm{a} \circ \mathrm{b}$؟ تجيب القراءة الأولى
بنعم، والثانية بلا.

وتسمى فرادة الطريق الذي يُبنى به الحد السليم من حدود سليمة أخرى
\emph{قابلية القراءة الفريدة}. ولما كانت التعريفات الاستقرائية
للعلاقات على هذه الكائنات ذات شأن، وجب أن نثبت تحقق هذه الخاصية.

\begin{prop}
  لنفترض أن $t$ حد سليم. عندئذ إما أن يكون $t$ حرفاً منفرداً، وإما أن
  يوجد حدان سليمان محددان على نحو فريد، $s_1$ و$s_2$، بحيث
  $t = [s_1 \circ s_2]$.
\end{prop}

\begin{proof}
  إذا كان $t$ حرفًا منفردًا حصل الشرط. فلنفرض إذن أن $t$ ليس حرفًا
  منفردًا. فيلزم من التعريف الاستقرائي أن يكون على صورة
  $[s_1 \circ s_2]$ لحدين سليمين $s_1$ و~$s_2$. ولم يبق إلا تفردهما؛
  أي إذا كان $t = [r_1 \circ r_2]$، لزم أن
  $s_1 = r_1$ و$s_2 = r_2$.

  فلنفرض أن $t = [s_1 \circ s_2]$ وأن $t = [r_1 \circ r_2]$ أيضًا،
  وأن $s_1$ و$s_2$ و$r_1$ و$r_2$ حدود سليمة. نريد إثبات
  $s_1 = r_1$ و$s_2 = r_2$. ولا بد أولًا من تطابق $s_1$ و$r_1$؛
  وإلا كان أحدهما قطعة ابتدائية صحيحة من الآخر، وذلك ممتنع بموجب
  \olref[sti]{prop:initial} ما دام $s_1$ و~$r_1$ حدين سليمين. وإذا
  ثبت $s_1 = r_1$ ظهر كذلك $s_2 = r_2$.
\end{proof}

وكما تُعرّف العلاقات، تُعرّف الدوال استقرائيًّا. فمن ذلك الدالة~$f$
التي تسند إلى الحد السليم أقصى عمق لأقواسه المتداخلة $[\dots]$،
وتعريفها:

\begin{defn}
  \ollabel{defn:depth} يُعرَّف \emph{عمق} الحد السليم، $f(t)$، استقرائياً
  كما يأتي:
  \[
  f(t) = \begin{cases}
    0 & \text{ إذا كان $t$ حرفاً}\\
    \max(f(s_1), f(s_2)) + 1 & \text{ إذا كان $t = [s_1 \circ s_2]$.}
  \end{cases}
  \]
\end{defn}

ومن أمثلة حسابها:
\begin{align*}
  f([\mathrm{a} \circ \mathrm{b}]) & =
    \max(f(\mathrm{a}),f(\mathrm{b})) + 1 \\
  &= \max(0, 0) + 1 = 1 \text{، و}\\
  f([[\mathrm{a} \circ \mathrm{b}] \circ \mathrm{c}]) & =
    \max(f([\mathrm{a} \circ \mathrm{b}]), f(\mathrm{c})) + 1 \\
  & = \max(1,0) + 1 = 2.
\end{align*}

ومعلوم أن $s_1$ و$s_2$ حدان سليمان، وأن كل حد سليم إما حرف وإما على
الصورة $[s_1 \circ s_2]$. ويعود التفرد هنا فيكون شرطًا لازمًا؛ ولبيان
ذلك نرجع إلى الحدود الخالية من الأقواس المتقدمة. فيكون «التعريف»
النظير:
\[
  g(t) =
  \begin{cases}
    0 & \text{ إذا كان $t$ حرفاً}\\
   \max(g(s_1), g(s_2)) + 1 & \text{ إذا كان $t = s_1 \circ s_2$.}
  \end{cases}
\]
ثم ننظر في الحد الخالي من الأقواس
$\mathrm{a} \circ \mathrm{b} \circ \mathrm{c} \circ \mathrm{d}$.
فله أكثر من قراءة؛ منها أن يؤخذ على أنه
$s_1 \circ s_2$، حيث
\begin{align*}
  s_1 & = \mathrm{a} \text{ و} &
  s_2 & = \mathrm{b} \circ \mathrm{c} \circ \mathrm{d},
\intertext{أو على أنه $r_1 \circ r_2$، حيث}
  r_1 & = \mathrm{a} \circ \mathrm{b} \text{ و} &
  r_2 &= \mathrm{c} \circ \mathrm{d}.
\end{align*}
وحساب $g$ على القراءة الأولى يعطي:
\begin{align*}
  g(s_1 \circ s_2) & =
  \max(g(\mathrm{a}), g(\mathrm{b} \circ \mathrm{c} \circ \mathrm{d})) + 1 \\ & =
  \max(0,2) + 1 = 3
  \intertext{بينما نحصل بحسب القراءة الأخرى على}
  g(r_1 \circ r_2) & =
  \max(g(\mathrm{a} \circ \mathrm{b}), g(\mathrm{c} \circ \mathrm{d})) + 1 \\ &
  = \max(1,1) + 1 = 2.
\end{align*}
ولما كان من شرط الدالة أن تعطي قيمة واحدة، لم يكن «تعريفنا» لـ~$g$
تعريفًا لدالة أصلًا.

\begin{prob}
  ضع تعريفًا استقرائيًّا للدالة $l$، حيث $l(t)$ عدد الرموز في الحد
  السليم~$t$.
\end{prob}

\begin{prob}
  أثبت بالاستقراء البنيوي على الحدود السليمة $t$ أن $f(t) < l(t)$ (حيث
  $l(t)$ عدد الرموز في~$t$، و$f(t)$ عمق $t$ كما عُرّف في
  \olref[mth][ind][rel]{defn:depth}).
\end{prob}
