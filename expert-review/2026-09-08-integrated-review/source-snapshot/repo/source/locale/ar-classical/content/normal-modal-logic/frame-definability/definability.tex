\olsection{قابلية تعريف الأطر}

لم تصح عكوس~\olref[acc]{thm:soundschemas} في النماذج؛ ولكنها تصح
متى جعلنا الكلام في الأطر. فالخصائص المذكورة
في~\olref[acc]{thm:soundschemas} تتميز بأن صلاحية !!a{formula}
في \emph{إطار} توجب على علاقة وصوله الخاصية المقابلة.
ولهذا نقول إن تلك ال!!{formula}s \emph{تعرّف} أصناف الأطر المتصفة
بتلك الخصائص.

\begin{defn}
  إذا كان~$\mClass{F}$ صنفًا من الأطر، نقول إن~$!A$
  \emph{تعرّف}~$\mClass{F}$ إذا وفقط إذا كان
  $\mModel{F} \Entails !A$ لجميع الأطر~$\mModel{F} \in \mClass{F}$
  ولها وحدها.
\end{defn}

نشرع الآن في إثبات نتائج قابلية التعريف الكاملة للأطر.

\begin{thm}\ollabel{thm:fullCorrespondence}
إذا كانت ال!!{formula} في الجانب الأيمن من~\olref[acc]{tab:five}
صالحة في إطار~$\mModel{F}$، كانت لـ$\mModel{F}$ الخاصية في الجانب
الأيسر.
\end{thm}

\begin{proof}
  \begin{enumerate}
  \item افترض أن~\Ax{D} صالحة في~$\mModel{F} = \tuple{W, R}$، أي
    $\mModel{F} \Entails \Box p \lif \Diamond p$. وليكن
    $\mModel{M} = \tuple{W, R, V}$ نموذجًا مبنيًا على~$\mModel{F}$،
    وليكن~$w \in W$. نحتاج إلى إثبات وجود~$v$ بحيث~$Rwv$. افترض خلاف
    ذلك؛ عندئذ يكون كل من~$\mSat{M}{\Box !A}[w]$ و
    $\mSat/{M}{\Diamond !A}[w]$ لأي~$!A$، ومنها~$p$. لكن عندئذ يكون
    $\mSat/{M}{\Box p \lif \Diamond p}[w]$، مناقضًا للفرض
    $\mModel{F} \Entails \Box p \lif \Diamond p$.
  \item افترض أن~\Ax{T} صالحة في~$\mModel{F}$، أي
    $\mModel{F} \Entails \Box p \lif p$. وليكن~$w \in W$ عالمًا
    اعتباطيًا؛ نحتاج إلى إثبات~$Rww$. اجعل~$u \in V(p)$ إذا وفقط إذا
    كان~$Rwu$ (وإذا كان~$q$ غير~$p$، فليكن~$V(q)$ اعتباطيًا، ولنقل
    $V(q) = \emptyset$). وليكن~$\mModel{M} = \tuple{W, R, V}$. بحسب
    البناء، لكل~$u$ يحقق~$Rwu$ يكون~$\mSat{M}{p}[u]$، ومن ثم
    $\mSat{M}{\Box p}[w]$. لكن بحسب الفرض تكون~$\Box p \lif p$ صادقة
    في~$w$، فيكون~$\mSat{M}{p}[w]$؛ وبتعريف~$V$ لا يمكن هذا إلا إذا
    كان~$Rww$.
  \item نبرهن المعاكس النقيض: افترض أن~$\mModel{F}$ غير متناظر، ونثبت
    أن~\Ax{B}، أي~$p \lif \Box\Diamond p$، ليست صالحة في
    $\mModel{F}= \tuple{W, R}$. فإذا كان~$\mModel{F}$ غير متناظر،
    وُجد~$u$ و$v \in W$ بحيث~$Ruv$ لكن ليس~$Rvu$. عرّف~$V$ بحيث
    $w \in V(p)$ إذا وفقط إذا لم يكن~$Rvw$ (واجعل~$V$ اعتباطيًا فيما
    عدا ذلك). وليكن~$\mModel{M} =\tuple{W, R, V}$. الآن، بتعريف~$V$،
    يكون~$\mSat{M}{p}[w]$ لكل~$w$ لا يحقق~$Rvw$، وبخاصة
    $\mSat{M}{p}[u]$ لأن~$Rvu$ لا يصدق. كذلك، لما كان~$Rvw$ إذا وفقط
    إذا كان~$w \notin V(p)$، فلا يوجد~$w$ يحقق~$Rvw$ و
    $\mSat{M}{p}[w]$ معًا، ومن ثم~$\mSat/{M}{\Diamond p}[v]$. ولما
    كان~$Ruv$، كان أيضًا~$\mSat/{M}{\Box\Diamond p}[u]$. فيلزم
    $\mSat/{M}{p \lif \Box\Diamond p}[u]$، ومن ثم لا تكون~\Ax{B}
    صالحة في~$\mModel{F}$.
  \item افترض أن~\Ax{4} صالحة في~$\mModel{F} = \tuple{W,R}$، أي
    $\mModel{F} \Entails \Box p \lif \Box\Box p$، ولتكن~$u$ و$v$
    و$w \in W$ عوالم اعتباطية بحيث~$Ruv$ و$Rvw$؛ نحتاج إلى إثبات
    $Ruw$. عرّف~$V$ بحيث~$z \in V(p)$ إذا وفقط إذا كان~$Ruz$ (واجعل
    $V$ اعتباطيًا فيما عدا ذلك). وليكن~$\mModel{M} =\tuple{W, R, V}$.
    بتعريف~$V$، يكون~$\mSat{M}{p}[z]$ لكل~$z$ يحقق~$Ruz$، ومن ثم
    $\mSat{M}{\Box p}[u]$. لكن بحسب الفرض~\Ax{4} تكون
    $\Box p \lif \Box \Box p$ صادقة في~$u$، فيكون
    $\mSat{M}{\Box \Box p}[u]$. ولما كان~$Ruv$ و$Rvw$، كان
    $\mSat{M}{p}[w]$؛ وبتعريف~$V$ لا يمكن هذا إلا إذا كان~$Ruw$، كما
    أُريد.
  \item نسلك طريق المعاكس النقيض، فنفترض أن الإطار
    $\mModel{F} = \tuple{W, R}$ غير إقليدي، ونثبت أنه يفنّد~\Ax{5}،
    أي~$\mModel{F} \Entails/ \Diamond p \lif \Box\Diamond p$. افترض
    وجود عوالم~$u$ و$v$ و$w \in W$ بحيث~$Rwu$ و$Rwv$ لكن ليس~$Ruv$.
    عرّف~$V$ بحيث، لكل عالم~$z$، يكون~$z \in V(p)$ إذا وفقط إذا كان
    \emph{غير} صحيح أن~$Ruz$. وليكن~$\mModel{M} =\tuple{W, R, V}$.
    عندئذ، بحسب الفرض،~$\mSat{M}{p}[v]$، ولما كان~$Rwv$، كان أيضًا
    $\mSat{M}{\Diamond p}[w]$. لكن لا يوجد عالم~$y$ يحقق~$Ruy$ و
    $\mSat{M}{p}[y]$، ولذلك~$\mSat/{M}{\Diamond p}[u]$. ولما كان
    $Rwu$، لزم~$\mSat/{M}{\Box\Diamond p}[w]$، فتفشل~\Ax{5}، أي
    $\Diamond p \lif \Box\Diamond p$، في~$w$.
  \end{enumerate}
\end{proof}

وبين حالة~\Ax{D} وسائر الحالات فرق: لم نحتج فيها إلى اختيار خاص
للتقييم~$V$. فالحاصل أن~$\mSat{M}{\Ax{D}}$ يوجب تسلسل
$\mModel{M}$ نفسه. ولهذا كانت~\Ax{D} مميزة لصنف \emph{النماذج}
التسلسلية، لا لصنف الأطر وحده.

\begin{cor}\ollabel{prop:D-serial}
  كل نموذج تكون فيه~\Ax{D} صادقة تسلسلي.
\end{cor}

\begin{cor}
كل !!{formula} في الجانب الأيمن من~\olref[acc]{tab:five} تعرّف صنف
الأطر ذات الخاصية في الجانب الأيسر.
\end{cor}

\begin{proof}
  الاتجاه الأول هو ما ثبت في~\olref[acc]{thm:soundschemas}:
  خاصية الجانب الأيسر في النموذج تكفي لصدق ال!!{formula}
  في الجانب الأيمن. فإذا اتصف إطار~$\mModel{F}$ بتلك الخاصية،
  كانت ال!!{formula} صالحة في~$\mModel{F}$.
  وأما الاتجاه الآخر فتكفلت به~\olref{thm:fullCorrespondence}:
  صلاحية !!a{formula} من الجانب الأيمن في~$\mModel{F}$
  تقتضي أن تكون لـ$\mModel{F}$ خاصية الجانب الأيسر.
\end{proof}

\begin{prob}
أثبت أنه إذا كانت ال!!{formula} في الجانب الأيمن من
\olref[nml][frd][acc]{tab:anotherfive} صالحة في إطار~$\mModel{F}$،
كانت لـ$\mModel{F}$ الخاصية في الجانب الأيسر. ولفعل ذلك، تأمل إطارًا
\emph{لا} يحقق الخاصية في الجانب الأيسر، وعرّف~$V$ مناسبة تجعل
ال!!{formula} في الجانب الأيمن كاذبة في عالم ما.
\end{prob}

ويتيح~\olref{thm:fullCorrespondence} ضم الخصائص بعضها إلى بعض.
فإذا صلحت~\Ax{B} و\Ax{4} معًا في~$\mModel{F}$، كان متناظرًا
ومتعديًا. وعلى هذا تُوصف منطقات موجهية كثيرة مهمة:
نأخذ مجموعة ال!!{formula}s الصالحة في جميع الأطر الجامعة
لخصائص معينة، وهي نفسها مجموعة ال!!{formula}s الصالحة في
الأطر التي تصلح فيها ال!!{formula}s المعرّفة لتلك الخصائص.
فـ$\Log{S4}$، مثلًا، مجموعة جميع ال!!{formula}s الصالحة في
الأطر الانعكاسية المتعدية، أي الأطر التي تصلح فيها~\Ax{T} و\Ax{4}.
وأما~$\Log{S5}$ فمجموعة الصيغ الصالحة في الأطر الانعكاسية
المتناظرة الإقليدية، وهي الأطر التي تصلح فيها~\Ax{T} و\Ax{B} و\Ax{5}.

ولذلك تقابل العلاقات بين خصائص~$R$ علاقات بين ال!!{formula}s
المعرّفة لها. فالانعكاسية تقتضي التسلسل، ومن ثم إذا صلحت~\Ax{T}
في إطار صلحت فيه~\Ax{D}. لكن هذا \emph{ليس} لزومًا بين الصيغ
في عالم واحد؛ فلا يلزم من~$\mSat{M}{\Ax{T}}[w]$
أن يكون~$\mSat{M}{\Ax{D}}[w]$. وفيما يأتي جملة من علاقات الخصائص.

\begin{prop}\ollabel{prop:relation-facts}
  لتكن~$R$ علاقة ثنائية على مجموعة~$W$؛ عندئذ:
  \begin{enumerate}
  \item إذا كانت~$R$ انعكاسية، فهي تسلسلية.
  \item إذا كانت~$R$ متناظرة، فهي متعدية إذا وفقط إذا كانت إقليدية.
  \item إذا كانت~$R$ متناظرة أو إقليدية، فهي موجهة بضعف (لها ``خاصية
    الماسة'').
  \item إذا كانت~$R$ إقليدية، فهي متصلة بضعف.
  \item إذا كانت~$R$ دالية، فهي تسلسلية.
  \end{enumerate}
\end{prop}

\begin{prob}
  أثبت~\olref[nml][frd][def]{prop:relation-facts}.
\end{prob}
