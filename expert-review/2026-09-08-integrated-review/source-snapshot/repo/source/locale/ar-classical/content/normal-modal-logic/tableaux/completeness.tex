\olsection{تمام \Log{K}}

\begin{explain}
  ولإثبات تمام طريقة ال!!{tableau}s يكفي أن نبين أنه متى لم يوجد
  !!{tableau} مغلق لـ~$\Gamma \Proves !A$، كان~$\Gamma \Entails/ !A$؛
  أي أمكن بناء نموذج مضاد. ومعنى عدم وجوده أن كل طريق إلى بنائه لا يفضي
  إلى الإغلاق. فيكفي أن نختار من تلك الطرق طريقًا
  \emph{منتظمًا يستنفد القواعد}: فإن امتنع الإغلاق عمومًا امتنع فيه
  أيضًا، ولو وجد !!{tableau} مغلق لأفضى هذا الطريق المستنفد إلى
  الإغلاق. ونجد في فرع مفتوح من ال!!{tableau} الناتج جميع ما يحتاج إليه
  بناء النموذج المضاد. نأخذ البادئات المستعملة في ذلك الفرع عوالم،
  ونجعل !!a{propositional variable}~$p$ صادقة عند~$\sigma$ إذا وفقط إذا
  وردت~$\sFmla{\True}{p}[\sigma]$ في الفرع.
\end{explain}

\begin{defn}
  الفرع التام في !!a{tableau} هو الذي لا تنقصه نتائج القواعد الواجبة
  عليه: فمتى احتوى !!{formula} ذات بادئة~$\sFmla{S}{!A}[\sigma]$
  تصلح لتطبيق قاعدة، احتوى كذلك:
  \begin{enumerate}
    \item جميع ال!!{formula}s ذات البادئات الناتجة عن القاعدة، إن
      كانت قاعدة قضية لا تشعّب الفرع؛
    \item واحدة من !!{formula}s النتيجة على الأقل، إن كانت القاعدة
      القضية تشعّب الفرع؛
    \item نتيجة ممكنة واحدة على الأقل، إن كانت القاعدة الجهوية
      تقتضي بادئة جديدة؛
    \item النتيجة عند كل بادئة مستعملة في الفرع تصلح لتطبيق القاعدة،
      إن كانت القاعدة الجهوية تشترط بادئة مستعملة.
  \end{enumerate}
\end{defn}

\begin{explain}
فمن ذلك أن الفرع التام يحتوي~$\sFmla{\True}{!B}[\sigma]$
و$\sFmla{\True}{!C}[\sigma]$ متى احتوى
$\sFmla{\True}{!B \land !C}[\sigma]$. ومتى احتوى
$\sFmla{\True}{!B \lor !C}[\sigma]$ لم يخل من واحدة على الأقل من
$\sFmla{\True}{!B}[\sigma]$ و$\sFmla{\True}{!C}[\sigma]$.
وإن احتوى
\iftag{prvBox} {$\sFmla{\False}{\Box !B}[\sigma]$}
{$\sFmla{\True}{\Diamond !B}[\sigma]$} وجب أن يحتوي
\iftag{prvBox} {$\sFmla{\False}{!B}[\sigma.n]$}
{$\sFmla{\True}{!B}[\sigma.n]$} لعدد~$n$ واحد على الأقل.
وأما إذا احتوى
\iftag{prvBox} {$\sFmla{\True}{\Box !B}[\sigma]$}
{$\sFmla{\False}{\Diamond !B}[\sigma]$} فيجب أن يحتوي
\iftag{prvBox} {$\sFmla{\True}{!B}[\sigma.n]$}
{$\sFmla{\False}{!B}[\sigma.n]$} لكل~$n$ تكون~$\sigma.n$
مستعملة في الفرع.
\end{explain}

\begin{prop}\ollabel{prop:complete-tableau}
  لكل $\Gamma$ منتهية !!a{tableau} يكون كل فرع فيه إما مغلقًا وإما تامًا.
\end{prop}

\begin{proof}
  نبدأ بفرع مفتوح في !!a{tableau} لـ~$\Gamma$.
  فيه عدد منتهٍ من ال!!{formula}s ذات البادئات التي تصلح لإجراء
  قاعدة. فنرتب هذه ال!!{formula}s ترتيبًا ثابتًا، كترتيبها من أعلى
  إلى أسفل، ثم نجري على كل واحدة لم تستوفِ الشروط (1)--(4) ما
  يلزمها من القواعد. فإن حدث تشعيب واصلنا العمل عند طرف كل فرع
  ناتج، مستوفين ال!!{formula}s الباقية. ولتناهي عدد ال!!{formula}s
  ذات البادئات وعدد البادئات المستعملة في هذه المرحلة، تنتهي هذه
  الدورة إلى فروع تمدّ الفرع الأول، وإن كثرت. ثم نعيد العمل في كل
  واحد منها، ونكرر ذلك، ولا نعيد قاعدةً قد استُوفيت نتائجها المطلوبة.

  ولا يمتد هذا العمل إلى غير نهاية. فإن الصيغ الجزئية للصيغ الابتدائية
  منتهية، وما يظهر عند كل بادئة صيغ موقعة منها. ثم إن كل بادئة جديدة
  تولدها قاعدة جهوية تحمل صيغة دون صيغة بادئتها الأم في العمق الجهوي؛
  فلا يجاوز طول سلاسل البادئات أكبر عمق جهوي في~$\Gamma$. ولكل بادئة
  عدد منتهٍ من الصيغ والقواعد. فتتناهى البادئات والصيغ المولدة، وتنتهي
  العملية إلى فروع مغلقة أو مفتوحة تامة.
\end{proof}

\begin{thm}[التمام]
  \ollabel{thm:tableau-completeness}
  إذا كانت $\Gamma$ منتهية ولم يكن لها !!{tableau} مغلق، فإن $\Gamma$
  قابلة للإرضاء.
\end{thm}

\begin{proof}
إن خلت~$\Gamma$ أرضاها أي نموذج ذي عالم واحد؛ فلنفرضها غير خالية.
بمقتضى القضية يوجد لـ~$\Gamma$ !!a{tableau} كل فرع فيه مغلق أو تام. ولما لم يوجد
لها !!{tableau} مغلق، كان لها !!a{tableau} فيه فرع مفتوح وتام.
نأخذ~$\Delta$ مجموعة ال!!{formula}s ذات البادئات في هذا الفرع،
و$P(\Delta)$ مجموعة ما ورد فيه من البادئات.
ولعدم خلو~$\Gamma$ لا تخلو~$P(\Delta)$، فتصلح مجموعةً لعوالم نموذج.
% تصحيح برهاني (0468-I3): فُصلت حالة Γ الخالية حتى لا ينتج نموذج بمجموعة عوالم خالية.

ثم نبني~$\mModel{M(\Delta)} = \tuple{P(\Delta), R, V}$، جاعلين
عوالمه البادئات الواقعة في~$\Delta$، وعلاقة الإتاحة فيه:
\[
R\sigma\sigma' \quad \text{إذا وفقط إذا} \quad
\sigma'=\sigma.n \quad \text{لبعض~$n$}
\]
ويُعطى إسناده بـ
\[
V(p) = \Setabs{\sigma}{\sFmla{\True}{p}[\sigma] \in \Delta}.
\]
وموضع البرهان بالاستقراء على~$!A$ أمران: إذا كان
$\sFmla{\True}{!A}[\sigma] \in
\Delta$ كان~$\mSat{M(\Delta)}{!A}[\sigma]$؛ وإذا كان
$\sFmla{\False}{!A}[\sigma] \in \Delta$ كان
$\mSat/{M(\Delta)}{!A}[\sigma]$.

\begin{enumerate}
  \item \indcase{!A}{p}{إذا كان $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$،
    فإن $\sigma \in V(p)$ (بحسب تعريف~$V$)، ومن ثم
    $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

    وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإن
    $\sFmla{\True}{\indfrm}[\sigma] \notin \Delta$، وإلا لكان الفرع
    مغلقًا. ولذلك $\sigma \notin V(p)$، ومن ثم
    $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}
  \item \indcase{!A}{\lnot !B}{\iftag{probNot}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فإن
      $\sFmla{\False}{!B}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، $\mSat/{M(\Delta)}{!B}[\sigma]$، ومن ثم
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإن
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، $\mSat{M(\Delta)}{!B}[\sigma]$، ومن ثم
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
  \item \indcase{!A}{!B \land !C}{\iftag{probAnd}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فإن كلًا من
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ و
      $\sFmla{\True}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، $\mSat{M(\Delta)}{!B}[\sigma]$ و
      $\mSat{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإما
      أن يكون $\sFmla{\False}{!B}[\sigma] \in \Delta$ أو يكون
      $\sFmla{\False}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، إما $\mSat/{M(\Delta)}{!B}[\sigma]$ أو
      $\mSat/{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
  \item \indcase{!A}{!B \lor !C}{\iftag{probOr}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فإما أن يكون
      $\sFmla{\True}{!B}[\sigma] \in \Delta$ أو يكون
      $\sFmla{\True}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، إما $\mSat{M(\Delta)}{!B}[\sigma]$ أو
      $\mSat{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإن
      كلًا من $\sFmla{\False}{!B}[\sigma] \in \Delta$ و
      $\sFmla{\False}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، يتحقق كل من $\mSat/{M(\Delta)}{!B}[\sigma]$ و
      $\mSat/{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
  \item \indcase{!A}{!B \lif !C}{\iftag{probIf}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فإما أن يكون
      $\sFmla{\False}{!B}[\sigma] \in \Delta$ أو يكون
      $\sFmla{\True}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، إما $\mSat/{M(\Delta)}{!B}[\sigma]$ أو
      $\mSat{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإن
      كلًا من $\sFmla{\True}{!B}[\sigma] \in \Delta$ و
      $\sFmla{\False}{!C}[\sigma] \in \Delta$ لأن الفرع تام. وبفرض
      الاستقراء، يتحقق كل من $\mSat{M(\Delta)}{!B}[\sigma]$ و
      $\mSat/{M(\Delta)}{!C}[\sigma]$. ومن ثم
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
  \item \indcase{!A}{\Box !B}{\iftag{probBox}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فإن
      $\sFmla{\True}{!B}[\sigma.n] \in \Delta$ لكل $\sigma.n$
      مستعملة على الفرع، لأن الفرع تام؛ أي لكل $\sigma'
      \in P(\Delta)$ بحيث $R\sigma\sigma'$. وبفرض الاستقراء،
      $\mSat{M(\Delta)}{!B}[\sigma']$ لكل $\sigma'$ بحيث
      $R\sigma\sigma'$. ومن ثم $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فلبعض
      $\sigma.n$ يكون $\sFmla{\False}{!B}[\sigma.n] \in \Delta$ لأن
      الفرع تام. وبفرض الاستقراء،
      $\mSat/{M(\Delta)}{!B}[\sigma.n]$. ولأن
      $R\sigma(\sigma.n)$، توجد~$\sigma'$ بحيث
      $\mSat/{M(\Delta)}{!B}[\sigma']$. ومن ثم
      $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
  \item \indcase{!A}{\Diamond !B}{\iftag{probDiamond}{تمرين.}{إذا كان
      $\sFmla{\True}{\indfrm}[\sigma] \in \Delta$، فلبعض
      $\sigma.n$ يكون $\sFmla{\True}{!B}[\sigma.n] \in \Delta$ لأن
      الفرع تام. وبفرض الاستقراء، $\mSat{M(\Delta)}{!B}[\sigma.n]$.
      ولأن $R\sigma(\sigma.n)$، توجد~$\sigma'$ بحيث
      $\mSat{M(\Delta)}{!B}[\sigma']$. ومن ثم
      $\mSat{M(\Delta)}{\indfrm}[\sigma]$.

      وإذا كان $\sFmla{\False}{\indfrm}[\sigma] \in \Delta$، فإن
      $\sFmla{\False}{!B}[\sigma.n] \in \Delta$ لكل $\sigma.n$
      مستعملة على الفرع، لأن الفرع تام؛ أي لكل $\sigma'
      \in P(\Delta)$ بحيث $R\sigma\sigma'$. وبفرض الاستقراء،
      $\mSat/{M(\Delta)}{!B}[\sigma']$ لكل $\sigma'$ بحيث
      $R\sigma\sigma'$. ومن ثم $\mSat/{M(\Delta)}{\indfrm}[\sigma]$.}}
\end{enumerate}
نأخذ الآن~$f$ دالة الهوية على~$P(\Delta)$، أي~$f(\sigma)=\sigma$.
فهذه~$f$ تأويل للبادئات في~$\mModel{M(\Delta)}$، والاستقراء المتقدم
يثبت أن~$\mModel{M(\Delta)}$ يرضي~$\Delta$ بالنسبة إلى~$f$.
ومن~$\Gamma \subseteq \Delta$ ينتج~$\mSat{M(\Delta)}{\Gamma}[f]$.
\end{proof}

\begin{cor}
\ollabel{cor:entailment-completeness}
إذا كانت $\Gamma$ منتهية وكان $\Gamma \Entails !A$، فإن
$\Gamma \Proves !A$.
\end{cor}

\begin{cor}
\ollabel{cor:weak-completeness}
إذا كانت $!A$ صادقة في جميع النماذج، فإن $\Proves !A$.
\end{cor}
