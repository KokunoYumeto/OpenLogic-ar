\olsection{دوال المزاوجة والرموز}

\begin{explain}
طريقة كانتور المتعرجة تظهر للعين قابلية $\Nat^n$ للتعداد.
ولنخص مصفوفة $\Nat^2$ بالنظر. بتتبع المسار وعدّ مواضعه يتبين
أن $\tuple{1,2}$ يقابله العدد~$7$. فهل لنا إلى حساب ذلك من
غير تتبع سبيل؟ المطلوب \emph{معكوس} التعداد المتعرج
$g\colon \Nat^2 \to \Nat$، بحيث
\[
g(\tuple{0,0}) = 0, \;
g(\tuple{0,1}) = 1, \;
g(\tuple{1,0}) = 2, \; \dots,
g(\tuple{1,2}) = 7, \; \dots
\]
فبه نعيّن موضع $\tuple{n, m}$ في التعداد تعيينًا مباشرًا.

وتعريف $g$ المباشر يبنى على ملاحظتين. إحداهما أن خانة الصف $n$
والعمود $m$ إذا كانت قيمتها~$v$، كانت قيمة خانة الصف $(n+1)$
والعمود $(m-1)$ هي $v + 1$. والأخرى أن الصف الأول الأعداد
المثلثية $0$، $1$، $3$، $6$، وهكذا. والعدد المثلثي ذو الرتبة $k$
هو مجموع الطبيعيّات $\leq k$، ويساوي
$k(k+1)/2$. فنجمع الملاحظتين في الدالة:
\[
  g(n,m) = \frac{(n+m+1)(n+m)}{2} + n
\]
\textbf{تنبيه تصحيحي على الأصل (C031-01).} الحد $k(k+1)/2$ يجمع الطبيعيّات التي
لا تتجاوز $k$، لا التي تقل عنه؛ فأصلحنا حد الجمع وأبقينا دالة المزاوجة.
ونختصر كثيرًا بـ$g(n, m)$ بدل $g(\tuple{n, m})$ تخفيفًا للرسم.
والمعنى: احسب أولًا العدد المثلثي ذا الرتبة $(n+m)$،
ثم زد عليه $n$. فيعطي ذلك قيم المصفوفة المطلوبة؛ ومنه أن
$\tuple{1, 2}$ يقابل $\frac{4 \times 3}{2} +
1 = 7$.

فـ$g$ \emph{معكوس} تعداد لمجموعة أزواج، واسم هذا الصنف
\emph{دوال المزاوجة}.
\end{explain}

\begin{defn}[دالة المزاوجة]
الدالة $f\colon A \times B \to \Nat$ \emph{دالة مزاوجة} حسابية
متى كانت $f$ متباينة. ونقول إن $f$ \emph{ترمّز} $A \times B$،
وإن $f(x,y)$ \emph{رمز} $\tuple{x,y}$.
\end{defn}

\begin{explain}
فبدالة المزاوجة نرمّز أزواج الطبيعيّات مثلًا؛ إذ يمثل كل
\emph{زوج} عدد \emph{واحد}. وبمعكوسها \emph{نفك الترميز}،
فنعرف أي زوج دل عليه العدد.
\end{explain}

\begin{prob}
ضع تعدادًا لجميع الأعداد النسبية غير السالبة.
\end{prob}

\begin{prob}
برهن على أن $\Rat$ هي !!{enumerable}، مستحضرًا أن كل نسبي
يكتب $z/m$ حيث $z \in \Int$ و$m \in \Nat^+$.
\end{prob}

\begin{prob}
ضع تعدادًا للمجموعة $\Bin^*$.
\end{prob}

\begin{prob}
تقدم في المنطق التمهيدي أن كل جدول صدق ممكن يعبّر عن دالة صدق؛
فدوال الصدق هي جميع الدوال $\Bin^k \to \Bin$ لبعض~$k$.
برهن على قابلية مجموعة دوال الصدق كلها للتعداد.
\end{prob}

\begin{prob}
خذ مجموعة غير منتهية !!{enumerable} كيفما اتفقت؛ وأثبت أن مجموعة
جزئياتها المنتهية كلها هي !!{enumerable}.
\end{prob}

\begin{prob}
يقال إن جزئية $A$ من $\Nat$ \emph{متممتها منتهية} إذا وفقط إذا وجدت
جزئية منتهية $F\subseteq\Nat$ بحيث $A=\Nat\setminus F$؛ أي إذا وفقط إذا
كانت $\Nat\setminus A$ منتهية. ولتكن $I$ المجموعة التي !!{element}sها
جزئيات $\Nat$ المنتهية وذات المتممات المنتهية، ولا غيرها. أثبت أن $I$
هي !!{enumerable}.
\end{prob}

\begin{editorial}
\textbf{تنبيه تصحيحي على الأصل (OLSIZ-002).} صُرّح بأن المتممة مأخوذة
داخل $\Nat$ لجزئية منتهية منها؛ فقد اضطرب نظم العبارة في الأصل،
وعينت العبارة المكافئة في النص الإنجليزي هذا المعنى.
\end{editorial}

\begin{prob}
برهن على أن اتحاد عائلة !!{enumerable} من مجموعات !!{enumerable}
هو مجموعة !!{enumerable}. أي إذا أعطيت $A_1$، $A_2$، \dots{}، وكانت كل
$A_i$ !!{enumerable}، فإن اتحادها $\bigcup_{i=1}^\infty
A_i$ هو مجموعة !!{enumerable} أيضًا. [تنبيه: المسألة صعبة!]
\end{prob}

\begin{prob}
خذ دالة مزاوجة $f \colon A \times B \to \Nat$ كيفما اتفقت.
رتّب أعضاء $\ran{f}$ تصاعديًا وأعد فهرستها، ثم فك رموزها بمعكوس $f$
على مداها، وبرهن أن ذلك يتيح بناء تعداد لـ$A \times B$ إذا كان حاصل
الضرب غير خال. وبيّن حكم الحالة الخالية على حدة؛ فإن كان الحاصل منتهيًا
غير خال كرر آخر عنصر عند اعتماد تعريف التعداد بدالة شاملة من $\PosInt$،
أو افهرس عناصره بمقطع ابتدائي عند اعتماد التعريف البديل.
\end{prob}

\textbf{تنبيه تصحيحي على الأصل (C031-02).} ليس معكوس دالة المزاوجة، من حيث هو،
تعدادًا؛ لأن مجاله $\ran{f}$ قد لا يكون $\Nat$ ولا $\PosInt$ ولا مقطعًا
ابتدائيًا. ولذلك صُرّح بإعادة فهرسة الرموز وبالحالتين المنتهية والخالية.

\begin{prob}
عيّن دالة ترمّز المجموعة $\Nat^3$.
\end{prob}
