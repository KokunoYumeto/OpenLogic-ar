\iftag{FOL}{%
\begin{defn}[نموذج الحدود]
\ollabel{defn:termmodel}
إذا كانت $\OLId{greek-variable}{Gamma}^*$ مجموعة !!a{complete} ومتسقة ومشبعة من ال!!{sentence}s
في لغة~$\Lang L$، سمينا \emph{نموذج الحدود}~$\Struct M(\OLId{greek-variable}{Gamma}^*)$ الخاص
بـ$\OLId{greek-variable}{Gamma}^*$ ال!!{structure} التي نحددها بهذه البنود:
\begin{enumerate}
\item نجعل ال!!{domain}~$\Domain{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}$ مجموعة الحدود المغلقة
  كلها في~$\Lang L$.
\item نأخذ !!a{constant} $\OLId{latin-ordinary}{c}$؛ فتفسيره $\OLId{latin-ordinary}{c}$ بعينه:
  $\Assign{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} = \OLId{latin-ordinary}{c}$.
\item نفسر ال!!{function}~$\OLId{latin-ordinary}{f}$ بالدالة التي ترسل الحدود المغلقة
  $\OLId{latin-ordinary}{t}_\OLMathNumeral{1}$، \dots، $\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}$ إلى الحد المغلق
  $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n})$؛ أي:
\[
\Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}) = \OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1},\dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n})
\]
\item إذا كان $\OLId{latin-looped}{R}$ !!{predicate} ذا $\OLId{latin-ordinary}{n}$ مواضع، فإن
  \[
  \tuple{\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \dots,
  \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}} \in \Assign{\OLId{latin-looped}{R}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} \text{ إذا وفقط إذا } \Atom{R}{t_1, \dots,
    t_n} \in \OLId{greek-variable}{Gamma}^*.
  \]
\end{enumerate}
\end{defn}
}{%
\begin{defn}
\ollabel{defn:termmodel}
متى كانت $\OLId{greek-variable}{Gamma}^*$ مجموعة متسقة كاملة من ال!!{formula}s، عرّفنا التقييم الآتي:
\[
\pAssign \OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)(\OLId{latin-ordinary}{p}) = \begin{cases}
  \True & \text{إذا كان $p \in \Gamma^*$}\\
  \False & \text{إذا كان $p \notin \Gamma^*$}
  \end{cases}
\]
\end{defn}
}

\iftag{FOL}{%
نثبت أولًا أن التفسير يفي بما وعدنا به، أي $\Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} = \OLId{latin-ordinary}{t}$.

\begin{lem}
\ollabel{lem:val-in-termmodel} ليكن $\Struct M(\OLId{greek-variable}{Gamma}^*)$ نموذج الحدود
في \olref{defn:termmodel}؛ عندئذ $\Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} = \OLId{latin-ordinary}{t}$ لكل حد
مغلق~$\OLId{latin-ordinary}{t}$.
\end{lem}
\begin{proof}
 نستقرئ تركيب $\OLId{latin-ordinary}{t}$. ففي الأساس يكون $\OLId{latin-ordinary}{t}$ !!a{constant}، وقيمته معلومة
 من تعريف نموذج الحدود. ولخطوة الاستقراء خذ حدودًا مغلقة
 $\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \ldots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}$ تحقق $\Value{\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{i}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} = \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{i}$، وليكن
 $\OLId{latin-ordinary}{f}$ !!{function} له $\OLId{latin-ordinary}{n}$ حجج. عندئذ نحسب:
\begin{align*}
\Value{\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1},\ldots,\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n})}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)} &= \Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}(\Value{\OLId{latin-ordinary}{t}_\OLMathNumeral{1}}
 {\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)},
\ldots, \Value{\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}) \\
&= \Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}) \\
&= \OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1},\dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}),
\end{align*}
ومن ثم، بالاستقراء، يصح ذلك لكل حد مغلق~$\OLId{latin-ordinary}{t}$.
\end{proof}
}

\begin{lem}[لمّة الصدق]
\ollabel{lem:truth} \iftag{FOL}{افترض أن $!A$ لا تحتوي~$\eq$. عندئذ}{}
$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!A}$ إذا وفقط إذا
كان $!A \in \OLId{greek-variable}{Gamma}^*$.
\end{lem}

\begin{proof}
نجري الاستقراء على $!A$، ونثبت وجهي التكافؤ في كل حالة معًا.
\begin{enumerate}
\tagitem{prvFalse}{\indcase{!A}{\lfalse}
  {$\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{\lfalse}}{\pSat/{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}{\lfalse}}$
    هو حكم تعريف الإرضاء؛ ويقابله $\lfalse \notin \OLId{greek-variable}{Gamma}^*$، إذ
    اتساق $\OLId{greek-variable}{Gamma}^*$ يمنع اشتمالها على الباطل.}}{}

\tagitem{prvTrue}{\indcase{!A}{\ltrue}
  {$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{\ltrue}$
    يقتضيه تعريف الإرضاء؛ وكذلك $\ltrue \in \OLId{greek-variable}{Gamma}^*$، لاجتماع
    الاتساق وكون $\OLId{greek-variable}{Gamma}^*$ !!{complete} مع $\OLId{greek-variable}{Gamma}^* \Proves \ltrue$.}}{}

\tagitem{FOL}{\indcase{!A}{R(t_1, \dots, t_n)}
  {$\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{\Atom{R}{t_1, \dots, t_n}}$ إذا وفقط إذا كان $\tuple{\OLId{latin-ordinary}{t}_\OLMathNumeral{1},
      \dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}} \in \Assign{\OLId{latin-looped}{R}}{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}$ (بحسب تعريف الإرضاء)
    إذا وفقط إذا كان $\OLId{latin-looped}{R}(\OLId{latin-ordinary}{t}_\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}) \in \OLId{greek-variable}{Gamma}^*$ (بحسب إنشاء
    $\Struct M(\OLId{greek-variable}{Gamma}^*)$).}}{\indcase{!A}{p}{$\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}{\OLId{latin-ordinary}{p}}$
    إذا وفقط إذا كان $\pAssign \OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)(\OLId{latin-ordinary}{p}) = \True$ (بحسب تعريف
    الإرضاء)، وذلك إذا وفقط إذا كان $\OLId{latin-ordinary}{p} \in \OLId{greek-variable}{Gamma}^*$ (بحسب إنشاء
    $\pAssign \OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)$).}}

\tagitem{prvNot}{%
  \indcase{!A}{\lnot !B}
    {$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{\indfrm}$ إذا وفقط إذا تحقق
    $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{!B}}{\pSat/{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}{!B}}$
    بتعريف الإرضاء. ثم ينقلنا فرض الاستقراء إلى أن
    $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{!B}}{\pSat/{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}{!B}}$ إذا
    وفقط إذا كان $!B \notin \OLId{greek-variable}{Gamma}^*$. والاتساق مع كون $\OLId{greek-variable}{Gamma}^*$
    !!{complete} يقتضيان أن $!B \notin \OLId{greek-variable}{Gamma}^*$ يكافئ
    $\lnot !B \in \OLId{greek-variable}{Gamma}^*$.}}{}

\tagitem{prvAnd}{%
\iftag{probAnd}{%
  \indcase!{!A}{!B \land !C}{}}{%
  \indcase{!A}{!B \land
    !C}{$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{\indfrm}$
    إذا وفقط إذا كان كل من
    $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!B}$ و
    $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!C}$ متحققًا
    بتعريف الإرضاء؛ ويحوّل فرض الاستقراء هذا إلى اجتماع
    $!B \in \OLId{greek-variable}{Gamma}^*$ و$!C \in \OLId{greek-variable}{Gamma}^*$. ثم تفيد
    \olref[ccs]{prop:ccs}\olref[ccs]{prop:ccs-and}، يتحقق ذلك إذا وفقط
    إذا كان $(!B \land !C) \in \OLId{greek-variable}{Gamma}^*$.}}}{}

\tagitem{prvOr}{%
\iftag{probOr}{%
  \indcase!{!A}{!B \lor !C}{}}{%
  \indcase{!A}{!B \lor !C}
    {$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{\indfrm}$
    إذا وفقط إذا تحقق
    $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!B}$ أو
    $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!C}$ بتعريف
    الإرضاء؛ وبفرض الاستقراء يكافئ ذلك $!B \in \OLId{greek-variable}{Gamma}^*$ أو
    $!C \in \OLId{greek-variable}{Gamma}^*$. وهذان البديلان يكافئان
    $(!B \lor !C) \in \OLId{greek-variable}{Gamma}^*$، بمقتضى
    \olref[ccs]{prop:ccs}\olref[ccs]{prop:ccs-or}.}}}{}

\tagitem{prvIf}{%
\iftag{probIf}{%
  \indcase!{!A}{!B \lif !C}{}}{%
  \indcase{!A}{!B \lif !C}
    {$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{\indfrm}$
    إذا وفقط إذا تحقق
    $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}}{\pSat/{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!B}$ أو
    $\iftag{FOL}{\Sat{\OLId{latin-looped}{M} (\OLId{greek-variable}{Gamma}^*)}}{\pSat{\OLId{latin-ordinary}{v}(\OLId{greek-variable}{Gamma}^*)}}{!C}$ بتعريف
    الإرضاء؛ وبتطبيق فرض الاستقراء نجد $!B \notin \OLId{greek-variable}{Gamma}^*$ أو
    $!C \in \OLId{greek-variable}{Gamma}^*$. وذلك مكافئ لـ
    $(!B \lif !C) \in \OLId{greek-variable}{Gamma}^*$، بمقتضى
    \olref[ccs]{prop:ccs}\olref[ccs]{prop:ccs-if}.}}}{}

\iftag{FOL}{%
\tagitem{prvAll}{%
  \iftag{probAll}{%
    \indcase!{!A}{\lforall[x][!B(x)]}{}}{%
    \indcase{!A}{\lforall[x][!B(x)]}{$\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{\indfrm}$ إذا
      وفقط إذا كان $\Sat{\OLId{latin-looped}{M}(\OLId{greek-variable}{Gamma}^*)}{!B(\OLId{latin-ordinary}{t})}$ لكل حد مغلق~$\OLId{latin-ordinary}{t}$
