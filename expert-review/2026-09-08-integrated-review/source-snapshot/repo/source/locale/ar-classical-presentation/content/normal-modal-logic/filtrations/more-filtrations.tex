\olsection{الترشيحات وخواص النفاذ}

قد تقدم أن الترشيح يحفظ الكلية والتسلسل والانعكاسية. وأما التناظر
والتعدي فلا يحفظهما كل ترشيح، وإن أمكن اختيار ترشيحات تحفظهما في الحالات
المناسبة. وسبيل ذلك أن نبدأ بتعريف الترشيح الأخشن، ثم نزيد قيودًا على
$\OLId{latin-looped}{R}^*$. ولتكن~$\OLId{greek-variable}{Gamma}$ مغلقة تحت أخذ ال!!{formula}s الجزئية.
نعتبر الشروط~$\OLId{latin-looped}{C}_\OLId{latin-ordinary}{i}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$ المبينة في~\olref{tab:Cn-filtrations}
للعالمين~$\OLId{latin-ordinary}{u}$ و$\OLId{latin-ordinary}{v}$ من نموذج~$\mModel{M}=\tuple{\OLId{latin-looped}{W},\OLId{latin-looped}{R},\OLId{latin-looped}{V}}$،
ونعيّن~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ باجتماع بعض هذه الشروط. فإن جعلنا
$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ مكافئة للشرط~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$ وحده، كان الحاصل الترشيح الأخشن
بعينه. وإن جعلنا~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ مكافئة لاجتماع~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$
و$\OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$، أمكن الحصول، في الحالات التي ستبيّن، على ترشيح آخر.
وهو «أدق» بمعنى أن الأزواج المحققة لـ~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$ و$\OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$
لا تزيد على الأزواج المحققة لـ~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$ وحده.

\begin{thm}\ollabel{thm:more-filtrations}
  ليكن~$\mModel{M}=\tuple{\OLId{latin-looped}{W},\OLId{latin-looped}{R},\OLId{latin-looped}{V}}$ نموذجًا، ولتكن~$\OLId{greek-variable}{Gamma}$ مغلقة تحت
  أخذ ال!!{formula}s الجزئية. ولتُعرّف~$\OLId{latin-looped}{W}^*$ و$\OLId{latin-looped}{V}^*$ كما في
  \olref[fil]{defn:filtration}. عندئذ:
  \begin{enumerate}
  \item افترض أن~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ إذا وفقط إذا كان
    $\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$. عندئذ تكون~$\OLId{latin-looped}{R}^*$ متناظرة، ويكون
    $\mModel{M^*}=\tuple{\OLId{latin-looped}{W}^*,\OLId{latin-looped}{R}^*,\OLId{latin-looped}{V}^*}$ ترشيحًا إذا كان~$\mModel{M}$
    متناظرًا.
  \item افترض أن~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ إذا وفقط إذا كان
    $\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{3}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$. عندئذ تكون~$\OLId{latin-looped}{R}^*$ متعدية، ويكون
    $\mModel{M^*}=\tuple{\OLId{latin-looped}{W}^*,\OLId{latin-looped}{R}^*,\OLId{latin-looped}{V}^*}$ ترشيحًا إذا كان~$\mModel{M}$
    متعديًا.
  \item افترض أن~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ إذا وفقط إذا كان
    $\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{3}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{4}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$. عندئذ تكون
    $\OLId{latin-looped}{R}^*$ متناظرة ومتعدية، ويكون~$\mModel{M^*}=\tuple{\OLId{latin-looped}{W}^*,\OLId{latin-looped}{R}^*,\OLId{latin-looped}{V}^*}$
    ترشيحًا إذا كان~$\mModel{M}$ متناظرًا ومتعديًا.
  \item افترض أن~$\OLId{latin-looped}{R}^*$ معرّفة بحيث يكون~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ إذا وفقط إذا كان
    $\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{3}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\land \OLId{latin-looped}{C}_\OLMathNumeral{4}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$. عندئذ تكون~$\OLId{latin-looped}{R}^*$ متعدية
    وإقليدية، ويكون~$\mModel{M^*}=\tuple{\OLId{latin-looped}{W}^*,\OLId{latin-looped}{R}^*,\OLId{latin-looped}{V}^*}$ ترشيحًا إذا
    كان~$\mModel{M}$ متعديًا وإقليديًا.
  \end{enumerate}
\end{thm}

\begin{proof}
  \begin{enumerate}
    \item أما تناظر~$\OLId{latin-looped}{R}^*$ فينتج من التكافؤين
      $\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\Leftrightarrow \OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{v},\OLId{latin-ordinary}{u})$ و
      $\OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})\Leftrightarrow \OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{v},\OLId{latin-ordinary}{u})$. وأما كونها ترشيحًا، فلنفرض
      $\mModel{M}$ متناظرًا ونثبت أن~$\mModel{M^*}$ ترشيح عبر~$\OLId{greek-variable}{Gamma}$.
      فالشرط~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$ كافٍ لتحقيق
      \iftag{prvBox}{%
        \iftag{prvDiamond}{%
          \olref[fil]{defn:filtration-R2} و
          \olref[fil]{defn:filtration-R3} من
          \olref[fil]{defn:filtration}}{%
          \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R2}}}{%
        \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R3}}.
      فلا يبقى إلا التحقق من
      \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R1}، أي
      إن~$Ruv$ يستلزم~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$.

      ولنفرض~$Ruv$؛ فإثبات~$\OLId{latin-looped}{R}^*[\OLId{latin-ordinary}{u}][\OLId{latin-ordinary}{v}]$ مردّه إلى إثبات~$\OLId{latin-looped}{C}_\OLMathNumeral{1}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$
      و$\OLId{latin-looped}{C}_\OLMathNumeral{2}(\OLId{latin-ordinary}{u},\OLId{latin-ordinary}{v})$. أما~$\OLId{latin-looped}{C}_\OLMathNumeral{1}$: \iftag{prvBox}{فإذا كان~$\Box!A
        \in\OLId{greek-variable}{Gamma}$ و$\mSat{\OLId{latin-looped}{M}}{\Box!A}[\OLId{latin-ordinary}{u}]$، كان أيضًا
        $\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{v}]$ (لأن~$Ruv$).\iftag{prvDiamond}{ وبالمثل،
        }{}}{}\iftag{prvDiamond}{إذا كان~$\Diamond!A \in \OLId{greek-variable}{Gamma}$
        و$\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{v}]$، كان~$\mSat{\OLId{latin-looped}{M}}{\Diamond!A}[\OLId{latin-ordinary}{u}]$ لأن
        $Ruv$.}{} وأما~$\OLId{latin-looped}{C}_\OLMathNumeral{2}$: \iftag{prvBox}{فإذا كان~$\Box!A \in
        \OLId{greek-variable}{Gamma}$ و$\mSat{\OLId{latin-looped}{M}}{\Box!A}[\OLId{latin-ordinary}{v}]$، استلزم~$Ruv$ أن~$Rvu$
        بالتناظر، ومن ثم~$\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{u}]$.\iftag{prvDiamond}{
        وبالمثل، }{}}{}\iftag{prvDiamond}{إذا كان~$\Diamond!A
        \in\OLId{greek-variable}{Gamma}$ و$\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{u}]$، كان
        $\mSat{\OLId{latin-looped}{M}}{\Diamond!A}[\OLId{latin-ordinary}{v}]$ (لأن~$Rvu$ بالتناظر).}{}
    \item تمرين.
    \item تمرين.
    \item تمرين.
  \end{enumerate}
\end{proof}
