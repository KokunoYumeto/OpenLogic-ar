\olsection{التمثيل الأحادي للأعداد}

\begin{explain}
إنما تعمل آلات تورنغ في سلاسل الرموز على أشرطتها. ويتنوع ما تمثله هذه السلاسل من دخل وخرج بتنوع الأبجدية. ويهمنا منها خاصة ما يحسب الدوال \emph{الحسابية}، أي الدوال على الأعداد الطبيعية. ومن أيسر تمثيلات الأعداد الصحيحة الموجبة سلسلة تكرر رمزًا واحدًا $\TMstroke$. فنضع، إذا كان $\OLId{latin-ordinary}{n} \in \Nat$، أن $\TMstroke^\OLId{latin-ordinary}{n}$ خالية عند $\OLId{latin-ordinary}{n}=\OLMathNumeral{0}$، وإلا فهي سلسلة من $\OLId{latin-ordinary}{n}$ علامات بالضبط، كلها $\TMstroke$.
\end{explain}

\begin{defn}[الحساب]
تكون آلة تورنغ~$\OLId{latin-looped}{M}$ \emph{حاسبة} للدالة $\OLId{latin-ordinary}{f}\colon\Nat^\OLId{latin-ordinary}{k}\to\Nat$ إذا وفقط إذا كانت~$\OLId{latin-looped}{M}$، لكل $\OLId{latin-ordinary}{n}_\OLMathNumeral{1},\dots,\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k}\in\Nat$، تقف على الدخل
\[
\TMstroke^{\OLId{latin-ordinary}{n}_\OLMathNumeral{1}} \TMblank \TMstroke^{\OLId{latin-ordinary}{n}_\OLMathNumeral{2}} \TMblank \dots \TMblank \TMstroke^{\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k}}
\]
ويكون خرجها $\TMstroke^{\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}_\OLMathNumeral{1},\dots,\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k})}$.
\end{defn}

\begin{prob}
ضع تعريفًا لكون آلة تورنغ~$\OLId{latin-looped}{M}$ تحسب الدالة $\OLId{latin-ordinary}{f}\colon\Nat^\OLId{latin-ordinary}{k}\to\Nat^\OLId{latin-ordinary}{m}$.
\end{prob}

\begin{prob}
سجل تهيئات الآلة في \olref[tur][mac][una]{ex:adder} على الدخل~$\tuple{\OLMathNumeral{3},\OLMathNumeral{2}}$. وبين ما يحدث لها عند حساب $\OLMathNumeral{0}+\OLMathNumeral{0}$.
\end{prob}

\begin{explain}
رأينا في \olref[rep]{ex:doubler} آلة المضاعفة: تأخذ سلسلة من~$\TMstroke$ وتقف وقد كتبت ضعف علاماتها. غير أن في خرجها رموز $\TMblank$ على يسار كتلة $\TMstroke$ المضاعفة؛ ولذلك لا تكون حاسبة للدالة $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{x})=\OLMathNumeral{2}\OLId{latin-ordinary}{x}$ بحسب التعريف، وإن أوحى المثال بذلك. ونبين لإصلاح هذا وجهين.
\end{explain}

وصفنا في \olref[tur][mac][una]{ex:mover} تركيب آلة المضاعفة الأصلية في \olref[tur][mac][rep]{ex:doubler} مع آلة النقل في \olref[tur][mac][una]{fig:mover}. فماذا يحدث عند تشغيل المركبة على~$\OLId{latin-ordinary}{x}=\OLMathNumeral{0}$، أي على شريط خال؟ وكيف تعدلها لتقف حينئذ بخرج~$\OLMathNumeral{2}\OLId{latin-ordinary}{x}=\OLMathNumeral{0}$؟ ينبغي أن تكفيك حالة واحدة وانتقال واحد زائدان.
% تصحيح مصدر (C258-01): يحيل التركيب الموصوف إلى آلة المضاعفة الأصلية، لا الرسم المصحح اللاحق.
\end{prob}

\begin{prob}
\emph{الطرح:} أنشئ آلة تورنغ، دخلها سلسلتان غير خاليتين من العلامات طولهما $\OLId{latin-ordinary}{n}$ و~$\OLId{latin-ordinary}{m}$، مع $\OLId{latin-ordinary}{n}>\OLId{latin-ordinary}{m}$، فتحسب $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n},\OLId{latin-ordinary}{m})=\OLId{latin-ordinary}{n}-\OLId{latin-ordinary}{m}$.
\end{prob}

\begin{prob}
\emph{المساواة:} أنشئ آلة تورنغ لحساب الدالة
\[
\fn{equality}(\OLId{latin-ordinary}{n},\OLId{latin-ordinary}{m}) =
\begin{cases}
  \text{1} & \text{إذا كان~$n = m$} \\
  \text{0} & \text{إذا كان~$n \neq m$}
\end{cases}
\]
على أن~$\OLId{latin-ordinary}{n}$ و~$\OLId{latin-ordinary}{m} \in \PosInt$.
\end{prob}

\begin{prob}
أنشئ آلة تورنغ لحساب $\min(\OLId{latin-ordinary}{x},\OLId{latin-ordinary}{y})$، حيث $\OLId{latin-ordinary}{x}$ و$\OLId{latin-ordinary}{y}$ صحيحان موجبان، يمثلهما على الشريط سلسلتا $\TMstroke$ بينهما $\TMblank$ واحدة. ولك أن تزيد في الأبجدية رموزًا تعينك على البناء.

والدالة $\min$ تأخذ أصغر وسيط؛ فيكون $\min(\OLMathNumeral{3},\OLMathNumeral{5})=\OLMathNumeral{3}$ و$\min(\OLMathNumeral{20},\OLMathNumeral{16})=\OLMathNumeral{16}$ و$\min(\OLMathNumeral{4},\OLMathNumeral{4})=\OLMathNumeral{4}$، وعلى هذا القياس.
\end{prob}

\begin{defn}
تكون آلة تورنغ~$\OLId{latin-looped}{M}$ حاسبة للدالة الجزئية $\OLId{latin-ordinary}{f}\colon\Nat^\OLId{latin-ordinary}{k}\pto\Nat$ إذا وفقط إذا تحقق الأمران:
\begin{enumerate}
\item تقف $\OLId{latin-looped}{M}$ على الدخل $\TMstroke^{\OLId{latin-ordinary}{n}_\OLMathNumeral{1}}\concat\TMblank\concat
    \dots \concat\TMblank\concat\TMstroke^{\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k}}$ بخرج $\TMstroke^{\OLId{latin-ordinary}{m}}$ متى كان $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}_\OLMathNumeral{1},\dots,\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k})=\OLId{latin-ordinary}{m}$.
\item إما أن تستمر $\OLId{latin-looped}{M}$ بلا توقف، وإما أن تقف بخرج ليس كتلة واحدة من $\TMstroke$، متى كانت $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}_\OLMathNumeral{1},\dots,\OLId{latin-ordinary}{n}_\OLId{latin-ordinary}{k})$ غير معرفة.
\end{enumerate}
\end{defn}
