\olsection{في التطبيع}

نثبت في هذا القسم أنه يمكن، وفق ترتيب اختزال مناسب، اختزال أي استنتاج إلى
استنتاج عادي، وتسمى هذه \emph{خاصية التطبيع}. وسنستعمل التناظر بين القضايا
والأنماط: فنبيّن أن كل حد برهان يمكن اختزاله إلى صيغة عادية؛ ومن ذلك ينتج
تطبيع~!!{derivation}s الاستنباط الطبيعي.

نعرّف أولًا بعض الدوال التي تقيس تعقيد الحدود. ويُعرَّف
\emph{طول}~$\len{!A}$ لكل~!!a{formula} على هذا الوجه:
\begin{align*}
  \len{\OLId{latin-ordinary}{p}} &= \OLMathNumeral{0}\\
  \len{\lfalse} &= \OLMathNumeral{0}\\
  \len{!A \land !B} &= \len{!A} + \len{!B} + \OLMathNumeral{1}\\
  \len{!A \lor !B} &= \len{!A} + \len{!B} + \OLMathNumeral{1} \\
  \len{!A \lif !B} &= \len{!A} + \len{!B} + \OLMathNumeral{1}.
\end{align*}
ويقاس تعقيد الردكس~$\OLId{latin-looped}{M}$ بـ\emph{رتبة القطع}~$\cutrank{\OLId{latin-looped}{M}}$:
\begin{align*}
  \cutrank{(\lambd[\typeof{\OLId{latin-ordinary}{x}}{!A}][\typeof{\OLId{latin-looped}{N}}{!B}])\OLId{latin-looped}{Q}} &=
  \len{!A} + \len{!B} + \OLMathNumeral{1} \\
  \cutrank{\ande{\OLId{latin-ordinary}{i}}{\andi{\typeof{\OLId{latin-looped}{M}}{!A}}{\typeof{\OLId{latin-looped}{N}}{!B}}}} &=
  \len{!A} + \len{!B} + \OLMathNumeral{1} \\
  \cutrank{\ore{\ori{\OLId{latin-ordinary}{i}}{!A_{\OLId{latin-ordinary}{i}}}{\typeof{\OLId{latin-looped}{M}}{!A_\OLId{latin-ordinary}{i}}}}
    {\typeof{\OLId{latin-ordinary}{x}_\OLMathNumeral{1}}{!A_\OLMathNumeral{1}}}{\typeof{\OLId{latin-looped}{N}_\OLMathNumeral{1}}{!C}}
    {\typeof{\OLId{latin-ordinary}{x}_\OLMathNumeral{2}}{!A_\OLMathNumeral{2}}}{\typeof{\OLId{latin-looped}{N}_\OLMathNumeral{2}}{!C}}} &=
  \len{!A_\OLMathNumeral{1}} + \len{!A_\OLMathNumeral{2}} + \OLMathNumeral{1}.
\end{align*}
ويقاس تعقيد حد البرهان بأعلى رتبة لردكس فيه، وبالصفر إذا كان عاديًا:
\[
  \maxrank{\OLId{latin-looped}{M}} = \max\bigl(\{\cutrank{\OLId{latin-looped}{N}} \mid \OLId{latin-looped}{N} \text{ حد جزئي من } \OLId{latin-looped}{M}
    \text{ وهو ردكس}\}\cup\{\OLMathNumeral{0}\}\bigr).
\]

\begin{lem}\ollabel{lem:subst}
  إذا كان~$\Subst{\OLId{latin-looped}{M}}{\typeof{\OLId{latin-looped}{N}}{!A}}{\typeof{\OLId{latin-ordinary}{x}}{!A}}$ ردكسًا وكان
  $\OLId{latin-looped}{M}\not\ident \OLId{latin-ordinary}{x}$، تحققت إحدى الحالات الآتية:
  \begin{enumerate}
  \item يكون~$\OLId{latin-looped}{M}$ نفسه ردكسًا؛ أو
  \item يكون~$\OLId{latin-looped}{M}$ من الصورة~$\ande{\OLId{latin-ordinary}{i}}{\OLId{latin-ordinary}{x}}$، ويكون~$\OLId{latin-looped}{N}$ من الصورة
    $\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}$؛
  \item يكون~$\OLId{latin-looped}{M}$ من الصورة~$\ore{\OLId{latin-ordinary}{x}}{\OLId{latin-ordinary}{x}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-ordinary}{x}_\OLMathNumeral{2}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}$، ويكون~$\OLId{latin-looped}{N}$
    من الصورة~$\ori{\OLId{latin-ordinary}{i}}{}{\OLId{latin-looped}{Q}}$؛
  \item يكون~$\OLId{latin-looped}{M}$ من الصورة~$xQ$، ويكون~$\OLId{latin-looped}{N}$ من الصورة~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{P}]$.
  \end{enumerate}
  في الحالة الأولى
  $\cutrank{\Subst{\OLId{latin-looped}{M}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}}=\cutrank{\OLId{latin-looped}{M}}$؛ وفي الحالات الأخرى
  $\cutrank{\Subst{\OLId{latin-looped}{M}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}}=\len{!A}$.
\end{lem}
\begin{proof}
  البرهان بالاستقراء في~$\OLId{latin-looped}{M}$.
  \begin{enumerate}
  \item إذا كان~$\OLId{latin-looped}{M}$ متغيرًا مفردًا~$\OLId{latin-ordinary}{y}$ وكان~$\OLId{latin-ordinary}{y}\not\ident \OLId{latin-ordinary}{x}$، فإن
    $\Subst{\OLId{latin-ordinary}{y}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}$ هو~$\OLId{latin-ordinary}{y}$، فلا يكون ردكسًا.
  \item إذا كان~$\OLId{latin-looped}{M}$ من إحدى الصور~$\andi{\OLId{latin-looped}{N}_\OLMathNumeral{1}}{\OLId{latin-looped}{N}_\OLMathNumeral{2}}$ أو
    $\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}]$ أو~$\ori{\OLId{latin-ordinary}{i}}{!A}{\OLId{latin-looped}{N}}$، فإن
    $\Subst{\OLId{latin-looped}{M}}{\typeof{\OLId{latin-looped}{N}}{!A}}{\typeof{\OLId{latin-ordinary}{x}}{!A}}$ يكون من الصورة نفسها،
    ومن ثم لا يكون ردكسًا.
  \item إذا كان~$\OLId{latin-looped}{M}$ من الصورة~$\ande{\OLId{latin-ordinary}{i}}{\OLId{latin-looped}{P}}$، فهناك حالتان.
    \begin{enumerate}
      \item إذا كان~$\OLId{latin-looped}{P}$ من الصورة~$\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}$، كان
        $\OLId{latin-looped}{M}\ident\ande{\OLId{latin-ordinary}{i}}{\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}}$ ردكسًا، ومن الواضح أن
        \[
        \Subst{\OLId{latin-looped}{M}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}\ident
        \ande{\OLId{latin-ordinary}{i}}{\andi{\Subst{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}}{\Subst{\OLId{latin-looped}{P}_\OLMathNumeral{2}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}}}
        \]
        ردكس كذلك. ورتبتا القطع متساويتان.
      \item إذا كان~$\OLId{latin-looped}{P}$ متغيرًا مفردًا، فلا بد أن يكون~$\OLId{latin-ordinary}{x}$ لكي يجعل
        التعويض ردكسًا، ولا بد أن يكون~$\OLId{latin-looped}{N}$ من الصورة~$\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}$.
        انظر الآن في
        \[
        \Subst{\OLId{latin-looped}{M}}{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}}\ident
        \Subst{\ande{\OLId{latin-ordinary}{i}}{\OLId{latin-ordinary}{x}}}{\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}}{\OLId{latin-ordinary}{x}},
        \]
        وهو~$\ande{\OLId{latin-ordinary}{i}}{\andi{\OLId{latin-looped}{P}_\OLMathNumeral{1}}{\OLId{latin-looped}{P}_\OLMathNumeral{2}}}$. ورتبة قطعه تساوي~$\len{!A}$.
    \end{enumerate}
  \end{enumerate}
  وحالتا~$\ore{\OLId{latin-looped}{N}}{\OLId{latin-ordinary}{x}_\OLMathNumeral{1}}{\OLId{latin-looped}{N}_\OLMathNumeral{1}}{\OLId{latin-ordinary}{x}_\OLMathNumeral{2}}{\OLId{latin-looped}{N}_\OLMathNumeral{2}}$ و$PQ$ مماثلتان.
\end{proof}

\begin{lem}
  إذا اختُزل الردكس الجذري~$\OLId{latin-looped}{M}$ إلى~$\OLId{latin-looped}{M}'$، وكان
  $\cutrank{\OLId{latin-looped}{M}}>\cutrank{\OLId{latin-looped}{N}}$ لكل حد جزئي أصلي~$\OLId{latin-looped}{N}$ من~$\OLId{latin-looped}{M}$ يكون ردكسًا، فإن
  $\cutrank{\OLId{latin-looped}{M}}>\maxrank{\OLId{latin-looped}{M}'}$.
\end{lem}

\begin{proof}
  نبرهن بتمييز الحالات.
  \begin{enumerate}
  \item إذا كان~$\OLId{latin-looped}{M}$ من الصورة~$\ande{\OLId{latin-ordinary}{i}}{\andi{\OLId{latin-looped}{M}_\OLMathNumeral{1}}{\OLId{latin-looped}{M}_\OLMathNumeral{2}}}$، فإن
    $\OLId{latin-looped}{M}'=\OLId{latin-looped}{M}_\OLId{latin-ordinary}{i}$. وكل حد جزئي من~$\OLId{latin-looped}{M}_\OLId{latin-ordinary}{i}$ حد جزئي أصلي من~$\OLId{latin-looped}{M}$، فتنتج الدعوى.
  \item إذا كان~$\OLId{latin-looped}{M}$ من الصورة
    $(\lambd[\typeof{\OLId{latin-ordinary}{x}}{!A}][\OLId{latin-looped}{N}])\typeof{\OLId{latin-looped}{Q}}{!A}$، فإن
    $\OLId{latin-looped}{M}'=\Subst{\OLId{latin-looped}{N}}{\typeof{\OLId{latin-looped}{Q}}{!A}}{\typeof{\OLId{latin-ordinary}{x}}{!A}}$. وقد يأتي أي ردكس في
    $\OLId{latin-looped}{M}'$ من ردكس في~$\OLId{latin-looped}{N}$ أو في~$\OLId{latin-looped}{Q}$ مع بقاء رتبة القطع نفسها، أو قد ينشأ من
    التعويض وتكون رتبته~$\len{!A}$. والرتب الموروثة أصغر من
    $\cutrank{\OLId{latin-looped}{M}}$ بالفرض، كما أن~$\len{!A}<\cutrank{\OLId{latin-looped}{M}}$ بالتعريف.
  \item إذا كان~$\OLId{latin-looped}{M}$ من الصورة
    \[
    \ore{\ori{\OLId{latin-ordinary}{i}}{}{\typeof{\OLId{latin-looped}{N}}{!A_\OLId{latin-ordinary}{i}}}}
        {\typeof{\OLId{latin-ordinary}{x}_\OLMathNumeral{1}}{!A_\OLMathNumeral{1}}}{\typeof{\OLId{latin-looped}{N}_\OLMathNumeral{1}}{!C}}
        {\typeof{\OLId{latin-ordinary}{x}_\OLMathNumeral{2}}{!A_\OLMathNumeral{2}}}{\typeof{\OLId{latin-looped}{N}_\OLMathNumeral{2}}{!C}},
    \]
    فإن~$\OLId{latin-looped}{M}'\ident\Subst{\OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}}{\OLId{latin-looped}{N}}{\typeof{\OLId{latin-ordinary}{x}_\OLId{latin-ordinary}{i}}{!A_\OLId{latin-ordinary}{i}}}$. وقد يأتي أي ردكس
    في~$\OLId{latin-looped}{M}'$ من~$\OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}$ أو من الحد المحقون~$\OLId{latin-looped}{N}$ مع بقاء رتبته، أو قد ينشأ من
    التعويض وتكون رتبته~$\len{!A_\OLId{latin-ordinary}{i}}$. والرتب الموروثة أصغر من
    $\cutrank{\OLId{latin-looped}{M}}$ بالفرض، كما أن~$\len{!A_\OLId{latin-ordinary}{i}}<\cutrank{\OLId{latin-looped}{M}}$ بالتعريف.
  \end{enumerate}
\end{proof}

\begin{thm}
  لكل حد برهان صحيح النوع متتالية اختزال منتهية إلى صيغة عادية؛ ولكل
  !!{derivation}s صحيحة النوع متتالية اختزال منتهية إلى~!!{derivation}s
  عادية.
\end{thm}

\begin{proof}
  تنتج الدعوى الثانية من الأولى. ولإثبات الأولى، ليكن~$\OLId{latin-looped}{M}$ حد برهان صحيح
  النوع، ولتكن~$\delta$~!!a{derivation} لتنميطه في~$\Log{tN3i}$ بالنسبة
  إلى سياق~$\OLId{greek-variable}{Gamma}$. ننقل أولًا~$\delta$، مع إبقاء الحد~$\OLId{latin-looped}{M}$، إلى
  !!a{derivation}~$\widehat\delta$ في~$\Log{tN2i}$. وسبيل ذلك الاستقراء في
  ارتفاع~$\delta$: نستبدل البديهية~$\OLId{greek-variable}{Gamma}\Sequent \OLId{latin-ordinary}{x}:!A$ بالبديهية
  $\OLId{latin-ordinary}{x}:!A\Sequent \OLId{latin-ordinary}{x}:!A$، ثم نعمل في كل خطوة القاعدة المناظرة، فنجمع سياقات
  المقدمات ونحذف منها الفرض المفرغ. فيكون سياق كل نتيجة على التحقيق
  تصريحات المتغيرات الحرة في حدها؛ وتشمل حالة الحذف التي لا يقع فيها
  المتغير المفرغ حرًا اتفاقية جواز إعمال القاعدة من غير تفريغ فرض.

  فإذا حذفنا وسوم الحدود من~$\widehat\delta$ حصلنا على~!!a{proof}
  $\widehat\delta^\circ$ في~$\Log{N2i}$. وبمبرهنة التطبيع المتقدمة
  \olref[pt][nor][thm]{thm:normalization-N2i} توجد متتالية منتهية من
  تحويلات التبديل والاختزال تنقله إلى~!!a{proof} عادي~$\delta'$. وهذه
  التحويلات معرفة مباشرةً للنسق~$\Log{tN2i}$؛ وبحسب المطابقة الصريحة بين
  تلك التحويلات واختزال حدود البرهان في قسم حفظ النوع
  \olref[int][pty][tpr]{sec}، ترفع المتتالية نفسها إلى البراهين الموسومة،
  ويغير كل تحويل حدها باختزال أحادي~$\redone$. فيوجد لذلك حد~$\OLId{latin-looped}{M}'$ بحيث
  $\OLId{latin-looped}{M}\red \OLId{latin-looped}{M}'$ وتوسم به~$\delta'$. ولو اشتمل~$\OLId{latin-looped}{M}'$ على ردكس لقابله تحويل في
  $\delta'$، خلاف عاديته؛ فـ$\OLId{latin-looped}{M}'$ في صيغة عادية. وكل~!!a{derivation} صحيحة
  تقبل وسمًا بحد برهان صحيح النوع، فتنتج الدعوى الثانية أيضًا.
\end{proof}

\begin{editorial}
  لا يعمل هذا البرهان استقراء الرتبة القصوى الوارد في النص المصدر؛ فإن ذلك
  الاستقراء لا يضمن نقصان عدد الردكسات ذات الرتبة القصوى بعد التعويض. وقد
  استُعيض عنه هنا بمبرهنة التطبيع المثبتة للنسق~$\Log{N2i}$ بعد النقل
  الصريح من~$\Log{tN3i}$ إلى~$\Log{tN2i}$؛ ثم رفعت تحويلاتها إلى البراهين
  الموسومة بالمطابقة المثبتة في قسم حفظ النوع.
\end{editorial}

يضمن التطبيع أن لكل حد صحيح النوع متتالية اختزال منتهية واحدة على الأقل إلى
صيغة عادية؛ ومن ثم تتوقف بعض استراتيجيات الاختزال المطبِّعة. ويضمن حفظ النوع
أن القيمة الناتجة من النوع الصحيح. فمثلًا، إذا كان~$\OLId{latin-looped}{N}$ من النوع
$!A\lif !B$ وكان~$\OLId{latin-looped}{M}$ من النوع~$!A$، فإن اختزالًا مطبِّعًا لـ~$NM$ ينتج
صيغة عادية~$\OLId{latin-looped}{N}'$ من النوع~$!B$.

وهناك نتيجة أقوى لا تثبتها الحجة السابقة، هي \emph{التطبيع القوي}: كل
متتالية من خطوات الاختزال الأحادية تبدأ بحد صحيح النوع منتهية. وينتج من ذلك،
على وجه الخصوص، أن كل استراتيجية اختزال تتوقف. ويتطلب البرهان حجة إضافية،
مثل حجة مرشحات القابلية للاختزال.

ولمعرفة متى تؤدي طرائق تنفيذ الحساب المختلفة إلى القيمة \emph{نفسها}، ننظر
في خاصية \emph{التلاقي}، أو خاصية تشيرش--روسر. إذا كان~$\OLId{latin-looped}{N}\red \OLId{latin-looped}{N}_\OLMathNumeral{1}$ و
$\OLId{latin-looped}{N}\red \OLId{latin-looped}{N}_\OLMathNumeral{2}$، فالتلاقي يعني وجود حد~$\OLId{latin-looped}{N}'$ بحيث~$\OLId{latin-looped}{N}_\OLMathNumeral{1}\red \OLId{latin-looped}{N}'$ و
$\OLId{latin-looped}{N}_\OLMathNumeral{2}\red \OLId{latin-looped}{N}'$. وتذكّر أن~$\OLId{latin-looped}{N}\red \OLId{latin-looped}{N}'$ يعني أن~$\OLId{latin-looped}{N}'$ ينتج من صفر أو أكثر من
خطوات الاختزال~$\redone$. فإذا كان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ عاديًا، فالحد الوحيد~$\OLId{latin-looped}{N}'$ الذي
يحقق~$\OLId{latin-looped}{N}_\OLMathNumeral{1}\red \OLId{latin-looped}{N}'$ هو~$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ نفسه، باستعمال صفر من الخطوات. ومن ثم، إذا
كان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ و$\OLId{latin-looped}{N}_\OLMathNumeral{2}$ عاديين، كان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}=\OLId{latin-looped}{N}'=\OLId{latin-looped}{N}_\OLMathNumeral{2}$. وإذا افترضنا أن علاقة الخطوة
الواحدة~$\redone$ مطبِّعة بقوة، انتهت كل طريقة للاختزال إلى صيغة عادية؛
وإذا افترضنا التلاقي أيضًا، كانت أي صيغتين عاديتين ناتجتين متساويتين. وعندئذ
ينتهي الحساب دائمًا وتكون قيمته مستقلة عن طريقة إجرائه.

غالبًا ما يصعب إثبات تلاقي علاقة، غير أنه يسهل إثبات الشرط الأضعف الآتي:

\begin{prop}[خاصية تشيرش--روسر الضعيفة]
إذا كان~$\OLId{latin-looped}{N}\redone \OLId{latin-looped}{N}_\OLMathNumeral{1}$ و$\OLId{latin-looped}{N}\redone \OLId{latin-looped}{N}_\OLMathNumeral{2}$، وُجد حد~$\OLId{latin-looped}{N}'$ بحيث
$\OLId{latin-looped}{N}_\OLMathNumeral{1}\red \OLId{latin-looped}{N}'$ و$\OLId{latin-looped}{N}_\OLMathNumeral{2}\red \OLId{latin-looped}{N}'$.
\end{prop}

\begin{proof}
  إذا تطابق موضعا الردكس المقلَّصان، كان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}=\OLId{latin-looped}{N}_\OLMathNumeral{2}$. وإذا كانا منفصلين،
  تبادل التقليصان موضعيهما. وإذا كان أحدهما داخل الآخر، نفحص قاعدة الاختزال
  الخارجية: فيُقلَّص المتبقي المحتفظ به بعد الخطوة الخارجية، ولا يحتاج
  المتبقي الممحُو إلى خطوة، وتُقلَّص جميع المتبقيات التي يكررها تعويض بيتا أو
  تعويض الحالة. ثم يضمن توافق الاختزال مع التعويض المتفادي لالتقاط المتغيرات
  وجود مختزَل مشترك.
\end{proof}

\begin{lem}[مبرهنة نيومان]
إذا كانت~$\redone$ مطبِّعة بقوة وتحقق خاصية تشيرش--روسر الضعيفة، فإنها
متلاقية.
\end{lem}

\begin{proof}
  لنفرض أن لـ~$\OLId{latin-looped}{N}$ صيغتين عاديتين مختلفتين~$\OLId{latin-looped}{U},\OLId{latin-looped}{V}$، واكتب الخطوتين الأوليين
  كما يأتي:
  \begin{align*}
    \OLId{latin-looped}{N} &\redone \OLId{latin-looped}{N}_\OLMathNumeral{1}\red \OLId{latin-looped}{U},\\
    \OLId{latin-looped}{N} &\redone \OLId{latin-looped}{N}_\OLMathNumeral{2}\red \OLId{latin-looped}{V}.
  \end{align*}
  يعطي التلاقي المحلي حدًا~$\OLId{latin-looped}{P}$ بحيث~$\OLId{latin-looped}{N}_\OLMathNumeral{1}\red \OLId{latin-looped}{P}$ و$\OLId{latin-looped}{N}_\OLMathNumeral{2}\red \OLId{latin-looped}{P}$ (وهذا تافه
  إذا كان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}=\OLId{latin-looped}{N}_\OLMathNumeral{2}$)، ويعطي الانتهاء~$\OLId{latin-looped}{P}\red \OLId{latin-looped}{W}$ لصيغة عادية~$\OLId{latin-looped}{W}$. ولأن
  $\OLId{latin-looped}{U}\ne \OLId{latin-looped}{V}$، يختلف~$\OLId{latin-looped}{W}$ عن واحدة منهما على الأقل؛ ولذلك يكون المختزَل الأحادي
  المقابل~$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ أو~$\OLId{latin-looped}{N}_\OLMathNumeral{2}$ ذا صيغتين عاديتين مختلفتين من جديد. ويؤدي تكرار
  ذلك إلى سلسلة لامتناهية من خطوات~$\redone$، مناقضًا التطبيع القوي. ومن ثم
  تكون الصيغ العادية وحيدة؛ ومع الانتهاء تستلزم وحدانية الصيغ العادية
  التلاقي.
\end{proof}
