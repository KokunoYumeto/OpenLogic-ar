\olsection{اختزال~$\beta$ المتوازي}

نعرّف الآن \emph{اختزال~$\beta$ المتوازي}، ثم نقيم البرهان على أنه ذو خاصية تشيرش--روسر.

\begin{defn}[اختزال~$\beta$ المتوازي، $\bredpar$] \ollabel{defn:bredpar}
  للاختزال المتوازي ($\bredpar$) على الحدود التعريف الاستقرائي التالي:
  \begin{enumerate}
    \item \ollabel{defn:bredpar1} $\OLId{latin-ordinary}{x} \bredpar \OLId{latin-ordinary}{x}$.
    \item \ollabel{defn:bredpar2} من~$\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$ يلزم
      $\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}] \bredpar \lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}']$.
    \item \ollabel{defn:bredpar3} من~$\OLId{latin-looped}{P} \bredpar \OLId{latin-looped}{P}'$ و
      $\OLId{latin-looped}{Q} \bredpar \OLId{latin-looped}{Q}'$ يلزم~$PQ \bredpar \OLId{latin-looped}{P}'\OLId{latin-looped}{Q}'$.
    \item \ollabel{defn:bredpar4} من~$\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$ و
      $\OLId{latin-looped}{Q} \bredpar \OLId{latin-looped}{Q}'$ يلزم
      $(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q} \bredpar \Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{Q}'}{\OLId{latin-ordinary}{x}}$.
  \end{enumerate}
\end{defn}

يجوز في اختزال~$\beta$ المتوازي أن نقلّص أي عدد من مواضع الاختزال في خطوة واحدة.
لكنه يخالف اختزال~$\beta$ في أن هذه الخطوة لا تتناول إلا مواضع الحد الأصلي،
ولا تتناول ما ينشأ عنها من مواضع جديدة بعد اختزال~$\beta$ المتوازي.
فالحد~$(\lambd[\OLId{latin-ordinary}{f}][fx])(\lambd[\OLId{latin-ordinary}{y}][\OLId{latin-ordinary}{y}])$، مثلًا، لا ينتقل في خطوة متوازية إلا إلى نفسه أو إلى~$(\lambd[\OLId{latin-ordinary}{y}][\OLId{latin-ordinary}{y}])\OLId{latin-ordinary}{x}$،
ولا يبلغ~$\OLId{latin-ordinary}{x}$ في تلك الخطوة. ومع أنه يختزل بحسب~$\beta$ إلى~$\OLId{latin-ordinary}{x}$،
فإن موضع هذا الاختزال لا يظهر إلا بعد خطوة من اختزال~$\beta$ المتوازي.
فلا بد من خطوة ثانية من اختزال~$\beta$ المتوازي للوصول إلى~$\OLId{latin-ordinary}{x}$.

\begin{thm}\ollabel{thm:refl}
  $\OLId{latin-looped}{M} \bredpar \OLId{latin-looped}{M}$.
\end{thm}

\begin{proof}
  إثبات ذلك تمرين.
\end{proof}

\begin{prob}
  أقم برهان \olref[lam][cr][pb]{thm:refl}.
\end{prob}

\begin{defn}[التطوير الكامل بحسب~$\beta$]\ollabel{defn:bcd}
  \emph{التطوير الكامل بحسب~$\beta$}، ورمزه $\bcd{\OLId{latin-looped}{M}}$، للحد~$\OLId{latin-looped}{M}$
  يحدد استقرائيًا كما يلي:
  \begin{align}
    \bcd{\OLId{latin-ordinary}{x}} &= \OLId{latin-ordinary}{x} \ollabel{defn:bcd1} \\
    \bcd{(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])} &= \lambd[\OLId{latin-ordinary}{x}][\bcd{\OLId{latin-looped}{N}}] \ollabel{defn:bcd2}\\
    \bcd{(PQ)} &= \bcd{\OLId{latin-looped}{P}}\bcd{\OLId{latin-looped}{Q}} && \text{إذا لم يكن~$P$ تجريد~$\lambd$}
    \ollabel{defn:bcd3} \\
    \bcd{((\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q})} &= \Subst{\bcd{\OLId{latin-looped}{N}}}{\bcd{\OLId{latin-looped}{Q}}}{\OLId{latin-ordinary}{x}} \ollabel{defn:bcd4}
  \end{align}
\end{defn}

التطوير الكامل بحسب~$\beta$، كما يدل الاسم، اختزال متواز يستوفي جميع المواضع.
ففي اختزال~$\beta$ المتوازي يجوز أن نترك بعض مواضع الاختزال،
وأما التطوير الكامل فلا يترك منها شيئًا.
ولذلك يكون التطوير الكامل للحد~$(\lambd[\OLId{latin-ordinary}{f}][fx])(\lambd[\OLId{latin-ordinary}{y}][\OLId{latin-ordinary}{y}])$ هو~$(\lambd[\OLId{latin-ordinary}{y}][\OLId{latin-ordinary}{y}])\OLId{latin-ordinary}{x}$، ولا يكون الحد نفسه.

\begin{editorial}
  يبقى في هذا التعريف أن طريقة تعريف الدوال عوديًا على حدود~$\lambd$ لم تعرض بعد.
  وستستدرك هذه المسألة فيما بعد.
\end{editorial}

\begin{lem}\ollabel{lem:comp}
  من~$\OLId{latin-looped}{M} \bredpar \OLId{latin-looped}{M}'$ و$\OLId{latin-looped}{R} \bredpar \OLId{latin-looped}{R}'$ يلزم~$\Subst{\OLId{latin-looped}{M}}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}}
  \bredpar \Subst{\OLId{latin-looped}{M}'}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}$.
\end{lem}

\begin{proof}
  نستقرئ~!!{derivation} العبارة~$\OLId{latin-looped}{M} \bredpar \OLId{latin-looped}{M}'$.
  وحيث تدعو الحاجة نغير أسماء الممثلين بتحويل~$\OLId{greek-variable}{alpha}$ قبل تطبيق حالات الإحلال،
  بحيث يحقق متغير التجريد~$\OLId{latin-ordinary}{x} \ne \OLId{latin-ordinary}{y}$، ولا يقع~$\OLId{latin-ordinary}{x}$ حرًا في~$\OLId{latin-looped}{R}$ ولا في~$\OLId{latin-looped}{R}'$.
  \begin{enumerate}
    \item إذا كانت الخطوة الأخيرة \olref{defn:bredpar1}، فالإثبات تمرين.
    \item إذا كانت الخطوة الأخيرة \olref{defn:bredpar2}، كان~$\OLId{latin-looped}{M}$
      هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}]$، و$\OLId{latin-looped}{M}'$ هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}']$، مع
      $\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$. فالمطلوب
      $\Subst{(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}} \bredpar
      \Subst{(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}'])}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}$، أو، بعبارة أخرى،
      $\lambd[\OLId{latin-ordinary}{x}][\Subst{\OLId{latin-looped}{N}}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}}] \bredpar
      \lambd[\OLId{latin-ordinary}{x}][\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}]$. وهذا يلزم من
      \olref{defn:bredpar2} وفرض الاستقراء مباشرة.
    \item إذا كانت الخطوة الأخيرة \olref{defn:bredpar3}، فالإثبات تمرين.
    \item إذا كانت الخطوة الأخيرة \olref{defn:bredpar4}، كان~$\OLId{latin-looped}{M}$
      هو~$(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q}$، و$\OLId{latin-looped}{M}'$ هو~$\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{Q}'}{\OLId{latin-ordinary}{x}}$. فالمطلوب
      $\Subst{((\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q})}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}} \bredpar
      \Subst{\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{Q}'}{\OLId{latin-ordinary}{x}}}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}$، أي
      $(\lambd[\OLId{latin-ordinary}{x}][\Subst{\OLId{latin-looped}{N}}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}}])\Subst{\OLId{latin-looped}{Q}}{\OLId{latin-looped}{R}}{\OLId{latin-ordinary}{y}} \bredpar
      \Subst{\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}}{\Subst{\OLId{latin-looped}{Q}'}{\OLId{latin-looped}{R}'}{\OLId{latin-ordinary}{y}}}{\OLId{latin-ordinary}{x}}$، وهو حاصل من
      \olref{defn:bredpar4} وفرض الاستقراء.
  \end{enumerate}
\end{proof}

\begin{prob}
  أتمّ برهان \olref[lam][cr][pb]{lem:comp}.
\end{prob}

\begin{lem}\ollabel{lem:cont}
  من~$\OLId{latin-looped}{M} \bredpar \OLId{latin-looped}{M}'$ يلزم~$\OLId{latin-looped}{M}' \bredpar \bcd{\OLId{latin-looped}{M}}$.
\end{lem}
\begin{proof}
  نستقرئ~!!{derivation} العبارة~$\OLId{latin-looped}{M} \bredpar \OLId{latin-looped}{M}'$.
  \begin{enumerate}
    \item إذا كانت القاعدة الأخيرة \olref{defn:bredpar1}، فالإثبات تمرين.
    \item إذا كانت القاعدة الأخيرة \olref{defn:bredpar2}، كان~$\OLId{latin-looped}{M}$
      هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}]$، و$\OLId{latin-looped}{M}'$ هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}']$، مع
      $\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$. ونطلب~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}'] \bredpar
      \bcd{(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])}$، أي~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}'] \bredpar
      \lambd[\OLId{latin-ordinary}{x}][\bcd{\OLId{latin-looped}{N}}]$ بحسب \olref{defn:bcd2}.
      ويحصل ذلك من \olref{defn:bredpar2} وفرض الاستقراء.
    \item إذا كانت القاعدة الأخيرة \olref{defn:bredpar3}، كان~$\OLId{latin-looped}{M}$ هو~$PQ$،
      و$\OLId{latin-looped}{M}'$ هو~$\OLId{latin-looped}{P}'\OLId{latin-looped}{Q}'$، لحدود~$\OLId{latin-looped}{P}$ و$\OLId{latin-looped}{Q}$ و$\OLId{latin-looped}{P}'$ و$\OLId{latin-looped}{Q}'$ تحقق
      $\OLId{latin-looped}{P} \bredpar \OLId{latin-looped}{P}'$ و$\OLId{latin-looped}{Q} \bredpar \OLId{latin-looped}{Q}'$. ويفيد فرض الاستقراء
      $\OLId{latin-looped}{P}' \bredpar \bcd{\OLId{latin-looped}{P}}$ و$\OLId{latin-looped}{Q}' \bredpar \bcd{\OLId{latin-looped}{Q}}$.
      \begin{enumerate}
        \item إن كان~$\OLId{latin-looped}{P}$ هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}]$ لبعض~$\OLId{latin-ordinary}{x}$ و$\OLId{latin-looped}{N}$،
          فلا بد أن يكون~$\OLId{latin-looped}{P}'$ هو~$\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}']$ لبعض~$\OLId{latin-looped}{N}'$ مع
          $\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$. ومن فرض الاستقراء~$\OLId{latin-looped}{N}' \bredpar \bcd{\OLId{latin-looped}{N}}$
          و$\OLId{latin-looped}{Q}' \bredpar \bcd{\OLId{latin-looped}{Q}}$، فينتج
          $(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}'])\OLId{latin-looped}{Q}' \bredpar
          \Subst{\bcd{\OLId{latin-looped}{N}}}{\bcd{\OLId{latin-looped}{Q}}}{\OLId{latin-ordinary}{x}}$ بالقاعدة \olref{defn:bredpar4}.
        \item وإن لم يكن~$\OLId{latin-looped}{P}$ تجريد~$\lambd$، حصلنا على
          $\OLId{latin-looped}{P}'\OLId{latin-looped}{Q}' \bredpar \bcd{\OLId{latin-looped}{P}}\bcd{\OLId{latin-looped}{Q}}$ بالقاعدة \olref{defn:bredpar3}.
          وطرفها الأيمن هو~$\bcd{PQ}$ بموجب \olref{defn:bcd3}.
      \end{enumerate}
    \item إذا كانت القاعدة الأخيرة \olref{defn:bredpar4}، كان~$\OLId{latin-looped}{M}$ هو
      $(\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q}$، و$\OLId{latin-looped}{M}'$ هو~$\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{Q}'}{\OLId{latin-ordinary}{x}}$ لبعض~$\OLId{latin-ordinary}{x}$ و$\OLId{latin-looped}{N}$
      و$\OLId{latin-looped}{Q}$ و$\OLId{latin-looped}{N}'$ و$\OLId{latin-looped}{Q}'$، مع~$\OLId{latin-looped}{N} \bredpar \OLId{latin-looped}{N}'$ و$\OLId{latin-looped}{Q} \bredpar \OLId{latin-looped}{Q}'$.
      ومن فرض الاستقراء~$\OLId{latin-looped}{N}' \bredpar \bcd{\OLId{latin-looped}{N}}$ و
      $\OLId{latin-looped}{Q}' \bredpar \bcd{\OLId{latin-looped}{Q}}$. فتفيد \olref{lem:comp}
      $\Subst{\OLId{latin-looped}{N}'}{\OLId{latin-looped}{Q}'}{\OLId{latin-ordinary}{x}} \bredpar \Subst{\bcd{\OLId{latin-looped}{N}}}{\bcd{\OLId{latin-looped}{Q}}}{\OLId{latin-ordinary}{x}}$،
      والطرف الأيمن هنا هو~$\bcd{((\lambd[\OLId{latin-ordinary}{x}][\OLId{latin-looped}{N}])\OLId{latin-looped}{Q})}$ نفسه.
  \end{enumerate}
\end{proof}

\begin{prob}
  أتمّ برهان \olref[lam][cr][pb]{lem:cont}.
\end{prob}

\begin{thm}\ollabel{thm:cr}
  العلاقة~$\bredpar$ ذات خاصية تشيرش--روسر.
\end{thm}

\begin{proof}
  هذه نتيجة مباشرة لـ\olref{lem:cont}.
\end{proof}
