\olsection{\usetoken{P}{derivation} بحساب التتابعيات في \Log{K}}

\iftag{prvBox}{

\begin{ex}
  هذه !!{derivation} بحساب التتابعيات يتبين بها أن
  $\Proves (\Box!A \land \Box!B) \lif \Box (!A \land !B)$.
  \begin{prooftree}
    \Axiom$!A \fCenter !A$
    \doubleLine
    \UnaryInf$!B, !A \fCenter !A$   
    \Axiom$!B \fCenter !B$
    \doubleLine
    \UnaryInf$!B, !A \fCenter !B$   
    \RightLabel{\RightR{\land}}
    \BinaryInf$!B, !A \fCenter !A \land !B$    
    \RightLabel{$\Box$}
    \UnaryInf$\Box!B, \Box!A \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\land}}
    \UnaryInf$\Box!A \land \Box!B, \Box!A \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\Exchange}}
    \UnaryInf$\Box!A, \Box!A \land \Box!B \fCenter \Box
    (!A \land !B)$
    \RightLabel{\LeftR{\land}}
    \UnaryInf$\Box!A \land \Box!B, \Box!A \land \Box!B \fCenter \Box (!A \land !B)$
    \RightLabel{\LeftR{\Contraction}}
    \UnaryInf$\Box!A \land \Box!B \fCenter \Box (!A \land !B)$
    \RightLabel{\RightR{\lif}}
    \UnaryInf$\fCenter (\Box!A \land \Box!B)
    \lif \Box (!A \land !B)$
  \end{prooftree}
\end{ex}
}{}

      \BinaryInf$!A \lor !B \fCenter !A, !B$    
      \RightLabel{$\Diamond$}
      \UnaryInf$\Diamond(!A \lor !B) \fCenter 
      \Diamond !A, \Diamond!B$
      \RightLabel{\RightR{\lor}}

\begin{prob}
  التمس في~$\Log{K}$ براهين بحساب التتابعيات لل!!{formula}s الآتية:
  \begin{enumerate}
    \item $\Box \lnot \OLId{latin-ordinary}{p} \lif \Box(\OLId{latin-ordinary}{p} \lif \OLId{latin-ordinary}{q})$
    \item $(\Box \OLId{latin-ordinary}{p} \lor \Box \OLId{latin-ordinary}{q}) \lif \Box(\OLId{latin-ordinary}{p} \lor \OLId{latin-ordinary}{q})$
    \item $\Diamond \OLId{latin-ordinary}{p} \lif \Diamond(\OLId{latin-ordinary}{p} \lor \OLId{latin-ordinary}{q})$
    \item $\Box(\OLId{latin-ordinary}{p} \land \OLId{latin-ordinary}{q}) \lif \Box \OLId{latin-ordinary}{p}$
  \end{enumerate}
\end{prob}
