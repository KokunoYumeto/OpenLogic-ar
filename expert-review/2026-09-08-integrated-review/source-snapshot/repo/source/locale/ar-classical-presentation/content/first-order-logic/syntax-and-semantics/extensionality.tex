\olsection{الامتدادية}

\begin{explain}
الامتدادية، ويقال لها أحيانًا وثاقة الصلة، مؤداها في العبارة غير
الصورية أن إرضاء !!{formula}~$!A$ في
!!a{structure}~$\Struct M$ بحسب إسناد ل!!a{variable}~$\OLId{latin-ordinary}{s}$
لا يتأثر إلا بحجم ال!!{domain}، وبما تعيّنه
$\Struct M$ و$\OLId{latin-ordinary}{s}$ لرموز اللغة التي وردت بالفعل في~$!A$.

ومن ثمرتها أن ال!!{structure}s
~$\Struct M$ و$\Struct M'$ إذا اتفقتا في تأويل رموز جملة~$!A$
وكان مجالهما واحدًا، اتفقت~$\Struct M$
و$\Struct M'$ في الحكم على صدق~$!A$ أو عدم صدقها.
\end{explain}

\begin{prop}[الامتدادية]
  \ollabel{prop:extensionality}
  لتكن $!A$ !!a{formula}، ولتكن $\Struct M_\OLMathNumeral{1}$ و$\Struct M_\OLMathNumeral{2}$
  !!{structure}s بحيث $\Domain{\OLId{latin-looped}{M}_\OLMathNumeral{1}} = \Domain{\OLId{latin-looped}{M}_\OLMathNumeral{2}}$، وليكن $\OLId{latin-ordinary}{s}$
  إسناد متغيرات على~$\Domain{\OLId{latin-looped}{M}_\OLMathNumeral{1}} = \Domain{\OLId{latin-looped}{M}_\OLMathNumeral{2}}$. إذا كان
  $\Assign{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}_\OLMathNumeral{1}} = \Assign{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}_\OLMathNumeral{2}}$،
  و$\Assign{\OLId{latin-looped}{R}}{\OLId{latin-looped}{M}_\OLMathNumeral{1}}=\Assign{\OLId{latin-looped}{R}}{\OLId{latin-looped}{M}_\OLMathNumeral{2}}$، و
  $\Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}_\OLMathNumeral{1}} = \Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}_\OLMathNumeral{2}}$ لكل !!{constant}~$\OLId{latin-ordinary}{c}$،
  ورمز علاقة~$\OLId{latin-looped}{R}$، و!!{function}~$\OLId{latin-ordinary}{f}$ ترد في~$!A$، فإن
  $\Sat{\OLId{latin-looped}{M}_\OLMathNumeral{1}}{!A}[\OLId{latin-ordinary}{s}]$ إذا وفقط إذا كان $\Sat{\OLId{latin-looped}{M}_\OLMathNumeral{2}}{!A}[\OLId{latin-ordinary}{s}]$.
\end{prop}

\begin{proof}
  % تصحيح مصدر (0165-I1): قُيّد استقراء الحدود بما يرد في A؛ فالفرض لا يسوّي تأويل الرموز التي لا ترد فيها.
  ابدأ بالاستقراء على بنية كل حد~$\OLId{latin-ordinary}{t}$ يرد في~$!A$ لإثبات أن قيمته
  واحدة في البنيتين، أي $\Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}_\OLMathNumeral{1}}[\OLId{latin-ordinary}{s}] = \Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}_\OLMathNumeral{2}}[\OLId{latin-ordinary}{s}]$. ثم أجرِ الاستقراء
  على~$!A$ لإثبات القضية، مستعينًا بهذه النتيجة في أساس الاستقراء،
  حيث تكون $!A$ ذرية.
\end{proof}

\begin{prob}
فصّل برهان \olref[fol][syn][ext]{prop:extensionality} بالتفصيل.
\end{prob}

\begin{cor}[امتدادية ال!!^{sentence}s]
  \ollabel{cor:extensionality-sent}
  لتكن $!A$ !!a{sentence}، ولتكن $\Struct{M_1}$ و$\Struct{M_2}$ كما
  في~\olref{prop:extensionality}. عندئذ
  $\Sat{\OLId{latin-looped}{M}_\OLMathNumeral{1}}{!A}$ إذا وفقط إذا كان $\Sat{\OLId{latin-looped}{M}_\OLMathNumeral{2}}{!A}$.
\end{cor}

\begin{proof}
تلزم من \olref{prop:extensionality} باستعمال
\olref[ass]{cor:sat-sentence}.
\end{proof}

وكذلك لا تتوقف قيمة الحد، ولا إرضاء !!a{structure} ل!!a{formula}
أو عدمه، إلا على قيم الحدود الجزئية المعنية.

\begin{prop}\ollabel{prop:ext-terms}
لتكن $\Struct M$ !!a{structure}، وليكن $\OLId{latin-ordinary}{t}$ و$\OLId{latin-ordinary}{t}'$ حدين، وليكن $\OLId{latin-ordinary}{s}$
إسناد متغيرات. عندئذ $\Value{\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}] =
\Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}]$.
\end{prop}

\begin{proof}
بالاستقراء على~$\OLId{latin-ordinary}{t}$.
\begin{enumerate}
\item أما إذا كان $\OLId{latin-ordinary}{t}$ رمزًا ثابتًا، وفرضنا $\OLId{latin-ordinary}{t}\ident \OLId{latin-ordinary}{c}$، فإن
  $\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}} = \OLId{latin-ordinary}{c}$، و$\Value{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}] = \Assign{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}} =
  \Value{\OLId{latin-ordinary}{c}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}]$.

\item وأما إذا كان $\OLId{latin-ordinary}{t}$ متغيرًا غير~$\OLId{latin-ordinary}{x}$، وفرضنا $\OLId{latin-ordinary}{t} \ident \OLId{latin-ordinary}{y}$، فإن
  $\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}} = \OLId{latin-ordinary}{y}$، و$\Value{\OLId{latin-ordinary}{y}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}] =
  \Value{\OLId{latin-ordinary}{y}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}]$ لأن
  $\varAssign{\OLId{latin-ordinary}{s}}{\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}}{\OLId{latin-ordinary}{x}}$.

\item وأما إذا كان $\OLId{latin-ordinary}{t} \ident \OLId{latin-ordinary}{x}$، فإن $\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}} = \OLId{latin-ordinary}{t}'$. لكن
  $\Value{\OLId{latin-ordinary}{x}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}] = \Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]$
  بحسب تعريف~$\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}$.

\item ويبقى أن يكون $\OLId{latin-ordinary}{t} \ident \Atom{f}{t_1,\dots,t_n}$، وعندئذ:
\begin{multline*}
  \Value{\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]  = \\
  \begin{aligned}[\OLId{latin-ordinary}{b}]
& \Value{\Atom{f}{\Subst{t_1}{t'}{x}, \dots, \Subst{t_n}{t'}{x}}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]\\
& \qquad    \text{ بحسب تعريف $\Subst{t}{t'}{x}$}\\
& = \Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}}(\Value{\Subst{\OLId{latin-ordinary}{t}_\OLMathNumeral{1}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}], \dots,
    \Value{\Subst{\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}])\\
    & \qquad  \text{ بحسب تعريف $\Value{\Atom{f}{\dots}}{M}[s]$}\\
& = \Assign{\OLId{latin-ordinary}{f}}{\OLId{latin-looped}{M}}(\Value{\OLId{latin-ordinary}{t}_\OLMathNumeral{1}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}], \dots,
   \Value{\OLId{latin-ordinary}{t}_\OLId{latin-ordinary}{n}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}])\\
& \qquad    \text{ بفرضية الاستقراء}\\
& = \Value{\OLId{latin-ordinary}{t}}{\OLId{latin-looped}{M}}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}]
    \text{ بحسب تعريف $\Value{\Atom{f}{\dots}}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]$}
  \end{aligned}
\end{multline*}
\end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:ext-formulas} لتكن $\Struct M$
!!a{structure}، ولتكن $!A$ !!a{formula}، وليكن $\OLId{latin-ordinary}{t}'$~حدًا، وليكن
$\OLId{latin-ordinary}{s}$~إسناد متغيرات. عندئذ
$\Sat{\OLId{latin-looped}{M}}{\Subst{!A}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}}[\OLId{latin-ordinary}{s}]$ إذا وفقط إذا كان
$\Sat{\OLId{latin-looped}{M}}{!A}[\Subst{\OLId{latin-ordinary}{s}}{\Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]}{\OLId{latin-ordinary}{x}}]$.
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\begin{prob}
أثبت \olref[fol][syn][ext]{prop:ext-formulas}.
\end{prob}

\begin{explain}
  وبيان المراد من
  \cref{fol:syn:ext:prop:ext-terms,fol:syn:ext:prop:ext-formulas}
  أن يكون معنا حد~$\OLId{latin-ordinary}{t}$ أو !!a{formula}~$!A$، وحد آخر~$\OLId{latin-ordinary}{t}'$،
  ونريد تعيين قيمة $\Subst{\OLId{latin-ordinary}{t}}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}$، أو الحكم على إرضاء
  $\Subst{!A}{\OLId{latin-ordinary}{t}'}{\OLId{latin-ordinary}{x}}$ في !!a{structure}~$\Struct M$
  بالنسبة إلى إسناد ل!!a{variable}~$\OLId{latin-ordinary}{s}$. ولنا في ذلك طريقان:
  أن نقدّم الإحلال ثم ننظر في القيمة أو الإرضاء بالنسبة إلى
  $\Struct{M}$ و$\OLId{latin-ordinary}{s}$؛ أو أن نقدّم حساب
  $\OLId{latin-ordinary}{m} = \Value{\OLId{latin-ordinary}{t}'}{\OLId{latin-looped}{M}}[\OLId{latin-ordinary}{s}]$، وهي قيمة الحد~$\OLId{latin-ordinary}{t}'$ في
  $\Struct{M}$ بحسب~$\OLId{latin-ordinary}{s}$، ثم نبدّل إسناد ال!!{variable}s
  إلى~$\Subst{\OLId{latin-ordinary}{s}}{\OLId{latin-ordinary}{m}}{\OLId{latin-ordinary}{x}}$، وننظر في قيمة~$\OLId{latin-ordinary}{t}$ في~$\Struct{M}$
  مع~$\Subst{\OLId{latin-ordinary}{s}}{\OLId{latin-ordinary}{m}}{\OLId{latin-ordinary}{x}}$، أو في صدق
  $\Sat{\OLId{latin-looped}{M}}{!A}[\Subst{\OLId{latin-ordinary}{s}}{\OLId{latin-ordinary}{m}}{\OLId{latin-ordinary}{x}}]$.
  وبمقتضى \Cref{fol:syn:ext:prop:ext-terms,fol:syn:ext:prop:ext-formulas}
  لا يختلف الجواب باختلاف الطريقين.
\end{explain}
