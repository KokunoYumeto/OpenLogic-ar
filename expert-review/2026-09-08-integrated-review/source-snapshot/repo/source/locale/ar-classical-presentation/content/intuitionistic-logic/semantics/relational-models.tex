\olsection{\usetoken{P}{relational model} للمنطق الحدسي}

لا تضبط دلالة المنطق القضي الحدسي حتى نحد نموذجًا تقوّم ال!!{formula}s
بالإضافة إليه؛ ومن هذا الحد تستخرج الصلاحية والاستلزام وسائر المفاهيم
الدلالية. وال!!{relational model}s أحد أصول هذه الدلالة.

\begin{defn}
  !!^a{relational model} للمنطق القضي الحدسي هو الثلاثي
  $\mModel{M}=\tuple{\OLId{latin-looped}{W},\OLId{latin-looped}{R},\OLId{latin-looped}{V}}$، وفيه:
  \begin{enumerate}
  \item $\OLId{latin-looped}{W}$ مجموعة غير فارغة؛
  \item $\OLId{latin-looped}{R}$ ترتيب جزئي على~$\OLId{latin-looped}{W}$ (أي علاقة ثنائية انعكاسية ومضادة للتناظر
    ومتعدية)؛
  \item $\OLId{latin-looped}{V}$ دالة تسند إلى كل~!!{propositional
    variable}~$\OLId{latin-ordinary}{p}$ مجموعة جزئية من~$\OLId{latin-looped}{W}$؛
  \item $\OLId{latin-looped}{V}$ رتيبة بالنسبة إلى~$\OLId{latin-looped}{R}$؛ أي إذا كان~$\OLId{latin-ordinary}{w} \in \OLId{latin-looped}{V}(\OLId{latin-ordinary}{p})$ و$Rww'$،
    كان~$\OLId{latin-ordinary}{w}' \in \OLId{latin-looped}{V}(\OLId{latin-ordinary}{p})$.
  \end{enumerate}
\end{defn}

\begin{defn}\ollabel{defn:true-at-w}
  \emph{صدق~$!A$ عند~$\OLId{latin-ordinary}{w}$ في~$\mModel{M}$}، ورمزه
  $\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{w}]$، يحد استقرائيًا بما يأتي:
  \begin{enumerate}
  \item \indcase{!A}{p}{$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا كان
    $\OLId{latin-ordinary}{w} \in \OLId{latin-looped}{V}(\OLId{latin-ordinary}{p})$.}
  \item \indcase{!A}{\lfalse}{لا يتحقق~$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$}.
  \item \indcase{!A}{\lnot !B}{$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا
    لم توجد~$\OLId{latin-ordinary}{w}'$ بحيث~$Rww'$ ويتحقق~$\mSat{\OLId{latin-looped}{M}}{!B}[\OLId{latin-ordinary}{w}']$}.
  \item \indcase{!A}{!B \land !C}{$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا
    تحقق~$\mSat{\OLId{latin-looped}{M}}{!B}[\OLId{latin-ordinary}{w}]$ و$\mSat{\OLId{latin-looped}{M}}{!C}[\OLId{latin-ordinary}{w}]$}.
  \item \indcase{!A}{!B \lor !C}{$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا
    تحقق~$\mSat{\OLId{latin-looped}{M}}{!B}[\OLId{latin-ordinary}{w}]$ أو~$\mSat{\OLId{latin-looped}{M}}{!C}[\OLId{latin-ordinary}{w}]$ (أو كلاهما)}.
  \item \indcase{!A}{!B \lif !C}{$\mSat{\OLId{latin-looped}{M}}{\indfrm}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا
    كان، لكل~$\OLId{latin-ordinary}{w}'$ بحيث~$Rww'$، لا يتحقق~$\mSat{\OLId{latin-looped}{M}}{!B}[\OLId{latin-ordinary}{w}']$ أو يتحقق
    $\mSat{\OLId{latin-looped}{M}}{!C}[\OLId{latin-ordinary}{w}']$} (أو كلاهما).
  \end{enumerate}
  ونكتب~$\mSat/{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{w}]$ لنفي~$\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{w}]$. وإذا كانت
  $\OLId{greek-variable}{Gamma}$ مجموعة من ال!!{formula}s، فقولنا~$\mSat{\OLId{latin-looped}{M}}{\OLId{greek-variable}{Gamma}}[\OLId{latin-ordinary}{w}]$ معناه
  $\mSat{\OLId{latin-looped}{M}}{!B}[\OLId{latin-ordinary}{w}]$ لكل~$!B \in \OLId{greek-variable}{Gamma}$.
\end{defn}

\begin{prob}
  بيّن أنه بحسب~\olref[int][sem][rel]{defn:true-at-w}،
  $\mSat{\OLId{latin-looped}{M}}{\lnot !A}[\OLId{latin-ordinary}{w}]$ إذا وفقط إذا كان
  $\mSat{\OLId{latin-looped}{M}}{!A \lif \lfalse}[\OLId{latin-ordinary}{w}]$.
\end{prob}

\begin{prop}\ollabel{prop:true-monotonic}
  الصدق عند العوالم رتيب بالنسبة إلى~$\OLId{latin-looped}{R}$؛ أي إذا كان
  $\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{w}]$ و$Rww'$، كان~$\mSat{\OLId{latin-looped}{M}}{!A}[\OLId{latin-ordinary}{w}']$.
\end{prop}

\begin{proof}
  تمرين.
\end{proof}

\begin{prob}
  أقم برهان~\olref[int][sem][rel]{prop:true-monotonic}.
\end{prob}
