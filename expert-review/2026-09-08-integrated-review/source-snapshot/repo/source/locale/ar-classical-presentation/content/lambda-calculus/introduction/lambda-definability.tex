\olsection{الدوال الحسابية \usetoken{S}{lambda definable}}

بأي معنى يكون حساب لامبدا نموذجًا للحساب؟
لا يستبين هذا أول النظر؛ إذ لا بد، قبل الكلام على حساب
الدوال على الأعداد الطبيعية، من تمثيل تلك الأعداد فيه تمثيلًا
ملائمًا. والتمثيل الآتي يؤدي الغرض على نحو قد لا يُتوقع.

\begin{defn}
لكل عدد طبيعي~$\OLId{latin-ordinary}{n}$ نجعل \emph{عدد تشيرش}~$\num{\OLId{latin-ordinary}{n}}$
هو حد لامبدا~$\lambd[\OLId{latin-ordinary}{x}][\lambd[\OLId{latin-ordinary}{y}][(\OLId{latin-ordinary}{x}(\OLId{latin-ordinary}{x}(\OLId{latin-ordinary}{x}(\dots \OLId{latin-ordinary}{x}(\OLId{latin-ordinary}{y})))))]]$،
وفيه~$\OLId{latin-ordinary}{n}$ من ورود~$\OLId{latin-ordinary}{x}$ في المجموع.
\end{defn}

هذه الحدود~$\num{\OLId{latin-ordinary}{n}}$ «مكررات»: إذا دخلت عليها~$\OLId{latin-ordinary}{f}$،
أعادت~$\num{\OLId{latin-ordinary}{n}}$ الدالة التي ترسل~$\OLId{latin-ordinary}{y}$ إلى~$\OLId{latin-ordinary}{f}^\OLId{latin-ordinary}{n}(\OLId{latin-ordinary}{y})$.
وكل تمثيل عددي منها سوي. بذلك يمكن ضبط معنى أن يحسب
حد لامبدا دالة على الأعداد الطبيعية.

\begin{defn}
لتكن~$\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{x}_\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}})$ دالة جزئية ذات~$\OLId{latin-ordinary}{k}$ مواضع
من~$\Nat$ إلى~$\Nat$. نقول إن حد~$\lambd$~$\OLId{latin-looped}{F}$
\emph{!!{lambda define}s}~$\OLId{latin-ordinary}{f}$ إذا وفقط إذا تحقق الشرط الآتي
لكل متتالية أعداد طبيعية~$\OLId{latin-ordinary}{n}_\OLMathNumeral{0}$، \dots،~$\OLId{latin-ordinary}{n}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}$:
\[
\OLId{latin-looped}{F}\, \num{\OLId{latin-ordinary}{n}_\OLMathNumeral{0}}\, \num{\OLId{latin-ordinary}{n}_\OLMathNumeral{1}} \dots \num{\OLId{latin-ordinary}{n}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}} \red \num{\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}_\OLMathNumeral{0}, \OLId{latin-ordinary}{n}_\OLMathNumeral{1}, \dots,
  \OLId{latin-ordinary}{n}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}})}
\]
متى كانت~$\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}_\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{n}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}})$ معرّفة؛ وأما إذا لم تكن
معرّفة فلا صورة سوية للحد
$\OLId{latin-looped}{F}\, \num{\OLId{latin-ordinary}{n}_\OLMathNumeral{0}}\, \num{\OLId{latin-ordinary}{n}_\OLMathNumeral{1}} \dots \num{\OLId{latin-ordinary}{n}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}}$.
\end{defn}

\begin{thm}
\ollabel{thm:lambda-def}
الدالة~$\OLId{latin-ordinary}{f}$ جزئية قابلة للحساب إذا وفقط إذا كانت
!!{lambda defined} بحد من حدود لامبدا.
\end{thm}

\begin{explain}
في هذه المبرهنة أمر يستوقف النظر: حساب لامبدا نموذج بسيط
للحساب، لا عملية فيه إلا التجريد والتطبيق!{}
ومع قلة هذه الأدوات يمكن أن ننفذ به كل إجراء حسابي.
\end{explain}
