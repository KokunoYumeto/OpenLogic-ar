\olsection{الدوال \usetoken{S}{lambda definable} مغلقة تحت التركيب}

\begin{lem}
التركيب يحفظ كون الدوال~!!{lambda definable}.
\end{lem}

\begin{proof}
لتكن~$\OLId{latin-ordinary}{f}$ حاصلة بالتركيب من~$\OLId{latin-ordinary}{h}$ و$\OLId{latin-ordinary}{g}_\OLMathNumeral{0}$، \dots،~$\OLId{latin-ordinary}{g}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}$.
ولنفرض أن~$\OLId{latin-ordinary}{h}$ و$\OLId{latin-ordinary}{g}_\OLMathNumeral{0}$، \dots،~$\OLId{latin-ordinary}{g}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}$ هي !!{lambda defined}
بالحدود~$\OLId{latin-looped}{H}$ و$\OLId{latin-looped}{G}_\OLMathNumeral{0}$، \dots،~$\OLId{latin-looped}{G}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}$، بالترتيب. فالمطلوب حد~$\OLId{latin-looped}{F}$
!!{lambda define}s~$\OLId{latin-ordinary}{f}$، ويحصل المطلوب بتعريف~$\OLId{latin-looped}{F}$ على الوجه الآتي:
\[
\OLId{latin-looped}{F}(\OLId{latin-ordinary}{x}_\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{l}-\OLMathNumeral{1}}) = \OLId{latin-looped}{H}(\OLId{latin-looped}{G}_\OLMathNumeral{0}(\OLId{latin-ordinary}{x}_\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{l}-\OLMathNumeral{1}}),
\dots, \OLId{latin-looped}{G}_{\OLId{latin-ordinary}{k}-\OLMathNumeral{1}}(\OLId{latin-ordinary}{x}_\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{l}-\OLMathNumeral{1}})).
\]
فهذا يبيّن ملاءمة لغة حساب لامبدا لتمثيل تركيب الدوال.
\end{proof}
