\olsection{قواعد القضايا لمنطقات مختارة}

العمدة في تعيين قواعد رابط ما في حساب تتابعيات ذي جوانب~$\OLId{latin-ordinary}{n}$ هي
دالة الصدق المميزة لذلك الرابط وحدها. فإن اشترك منطقان أو أكثر في
هذه الدالة، اشتركا في قواعد التتابعيات ذات الجوانب~$\OLId{latin-ordinary}{n}$ الخاصة
بالرابط، وإن اختلفا من وجوه أخرى.

\subsection{قواعد~$\lnot$}

أما~$\lnot$ في منطقي~\L ukasiewicz وكلايني وفي صورهما المختلفة،
فقواعده كما يأتي.

\begin{defish}
  \begin{center}
\AxiomC{$ \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}, !A $}
\RightLabel{\iR{\lnot}{\False}}
\UnaryInfC{$ \lnot !A, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
\DisplayProof
\\[2ex]
\AxiomC{$ \OLId{greek-variable}{Gamma} \nSequent !A, \Pi \nSequent \OLId{greek-variable}{Delta} $}
\RightLabel{\iR{\lnot}{\Undef}}
\UnaryInfC{$\OLId{greek-variable}{Gamma} \nSequent \lnot !A, \Pi \nSequent \OLId{greek-variable}{Delta}$}
\DisplayProof
\\[2ex]
\AxiomC{$!A, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
\RightLabel{\iR{\lnot}{\True}}
\UnaryInfC{$\OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta},  \lnot !A$}
\DisplayProof
  \end{center}
\end{defish}

تنطبق القواعد الآتية لـ$\lnot$ على منطق غودل.

\begin{defish}
\AxiomC{$ \OLId{greek-variable}{Gamma} \nSequent !A, \Pi \nSequent \OLId{greek-variable}{Delta}, !A $}
\RightLabel{\iR{\lnot}{\False}[\LogGod]}
\UnaryInfC{$ \lnot !A, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
\DisplayProof
\hfill
\AxiomC{$!A, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
\RightLabel{\iR{\lnot}{\True}[\LogGod]}
\UnaryInfC{$\OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta},  \lnot !A$}
\DisplayProof
\hfill
\end{defish}

وأما~$\lnot !A$ فلا سبيل إلى أن تكون قيمته~$\Undef$ في منطق غودل؛
ولذلك خلا الموضع الأوسط من قاعدة خاصة به.

\subsection{قواعد~$\land$}

هذه قواعد~$\land$ في منطق~\L ukasiewicz ومنطق كلايني القوي ومنطق
غودل.

\begin{defish}
\begin{center}
  \AxiomC{$!A, !B, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
  \RightLabel{$\iR{\land}{\False}$}
  \UnaryInfC{$!A \land !B, \OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}$}
  \DisplayProof
\\[2ex]
  \AxiomC{$\OLId{greek-variable}{Gamma} \nSequent !A, \Pi \nSequent !A, \OLId{greek-variable}{Delta}$}
  \AxiomC{$\OLId{greek-variable}{Gamma} \nSequent !B, \Pi \nSequent !B, \OLId{greek-variable}{Delta}$}
  \AxiomC{$\OLId{greek-variable}{Gamma} \nSequent !A, !B, \Pi \nSequent \OLId{greek-variable}{Delta}$}
  \RightLabel{$\iR\land\Undef$}
  \TrinaryInfC{$\OLId{greek-variable}{Gamma} \nSequent !A \land !B, \Pi \nSequent \OLId{greek-variable}{Delta}$}
  \DisplayProof
  \\[2ex]
  \AxiomC{$\OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}, !A$}
  \AxiomC{$\OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}, !B$}
  \RightLabel{$\iR\land\True$}
  \BinaryInfC{$\OLId{greek-variable}{Gamma} \nSequent \Pi \nSequent \OLId{greek-variable}{Delta}, !A \land !B$}
  \DisplayProof
\end{center}
\end{defish}
