\olsection{الاستنباط الطبيعي}

الاستنباط الطبيعي إذا حذفت منه قواعد~$\FalseCl$ نظام معياري من أنظمة
!!{derivation} للمنطق الحدسي. وسنورد قواعده، ثم نبين وجهها على تأويل BHK:
فكل قاعدة تنقلنا من وجود بناءات لمقدماتها إلى وجود بناء لنتيجتها.

ولما كانت !!{derivation}s الاستنباط الطبيعي قد تشتمل على افتراضات غير
مصروفة، حملنا !!a{derivation} لــ$!A$ من افتراضات
!!{undischarged}~$\OLId{greek-variable}{Gamma}$ على دالة تنقل بناءات كل~$!B \in \OLId{greek-variable}{Gamma}$ إلى
بناء لــ$!A$. فإذا وجدت !!a{derivation} لــ$!A$ لا افتراض فيها
!!{undischarged}، وجد بناء لــ$!A$ بحسب BHK. وسنسقط~$\OLId{greek-variable}{Gamma}$ حيث لا
يحتاج إليها البيان.

والافتراض~$!A$ وحده !!a{derivation} لــ$!A$ من الافتراض
!!{undischarged}~$!A$؛ وهذا موافق لتأويل BHK، لأن دالة الهوية تنقل كل
بناء لــ$!A$ إلى بناء لــ$!A$.

\subsection{العطف}

\begin{defish}
\AxiomC{$!A$}
\AxiomC{$!B$}
\RightLabel{\Intro{\land}}
\BinaryInfC{$!A \land !B$}
\DisplayProof
\hfill
\begin{tabular}{l}
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!A$}
\DisplayProof
\\[3ex]
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!B$}
\DisplayProof
\end{tabular}
\end{defish}

إذا كان لدينا البناءان~$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ و$\OLId{latin-looped}{N}_\OLMathNumeral{2}$ لــ$!A_\OLMathNumeral{1}$ و$!A_\OLMathNumeral{2}$ على الترتيب،
كان لدينا بناء لــ$!A_\OLMathNumeral{1} \land !A_\OLMathNumeral{2}$، وهو الزوج~$\tuple{\OLId{latin-looped}{N}_\OLMathNumeral{1}, \OLId{latin-looped}{N}_\OLMathNumeral{2}}$.

وبناء~$!A_\OLMathNumeral{1} \land !A_\OLMathNumeral{2}$ في تأويل BHK زوج~$\tuple{\OLId{latin-looped}{N}_\OLMathNumeral{1}, \OLId{latin-looped}{N}_\OLMathNumeral{2}}$. فإذا
أعطي هذا الزوج حصل بناء كل معطوف: فــ$\OLId{latin-looped}{N}_\OLMathNumeral{1}$ بناء لــ$!A_\OLMathNumeral{1}$، و$\OLId{latin-looped}{N}_\OLMathNumeral{2}$ بناء
لــ$!A_\OLMathNumeral{2}$.

\subsection{الشرط}

\begin{defish}
  \AxiomC{$\Discharge{!A}{\OLId{latin-ordinary}{u}}$}
  \DeduceC{$!B$}
  \DischargeRule{$\Intro{\lif}$}{u}
  \UnaryInfC{$!A \lif !B$}
  \DisplayProof
\hfill
  \AxiomC{$!A \lif !B$}
  \AxiomC{$!A$}
  \RightLabel{$\Elim{\lif}$}
  \BinaryInfC{$!B$}
  \DisplayProof
\end{defish}

إذا قامت !!a{derivation} لــ$!B$ من الافتراض
!!{undischarged}~$!A$، قامت دالة~$\OLId{latin-ordinary}{f}$ تنقل بناءات~$!A$ إلى بناءات~$!B$.
وهذه الدالة بعينها بناء لــ$!A \lif !B$؛ فإذا كان لمقدمة~$\Intro\lif$
بناء مشروط ببناء لــ$!A$، كان للنتيجة~$!A \lif !B$ بناء.

وأما الحذف، فإذا وجد بناء~$\OLId{latin-looped}{N}$ لــ$!A$ وبناء~$\OLId{latin-ordinary}{f}$ لــ$!A \lif !B$، كانت
الدالة التي هي بناء~$!A \lif !B$ ناقلة لبناءات~$!A$ إلى بناءات~$!B$؛
فكان~$\OLId{latin-ordinary}{f}(\OLId{latin-looped}{N})$ بناء لــ$!B$، وهو بناء نتيجة~$\Elim\lif$.

\subsection{الفصل}

\begin{defish}
\begin{tabular}{l}
\AxiomC{$!A$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\\[3ex]
\AxiomC{$!B$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\end{tabular}
\hfill
\AxiomC{$!A \lor !B$}
\AxiomC{$\Discharge{!A}{\OLId{latin-ordinary}{n}}$}
\DeduceC{$!C$}
\AxiomC{$\Discharge{!B}{\OLId{latin-ordinary}{n}}$}
\DeduceC{$!C$}
\DischargeRule{\Elim{\lor}}{n}
\TrinaryInfC{$!C$}
\DisplayProof
\end{defish}

إذا كان~$\OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}$ بناء لــ$!A_\OLId{latin-ordinary}{i}$، صار الزوج~$\tuple{\OLId{latin-ordinary}{i}, \OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}}$ بناء لــ
$!A_\OLMathNumeral{1} \lor !A_\OLMathNumeral{2}$. وأما إذا أعطينا بناءً لــ$!A_\OLMathNumeral{1} \lor !A_\OLMathNumeral{2}$، أي زوجًا
$\tuple{\OLId{latin-ordinary}{i}, \OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}}$ يكون~$\OLId{latin-looped}{N}_\OLId{latin-ordinary}{i}$ فيه بناء لــ$!A_\OLId{latin-ordinary}{i}$، وأعطينا دالتين~$\OLId{latin-ordinary}{f}_\OLMathNumeral{1}$
و$\OLId{latin-ordinary}{f}_\OLMathNumeral{2}$ تنقلان بناءات~$!A_\OLMathNumeral{1}$ و$!A_\OLMathNumeral{2}$ على الترتيب إلى بناءات~$!C$، كان
$\OLId{latin-ordinary}{f}_\OLId{latin-ordinary}{i}(\OLId{latin-looped}{N}_\OLId{latin-ordinary}{i})$ بناء لــ$!C$، وهو نتيجة~$\Elim\lor$.

\subsection{المحال}

\begin{defish}
  \AxiomC{$\lfalse$}
  \RightLabel{\FalseInt}
  \UnaryInfC{$!A$}
  \DisplayProof
\end{defish}

إذا قامت !!a{derivation} لــ$\lfalse$ من الافتراضات
!!{undischarged}~$!B_\OLMathNumeral{1}$، \dots، $!B_\OLId{latin-ordinary}{n}$، قامت دالة
$\OLId{latin-ordinary}{f}(\OLId{latin-looped}{M}_\OLMathNumeral{1}, \dots, \OLId{latin-looped}{M}_\OLId{latin-ordinary}{n})$ تنقل بناءات~$!B_\OLMathNumeral{1}$، \dots، $!B_\OLId{latin-ordinary}{n}$ إلى بناء
لــ$\lfalse$. ولما لم يكن لــ$\lfalse$ بناء، امتنع اجتماع بناءات لكل
$!B_\OLMathNumeral{1}$، \dots، $!B_\OLId{latin-ordinary}{n}$. ومن ثم تصدق في~$\OLId{latin-ordinary}{f}$ القضية الآتية أيضًا:
\emph{إذا} كانت~$\OLId{latin-looped}{M}_\OLMathNumeral{1}$، \dots، $\OLId{latin-looped}{M}_\OLId{latin-ordinary}{n}$ بناءات لــ$!B_\OLMathNumeral{1}$،
\dots، $!B_\OLId{latin-ordinary}{n}$ على الترتيب، \emph{فإن}~$\OLId{latin-ordinary}{f}(\OLId{latin-looped}{M}_\OLMathNumeral{1}, \dots, \OLId{latin-looped}{M}_\OLId{latin-ordinary}{n})$ بناء
لــ$!A$.

\subsection{قواعد $\lnot$}

لما كانت~$\lnot !A$ معرّفة بــ$!A \lif \lfalse$، استغنينا من جهة
الدقة عن قواعد~$\lnot$. وإن أفردناها كانت كما يأتي:

\begin{defish}
\AxiomC{$\Discharge{!A}{\OLId{latin-ordinary}{n}}$}
\noLine
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{n}
\UnaryInfC{$\lnot !A$}
\DisplayProof
\hfill
\AxiomC{$\lnot !A$}
\AxiomC{$!A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\DisplayProof
\end{defish}


\subsection{أمثلة على \usetoken{P}{derivation}}

\begin{enumerate}
\item $\Proves !A \lif (\lnot !A \lif \lfalse)$، أي $\Proves !A
  \lif ((!A \lif \lfalse) \lif \lfalse)$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A}{\OLMathNumeral{2}}$}
    \AxiomC{$\Discharge{!A \lif \lfalse}{\OLMathNumeral{1}}$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \lif \lfalse)\lif \lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!A \lif \lfalse) \lif \lfalse$}
  \end{prooftree}

\item $\Proves ((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$
  \begin{prooftree}
    \AxiomC{$\Discharge{(!A \land !B) \lif !C}{\OLMathNumeral{3}}$}
    \AxiomC{$\Discharge{!A}{\OLMathNumeral{2}}$}
    \AxiomC{$\Discharge{!B}{\OLMathNumeral{1}}$}
    \RightLabel{\Intro\land}
    \BinaryInfC{$!A \land !B$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$!C$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!B \lif !C$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!B \lif !C)$}
    \DischargeRule{\Intro\lif}{3}
    \UnaryInfC{$((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$}
  \end{prooftree}

\item $\Proves \lnot(!A \land \lnot !A)$، أي $\Proves (!A \land (!A
  \lif \lfalse))\lif \lfalse$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{\OLMathNumeral{1}}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A \lif \lfalse$}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{\OLMathNumeral{1}}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \land (!A \lif \lfalse))\lif \lfalse$}
  \end{prooftree}

\item $\Proves \lnot\lnot(!A \lor \lnot !A)$، أي $\Proves ((!A \lor
  (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$
  \begin{prooftree}\hskip-3em
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{\OLMathNumeral{2}}$}
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{\OLMathNumeral{2}}$}
    \AxiomC{$\Discharge{!A}{\OLMathNumeral{1}}$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!A \lif \lfalse$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \insertBetweenHyps{\hskip -4em}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$((!A \lor (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$}
  \end{prooftree}
\end{enumerate}

\begin{prop}
  إذا كان~$\OLId{greek-variable}{Gamma} \Proves !A$ في الاستنباط الطبيعي الحدسي، كان
  $\OLId{greek-variable}{Gamma} \Proves !A$ في المنطق الكلاسيكي. وعلى الخصوص، كل~$!A$
  تكون مبرهنة حدسية هي مبرهنة كلاسيكية.
\end{prop}

\begin{proof}
  لأن كل قاعدة من قواعد الاستنباط الطبيعي الحدسي قاعدة في الاستنباط
  الطبيعي الكلاسيكي؛ فكل !!{derivation} حدسية !!a{derivation}
  كلاسيكية أيضًا.
\end{proof}

\begin{prob}
  أقم في المنطق الحدسي !!{derivation}s لل!!{formula}s الآتية:
  \begin{enumerate}
    \item $(\lnot !A \lor !B) \lif (!A \lif !B)$
    \item $\lnot\lnot\lnot !A \lif \lnot !A$
    \item $\lnot\lnot (!A \land !B) \liff (\lnot\lnot !A\land \lnot \lnot !B)$
    \item $\lnot (!A \lor !B) \liff (\lnot !A \land \lnot !B)$
    \item $(\lnot !A \lor \lnot !B) \lif \lnot (!A \land !B)$
    \item $\lnot\lnot ( !A \land !B) \lif  (\lnot\lnot !A \lor
    \lnot\lnot !B)$
  \end{enumerate}
\end{prob}
