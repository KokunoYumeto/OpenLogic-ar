\olsection{النماذج القانونية}

% تصحيح مصدر (0446-I1): يلزم اتساق النسق كيلا تخلو مجموعة العوالم.
نسمّي النموذج المخصوص بالنسق الجهوي النظامي المتسق~$\Sigma$،
وهو~$\mModel{M^\Sigma}$ الذي عوالمه جميع المجموعات
$\Sigma$-المتسقة التامة، \emph{النموذج القانوني}.
والمراد من اختيار علاقة إمكان الوصول~$\OLId{latin-looped}{R}^\Sigma$ والتقييم~$\OLId{latin-looped}{V}^\Sigma$
فيه أن تطابق الحقيقةُ العضويةَ: فتكون ال!!{formula}s الصادقة
عند العالم~$\OLId{greek-variable}{Delta}$ هي عين ال!!{formula}s المنتمية إلى المجموعة~$\OLId{greek-variable}{Delta}$.
وحدّ هذا النموذج ما يأتي.

\begin{defn}
  لتكن~$\Sigma$ منطقًا جهويًا نظاميًا متسقًا. \emph{النموذج القانوني} لـ
  $\Sigma$ هو~$\mModel{M}^\Sigma = \tuple{\OLId{latin-looped}{W}^\Sigma, \OLId{latin-looped}{R}^\Sigma,
    \OLId{latin-looped}{V}^\Sigma}$، حيث:
  \begin{enumerate}
  \item $\OLId{latin-looped}{W}^\Sigma = \Setabs{\OLId{greek-variable}{Delta}}{\OLId{greek-variable}{Delta} \text{ مجموعة
      $\Sigma$-متسقة تامة}}$.
  \item يتحقق~$\OLId{latin-looped}{R}^\Sigma \OLId{greek-variable}{Delta}\OLId{greek-variable}{Delta}'$ إذا وفقط إذا كان
    \iftag{prvBox}{$\Box^{-\OLMathNumeral{1}}\OLId{greek-variable}{Delta} \subseteq
      \OLId{greek-variable}{Delta}'$}{$\Diamond\OLId{greek-variable}{Delta}' \subseteq \OLId{greek-variable}{Delta}$}.
  \item $\OLId{latin-looped}{V}^\Sigma(\OLId{latin-ordinary}{p}) = \Setabs{\OLId{greek-variable}{Delta}}{\OLId{latin-ordinary}{p} \in \OLId{greek-variable}{Delta}}$.
  \end{enumerate}
\end{defn}
