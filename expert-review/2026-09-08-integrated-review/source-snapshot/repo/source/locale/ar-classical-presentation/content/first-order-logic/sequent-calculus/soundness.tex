\olsection{السلامة}

\begin{explain}
السلامة في نظام !!^a{derivation}، كحساب التتابعيات، أن يمتنع عليه
!!{derive} ما لا يصدق. فهي ضمان لأنظمة ال!!{derivation} من تجاوز الحق.
وبحسب المعنى البرهاني الذي نبحثه، نطلب مثلًا الأمور الآتية:
\begin{enumerate}
\item كل~$!A$ !!{derivable} تكون \iftag{FOL}{صالحة}{تحصيل حاصل}؛
\item إذا كانت !!a{sentence} !!{derivable} من جمل أخرى، فهي أيضًا
  نتيجة لها؛
\item إذا كانت مجموعة من ال!!{sentence}s غير متسقة، فهي غير قابلة
  للإرضاء.
\end{enumerate}
هذه خصائص لا يستغني عنها نظام !!a{derivation}. فإذا تخلف شيء منها كان نظام ال!!{derivation} معيبًا، لأنه يتيح
!!{derive} ما لا ينبغي إثباته. فإقامة السلامة لنظام ال!!{derivation} من أهم ما يلزم.

وكل هذه المعاني البرهانية مردّه إلى
!!{derivability} تتابعيات بعينها في حساب التتابعيات؛ فطريق إثبات
(\OLClassicalDigits{1})--(\OLClassicalDigits{3}) أن ننظر في دلالة التتابعيات ال!!{derivable}.
نعرّف أولًا صلاحية التتابعية، أي كونها \emph{صالحة}، ثم نثبت أن كل تتابعية
!!{derivable} صالحة. وحينئذ تأتي (\OLClassicalDigits{1})--(\OLClassicalDigits{3}) نتائج تابعة.
\end{explain}

\begin{defn}
\iftag{FOL}{!!^a{structure}~$\Struct
  M$}{!!^a{valuation}~$\pAssign{\OLId{latin-ordinary}{v}}$} \emph{تُرضي} تتابعية
$\OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}$ إذا وفقط إذا كان إما
$\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!A}$ لبعض $!A \in \OLId{greek-variable}{Gamma}$، أو
$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!A}$ لبعض $!A \in \OLId{greek-variable}{Delta}$.

تكون التتابعية \emph{صالحة} إذا وفقط إذا كانت كل
\iftag{FOL}{!!{structure}~$\Struct M$}{!!{valuation}~$\pAssign{\OLId{latin-ordinary}{v}}$}
ترضيها.
\end{defn}

\begin{thm}[السلامة]
\ollabel{thm:sequent-soundness} إذا كان~$\Log{LK}$ !!{derive}s
$\Theta \Sequent \Xi$، فإن~$\Theta \Sequent \Xi$ صالحة.
\end{thm}

\begin{proof}
خذ~$\pi$، وهو !!a{derivation} لـ~$\Theta \Sequent \Xi$، واستقرئ على عدد استدلالاته~$\OLId{latin-ordinary}{n}$ في~$\pi$.

أما إذا كان العدد~$\OLMathNumeral{0}$، فلا يشتمل~$\pi$ إلا على تتابعية ابتدائية.
والتتابعية~$!A \Sequent !A$ صالحة، إذ لكل
\iftag{FOL}{$\Struct M$}{$\pAssign{\OLId{latin-ordinary}{v}}$} إما
$\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!A}$ وإما
$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!A}$.

وأما إذا زاد العدد على~\OLClassicalDigits{0}، فالتفصيل بحسب الاستدلال الأخير.
مقدماته صالحة بفرض الاستقراء، لأن كل !!{derivation} مؤدٍّ إلى مقدمة منها يشتمل على أقل من~$\OLId{latin-ordinary}{n}$ استدلالات.

ننظر أولًا في الاستدلالات الممكنة ذات المقدمة الواحدة.
\begin{enumerate}
\item الاستدلال الأخير إضعاف. وعندئذ تكون~$\Theta \Sequent \Xi$
  إما $!A, \OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}$ (إذا كان الاستدلال الأخير
  \LeftR{\Weakening})، أو $\OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}, !A$ (إذا كان
  \RightR{\Weakening})، وينتهي ال!!{derivation} بإحدى الصورتين:
  \begin{prooftree}
    \AxiomC{}
    \Deduce$\OLId{greek-variable}{Gamma} \fCenter \OLId{greek-variable}{Delta}$
    \RightLabel{\LeftR{\Weakening}}
    \UnaryInf$!A, \OLId{greek-variable}{Gamma} \fCenter \OLId{greek-variable}{Delta}$
    \DisplayProof\qquad\bottomAlignProof
    \AxiomC{}
    \Deduce$\OLId{greek-variable}{Gamma} \fCenter \OLId{greek-variable}{Delta}$
    \RightLabel{\RightR{\Weakening}}
    \UnaryInf$\OLId{greek-variable}{Gamma} \fCenter \OLId{greek-variable}{Delta}, !A$
  \end{prooftree}
  وبفرض الاستقراء، تكون~$\OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}$ صالحة، أي إنه لكل
  \iftag{FOL}{!!{structure}~$\Struct{M}$}{!!{valuation}~$\pAssign{\OLId{latin-ordinary}{v}}$}،
  يوجد إما $!C \in \OLId{greek-variable}{Gamma}$ بحيث
  $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!C}$، أو $!C \in \OLId{greek-variable}{Delta}$ بحيث
  $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!C}$.
  
  إذا كان $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!C}$ لبعض
  $!C \in \OLId{greek-variable}{Gamma}$، فإن~$!C \in \Theta$ أيضًا؛ ففي الإضعاف الأيسر
  $\Theta = !A, \OLId{greek-variable}{Gamma}$، وأما في الإضعاف الأيمن فـ$\Theta=\OLId{greek-variable}{Gamma}$. ومن ثم يكون
  $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!C}$ لبعض $!C \in \Theta$.
  وبالمثل، إذا كان $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!C}$ لبعض
  $!C \in \OLId{greek-variable}{Delta}$، فإن~$!C \in \Xi$ أيضًا؛ ففي الإضعاف الأيسر
  $\Xi=\OLId{greek-variable}{Delta}$، وفي الإضعاف الأيمن $\Xi=\OLId{greek-variable}{Delta},!A$. ومن ثم يكون
  $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!C}$ لبعض $!C \in \Xi$.
  وبناء عليه تكون~$\Theta \Sequent \Xi$ صالحة.
  \textbf{تنبيه تصحيحي على الأصل (C081-01).} فُصلت حالتا الإضعاف؛ فالمعادلة
  $\Theta=!A,\OLId{greek-variable}{Gamma}$ لا تصدق إلا في الإضعاف الأيسر، أما في الإضعاف
  الأيمن فلا يتغير المقدم، ويزاد $!A$ في التالي.
\item الاستدلال الأخير هو~\LeftR{\lnot}: عندئذ تكون مقدمة الاستدلال
  الأخير~$\OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}, !A$، وتكون نتيجته
  $\lnot !A, \OLId{greek-variable}{Gamma} \Sequent \OLId{greek-variable}{Delta}$، أي إن ال!!{derivation} ينتهي

    \AxiomC{}
    \Deduce$\OLId{greek-variable}{Gamma} \fCenter \OLId{greek-variable}{Delta}, !A$

  (ب) يوجد $!C \in \OLId{greek-variable}{Delta}$ مع
  $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!C}$؛
  (ج) يتحقق $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!A}$.

  \iftag{FOL}{$\Struct{M}$ !!a{structure}}{$\pAssign{\OLId{latin-ordinary}{v}}$ !!a{valuation}}.

  $\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!C}$؛ ولذلك يتحقق أيضًا~$!C \in \Xi$.
  وأما (ج)، فمن~$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{!A}$ يلزم
  $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{\lnot !A}$.
  وحيث إن $\lnot !A \in \Theta$، فقد وجد~$!C \in \Theta$ بحيث
  $\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!C}$.
  فكل وجه يحقق شرط الإرضاء، وتثبت صلاحية

\begin{cor}
\ollabel{cor:entailment-soundness}
إذا كان~$\OLId{greek-variable}{Gamma} \Proves !A$، فإن~$\OLId{greek-variable}{Gamma} \Entails !A$.
\end{cor}

\begin{proof}

$\OLId{greek-variable}{Gamma}_\OLMathNumeral{0} \subseteq \OLId{greek-variable}{Gamma}$ و!!a{derivation} لـ
$\OLId{greek-variable}{Gamma}_\OLMathNumeral{0} \Sequent !A$. وبمقتضى \olref{thm:sequent-soundness}، فإن كل
\iftag{FOL}{!!{structure}~$\Struct{M}$}{!!{valuation}~$\pAssign{\OLId{latin-ordinary}{v}}$}
إما أن تكذّب بعض~$!B \in \OLId{greek-variable}{Gamma}_\OLMathNumeral{0}$ وإما أن تصدّق~$!A$.

$\iftag{FOL}{\Sat{\OLId{latin-looped}{M}}}{\pSat{\OLId{latin-ordinary}{v}}}{\OLId{greek-variable}{Gamma}}$، امتنع الوجه الأول، ولزم

\begin{cor}
\ollabel{cor:consistency-soundness}
إذا كانت~$\OLId{greek-variable}{Gamma}$ قابلة للإرضاء، فهي متسقة.
\end{cor}

\begin{proof}

$\OLId{greek-variable}{Gamma}_\OLMathNumeral{0} \subseteq \OLId{greek-variable}{Gamma}$ و!!a{derivation} لـ
$\OLId{greek-variable}{Gamma}_\OLMathNumeral{0} \Sequent \quad$. ومن \olref{thm:sequent-soundness} أن
$\OLId{greek-variable}{Gamma}_\OLMathNumeral{0} \Sequent \quad$ صالحة.
ومعناه أن لكل
\iftag{FOL}{!!{structure}~$\Struct{M}$}{!!{valuation}~$\pAssign{\OLId{latin-ordinary}{v}}$}
شاهدًا~$!C \in \OLId{greek-variable}{Gamma}_\OLMathNumeral{0}$ بحيث
$\iftag{FOL}{\Sat/{\OLId{latin-looped}{M}}}{\pSat/{\OLId{latin-ordinary}{v}}}{!C}$.

فلا توجد \iftag{FOL}{$\Struct{M}$}{$\pAssign{\OLId{latin-ordinary}{v}}$} ترضي
$\OLId{greek-variable}{Gamma}$؛ أي إن~$\OLId{greek-variable}{Gamma}$ غير قابلة للإرضاء.
\end{proof}
