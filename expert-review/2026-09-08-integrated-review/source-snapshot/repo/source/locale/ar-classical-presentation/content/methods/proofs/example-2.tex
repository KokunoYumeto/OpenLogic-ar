\olsection{مثال ثان}

\begin{prop}
إذا كان $\OLId{latin-looped}{A} \subseteq \OLId{latin-looped}{C}$، فإن $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) = \OLId{latin-looped}{C}$.
\end{prop}

\begin{proof}
  لنفرض أن $\OLId{latin-looped}{A} \subseteq \OLId{latin-looped}{C}$. والمطلوب أن نبين
  $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) = \OLId{latin-looped}{C}$.
  \begin{quote}
    أول ما يلحظ أن العبارة شرطية، وأن تكميمها الكلي مضمر؛ فهي صادقة
    لجميع المجموعات $\OLId{latin-looped}{A}$ و$\OLId{latin-looped}{C}$. فـ$\OLId{latin-looped}{A}$ و$\OLId{latin-looped}{C}$ اسمان لمجموعتين اعتباطيتين،
    وطريق إثبات الشرطية أن نفرض مقدمها ونستخرج تاليها.

    ثم نعمل بالفرض $\OLId{latin-looped}{A} \subseteq \OLId{latin-looped}{C}$. وبسط~$\subseteq$ يبين أن كل
    !!{element} للمجموعة~$\OLId{latin-looped}{A}$ هو !!{element} أيضاً للمجموعة~$\OLId{latin-looped}{C}$.
    فلنثبت هذه الحقيقة، إذ يتكرر استعمالها في البرهان.
  \end{quote}
  فبتعريف~$\subseteq$، ولما كان $\OLId{latin-looped}{A} \subseteq \OLId{latin-looped}{C}$، يكون لكل $\OLId{latin-ordinary}{z}$: إذا
  كان $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ كان $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$.
  \begin{quote}
    بهذا بسطنا ما في الفرض من تعريفات، فننتقل إلى النتيجة. ولإثبات
    $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) = \OLId{latin-looped}{C}$ نرتب الحجة كما في المثال السابق:
    نثبت أن كل !!{element} في $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$ هو
    !!a{element} أيضاً في~$\OLId{latin-looped}{C}$، ثم نثبت بالعكس أن كل !!{element} في
    $\OLId{latin-looped}{C}$ هو !!a{element} في $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. ويختصر القولان
    في~$\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) \subseteq \OLId{latin-looped}{C}$ و
    $\OLId{latin-looped}{C} \subseteq \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. (وهذا رجوع من العبارة
    المبسوطة إلى الاختصار، وفيه تيسير لقراءة البرهان.) ولأنهما اقتران
    وجب إثبات جزأيه. ففي الأول، أي كون كل !!{element} في
    $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$ !!a{element} أيضاً في~$\OLId{latin-looped}{C}$، نأخذ
    $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$ لـ$\OLId{latin-ordinary}{z}$ اعتباطي، ونثبت
    $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$. ثم يرد الفرض $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$، بتعريف
    $\cup$، إلى $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ أو
    $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$. وقد غدا هذا المسلك مألوفاً.
    \end{quote}
  ويكون $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) = \OLId{latin-looped}{C}$ إذا وفقط إذا كان
  $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) \subseteq \OLId{latin-looped}{C}$ و
  $\OLId{latin-looped}{C} \subseteq (\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}))$. فلنثبت أولاً
  $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) \subseteq \OLId{latin-looped}{C}$. لتكن
  $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. فإما أن $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ وإما أن
  $\OLId{latin-ordinary}{z} \in (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$.
  \begin{quote}
    بلغنا فصلاً، والمطلوب استخراج $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$ منه؛ فنقسم النظر إلى
    حالتين.
  \end{quote}
  الحالة \OLClassicalDigits{1}: $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$. ولما كان لكل $\OLId{latin-ordinary}{z}$: إذا كان $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ كان
  $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$، ثبت $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$.
  \begin{quote}
    استعملنا الحقيقة المستخرجة آنفاً من فرض القضية
    $\OLId{latin-looped}{A} \subseteq \OLId{latin-looped}{C}$. فتمت الحالة الأولى، وننتقل إلى الثانية:
    $\OLId{latin-ordinary}{z} \in (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. ويدل $\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$ على \emph{فرق}
    المجموعتين، أي مجموعة !!{element} التي في~$\OLId{latin-looped}{C}$ وليست
    !!{element} في~$\OLId{latin-looped}{A}$. وكل !!{element} في $\OLId{latin-looped}{C}$ غير منتم إلى~$\OLId{latin-looped}{A}$ هو،
    من باب أولى،~!!a{element} في~$\OLId{latin-looped}{C}$.
  \end{quote}
  الحالة \OLClassicalDigits{2}: $\OLId{latin-ordinary}{z} \in (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. ومعناه أن $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$ و
  $\OLId{latin-ordinary}{z} \notin \OLId{latin-looped}{A}$؛ فيثبت، بوجه خاص،~$\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$.
  \begin{quote}
    تم الاتجاه الأول؛ ونقصد الآن إلى الثاني، فنثبت
    $\OLId{latin-looped}{C} \subseteq \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. نأخذ $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$ ونبين
    $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$.
  \end{quote}
  ولتكن الآن $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$. فالمطلوب أن $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ أو
  $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$.
  \begin{quote}
    إن كل !!{element} في $\OLId{latin-looped}{A}$ هو !!{element} أيضاً في $\OLId{latin-looped}{C}$، وإن
    $\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$ جماعة الأشياء التي هي !!{element} في $\OLId{latin-looped}{C}$ وليست
    في $\OLId{latin-looped}{A}$؛ فمن شأن $\OLId{latin-ordinary}{z}$ أن يكون إما في $\OLId{latin-looped}{A}$ وإما في
    $\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$. وقد يخفى وجه اللزوم على من لم يسبق له تصور
    النتيجة، فنقيمه خطوة خطوة. نستعمل حقيقة ظاهرة لا تحتاج هنا إلى
    برهان: $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ أو $\OLId{latin-ordinary}{z} \notin \OLId{latin-looped}{A}$. وهذا هو «مبدأ الثالث المرفوع»:
    لكل عبارة~$\OLId{latin-ordinary}{p}$، إما أن تكون $\OLId{latin-ordinary}{p}$ صادقة وإما أن يكون نفيها صادقاً.
    (والعبارة $\OLId{latin-ordinary}{p}$ هنا هي $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$.) ولما كان هذا فصلاً عدنا إلى
    البرهان بالحالات.
  \end{quote}
  إما أن $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A}$ أو $\OLId{latin-ordinary}{z} \notin \OLId{latin-looped}{A}$. في الحالة الأولى
  $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$. وفي الثانية $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C}$ و
  $\OLId{latin-ordinary}{z} \notin \OLId{latin-looped}{A}$، ومن ثم $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}$. وعندئذ
  $\OLId{latin-ordinary}{z} \in \OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A})$.
  \begin{quote}
    وبهذا تم البرهان، إذ قد ثبت $\OLId{latin-looped}{A} \cup (\OLId{latin-looped}{C} \setminus \OLId{latin-looped}{A}) = \OLId{latin-looped}{C}$.
  \end{quote}
\end{proof}
