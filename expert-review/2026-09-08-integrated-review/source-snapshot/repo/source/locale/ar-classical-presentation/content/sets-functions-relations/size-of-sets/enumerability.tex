\olsection{التعدادات ومجموعات \usetoken{S}{enumerable}}

\begin{editorial}
هذا القسم في تعداد المجموعات، وتعريف التعداد فيه دالة شاملة من
$\PosInt$. ويسلك في البيان طريق التدرج لمن قلّ تمرسه بالرياضيات.
وفي \olref[enm-alt]{sec} بديل أوجز، لكنه يعرّف التعداد تعريفًا
آخر: تقابل مع $\Nat$ أو مع مقطع ابتدائي.
\end{editorial}

\begin{explain}
تقدمت أمثلة عيّنّا فيها المجموعات بسرد !!{element}sها.
فلنعمم القول: كيف نسرد !!{element}s مجموعة، ومتى يتيسر ذلك،
وإن كانت غير منتهية؟
\end{explain}

\begin{defn}[التعداد، بصيغة غير رسمية]
\emph{تعداد} المجموعة~$\OLId{latin-looped}{A}$، على التقريب، قائمة قد لا تتناهى من
!!{element}s~$\OLId{latin-looped}{A}$، يظهر كل !!{element} من $\OLId{latin-looped}{A}$ فيها عند موضع
رقمه منته. فمتى كان لـ$\OLId{latin-looped}{A}$ تعداد، سميت $\OLId{latin-looped}{A}$ \emph{!!{enumerable}}.
\end{defn}

\begin{explain}
ولضبط هذا المعنى ينبغي تمييز أمور:
\begin{enumerate}
\item التعداد قائمة لها أول، ولكل !!{element} سوى الأول
!!{element} واحد يسبقه مباشرة، ولا يكون بين !!{element} القائمة
الأول وأي !!{element} آخر إلا عدد منته من العناصر. فمن ثم لكل
!!{element} موضع منته: موضع !!{element} الأول~$\OLMathNumeral{1}$، والثاني~$\OLMathNumeral{2}$، وهكذا.
\item قد يختلف ترتيب ظهور !!{element}s المجموعة~$\OLId{latin-looped}{A}$ باختلاف تعدادها.
فـ$\OLMathNumeral{4}$، $\OLMathNumeral{1}$، $\OLMathNumeral{25}$، $\OLMathNumeral{16}$،~$\OLMathNumeral{9}$ تعداد لمجموعة المربعات الخمسة الأولى،
كما أن $\OLMathNumeral{1}$، $\OLMathNumeral{4}$، $\OLMathNumeral{9}$، $\OLMathNumeral{16}$،~$\OLMathNumeral{25}$ تعداد لها.
\item والتكرار لا يبطل التعداد: $\OLMathNumeral{1}$، $\OLMathNumeral{1}$، $\OLMathNumeral{2}$، $\OLMathNumeral{2}$، $\OLMathNumeral{3}$، $\OLMathNumeral{3}$،~\dots{}
تعدّد ما تعدده $\OLMathNumeral{1}$، $\OLMathNumeral{2}$، $\OLMathNumeral{3}$،~\dots{}.
\item غير أن الترتيب والتكرار \emph{معتبران} في تعيين التعداد نفسه.
فيجوز أن يبدأ تعداد الصحيحات الموجبة بـ$\OLMathNumeral{1}$، $\OLMathNumeral{2}$، $\OLMathNumeral{3}$، $\OLMathNumeral{1}$، \dots{}؛
إلا أن انتظامه يظهر أسهل في الصورة المألوفة $\OLMathNumeral{1}$، $\OLMathNumeral{2}$، $\OLMathNumeral{3}$، $\OLMathNumeral{4}$،~\dots
\item ولا بد من أول: فـ\dots، $\OLMathNumeral{3}$، $\OLMathNumeral{2}$، $\OLMathNumeral{1}$ ليست تعدادًا للصحيحات
الموجبة، إذ ليس فيها !!{element} أول. ولتر ذلك من التعريف التقريبي،
سل: «ما رقم الموضع الذي يظهر فيه العدد \OLClassicalDigits{76}؟»
\item وكذلك ليست $\OLMathNumeral{1}$، $\OLMathNumeral{3}$، $\OLMathNumeral{5}$، \dots، $\OLMathNumeral{2}$، $\OLMathNumeral{4}$، $\OLMathNumeral{6}$، \dots\@
تعدادًا لها؛ فمواضع الأزواج فيها $\infty + \OLMathNumeral{1}$، $\infty + \OLMathNumeral{2}$، $\infty +
\OLMathNumeral{3}$، وليست مواضع منتهية.
\item والخالية قابلة للتعداد؛ فتعدادها القائمة الخالية!{}
\end{enumerate}
\end{explain}

\begin{prop}
كل مجموعة $\OLId{latin-looped}{A}$ لها تعداد يمكن تعدادها بلا تكرار.
\end{prop}

\begin{proof}
ليكن لـ$\OLId{latin-looped}{A}$ تعداد $\OLId{latin-ordinary}{x}_\OLMathNumeral{1}$، $\OLId{latin-ordinary}{x}_\OLMathNumeral{2}$، \dots{}، وكل $\OLId{latin-ordinary}{x}_\OLId{latin-ordinary}{i}$ فيه !!{element}
من~$\OLId{latin-looped}{A}$. نحذف !!{element}s المكررة فنحصل على تعداد جديد:
نثبت $\OLId{latin-ordinary}{x}_\OLId{latin-ordinary}{i}$ إذا كان !!a{element} من~$\OLId{latin-looped}{A}$ لم يسبق بين
$\OLId{latin-ordinary}{x}_\OLMathNumeral{1}$، \dots، $\OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{i}-\OLMathNumeral{1}}$، ونحذف $\OLId{latin-ordinary}{x}_\OLId{latin-ordinary}{i}$ إذا سبق ظهوره بين
$\OLId{latin-ordinary}{x}_\OLMathNumeral{1}$، \dots،~$\OLId{latin-ordinary}{x}_{\OLId{latin-ordinary}{i}-\OLMathNumeral{1}}$.
\end{proof}

وقد أبانت هذه الحجة حاجتنا إلى حد أدق للتعداد ولمجموعات
!!{enumerable}، حتى يحكم فهمها والبرهان عليها. وهو الآتي.

\begin{defn}[التعداد، بصيغة رسمية]
لكل مجموعة $\OLId{latin-looped}{A} \neq \emptyset$ يسمى \emph{تعدادًا} لها كل دالة
!!{surjective} $\OLId{latin-ordinary}{f} \colon \PosInt \to \OLId{latin-looped}{A}$.
\end{defn}

\begin{explain}
وبيان تكافؤ الحد الصوري والحد التقريبي بالقوائم من وجهين.
أحدهما أن كل !!{surjective} من $\PosInt$ إلى~$\OLId{latin-looped}{A}$ تعدّد~$\OLId{latin-looped}{A}$،
إذ تعطي القائمة $\OLId{latin-ordinary}{f}(\OLMathNumeral{1})$، $\OLId{latin-ordinary}{f}(\OLMathNumeral{2})$، $\OLId{latin-ordinary}{f}(\OLMathNumeral{3})$، \dots.
ولأن $\OLId{latin-ordinary}{f}$ !!{surjective} فلا يفوتها !!{element} من~$\OLId{latin-looped}{A}$، بل يكون
قيمة~$\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n})$ لبعض~$\OLId{latin-ordinary}{n} \in \PosInt$. فكل !!{element} من $\OLId{latin-looped}{A}$ يظهر
عند موضع منته. وقد لا تكون الدالة !!{injective} فيقع التكرار،
وهو جائز كما تقدم.

والآخر أن القائمة الجامعة لكل !!{element}s~$\OLId{latin-looped}{A}$ تعطينا دالة
!!a{surjective} $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$: نجعل $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n})$
!!{element}ها ذا الرتبة $\OLId{latin-ordinary}{n}$، أو !!{element}ها الأخير إن لم يوجد
!!{element} ذو رتبة $\OLId{latin-ordinary}{n}$. وإنما يمتنع الحصول على دالة
!!a{surjective} بهذا الوجه إذا كانت $\OLId{latin-looped}{A}$ خالية، فقائمتها خالية.
فكل قائمة غير خالية تعيّن دالة !!a{surjective} $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$.
\end{explain}

\begin{defn}
  \ollabel{defn:enumerable}
المجموعة~$\OLId{latin-looped}{A}$ !!{enumerable} إذا وفقط إذا كانت خالية أو كان لها تعداد.
\end{defn}

\begin{ex}
تعدّد الصحيحات الموجبة ($\PosInt$) دالة الهوية $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}) = \OLId{latin-ordinary}{n}$،
وتعدّد الطبيعيّات $\Nat$ الدالة $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{n}) = \OLId{latin-ordinary}{n} - \OLMathNumeral{1}$.
\end{ex}

\begin{ex}
خذ الدالتين $\OLId{latin-ordinary}{f}\colon \PosInt \to \PosInt$ و$\OLId{latin-ordinary}{g} \colon \PosInt \to
\PosInt$ المحدّدتين بـ
\begin{align*}
\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}) & = \OLMathNumeral{2}\OLId{latin-ordinary}{n} \text{ و}\\
\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{n}) & = \OLMathNumeral{2}\OLId{latin-ordinary}{n} - \OLMathNumeral{1}
\end{align*}
فالأولى تعدّد الصحيحات الموجبة الزوجية، والثانية تعدّد الفردية.
وليست واحدة منهما تعدادًا لـ$\PosInt$؛ إذ ليست !!{surjective}.
\end{ex}

\begin{prob}
ضع تعدادًا للمربعات الموجبة $\OLMathNumeral{1}$، $\OLMathNumeral{4}$، $\OLMathNumeral{9}$، $\OLMathNumeral{16}$، \dots
\end{prob}

\begin{ex}
الدالة $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}) = (-\OLMathNumeral{1})^{\OLId{latin-ordinary}{n}} \lceil \frac{(\OLId{latin-ordinary}{n}-\OLMathNumeral{1})}{\OLMathNumeral{2}}\rceil$ تعدّد
الصحيحات. والرمز $\lceil \OLId{latin-ordinary}{x} \rceil$ \emph{دالة السقف}: تصعد بـ$\OLId{latin-ordinary}{x}$
إلى أقرب صحيح. فتعداد~$\Int$ بها يجعل $\OLId{latin-ordinary}{f}$ تنتقل في قيم $\Int$

\[
\begin{array}{c c c c c c c c}

\begin{prob}
أثبت أنه إذا كانت $\OLId{latin-looped}{A}$ و$\OLId{latin-looped}{B}$ !!{enumerable} كان $\OLId{latin-looped}{A} \cup \OLId{latin-looped}{B}$ كذلك.
فافرض دالتين، كل منهما !!{surjective}، هما $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$ و$\OLId{latin-ordinary}{g}\colon \PosInt \to \OLId{latin-looped}{B}$،
ثم عرّف دالة !!a{surjective}~$\OLId{latin-ordinary}{h}\colon \PosInt \to \OLId{latin-looped}{A} \cup \OLId{latin-looped}{B}$،
وأقم البرهان على كونها !!{surjective}. ولا تهمل كون $\OLId{latin-looped}{A}$ خالية
أو~$\OLId{latin-looped}{B} = \emptyset$.
\end{prob}

\begin{prob}
أثبت أنه إذا كان $\OLId{latin-looped}{B} \subseteq \OLId{latin-looped}{A}$ وكانت $\OLId{latin-looped}{A}$ !!{enumerable} كانت~$\OLId{latin-looped}{B}$ كذلك.

وعرّف دالة !!a{surjective} $\OLId{latin-ordinary}{g}\colon \PosInt \to \OLId{latin-looped}{B}$ ثم أثبت أنها
!!{surjective}. وما الحكم إذا كان $\OLId{latin-looped}{B} = \emptyset$؟
\end{prob}

\begin{prob}
برهن بالاستقراء على $\OLId{latin-ordinary}{n}$ أن $\OLId{latin-looped}{A}_\OLMathNumeral{1}$، $\OLId{latin-looped}{A}_\OLMathNumeral{2}$، \dots، $\OLId{latin-looped}{A}_\OLId{latin-ordinary}{n}$ إذا كانت

ولك استعمال أن $\OLId{latin-looped}{A}$ و~$\OLId{latin-looped}{B}$ إذا كانتا !!{enumerable} كان $\OLId{latin-looped}{A} \cup \OLId{latin-looped}{B}$ كذلك.
\end{prob}

قد يبدو البدء في سرد !!{element}s المجموعة بـ!!{element} الرتبة~$\OLMathNumeral{1}$
أقرب إلى الطبع، ولكن الرياضيين يؤثرون العد بالطبيعيّات~$\Nat$،
فيتحدثون عن !!{element}s القائمة ذات الرتب $\OLMathNumeral{0}$ ثم $\OLMathNumeral{1}$ ثم $\OLMathNumeral{2}$، وهكذا.
فبإزاء ذلك يجوز تعريف التعداد بأنه دالة !!a{surjective} من $\Nat$
إلى~$\OLId{latin-looped}{A}$. والتعريفان متكافئان.

\begin{prop}\ollabel{prop:enum-shift}
وجود !!a{surjection} $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$ يكافئ وجود
!!a{surjection} $\OLId{latin-ordinary}{g}\colon \Nat \to \OLId{latin-looped}{A}$.
\end{prop}

\begin{proof}
إذا أعطيت !!a{surjection} $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$ ضع $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{n}) = \OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}+\OLMathNumeral{1})$
لكل $\OLId{latin-ordinary}{n} \in \Nat$. فيظهر أن $\OLId{latin-ordinary}{g}\colon \Nat \to \OLId{latin-looped}{A}$ هي !!{surjective}.
وبالعكس، إذا أعطيت !!a{surjection} $\OLId{latin-ordinary}{g}\colon \Nat \to \OLId{latin-looped}{A}$
فضع $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{n}) = \OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{n}-\OLMathNumeral{1})$.
\end{proof}

ومن ذلك نحصل على التالي:

\begin{cor}\ollabel{cor:enum-nat}
المجموعة $\OLId{latin-looped}{A}$ !!{enumerable} إذا وفقط إذا كانت خالية أو وجدت دالة
!!a{surjective} $\OLId{latin-ordinary}{f}\colon \Nat \to \OLId{latin-looped}{A}$.
\end{cor}

تقدم إمكان حذف التكرار من قائمة !!{element}s~$\OLId{latin-looped}{A}$.
ويجري مثله في التعداد الصوري، لكن ضبطه وبرهانه أدق.
فالدالة $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$ لا بد أن تعرّف لكل $\OLId{latin-ordinary}{n} \in \PosInt$.
فإن كانت~$\OLId{latin-looped}{A}$ ذات عدد منته من !!{element}s، لم يمكن أن تعطى
!!{element}s~$\PosInt$ غير المتناهية قيمًا تستوفي !!{element}s~$\OLId{latin-looped}{A}$
ولا تعود إلى قيمة مرتين. فحين تنتهي القائمة الخالية من التكرار
يجب اختيار مجال آخر لـ$\OLId{latin-ordinary}{f}$ له عدد منته من !!{element}s.
وعدم التكرار يقتضي أن تكون $\OLId{latin-ordinary}{f}$ !!{injective}؛ وهي !!{surjective}
كذلك، فالمطلوب !!a{bijection} من منتهية $\{\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{n}\}$
أو من $\PosInt$ إلى~$\OLId{latin-looped}{A}$.

\begin{prop}\ollabel{prop:enum-bij}
إذا كانت $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$ !!{surjective}، أي تعدادًا لـ~$\OLId{latin-looped}{A}$،
وجدت !!a{bijection} $\OLId{latin-ordinary}{g}\colon \OLId{latin-looped}{Z} \to \OLId{latin-looped}{A}$، حيث $\OLId{latin-looped}{Z}$ إما~$\PosInt$
وإما $\{\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{n}\}$ لبعض~$\OLId{latin-ordinary}{n} \in \PosInt$.
\end{prop}

\begin{proof}
نبني $\OLId{latin-ordinary}{g}$ عوديًا، فنضع $\OLId{latin-ordinary}{g}(\OLMathNumeral{1}) = \OLId{latin-ordinary}{f}(\OLMathNumeral{1})$. وإذا تعيّنت $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$
جعلنا $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i}+\OLMathNumeral{1})$ أول قيمة في $\OLId{latin-ordinary}{f}(\OLMathNumeral{1})$، $\OLId{latin-ordinary}{f}(\OLMathNumeral{2})$، \dots{}
لم ترد في $\OLId{latin-ordinary}{g}(\OLMathNumeral{1})$، \dots، $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$، إن وجدت.
فإن كان عدد !!{element}s $\OLId{latin-looped}{A}$ هو $\OLId{latin-ordinary}{n}$ فقط، تعيّنت
$\OLId{latin-ordinary}{g}(\OLMathNumeral{1})$، \dots، $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{n})$ كلها، فحصلت $\OLId{latin-ordinary}{g}\colon \{\OLMathNumeral{1}, \dots, \OLId{latin-ordinary}{n}\} \to \OLId{latin-looped}{A}$.
وإن كانت !!{element}s $\OLId{latin-looped}{A}$ غير متناهية، فلكل $\OLId{latin-ordinary}{i}$ يوجد !!a{element}
من~$\OLId{latin-looped}{A}$ في $\OLId{latin-ordinary}{f}(\OLMathNumeral{1})$، $\OLId{latin-ordinary}{f}(\OLMathNumeral{2})$، \dots، لم يرد بين $\OLId{latin-ordinary}{g}(\OLMathNumeral{1})$، \dots، $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$.
فنحصل حينئذ على $\OLId{latin-ordinary}{g}\colon \PosInt \to \OLId{latin-looped}{A}$.

و$\OLId{latin-ordinary}{g}$ !!{surjective}، إذ كل عنصر من~$\OLId{latin-looped}{A}$ يظهر في
$\OLId{latin-ordinary}{f}(\OLMathNumeral{1})$، $\OLId{latin-ordinary}{f}(\OLMathNumeral{2})$، \dots{} لأن $\OLId{latin-ordinary}{f}$ !!{surjective}، فلا بد أن يؤخذ
قيمة لـ$\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$ عند بعض~$\OLId{latin-ordinary}{i}$. وهي !!{injective} أيضًا؛ فلو كان
$\OLId{latin-ordinary}{j} < \OLId{latin-ordinary}{i}$ و$\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{j}) = \OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$ لكانت $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i})$ قد سبقت بين
$\OLId{latin-ordinary}{g}(\OLMathNumeral{1})$، \dots، $\OLId{latin-ordinary}{g}(\OLId{latin-ordinary}{i}-\OLMathNumeral{1})$، خلافًا لتعريف~$\OLId{latin-ordinary}{g}$.
\end{proof}

\begin{cor}\ollabel{cor:enum-nat-bij}
المجموعة $\OLId{latin-looped}{A}$ !!{enumerable} إذا وفقط إذا كانت خالية أو وجدت
!!a{bijection} $\OLId{latin-ordinary}{f}\colon \OLId{latin-looped}{N} \to \OLId{latin-looped}{A}$، حيث $\OLId{latin-looped}{N} = \Nat$ أو
$\OLId{latin-looped}{N} = \{\OLMathNumeral{0}, \dots, \OLId{latin-ordinary}{n}\}$ لبعض $\OLId{latin-ordinary}{n} \in \Nat$.
\end{cor}

\begin{proof}
فـ$\OLId{latin-looped}{A}$ !!{enumerable} إذا وفقط إذا كانت $\OLId{latin-looped}{A}$ خالية أو وجدت
!!a{surjective} $\OLId{latin-ordinary}{f}\colon \PosInt \to \OLId{latin-looped}{A}$. وبـ\olref{prop:enum-bij}
تكافئ الحالة الأخيرة وجود دالة !!a{bijective}~$\OLId{latin-ordinary}{f}\colon \OLId{latin-looped}{Z} \to \OLId{latin-looped}{A}$،
