\olsection{التجاوز}

\begin{thm}
\ollabel{overspill} كل مجموعة جمل~$\OLId{greek-variable}{Gamma}$ لها نماذج منتهية لا حدّ لكبرها
يكون لها نموذج لا نهائي.
\end{thm}

\begin{proof}
لنزد لغة~$\OLId{greek-variable}{Gamma}$ ثوابت جديدة معدودة $\OLId{latin-ordinary}{c}_\OLMathNumeral{0}$، و$\OLId{latin-ordinary}{c}_\OLMathNumeral{1}$،
و\dots، ولنأخذ المجموعة $\OLId{greek-variable}{Gamma} \cup \{\OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{i} \neq \OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{j} : \OLId{latin-ordinary}{i} \neq
  \OLId{latin-ordinary}{j}\}$. ومعنى أن لـ~$\OLId{greek-variable}{Gamma}$ نماذج منتهية لا حدّ لكبرها أنه،
مهما كان $\OLId{latin-ordinary}{m} >\OLMathNumeral{0}$، وجد $\OLId{latin-ordinary}{n}\ge \OLId{latin-ordinary}{m}$ وكان لـ~$\OLId{greek-variable}{Gamma}$ نموذج عدديته~$\OLId{latin-ordinary}{n}$.
فتكون $\OLId{greek-variable}{Gamma} \cup \{\OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{i} \neq \OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{j} : \OLId{latin-ordinary}{i} \neq \OLId{latin-ordinary}{j}\}$ قابلة للإرضاء
منتهيًا. وبالتراص يوجد لـ~$\OLId{greek-variable}{Gamma} \cup \{\OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{i} \neq \OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{j} : \OLId{latin-ordinary}{i}
\neq \OLId{latin-ordinary}{j}\}$ نموذج~$\Struct M$. ولا يكون مجاله منتهيًا، لأنه يرضي
المتراجحات $\OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{i} \neq \OLId{latin-ordinary}{c}_\OLId{latin-ordinary}{j}$ كلها.
\end{proof}

\begin{prop}
\ollabel{inf-not-fo}
لا توجد جملة~$!A$ من أي لغة من الرتبة الأولى تصدق في
!!a{structure}~$\Struct M$ إذا وفقط إذا كان ال!!{domain}~$\Domain{\OLId{latin-looped}{M}}$
لل!!{structure} لا نهائيًا.
\end{prop}

\begin{proof}
لنفرض وجود~$!A$ على الصفة المذكورة. فيصدق نفيها~$\lnot !A$
في جميع ال!!{structure}s المنتهية، ولا يصدق في غيرها. فيكون
  له نماذج منتهية لا حدّ لكبرها، ولا يكون له نموذج لا نهائي،
خلافًا لـ \olref{overspill}.
\end{proof}
