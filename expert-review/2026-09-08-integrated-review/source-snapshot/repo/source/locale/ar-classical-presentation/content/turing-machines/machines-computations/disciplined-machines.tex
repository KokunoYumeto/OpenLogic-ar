\olsection{الآلات المنضبطة}

\begin{explain}
نظرنا في \olref[hal]{sec} في آلات لها حالة توقف معينة واحدة~$\OLId{latin-ordinary}{h}$، فلا تقف، إن وقفت، إلا في~$\OLId{latin-ordinary}{h}$. فهي من هذه الجهة «أشد انضباطًا» من مطلق آلات تورنغ. ويمكن فرض ضروب أخرى من الانضباط؛ إذ لم نمنع الآلة من محو علامة الطرف في الخانة~$\OLMathNumeral{0}$، ولا من محاولة التحرك يسارًا من الخانة~$\OLMathNumeral{0}$. وعند المحاولة يبقى الرأس، في تعريفنا، على الخانة~$\OLMathNumeral{0}$؛ أما تعريفات غيرنا فتجعله يتوقف. وربما احتجنا أيضًا إلى أن تقف الآلة، متى وقفت، على الخانة~$\OLMathNumeral{1}$.
\end{explain}

\begin{defn}\ollabel{defn:disciplined}
آلة تورنغ~$\OLId{latin-looped}{M}$ \emph{منضبطة} إذا وفقط إذا استوفت:
\begin{enumerate}
\item لها حالة توقف معينة وحيدة~$\OLId{latin-ordinary}{h}$؛
\item لا تقف إلا وهي تمسح الخانة~$\OLMathNumeral{1}$؛
\item لا تمحو $\TMendtape$ من الخانة~$\OLMathNumeral{0}$ أبدًا؛
\item لا تحاول التحرك يسارًا من الخانة~$\OLMathNumeral{0}$ أبدًا.
\end{enumerate}
\end{defn}

\begin{explain}
قد بينا أن كل آلة يمكن تعديلها مع حفظ سلوكها ليكون لها توقف في حالة مخصوصة. فنزيد حالة جديدة~$\OLId{latin-ordinary}{h}$، ونجعل $\delta(\OLId{latin-ordinary}{q},\sigma)=\tuple{\OLId{latin-ordinary}{h},\sigma,\TMstay}$ لكل زوج $\tuple{\OLId{latin-ordinary}{q},\sigma}$ لا تعرف عنده الدالة الأصلية~$\delta$. وكذلك يمكن، وإن طال برهانه، تحويل كل آلة تورنغ~$\OLId{latin-looped}{M}$ إلى آلة منضبطة~$\OLId{latin-looped}{M}'$، لها مدخلات التوقف نفسها والخرج نفسه. فإذا كانت الآلة تقف خارج الخانة~$\OLMathNumeral{1}$ مثلًا، ألحقنا بها تعليمات ترد الرأس يسارًا حتى علامة الطرف، ثم يمينًا خانة واحدة، ثم توقفه. وأما الوفاء بالشروط الأخرى فنترك طريقه للنظر.
\end{explain}

\begin{prop}\ollabel{prop:disciplined}
لكل آلة تورنغ~$\OLId{latin-looped}{M}$ آلة منضبطة~$\OLId{latin-looped}{M}'$، تقف بخرج~$\OLId{latin-looped}{O}$ متى وقفت $\OLId{latin-looped}{M}$ بخرج~$\OLId{latin-looped}{O}$، ولا تقف متى لم تقف~$\OLId{latin-looped}{M}$. ومن ثم فإن كل دالة $\OLId{latin-ordinary}{f}\colon\Nat^\OLId{latin-ordinary}{n}\to\Nat$ تحسبها آلة تورنغ تحسبها آلة منضبطة أيضًا.
\end{prop}

\begin{prob}\label{tur:mac:dis:prob:disc-succ}
أنشئ آلة منضبطة لحساب $\OLId{latin-ordinary}{f}(\OLId{latin-ordinary}{x})=\OLId{latin-ordinary}{x}+\OLMathNumeral{1}$.
\end{prob}

\begin{prob}\label{tur:mac:dis:prob:copier}
أنشئ آلة منضبطة تحول الدخل $\TMstroke^\OLId{latin-ordinary}{n}$ إلى الخرج $\TMstroke^\OLId{latin-ordinary}{n}\concat\TMblank\concat\TMstroke^\OLId{latin-ordinary}{n}$.
\end{prob}
