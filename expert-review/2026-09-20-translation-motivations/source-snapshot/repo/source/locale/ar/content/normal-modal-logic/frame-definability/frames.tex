\olsection{الأطر}

\begin{defn}
  \emph{الإطار} زوج~$\mModel{F} = \tuple{W,R}$ حيث~$W$ مجموعة عوالم
  غير خالية و$R$ علاقة ثنائية على~$W$. ويكون النموذج~$\mModel{M}$
  \emph{مبنيًا على} إطار~$\mModel{F} = \tuple{W,R}$ إذا وفقط إذا كان
  $\mModel{M} = \tuple{W, R, V}$ لتقييم ما~$V$.
\end{defn}

\begin{defn}
  إذا كان~$\mModel{F}$ إطارًا، نقول إن~$!A$ \emph{صالحة في
    $\mModel{F}$،} ونكتب~$\mModel{F} \Entails !A$، إذا كان
  $\mSat{M}{!A}$ لكل نموذج~$\mModel{M}$ مبني على~$\mModel{F}$.

  وإذا كان~$\mClass{F}$ صنفًا من الأطر، نقول إن~$!A$ \emph{صالحة في
    $\mClass{F}$،} ونكتب~$\mClass{F} \Entails !A$، إذا وفقط إذا كان
  $\mModel{F} \Entails !A$ لكل إطار~$\mModel{F} \in \mClass{F}$.
\end{defn}

سبب أهمية الأطر هو أن المطابقة بين المخططات وخصائص علاقة الوصول~$R$
تقع على مستوى الأطر، \emph{لا النماذج}. فمع أن~\Ax{T} صادقة في جميع
النماذج الانعكاسية، فليس كل نموذج تصدق فيه~\Ax{T} انعكاسيًا. لكن
الصحيح \emph{هو} أنه، فضلًا عن كون~\Ax{T} \emph{صالحة} في جميع
\emph{الأطر} الانعكاسية، يكون كل إطار تصلح فيه~\Ax{T} انعكاسيًا أيضًا.

\begin{rem}
الصلاحية في صنف من الأطر حالة خاصة من مفهوم الصلاحية في صنف من
النماذج:~$\mClass{F} \Entails !A$ إذا وفقط إذا كان
$\mClass{C} \Entails !A$، حيث~$\mClass{C}$ صنف جميع النماذج المبنية
على إطار من~$\mClass{F}$.

ومن الواضح أنه إذا كانت !!a{formula} أو مخطط صالحًا، أي صالحًا
بالنسبة إلى صنف \emph{جميع} النماذج، فهو صالح أيضًا بالنسبة إلى أي
صنف أطر~$\mClass{F}$.
\end{rem}
