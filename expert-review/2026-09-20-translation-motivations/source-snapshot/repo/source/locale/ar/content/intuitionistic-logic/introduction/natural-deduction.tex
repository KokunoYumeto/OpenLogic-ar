\olsection{الاستنباط الطبيعي}

الاستنباط الطبيعي من دون قواعد~$\FalseCl$ نظام !!{derivation} معياري
للمنطق الحدسي. ونعيد هنا ذكر القواعد ونبين دوافعها باستعمال تأويل BHK.
ويمكننا في كل حالة أن نفكر في القاعدة على أنها تتيح لنا استنتاج أن
للنتيجة بناءً إذا كان للمقدمات بناءات.

ولأن !!{derivation}s الاستنباط الطبيعي لها افتراضات غير مصروفة، ينبغي
أن نعد !!a{derivation} كهذه، ولنقل إنها لـ$!A$ من افتراضات
!!{undischarged}~$\Gamma$، دالة تحوّل بناءات جميع $!B \in \Gamma$ إلى
بناء لـ~$!A$. وإذا وُجدت !!a{derivation} لـ~$!A$ من غير افتراضات
!!{undischarged}، فثمة بناء لـ~$!A$ بمعنى تأويل BHK. لكننا سنحذف
$\Gamma$ حين لا نحتاج إليها في هذا النقاش.

الافتراض~$!A$ وحده !!a{derivation} لـ~$!A$ من الافتراض
!!{undischarged}~$!A$. وهذا يوافق تأويل BHK: فدالة الهوية على البناءات
تحوّل كل بناء لـ~$!A$ إلى بناء لـ~$!A$.

\subsection{العطف}

\begin{defish}
\AxiomC{$!A$}
\AxiomC{$!B$}
\RightLabel{\Intro{\land}}
\BinaryInfC{$!A \land !B$}
\DisplayProof
\hfill
\begin{tabular}{l}
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!A$}
\DisplayProof
\\[3ex]
\AxiomC{$!A \land !B$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!B$}
\DisplayProof
\end{tabular}
\end{defish}

افترض أن لدينا بناءين $N_1$ و$N_2$ لـ$!A_1$ و$!A_2$ على الترتيب.
عندئذ لدينا أيضًا بناء لـ$!A_1 \land !A_2$، وهو الزوج
$\tuple{N_1, N_2}$.

بناء $!A_1 \land !A_2$ في تأويل BHK هو زوج
$\tuple{N_1, N_2}$. فافترض أن لدينا مثل هذا الزوج. عندئذ لدينا أيضًا
بناء لكل من المعطوفين: $N_1$ بناء لـ~$!A_1$ و$N_2$ بناء لـ$!A_2$.

\subsection{الشرط}

\begin{defish}
  \AxiomC{$\Discharge{!A}{u}$}
  \DeduceC{$!B$}
  \DischargeRule{$\Intro{\lif}$}{u}
  \UnaryInfC{$!A \lif !B$}
  \DisplayProof
\hfill
  \AxiomC{$!A \lif !B$}
  \AxiomC{$!A$}
  \RightLabel{$\Elim{\lif}$}
  \BinaryInfC{$!B$}
  \DisplayProof
\end{defish}

إذا كانت لدينا !!a{derivation} لـ~$!B$ من الافتراض
!!{undischarged}~$!A$، فثمة دالة~$f$ تحوّل بناءات~$!A$ إلى بناءات
$!B$. وهذه الدالة نفسها بناء لـ$!A \lif !B$. ولذلك، إذا كان لمقدمة
$\Intro\lif$ بناء مشروط ببناء لـ~$!A$، فللنتيجة $!A \lif !B$ بناء.

ومن جهة أخرى، افترض وجود بناء~$N$ لـ$!A$ وبناء~$f$ لـ
$!A \lif !B$. إن بناء $!A \lif !B$ دالة تحوّل بناءات~$!A$ إلى
بناءات~$!B$. ولذلك يكون $f(N)$ بناء لـ~$!B$؛ أي إن لنتيجة
$\Elim\lif$ بناءً.

\subsection{الفصل}

\begin{defish}
\begin{tabular}{l}
\AxiomC{$!A$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\\[3ex]
\AxiomC{$!B$}
\RightLabel{\Intro{\lor}}
\UnaryInfC{$!A \lor !B$}
\DisplayProof
\end{tabular}
\hfill
\AxiomC{$!A \lor !B$}
\AxiomC{$\Discharge{!A}{n}$}
\DeduceC{$!C$}
\AxiomC{$\Discharge{!B}{n}$}
\DeduceC{$!C$}
\DischargeRule{\Elim{\lor}}{n}
\TrinaryInfC{$!C$}
\DisplayProof
\end{defish}

إذا كان لدينا بناء~$N_i$ لـ~$!A_i$، أمكننا تحويله إلى بناء
$\tuple{i, N_i}$ لـ~$!A_1 \lor !A_2$. ومن جهة أخرى، افترض أن لدينا
بناءً لـ~$!A_1 \lor !A_2$، أي زوجًا $\tuple{i, N_i}$ يكون فيه $N_i$
بناءً لـ~$!A_i$، ولدينا أيضًا دالتان $f_1$ و$f_2$ تحوّلان بناءات
$!A_1$ و$!A_2$، على الترتيب، إلى بناءات~$!C$. عندئذ يكون
$f_i(N_i)$ بناء لـ~$!C$، وهي نتيجة~$\Elim\lor$.

\subsection{المحال}

\begin{defish}
  \AxiomC{$\lfalse$}
  \RightLabel{\FalseInt}
  \UnaryInfC{$!A$}
  \DisplayProof
\end{defish}

إذا كانت لدينا !!a{derivation} لـ~$\lfalse$ من افتراضات
!!{undischarged}~$!B_1$، \dots، $!B_n$، فثمة دالة
$f(M_1, \dots, M_n)$ تحوّل بناءات $!B_1$، \dots، $!B_n$ إلى بناء
لـ~$\lfalse$. ولأن $\lfalse$ لا بناء لها، فلا يمكن أيضًا أن توجد
بناءات لجميع $!B_1$، \dots، $!B_n$. ومن ثم تملك $f$ أيضًا الخاصية
الآتية: \emph{إذا} كانت $M_1$، \dots، $M_n$ بناءات لـ$!B_1$،
\dots، $!B_n$ على الترتيب، \emph{فإن} $f(M_1, \dots, M_n)$ بناء
لـ~$!A$.

\subsection{قواعد $\lnot$}

لأن $\lnot !A$ معرّفة بأنها $!A \lif \lfalse$، فلا نحتاج، على وجه
الدقة، إلى قواعد لـ~$\lnot$. لكن لو أعطيناها لكانت هكذا:

\begin{defish}
\AxiomC{$\Discharge{!A}{n}$}
\noLine
\DeduceC{$\lfalse$}
\DischargeRule{\Intro{\lnot}}{n}
\UnaryInfC{$\lnot !A$}
\DisplayProof
\hfill
\AxiomC{$\lnot !A$}
\AxiomC{$!A$}
\RightLabel{\Elim{\lnot}}
\BinaryInfC{$\lfalse$}
\DisplayProof
\end{defish}


\subsection{أمثلة على \usetoken{P}{derivation}}

\begin{enumerate}
\item $\Proves !A \lif (\lnot !A \lif \lfalse)$، أي $\Proves !A
  \lif ((!A \lif \lfalse) \lif \lfalse)$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A}{2}$}
    \AxiomC{$\Discharge{!A \lif \lfalse}{1}$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \lif \lfalse)\lif \lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!A \lif \lfalse) \lif \lfalse$}
  \end{prooftree}

\item $\Proves ((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$
  \begin{prooftree}
    \AxiomC{$\Discharge{(!A \land !B) \lif !C}{3}$}
    \AxiomC{$\Discharge{!A}{2}$}
    \AxiomC{$\Discharge{!B}{1}$}
    \RightLabel{\Intro\land}
    \BinaryInfC{$!A \land !B$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$!C$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!B \lif !C$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$!A \lif (!B \lif !C)$}
    \DischargeRule{\Intro\lif}{3}
    \UnaryInfC{$((!A \land !B) \lif !C) \lif (!A \lif (!B \lif !C))$}
  \end{prooftree}

\item $\Proves \lnot(!A \land \lnot !A)$، أي $\Proves (!A \land (!A
  \lif \lfalse))\lif \lfalse$
  \begin{prooftree}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{1}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A \lif \lfalse$}
    \AxiomC{$\Discharge{!A \land (!A \lif \lfalse)}{1}$}
    \RightLabel{\Elim\land}
    \UnaryInfC{$!A$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$(!A \land (!A \lif \lfalse))\lif \lfalse$}
  \end{prooftree}

\item $\Proves \lnot\lnot(!A \lor \lnot !A)$، أي $\Proves ((!A \lor
  (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$
  \begin{prooftree}\hskip-3em
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{2}$}
    \AxiomC{$\Discharge{(!A \lor (!A \lif \lfalse))\lif \lfalse}{2}$}
    \AxiomC{$\Discharge{!A}{1}$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{1}
    \UnaryInfC{$!A \lif \lfalse$}
    \RightLabel{\Intro\lor}
    \UnaryInfC{$!A \lor (!A \lif \lfalse)$}
    \RightLabel{\Elim\lif}
    \insertBetweenHyps{\hskip -4em}
    \BinaryInfC{$\lfalse$}
    \DischargeRule{\Intro\lif}{2}
    \UnaryInfC{$((!A \lor (!A \lif \lfalse))\lif \lfalse)\lif \lfalse$}
  \end{prooftree}
\end{enumerate}

\begin{prop}
  إذا كان $\Gamma \Proves !A$ في الاستنباط الطبيعي للمنطق الحدسي،
  فإن $\Gamma \Proves !A$ في المنطق الكلاسيكي. وبوجه خاص، إذا كانت
  $!A$ مبرهنة حدسية، فهي مبرهنة كلاسيكية أيضًا.
\end{prop}

\begin{proof}
  كل قاعدة في الاستنباط الطبيعي الحدسي قاعدة أيضًا في الاستنباط
  الطبيعي الكلاسيكي، ولذلك تكون كل !!{derivation} في المنطق الحدسي
  !!a{derivation} في المنطق الكلاسيكي أيضًا.
\end{proof}

\begin{prob}
  أعط !!{derivation}s في المنطق الحدسي لل!!{formula}s الآتية:
  \begin{enumerate}
    \item $(\lnot !A \lor !B) \lif (!A \lif !B)$
    \item $\lnot\lnot\lnot !A \lif \lnot !A$
    \item $\lnot\lnot (!A \land !B) \liff (\lnot\lnot !A\land \lnot \lnot !B)$
    \item $\lnot (!A \lor !B) \liff (\lnot !A \land \lnot !B)$
    \item $(\lnot !A \lor \lnot !B) \lif \lnot (!A \land !B)$
    \item $\lnot\lnot ( !A \land !B) \lif  (\lnot\lnot !A \lor
    \lnot\lnot !B)$
  \end{enumerate}
\end{prob}
