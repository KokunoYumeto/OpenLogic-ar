\olsection{ال\usetoken{P}{valuation} والإرضاء}

\begin{defn}[!!^{valuation}s]
لتكن $\{\True, \False\}$ مجموعة قيمتي الصدق، «الصدق»
و«الكذب». يكون \emph{!!{valuation}} للغة $\Lang{L_0}$
دالةً~$\pAssign{v}$ تسند إما $\True$ أو $\False$ إلى
!!{propositional variable}s في اللغة، أي إن $\pAssign{v} \colon
\PVar \to \{\True, \False \}$.
\end{defn}

\begin{defn}
  إذا كان لدينا !!a{valuation}~$\pAssign{v}$، فعرّف دالة القيمة
  $\pValue{v} \colon \Frm[L_0] \to \{\True, \False \}$ استقرائيًا كما يأتي:
  \begin{align*}
    \iftag{prvFalse}{\pValue{v}(\lfalse) & = \False; \\}{}
    \iftag{prvTrue}{\pValue{v}(\ltrue) & = \True; \\}{}
    \pValue{v}(\Obj p_n) & = \pAssign{v}(\Obj p_n); \\
    \iftag{prvNot}{\pValue{v}(\lnot !A) & = \begin{cases}
        \True & \text{إذا كان } \pValue{v}(!A) = \False;\\
        \False & \text{فيما عدا ذلك.}
      \end{cases} \\ }{}
    \iftag{prvAnd}{\pValue{v}(!A \land !B) & = \begin{cases}
        \True &
        \text{إذا كان $\pValue{v}(!A) = \True$ و$\pValue{v}(!B) = \True$؛}\\
        \False &
        \text{إذا كان $\pValue{v}(!A) = \False$ أو $\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvOr}{\pValue{v}(!A \lor !B) & = \begin{cases}
        \True &
        \text{إذا كان $\pValue{v}(!A) = \True$ أو $\pValue{v}(!B) = \True$؛}\\
        \False &
        \text{إذا كان $\pValue{v}(!A) = \False$ و$\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIf}{\pValue{v}(!A \lif !B) & = \begin{cases}
        \True &
        \text{إذا كان $\pValue{v}(!A) = \False$ أو $\pValue{v}(!B) = \True$؛}\\
        \False &
        \text{إذا كان $\pValue{v}(!A) = \True$ و$\pValue{v}(!B) = \False$}.
      \end{cases}\\}{}
    \iftag{prvIff}{\pValue{v}(!A \liff !B) & = \begin{cases}
        \True &
        \text{إذا كان $\pValue{v}(!A) = \pValue{v}(!B)$؛}\\
        \False &
        \text{إذا كان $\pValue{v}(!A) \neq \pValue{v}(!B)$}.
      \end{cases}}{}
\end{align*}
\end{defn}

\begin{explain}
تقابل هذه البنود جداول الصدق الآتية:
\begin{center}
\iftag{prvNot}{
  \begin{tabular}{|c||c|} \hline
    $!A$ & $ \lnot !A$ \\
    \hline \hline
    $\True$ & $\False$ \\
    $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvAnd}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \land !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvOr}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lor !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\True$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\False$ \\
    \hline
  \end{tabular}
}{}%
\iftag{notprvNot,notprvAnd,notprvOr,notprvIf}{}{\\[1em]}
\iftag{prvIf}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \lif !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\True$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\iftag{prvIff}{
  \begin{tabular}{|cc||c|} \hline
    $!A$ & $!B$ & $!A \liff !B$ \\
    \hline \hline
    $\True$ & $\True$ & $\True$ \\
    $\True$ & $\False$ & $\False$ \\
    $\False$ & $\True$ & $\False$ \\
    $\False$ & $\False$ & $\True$ \\
    \hline
  \end{tabular}
}{}
\end{center}
\end{explain}

\begin{prob}
انظر في إضافة رابط ثلاثي إلى $\Lang{L_0}$ نرمز إليه بـ$\diamondsuit$،
ويُعطى حساب قيمته بـ
\begin{gather*}
  \pValue{v}(\diamondsuit( !A , !B , !C ) ) = \begin{cases}
    \pValue{v}( !B ) &
    \text{إذا كان $\pValue{v}(!A) = \True$؛}\\
    \pValue{v}( !C ) &
    \text{إذا كان $\pValue{v}(!A) = \False $}.
  \end{cases}
\end{gather*}
اكتب جدول الصدق لهذا الرابط.
\end{prob}

\begin{thm}[التحديد المحلي]
\ollabel{thm:LocalDetermination} افترض أن كلًا من $\pAssign{v_1}$ و
$\pAssign{v_2}$ هو واحد من ال!!{valuation}s، وأنهما يتفقان في قيم !!{propositional
variable}s واردة في $!A$، أي إن $\pAssign{v_1}(\Obj p_n) =
\pAssign{v_2}(\Obj p_n)$ كلما ورد $\Obj p_n$ في
!!{formula}~$!A$. وعندئذ تعطي دالتا القيمة $\pValue{v_1}$ و$\pValue{v_2}$ أيضًا
القيمة نفسها لـ~$!A$، أي إن $\pValue{v_1}(!A) = \pValue{v_2}(!A)$.
\end{thm}

\begin{proof}
بالاستقراء على $!A$.
\end{proof}

\begin{defn}[الإرضاء]
\ollabel{defn:satisfaction} يمكننا أن نعرّف استقرائيًا مفهوم
  \emph{إرضاء !!a{formula}~$!A$ بواسطة
  !!a{valuation}~$\pAssign{v}$}، أي $\pSat{v}{!A}$، كما يأتي.
  (ونكتب $\pSat/{v}{!A}$ للدلالة على «عدم تحقق $\pSat{v}{!A}$».)
\begin{enumerate}
\tagitem{prvFalse}{%
  \indcase{!A}{\lfalse}{$\pSat/{v}{\indfrm}$.}}{}

\tagitem{prvTrue}{%
  \indcase{!A}{\ltrue}{$\pSat{v}{\indfrm}$.}}{}

\item \indcase{!A}{\Obj p_i}{$\pSat{v}{\indfrm}$
  إذا وفقط إذا كان $\pAssign{v}(\Obj p_i) = \True$.}

\tagitem{prvNot}{%
  \indcase{!A}{\lnot !B}{$\pSat{v}{\indfrm}$ إذا وفقط إذا
    $\pSat/{v}{!B}$.}}{}

\tagitem{prvAnd}{%
  \indcase{!A}{(!B \land !C)}{$\pSat{v}{\indfrm}$ إذا وفقط إذا تحقق كل من $\pSat{v}{!B}$
    و$\pSat{v}{!C}$.}}{}

\tagitem{prvOr}{%
  \indcase{!A}{(!B \lor !C)}{$\pSat{v}{\indfrm}$ إذا وفقط إذا تحقق
    $\pSat{v}{!B}$ أو $\pSat{v}{!C}$ (أو تحققا معًا).}}{}

\tagitem{prvIf}{%
  \indcase{!A}{(!B \lif !C)}{$\pSat{v}{\indfrm}$ إذا وفقط إذا تحقق $\pSat/{v}{!B}$
    أو $\pSat{v}{!C}$ (أو تحققا معًا).}}{}

\tagitem{prvIff}{%
  \indcase{!A}{(!B \liff !C)}{$\pSat{v}{\indfrm}$ إذا وفقط إذا تحقق كل من
    $\pSat{v}{!B}$ و$\pSat{v}{!C}$، أو لم يتحقق $\pSat{v}{!B}$
    ولا $\pSat{v}{!C}$.}}{}
\end{enumerate}
إذا كانت $\Gamma$ مجموعة من !!{formula}s، فإن $\pSat{v}{\Gamma}$ يتحقق إذا وفقط إذا
تحقق $\pSat{v}{!A}$ لكل~$!A \in \Gamma$.
\end{defn}

\begin{prop}\ollabel{prop:sat-value}
  $\pSat{v}{!A}$ إذا وفقط إذا كان $\pValue{v}(!A) = \True$.
\end{prop}

\begin{proof}
  بالاستقراء على~$!A$.
\end{proof}

\end{prob}
