\olsection{\usetoken{P}{derivation} مع \usetoken{S}{identity}}

لكي نتيح $\eq$ في ال!!{derivation}s، نضيف ببساطة مخططات بديهيات جديدة.
ويبقى تعريف !!{derivation} و$\Proves$ كما هو؛ وكل ما هنالك أننا نجيز
البديهيات الجديدة أيضًا.

\begin{defn}[بديهيات !!{identity}]
\begin{align}
\ollabel{ax:id1} & \eq[t][t], \\
\ollabel{ax:id2} & \eq[t_1][t_2] \lif (!B(t_1) \lif
  !B(t_2)),
\end{align}
لكل حدود مغلقة $t$ و$t_1$ و$t_2$.
\end{defn}

\begin{prop}\ollabel{prop:sound}
  البديهيتان \olref{ax:id1} و\olref{ax:id2} صالحتان.
\end{prop}

\begin{proof}
  تمرين.
\end{proof}

\begin{prob}
أثبت \olref[fol][axd][ide]{prop:sound}.
\end{prob}

\begin{prop}\ollabel{prop:iden1}
 $\Gamma \Proves \eq[t][t]$ لكل حد مغلق $t$ ولكل مجموعة~$\Gamma$.
\end{prop}

\begin{prop}\ollabel{prop:iden2}
  لكل حدين مغلقين $t_1$ و$t_2$، إذا كان $\Gamma \Proves !A(t_1)$ و
  $\Gamma \Proves \eq[t_1][t_2]$، فإن $\Gamma \Proves !A(t_2)$.
\end{prop}

\begin{proof}
ال!!{formula}
\[
(\eq[t_1][t_2] \lif (!A(t_1) \lif !A(t_2)))
\]
حالة من~\olref{ax:id2}. وتلزم النتيجة بتطبيق~\MP مرتين.
\end{proof}
