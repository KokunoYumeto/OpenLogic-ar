\olsection{خواص الأطر الزمنية}

لما كانت نماذجنا الزمنية لا تفرض أي شروط على العلاقة~$\prec$، فإن البديهية
الوحيدة من بديهياتنا المألوفة التي تصح في جميع النماذج هي~$K$، أو نظيراها
$K_{\Gtemp}$ و$K_{\Htemp}$:
\begin{align*}
\tag{$K_{\Gtemp}$}  \Gtemp (p \to q) & \to (\Gtemp p \to \Gtemp q)\\
\tag{$K_{\Htemp}$}  \Htemp (p \to q) & \to (\Htemp p \to \Htemp q)
\end{align*}

أما إذا أردنا أن تفرض نماذجنا شروطًا أشد على كيفية تمثيل الزمن، كأن نضمن
أن~$\prec$ ترتيب خطي، فسنحصل على صلاحيات أخرى في نماذجنا.

\begin{table}[t]
  \begin{tabular}{| p{.48\textwidth} || p{.48\textwidth} |}
    \hline
    {\emph{إذا كانت~$\prec$ \dots}} & {\emph{فإن~\dots تصدق في~$\mModel{M}$:}} \\
    \hline \hline
    \emph{متعدية}: \newline
    $\forall u \forall v \forall w ((u \prec v \land v \prec w) \lif u \prec w)$ &
    $\Ftemp \Ftemp p \lif \Ftemp p$  \\
    \hline
    \emph{خطية}: \newline
    $\forall w \forall v (w \prec v \lor w = v \lor v \prec w)$ &
    $(\Ftemp \Ptemp p \lor \Ptemp \Ftemp p) \lif (\Ptemp p \lor p \lor \Ftemp p)$\\
    \hline
    \emph{كثيفة}: \newline
    $\forall w \forall v (w \prec v \to \exists u(w \prec u \land u \prec v))$ &
    $\Ftemp p \lif \Ftemp \Ftemp p$ \\
    \hline
    \emph{غير محدودة (في الماضي)}: \newline
    $\forall w \exists v( v \prec w)$ &
    $\Htemp p \to \Ptemp p$ \\
    \hline
    \emph{غير محدودة (في المستقبل)}: \newline
    $\forall w \exists v( w \prec v)$ &
    $\Gtemp p \to \Ftemp p$ \\
    \hline
  \end{tabular}
  \caption{بعض خواص المقابلة للأطر الزمنية.}
  \ollabel{tab:correspondence}
\end{table}

قد يبدو عدد من الخواص الواردة في~\olref{tab:correspondence} سمات مرغوبًا
فيها لنموذج يقصد به تمثيل الزمن. لكن يجدر التنبيه إلى أنه، مع إمكان فرض أي
شروط نشاء على العلاقة~$\prec$، لا توجد لكل الشروط !!{formula}s مقابلة
يمكن التعبير عنها في لغة المنطق الزمني. فمثلًا، لا تقابل اللاانعكاسية، أي الفكرة المعبر
عنها بـ~$\forall w \lnot (w \prec w)$، صيغة في المنطق الزمني.
