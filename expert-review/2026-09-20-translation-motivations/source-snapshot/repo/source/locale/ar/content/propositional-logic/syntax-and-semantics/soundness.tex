\olsection{سلامة منطق القضايا}

\begin{thm}[السلامة] 
\ollabel{thm:soundness} 
إذا كان $\Gamma\Proves!A$، فإن
$\Gamma \Entails !A$. 
\end{thm}

\begin{proof} 
بالاستقراء في اشتقاق $!A$. إذا كانت $!A$ مسلّمة منطقية، فإن $\Entails
!A$، ومن ثم، بالأولى، $\Gamma \Entails!A$. وبالمثل، إذا كانت $!A \in \Gamma$،
فإن $\Gamma \Entails!A$. ولخطوة الاستقراء، افترض أن $!A$ استُنتجت بقاعدة
\emph{الفصل} من $!C$ و$!C\lif!A$؛ وافترض أيضًا، بموجب فرضيتي الاستقراء، أن
$\Gamma \Entails !C$ و$\Gamma \Entails (!C\lif !A)$. ويعطي البند الثالث
في \olref[sem]{prop:semanticalfacts} النتيجة المطلوبة. 
\end{proof}

\begin{cor} 
إذا كانت $\Gamma$ قابلة للإرضاء، فهي متسقة. ومن ثم فإن
منطق القضايا متسق. 
\end{cor}
