\olsection{\usetoken{P}{formula} المزدوجة}

\begin{defn}\ollabel{def:duals}
  لكل واحدة من !!{formula}s \Ax{T} و\Ax{B} و\Ax{4} و
  \Ax{5} \emph{مزدوجة}، يرمز إليها بمعين ذي دليل سفلي، كما يأتي:
  \begin{align}
    \tag{\Ax{T_\Diamond}} p & \lif \Diamond p\\
    \tag{\Ax{B_\Diamond}} \Diamond\Box p & \lif p\\
    \tag{\Ax{4_\Diamond}} \Diamond\Diamond p & \lif \Diamond p\\
    \tag{\Ax{5_\Diamond}} \Diamond\Box p & \lif \Box p
    \end{align}
\end{defn}

تحصل كل واحدة من !!{formula}s المزدوجة أعلاه من !!{formula} المقابلة
بإحلال $\lnot p$ محل $p$، وأخذ العكس النقيض، واستبدال
$\lnot\Box\lnot$ بالصيغة $\Diamond$، واستبدال
$\lnot\Diamond\lnot$ بالصيغة~$\Box$.
أما \Ax{D}، أي $\Box!A \lif \Diamond!A$، فهي مزدوجة نفسها بهذا المعنى.

\begin{prop}\ollabel{prop:dualsys}
  لكل !!{formula}~$!A$ في \olref[nml][prf][dua]{def:duals}:
  $\Log{K}!A = \Log{K}!A_{\Diamond}$.
\end{prop}
\begin{proof}
  تمرين.
\end{proof}
\begin{prob}
  أثبت \olref[nml][prf][dua]{prop:dualsys}.
\end{prob}
