\olsection{مقدمة}

\begin{explain}
الاستقراء تقنية برهانية شائعة في الرياضيات والمنطق. ولعلك تعرفه من
الرياضيات، حيث يُستعمل الاستقراء على الأعداد الطبيعية. ويتبين أن أنواعًا
كثيرة مختلفة من مجموعات الأشياء الرياضية والمنطقية تصلح للاستقراء.

وعمومًا، يتيح الاستقراء إثبات دعوى كلية، أي بيان أن كل شيء من نوع معين
يتصف بخاصية ما. ويكون الاستقراء مفيدًا، بوجه خاص، عندما تكون طريقة
«الإدخال الكلي» في البرهان شديدة التعقيد. ولا يعمل الاستقراء إلا على
الأشياء الرياضية المنشأة بطرائق مخصوصة: فإذا كان كل عنصر في المجموعة
إما أساسيًا وإما مبنيًا من عناصر أساسية في عدد متناه من الخطوات، أمكننا
استعمال الاستقراء عليها. وبتعبير أدق:
\end{explain}

\begin{defn}
لتكن $f$ دالة $n$-موضعية. تكون المجموعة $S$ \emph{مغلقة} تحت $f$
إذا وفقط إذا كان $f(x_1,\dots,x_n) \in S$ كلما كانت
$x_1,\dots,x_n \in S$ وكانت قيمة $f$ معرفة لهذه المدخلات.
\end{defn}

\begin{explain}
فمثلًا، مجموعة الأعداد الطبيعية ($\mathbb{N}$) مغلقة تحت دالة التالي
$f(x) = x+1$. فإذا كان $x$ عددًا طبيعيًا، كان $x+1$ طبيعيًا أيضًا.
لكن الأعداد الطبيعية ليست مغلقة تحت دالة السابق $f(x) = x-1$، لأن
$0$ عدد طبيعي، في حين أن $-1$ ليس كذلك.
\end{explain}

\begin{defn}
يقال إن الخاصية $P$ \emph{محفوظة تحت} دالة $n$-موضعية $f$ إذا كانت،
لكل مدخلات $a_1,\dots,a_n$ في مجال $f$، الخصائص
$P(a_1),\dots,P(a_n)$ تستلزم $P(f(a_1,\dots,a_n))$.
\end{defn}

\begin{explain}
خاصية «كون العدد زوجيًا» محفوظة تحت الدالة $f(x) = x+2$، لأنه إذا
كان $x$ زوجيًا كان $x+2$ زوجيًا أيضًا. أما تحت دالة التالي فلا تُحفظ
هذه الخاصية، لأن $x+1$ فردي متى كان $x$ زوجيًا.

ينجح الاستقراء على الأعداد الطبيعية لأن كل عدد طبيعي يمكن بلوغه من
$0$ بتطبيق دالة التالي عددًا متناهيًا من المرات. وهذه سمة كل مجموعة
يمكننا إجراء الاستقراء عليها.
\end{explain}

\begin{defn}[الاستقراء على $\Nat$]
إذا كان $0$ يتصف بخاصية ما $P$، وكانت $P$ محفوظة تحت دالة التالي،
فإن كل عدد طبيعي يتصف بالخاصية $P$.
\end{defn}

\begin{ex} مجموع الأعداد الطبيعية من $0$ إلى $n$ هو $\frac{n(n+1)}{2}$. \\

\textbf{حالة الأساس.} مجموع الأعداد الطبيعية من $0$ إلى $0$ هو $0=
\frac{0\cdot 1}{2}$.\\

\textbf{خطوة الاستقراء.} افترض أن الخاصية تصدق على $k$. سنبين أنها
تصدق على $k+1$. وبعبارة أخرى، افترض أن $P(k)$ تصدق، أي إن
\[ 0 + 1 + \ldots + k = \frac{k(k+1)}{2} \]
وسنبين أن $P(k+1)$ تصدق، أي إن
\[ 0 + 1 + \ldots + k + (k+1) = \frac{(k+1)(k+2)}{2} \]
وذلك باستعمال فرض أن $P(k)$ تصدق.

\begin{align*}
0 + 1 + \ldots + k &= \frac{k(k+1)}{2} \tag{فرض الاستقراء} \\
0 + 1 + \ldots + k + (k+1) &= \frac{k(k+1)}{2} + (k+1)
\tag{إضافة $k+1$ إلى الطرفين} \\
&= \frac{k(k+1) + 2(k+1)}{2} \\
&= \frac{k^2 + k + 2k +2}{2} \\
&=\frac{(k+1)(k+2)}{2}
\end{align*}
وهكذا تتم خطوة الاستقراء، ونكون قد برهنا الدعوى.
\end{ex}

\begin{explain}
بالنسبة إلى !!{formula}s، تكون العناصر الأساسية هي !!{formula}s الذرية.
وتُنشأ !!{formula}s الأكثر تركيبًا بضم !!{formula}s أقل تعقيدًا بعضها إلى
بعض باستعمال الروابط. ويمكن النظر إلى كل رابط على أنه يقابل دالة على
!!{formula}s تُرجع !!{formula} جديدة بتطبيق الرابط على مدخلاتها. فمثلًا،
يمكن كتابة الدالة التي تصل بين اثنتين من !!{formula}s، هما $!A$ و$!B$، بالاقتران
على الصورة $\mathcal E _\land (!A, !B) = !A \land !B$. ويمكن أن نسمي
أمثال هذه \emph{دوال بناء الصيغ}. ومجموعة !!{formula}s مغلقة تحت هذه الدوال:
فكلما كان كل من $!A$ و$!B$ من !!{formula}s، كان كل من $\lnot !A$
و$!A \land !B$، وغيرهما، صيغةً كذلك.
\end{explain}

\begin{defn}[الاستقراء على !!{formula}s]
إذا كانت كل !!{formula} ذرية تتصف بالخاصية $P$، وكانت $P$ محفوظة
تحت الدوال التي تبني !!{formula}، فإن كل !!{formula} تتصف بالخاصية $P$.
\end{defn}

\begin{explain}
يقترح هذا التعريف «وصفة» للبراهين الاستقرائية على !!{formula}s. فإذا
طُلب منا بيان أن كل !!{formula} تتصف بالخاصية $P$، اتبعنا الخطوات الآتية:\\

\textbf{حالة الأساس.} لتكن $!A$ !!{formula} ذرية. [\ldots]
إذن تتصف $!A$ بالخاصية $P$.

\textbf{خطوة الاستقراء.} لتكن $!A$ و$!B$ من !!{formula}s، ولتتصف
كلتاهما بالخاصية $P$.

حالة $\lnot$: [\ldots] إذن تتصف $\lnot !A$ بالخاصية $P$.

حالة $\land$: [\ldots] إذن تتصف $!A \land !B$ بالخاصية $P$.

حالة $\lor$: [\ldots] إذن تتصف $!A \lor !B$ بالخاصية $P$.

حالة $\lif$: [\ldots] إذن تتصف $!A \lif !B$ بالخاصية $P$.

حالة $\liff$: [\ldots] إذن تتصف $!A \liff !B$ بالخاصية $P$.

حالة $\lforall[]$: [\ldots] إذن تتصف $\lforall[x] !A$ بالخاصية $P$.

حالة $\lexists[]$: [\ldots] إذن تتصف $\lexists[x] !A$ بالخاصية $P$. \\

إذن تتصف كل !!{formula} بالخاصية $P$.
\end{explain}
