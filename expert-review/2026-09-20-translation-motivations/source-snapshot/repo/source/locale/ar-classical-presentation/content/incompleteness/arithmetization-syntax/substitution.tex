\olsection{الإحلال}

الإحلال، كما تقدم، أن نجعل الحد~${\OLId{latin-ordinary}{t}}$ محل كل ورود حر
لـ!!a{variable}~${\OLId{latin-ordinary}{u}}$ في !!a{formula}~$!A$؛ ورمزه
$\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{u}}}$. فإذا نقلنا هذه العملية إلى أعداد غودل
لل!!{variable}s وال!!{formula}s والحدود، حصلت دالة
عودية بدائية.

\begin{prop}\ollabel{prop:subst-primrec}
يمكن تعيين دالة عودية بدائية $\fn{Subst}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}})$ تحقق
\[
\fn{Subst}(\Gn{!A}, \Gn{{\OLId{latin-ordinary}{t}}}, \Gn{{\OLId{latin-ordinary}{u}}}) = \Gn{\Subst{!A}{{\OLId{latin-ordinary}{t}}}{{\OLId{latin-ordinary}{u}}}}.
\]
\end{prop}

\begin{proof}
نبدأ بدالة مساعدة $\fn{hSubst}$، ونحدّها بالعودية البدائية:
\begin{multline*}
\begin{aligned}
\fn{hSubst}({\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}, {\OLId{latin-ordinary}{z}}, {\OLMathNumeral{0}}) & = \emptyseq \\
\fn{hSubst}({\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}, {\OLId{latin-ordinary}{z}}, {\OLId{latin-ordinary}{i}}+{\OLMathNumeral{1}}) & =
\end{aligned}\\
\begin{cases}
\fn{hSubst}({\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}, {\OLId{latin-ordinary}{z}}, {\OLId{latin-ordinary}{i}}) \concat {\OLId{latin-ordinary}{y}} & \text{إذا صدقت $\fn{FreeOcc}({\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{z}}, {\OLId{latin-ordinary}{i}})$} \\
\fn{append}(\fn{hSubst}({\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}, {\OLId{latin-ordinary}{z}}, {\OLId{latin-ordinary}{i}}), ({\OLId{latin-ordinary}{x}})_{{\OLId{latin-ordinary}{i}}}) & \text{في غير ذلك.}
\end{cases}
\end{multline*}
ثم نجعل $\fn{Subst}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}})$ مساويةً لـ
$\fn{hSubst}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}},\len{{\OLId{latin-ordinary}{x}}})$.
\end{proof}

\begin{prop}
\ollabel{prop:free-for}
العلاقة $\fn{FreeFor}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}})$ عودية بدائية؛ ومعناها أن الحد ذا
عدد غودل~${\OLId{latin-ordinary}{y}}$ !!{free for} المتغير ذي عدد غودل~${\OLId{latin-ordinary}{z}}$
في الصيغة ذات عدد غودل~${\OLId{latin-ordinary}{x}}$.
\end{prop}

\begin{proof} برهانه متروك للتمرين. \end{proof}

\begin{prob}
برهن \olref[inc][art][sub]{prop:free-for}.
\end{prob}
