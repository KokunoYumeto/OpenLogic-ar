\olsection{العلائق العودية البدائية}

\begin{defn}
العلاقة ${\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}})$ عودية بدائية متى كانت دالتها المميزة
\[
\Char{{\OLId{latin-looped}{R}}}(\vec {\OLId{latin-ordinary}{x}}) = \left\{
  \begin{array}{ll}
  {\OLMathNumeral{1}} & \mbox{إذا صدقت ${\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}})$} \\
  {\OLMathNumeral{0}} & \mbox{في غير ذلك}
  \end{array}
\right.
\]
عودية بدائية.
\end{defn}

فمعنى القول بعودية العلاقة ${\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}})$ البدائية أنها
تأخذ صورة $\Char{{\OLId{latin-looped}{R}}}(\vec {\OLId{latin-ordinary}{x}})={\OLMathNumeral{1}}$، وأن $\Char{{\OLId{latin-looped}{R}}}$ دالة
عودية بدائية لا تعطي، لأي دخل، إلا ${\OLMathNumeral{1}}$ أو ${\OLMathNumeral{0}}$. ومن أمثلة ذلك العلاقة
$\fn{IsZero}({\OLId{latin-ordinary}{x}})$، الصادقة إذا وفقط إذا كان ${\OLId{latin-ordinary}{x}}={\OLMathNumeral{0}}$؛ فإن دالتها المميزة هي
$\Char{\fn{IsZero}}$، وتعريفها بالعودية البدائية كما يأتي:
\begin{align*}
\Char{\fn{IsZero}}({\OLMathNumeral{0}}) & = {\OLMathNumeral{1}},\\
\Char{\fn{IsZero}}({\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}}) & = {\OLMathNumeral{0}}.
\end{align*}

ويجوز تركيب العلائق مع دوال عودية بدائية
أخرى، فلا تخرج بذلك عن العودية البدائية. ومن هذا نحصل على العلاقتين الآتيتين:
\begin{enumerate}
\item علاقة المساواة، ${\OLId{latin-ordinary}{x}}={\OLId{latin-ordinary}{y}}$، المعرفة بـ
$\fn{IsZero}(\left|{\OLId{latin-ordinary}{x}}-{\OLId{latin-ordinary}{y}}\right|)$.
\item علاقة الأصغر من أو المساوي، ${\OLId{latin-ordinary}{x}}\leq {\OLId{latin-ordinary}{y}}$، المعرفة بـ
$\fn{IsZero}({\OLId{latin-ordinary}{x}}\tsub {\OLId{latin-ordinary}{y}})$.
\end{enumerate}

\begin{prop}
  تحفظ العمليات البوليانية العودية البدائية للعلائق؛ فمتى
  كانت ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}})$ و${\OLId{latin-looped}{Q}}(\vec {\OLId{latin-ordinary}{x}})$ عوديتين بدائيتين، لزمت العودية البدائية
  للعلائق الآتية:
  \begin{enumerate}
  \item $\lnot {\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}})$
  \item ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}}) \land {\OLId{latin-looped}{Q}}(\vec {\OLId{latin-ordinary}{x}})$
  \item ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}}) \lor {\OLId{latin-looped}{Q}}(\vec {\OLId{latin-ordinary}{x}})$
  \item ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}}) \lif {\OLId{latin-looped}{Q}}(\vec {\OLId{latin-ordinary}{x}})$
  \end{enumerate}
\end{prop}

\begin{proof}
  لنفرض عودية ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}})$ و${\OLId{latin-looped}{Q}}(\vec {\OLId{latin-ordinary}{x}})$ البدائية؛ فدالتاهما
  المميزتان $\Char{{\OLId{latin-looped}{P}}}$ و$\Char{{\OLId{latin-looped}{Q}}}$ عوديتان بدائيتان. والمطلوب إثبات العودية
  البدائية للدوال المميزة لـ$\lnot {\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}})$ ولسائر العلائق المذكورة.
  \[
  \Char{\lnot {\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}) = \begin{cases}
    {\OLMathNumeral{0}} & \text{إذا كانت $\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}) = {\OLMathNumeral{1}}$}\\
    {\OLMathNumeral{1}} & \text{في غير ذلك}
  \end{cases}
  \]
  ويكفي لتعريف $\Char{\lnot {\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}})$ أن نجعلها
  ${\OLMathNumeral{1}}\tsub\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}})$.
  \[
  \Char{{\OLId{latin-looped}{P}} \land {\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}}) = \begin{cases}
    {\OLMathNumeral{1}} & \text{إذا كانت $\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}) = \Char{{\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}}) = {\OLMathNumeral{1}}$}\\
    {\OLMathNumeral{0}} & \text{في غير ذلك}
  \end{cases}
  \]
  وأما $\Char{{\OLId{latin-looped}{P}}\land {\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}})$ فتعريفها أن نجعلها
  $\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}})\cdot\Char{{\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}})$، أو بأنها
  $\fn{min}(\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}),\Char{{\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}}))$. وبالمثل،
  \begin{align*}
    \Char{{\OLId{latin-looped}{P}} \lor {\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}}) & = \fn{max}(\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}), \Char{{\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}})) \text{، و}\\
    \Char{{\OLId{latin-looped}{P}} \lif {\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}}) & = \fn{max}({\OLMathNumeral{1}} \tsub
  \Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}), \Char{{\OLId{latin-looped}{Q}}}(\vec {\OLId{latin-ordinary}{x}})).
  \end{align*}
\end{proof}

\begin{prop}
  والتكميم المحدود كذلك يحفظ العودية البدائية؛ فإذا
  كانت العلاقة ${\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})$ عودية بدائية، كانت العلاقتان
  \begin{align*}
    & \bforall{{\OLId{latin-ordinary}{z}} < {\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{z}})} \text{، و}\\
    & \bexists{{\OLId{latin-ordinary}{z}} < {\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{z}})}
  \end{align*}
  عوديتين بدائيتين. وتصدق $\bforall{{\OLId{latin-ordinary}{z}}<{\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$ على $\vec {\OLId{latin-ordinary}{x}}$
  و${\OLId{latin-ordinary}{y}}$ إذا وفقط إذا صدقت ${\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})$ لكل~${\OLId{latin-ordinary}{z}}$ أصغر من~${\OLId{latin-ordinary}{y}}$، وكذلك
  الحال في $\bexists{{\OLId{latin-ordinary}{z}}<{\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$.
\end{prop}

\begin{proof}
  اصطلاحنا أن $\bforall{{\OLId{latin-ordinary}{z}}<{\OLMathNumeral{0}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$ صادقة، إذ ليس
  ثمة ${\OLId{latin-ordinary}{z}}$ أصغر من~${\OLMathNumeral{0}}$، وأن
  $\bexists{{\OLId{latin-ordinary}{z}}<{\OLMathNumeral{0}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$ كاذبة. والمكمم الكلي المحدود بمنزلة
  جداء منته أو تكرار لأخذ الأصغر؛ فمتى وضعنا
  ${\OLId{latin-looped}{P}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}})\defiff\bforall{{\OLId{latin-ordinary}{z}}<{\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$، عرّفنا
  $\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}})$ هكذا:
  \begin{align*}
    \Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}, {\OLMathNumeral{0}}) & = {\OLMathNumeral{1}}\\
    \Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}+{\OLMathNumeral{1}}) & =
    \fn{min}(\Char{{\OLId{latin-looped}{P}}}(\vec {\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}}), \Char{{\OLId{latin-looped}{R}}}(\vec {\OLId{latin-ordinary}{x}}, {\OLId{latin-ordinary}{y}})).
  \end{align*}
  وأما التكميم الوجودي المحدود فيعرّف بمثل ذلك باستعمال $\fn{max}$. ولنا أيضًا
  أن نرده إلى التكميم الكلي المحدود، اعتمادًا على التكافؤ
  $\bexists{{\OLId{latin-ordinary}{z}}<{\OLId{latin-ordinary}{y}}}{{\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}\liff
  \lnot\bforall{{\OLId{latin-ordinary}{z}}<{\OLId{latin-ordinary}{y}}}{\lnot {\OLId{latin-looped}{R}}(\vec {\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{z}})}$. ومن المفيد هنا أن المكمم
  المحدود الذي صورته $\bexists{{\OLId{latin-ordinary}{x}}\leq {\OLId{latin-ordinary}{y}}}{\dots {\OLId{latin-ordinary}{x}}\dots}$ يكافئ
  $\bexists{{\OLId{latin-ordinary}{x}}<{\OLId{latin-ordinary}{y}}+{\OLMathNumeral{1}}}{\dots {\OLId{latin-ordinary}{x}}\dots}$.
\end{proof}

\begin{prob}
  أثبت العودية البدائية للعلاقة الثلاثية الحجج ${\OLId{latin-ordinary}{x}}\equiv {\OLId{latin-ordinary}{y}}\mod {\OLId{latin-ordinary}{n}}$ (التطابق بترديد
  ~ ${\OLId{latin-ordinary}{n}}$).
\end{prob}

ومما ينفعنا من الدوال العودية البدائية الدالة الشرطية
$\fn{cond}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}})$ المعرفة بـ
\begin{align*}
  \fn{cond}({\OLId{latin-ordinary}{x}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}}) & = \begin{cases}
  {\OLId{latin-ordinary}{y}} & \text{إذا كان ${\OLId{latin-ordinary}{x}}={\OLMathNumeral{0}}$} \\
  {\OLId{latin-ordinary}{z}} & \text{في غير ذلك}.
\end{cases}
\intertext{وهي معرفة عوديًا بـ}
\fn{cond}({\OLMathNumeral{0}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}}) & = {\OLId{latin-ordinary}{y}},\\
\fn{cond}({\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}},{\OLId{latin-ordinary}{y}},{\OLId{latin-ordinary}{z}}) & = {\OLId{latin-ordinary}{z}}.
\end{align*}
وبها يتبين جواز تعريف الدوال العودية البدائية بالحالات، متى كانت
العلائق التي تفصل تلك الحالات عودية بدائية:

\begin{prop}
إذا كانت ${\OLId{latin-ordinary}{g}}_{\OLMathNumeral{0}}(\vec {\OLId{latin-ordinary}{x}})$، …،~${\OLId{latin-ordinary}{g}}_{\OLId{latin-ordinary}{m}}(\vec {\OLId{latin-ordinary}{x}})$ دوال عودية بدائية، وكانت
${\OLId{latin-looped}{R}}_{\OLMathNumeral{0}}(\vec {\OLId{latin-ordinary}{x}})$، …،~${\OLId{latin-looped}{R}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}(\vec {\OLId{latin-ordinary}{x}})$ علائق عودية بدائية، فإن الدالة
${\OLId{latin-ordinary}{f}}$ المعرفة بـ
\[
{\OLId{latin-ordinary}{f}}(\vec {\OLId{latin-ordinary}{x}}) = \begin{cases}
    {\OLId{latin-ordinary}{g}}_{\OLMathNumeral{0}}(\vec {\OLId{latin-ordinary}{x}}) & \text{إذا صدقت ${\OLId{latin-looped}{R}}_{\OLMathNumeral{0}}(\vec{{\OLId{latin-ordinary}{x}}})$} \\
    {\OLId{latin-ordinary}{g}}_{\OLMathNumeral{1}}(\vec {\OLId{latin-ordinary}{x}}) & \text{إذا صدقت ${\OLId{latin-looped}{R}}_{\OLMathNumeral{1}}(\vec{{\OLId{latin-ordinary}{x}}})$ ولم تصدق ${\OLId{latin-looped}{R}}_{\OLMathNumeral{0}}(\vec{{\OLId{latin-ordinary}{x}}})$} \\
    \vdots & \\
    {\OLId{latin-ordinary}{g}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}(\vec {\OLId{latin-ordinary}{x}}) & \text{إذا صدقت ${\OLId{latin-looped}{R}}_{{\OLId{latin-ordinary}{m}}-{\OLMathNumeral{1}}}(\vec{{\OLId{latin-ordinary}{x}}})$ ولم تصدق أي
      من سابقاتها}
    \\
    {\OLId{latin-ordinary}{g}}_{\OLId{latin-ordinary}{m}}(\vec {\OLId{latin-ordinary}{x}}) & \mbox{في غير ذلك}
\end{cases}
\]
عودية بدائية أيضًا.
\end{prop}

\begin{proof}
  أما عند ${\OLId{latin-ordinary}{m}}={\OLMathNumeral{1}}$ فالدالة المطلوبة عين المعرّفة بـ
  \[
  {\OLId{latin-ordinary}{f}}(\vec {\OLId{latin-ordinary}{x}}) = \fn{cond}(\Char{\lnot {\OLId{latin-looped}{R}}_{\OLMathNumeral{0}}}(\vec {\OLId{latin-ordinary}{x}}),{\OLId{latin-ordinary}{g}}_{\OLMathNumeral{0}}(\vec {\OLId{latin-ordinary}{x}}),{\OLId{latin-ordinary}{g}}_{\OLMathNumeral{1}}(\vec
  {\OLId{latin-ordinary}{x}})).
  \]
  وأما إذا زاد ${\OLId{latin-ordinary}{m}}$ على~${\OLMathNumeral{1}}$، ركبنا تعريفات من هذا القبيل فحصل المطلوب.
\end{proof}
