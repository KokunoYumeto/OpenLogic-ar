\olsection{مجموعات ال\usetoken{P}{sentence} المتسقة الكاملة}

\begin{defn}[المجموعة الكاملة]
\ollabel{def:complete-set} تكون مجموعة~${\OLId{greek-variable}{Gamma}}$ من ال!!{sentence}s
\emph{!!{complete}} إذا وفقط إذا تحقق لكل !!{sentence}~$!A$ أن
$!A \in {\OLId{greek-variable}{Gamma}}$ أو $\lnot !A \in {\OLId{greek-variable}{Gamma}}$.
\end{defn}

\begin{explain}
مجموعات الجمل ال!!^{complete} تفصل في كل سؤال: فأيّ !!{sentence}~$!A$
أخذنا، وجدنا في ${\OLId{greek-variable}{Gamma}}$ حكمًا بصدق $!A$ أو كذبها. وليست فائدة
المجموعات ال!!{complete} مقصورة على برهان الاكتمال؛ فإن النظرية
ال!!{complete} إذا كانت قابلة للأكسمة كانت قابلة للقرار.
\end{explain}

\begin{explain}
تقوم المجموعات المتسقة ال!!^{complete} بدور أساس في البرهان؛ إذ سنضمن
أن كل مجموعة متسقة من ال!!{sentence}s~${\OLId{greek-variable}{Gamma}}$ تقع ضمن مجموعة متسقة
!!a{complete}~${\OLId{greek-variable}{Gamma}}^*$. ففي المجموعة المتسقة !!^a{complete}،
لكل !!{sentence}~$!A$، إما $!A$ وإما نفيها $\lnot !A$، ولا تجتمعان.
ويشمل هذا \iftag{FOL}{ال!!{sentence}s الذرية}{المتغيرات القضوية}؛
ومن ثم تتيح لنا المجموعة المتسقة !!a{complete}\iftag{FOL}{، في لغة
زِيدت فيها !!{constant}s على وجه مناسب،}{} بناء
\iftag{FOL}{!!a{structure}}{!!a{valuation}} نعيّن فيها
\iftag{FOL}{تفسير ال!!{predicate}s}{قيم صدق المتغيرات القضوية}
بحسب ما في~${\OLId{greek-variable}{Gamma}}^*$ من
\iftag{FOL}{ال!!{sentence}s الذرية}{ال!!{propositional variable}s}.
ثم نثبت أن هذه ال\iftag{FOL}{!!{structure}}{!!{valuation}} ترضي
ال!!{sentence}s كلها في~${\OLId{greek-variable}{Gamma}}^*$، وبذلك ترضي ما في~${\OLId{greek-variable}{Gamma}}$.
ولا يتم هذا الإثبات إلا بخواص منها أن $\lnot !A \in {\OLId{greek-variable}{Gamma}}^*$ إذا
وفقط إذا كان $!A \notin {\OLId{greek-variable}{Gamma}}^*$، وأن $(!A \lor !B) \in {\OLId{greek-variable}{Gamma}}^*$
إذا وفقط إذا كان $!A \in {\OLId{greek-variable}{Gamma}}^*$ أو $!B \in {\OLId{greek-variable}{Gamma}}^*$، ونظائر ذلك.
\end{explain}

وسنجري في البراهين الآتية على انعكاسية العلاقة $\Proves$ ورتابتها
وتعديها، وكثيرًا ما نكتفي باستعمالها من غير تنبيه (انظر

  \tagrefs{prfSC/{fol:seq:ptn:sec},prfND/{fol:ntd:ptn:sec},prfAX/{fol:axd:ptn:sec},prfTab/{fol:tab:ptn:sec}}}{%

\begin{prop}
\ollabel{prop:ccs}
افترض أن ${\OLId{greek-variable}{Gamma}}$ !!{complete} ومتسقة. عندئذ:
\begin{enumerate}
\item \ollabel{prop:ccs-prov-in} إذا كان ${\OLId{greek-variable}{Gamma}} \Proves !A$، فإن
  $!A \in {\OLId{greek-variable}{Gamma}}$.

\tagitem{prvAnd}{\ollabel{prop:ccs-and} $!A \land !B \in {\OLId{greek-variable}{Gamma}}$
  إذا وفقط إذا كان كل من $!A \in {\OLId{greek-variable}{Gamma}}$ و$!B \in {\OLId{greek-variable}{Gamma}}$.}{}

\tagitem{prvOr}{\ollabel{prop:ccs-or} $!A \lor !B \in {\OLId{greek-variable}{Gamma}}$ إذا
  وفقط إذا كان $!A \in {\OLId{greek-variable}{Gamma}}$ أو $!B \in {\OLId{greek-variable}{Gamma}}$.}{}

\tagitem{prvIf}{\ollabel{prop:ccs-if} $!A \lif !B \in {\OLId{greek-variable}{Gamma}}$ إذا
  وفقط إذا كان $!A \notin {\OLId{greek-variable}{Gamma}}$ أو $!B \in {\OLId{greek-variable}{Gamma}}$.}{}
\end{enumerate}
\end{prop}

\begin{proof}
فرضنا في جميع أجزاء البرهان أن ${\OLId{greek-variable}{Gamma}}$ !!{complete} مع اتساقها.
\begin{enumerate}
\item إذا كان ${\OLId{greek-variable}{Gamma}} \Proves !A$، فإن $!A \in {\OLId{greek-variable}{Gamma}}$.

ليكن ${\OLId{greek-variable}{Gamma}} \Proves !A$. فإن فرضنا $!A \notin
{\OLId{greek-variable}{Gamma}}$، اقتضى كون ${\OLId{greek-variable}{Gamma}}$ !!{complete} أن $\lnot !A \in {\OLId{greek-variable}{Gamma}}$.
فتفيد
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:prv:prop:explicit-inc},
    prfND/{fol:ntd:prv:prop:explicit-inc},
    prfAX/{fol:axd:prv:prop:explicit-inc},
    prfTab/{fol:tab:prv:prop:explicit-inc}}}{%
  \tagrefs{prfSC/{pl:seq:prv:prop:explicit-inc},
    prfND/{pl:ntd:prv:prop:explicit-inc},
    prfAX/{pl:axd:prv:prop:explicit-inc},
    prfTab/{pl:tab:prv:prop:explicit-inc}}},
أن ${\OLId{greek-variable}{Gamma}}$ غير متسقة، خلاف اتساق ${\OLId{greek-variable}{Gamma}}$ المفروض. فبطل
$!A \notin {\OLId{greek-variable}{Gamma}}$، وثبت $!A \in {\OLId{greek-variable}{Gamma}}$.

\tagitem{defAnd}{}{%
\iftag{probAnd}{تمرين.}{%
$!A \land !B \in {\OLId{greek-variable}{Gamma}}$ إذا وفقط إذا كان كل من $!A \in {\OLId{greek-variable}{Gamma}}$ و
$!B \in {\OLId{greek-variable}{Gamma}}$:

أما اللزوم من اليسار إلى اليمين، فليكن $!A \land !B \in {\OLId{greek-variable}{Gamma}}$.
بمقتضى
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-land},
    prfND/{fol:ntd:ppr:prop:provability-land},
    prfAX/{fol:axd:ppr:prop:provability-land},
    prfTab/{fol:tab:ppr:prop:provability-land}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-land},
    prfND/{pl:ntd:ppr:prop:provability-land},
    prfAX/{pl:axd:ppr:prop:provability-land},
    prfTab/{pl:tab:ppr:prop:provability-land}}}، البند~(\OLClassicalDigits{1})،
يلزم ${\OLId{greek-variable}{Gamma}} \Proves !A$ و${\OLId{greek-variable}{Gamma}} \Proves !B$؛ فتنقلنا
\olref{prop:ccs-prov-in} إلى $!A \in {\OLId{greek-variable}{Gamma}}$ و$!B \in {\OLId{greek-variable}{Gamma}}$،
وهو المطلوب.

وأما العكس، فإذا كان $!A \in {\OLId{greek-variable}{Gamma}}$ و$!B \in {\OLId{greek-variable}{Gamma}}$، أفاد
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-land},
    prfND/{fol:ntd:ppr:prop:provability-land},
    prfAX/{fol:axd:ppr:prop:provability-land},
    prfTab/{fol:tab:ppr:prop:provability-land}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-land},
    prfND/{pl:ntd:ppr:prop:provability-land},
    prfAX/{pl:axd:ppr:prop:provability-land},
    prfTab/{pl:tab:ppr:prop:provability-land}}}، البند~(\OLClassicalDigits{2})،
أن ${\OLId{greek-variable}{Gamma}} \Proves !A \land !B$؛ وبـ\olref{prop:ccs-prov-in}
يكون $!A \land !B \in {\OLId{greek-variable}{Gamma}}$.}}

\tagitem{defOr}{}{%
\iftag{probOr}{تمرين.}{%
نبدأ بإثبات أن $!A \lor !B \in {\OLId{greek-variable}{Gamma}}$ يقتضي $!A \in {\OLId{greek-variable}{Gamma}}$
أو $!B \in {\OLId{greek-variable}{Gamma}}$. فلو كان $!A \lor !B \in {\OLId{greek-variable}{Gamma}}$ مع
$!A \notin {\OLId{greek-variable}{Gamma}}$ و$!B \notin {\OLId{greek-variable}{Gamma}}$، للزم، لأن ${\OLId{greek-variable}{Gamma}}$
!!{complete}، أن $\lnot !A \in {\OLId{greek-variable}{Gamma}}$ و$\lnot !B \in {\OLId{greek-variable}{Gamma}}$.
فتفيد
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-lor},
    prfND/{fol:ntd:ppr:prop:provability-lor},
    prfAX/{fol:axd:ppr:prop:provability-lor},
    prfTab/{fol:tab:ppr:prop:provability-lor}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-lor},
    prfND/{pl:ntd:ppr:prop:provability-lor},
    prfAX/{pl:axd:ppr:prop:provability-lor},
    prfTab/{pl:tab:ppr:prop:provability-lor}}}،
البند (\OLClassicalDigits{1})، عدم اتساق ${\OLId{greek-variable}{Gamma}}$، وهو محال. فلا بد من
$!A \in {\OLId{greek-variable}{Gamma}}$ أو $!B \in {\OLId{greek-variable}{Gamma}}$.

وأما العكس، فليكن $!A \in {\OLId{greek-variable}{Gamma}}$ أو $!B \in {\OLId{greek-variable}{Gamma}}$؛ وعندئذ يفيد
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-lor},
    prfND/{fol:ntd:ppr:prop:provability-lor},
    prfAX/{fol:axd:ppr:prop:provability-lor},
    prfTab/{fol:tab:ppr:prop:provability-lor}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-lor},
    prfND/{pl:ntd:ppr:prop:provability-lor},
    prfAX/{pl:axd:ppr:prop:provability-lor},
    prfTab/{pl:tab:ppr:prop:provability-lor}}}، البند (\OLClassicalDigits{2})،
أن ${\OLId{greek-variable}{Gamma}} \Proves !A \lor !B$، فبمقتضى \olref{prop:ccs-prov-in}
ثبت $!A \lor !B \in {\OLId{greek-variable}{Gamma}}$، وتم هذا الاتجاه.}}

\iftag{probIf}{تمرين.}{%

المطلوب: $!A \in {\OLId{greek-variable}{Gamma}}$ و$!B \notin {\OLId{greek-variable}{Gamma}}$. فيجتمع
$!A \lif !B \in {\OLId{greek-variable}{Gamma}}$ مع $!A \in {\OLId{greek-variable}{Gamma}}$. وعليه يفيد

  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-lif},
    prfND/{fol:ntd:ppr:prop:provability-lif},
    prfAX/{fol:axd:ppr:prop:provability-lif},
    prfTab/{fol:tab:ppr:prop:provability-lif}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-lif},
    prfND/{pl:ntd:ppr:prop:provability-lif},
    prfAX/{pl:axd:ppr:prop:provability-lif},
    prfTab/{pl:tab:ppr:prop:provability-lif}}}، البند (\OLClassicalDigits{1})،
أن ${\OLId{greek-variable}{Gamma}} \Proves !B$؛ ومن \olref{prop:ccs-prov-in} يلزم
$!B \in {\OLId{greek-variable}{Gamma}}$، مناقضًا فرض $!B \notin {\OLId{greek-variable}{Gamma}}$.

وللعكس حالتان. أما حالة $!A \notin {\OLId{greek-variable}{Gamma}}$، فكون
${\OLId{greek-variable}{Gamma}}$ !!{complete} يقتضي $\lnot !A \in {\OLId{greek-variable}{Gamma}}$؛ فيفيد
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-lif},
    prfND/{fol:ntd:ppr:prop:provability-lif},
    prfAX/{fol:axd:ppr:prop:provability-lif},
    prfTab/{fol:tab:ppr:prop:provability-lif}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-lif},
    prfND/{pl:ntd:ppr:prop:provability-lif},
    prfAX/{pl:axd:ppr:prop:provability-lif},
    prfTab/{pl:tab:ppr:prop:provability-lif}}}، البند (\OLClassicalDigits{2})،
أن ${\OLId{greek-variable}{Gamma}} \Proves !A \lif !B$؛ وبالرجوع إلى
\olref{prop:ccs-prov-in} يثبت $!A \lif !B \in {\OLId{greek-variable}{Gamma}}$.

وأما حالة $!B \in {\OLId{greek-variable}{Gamma}}$، فنستعمل
\iftag{FOL}{%
  \tagrefs{prfSC/{fol:seq:ppr:prop:provability-lif},
    prfND/{fol:ntd:ppr:prop:provability-lif},
    prfTab/{fol:tab:ppr:prop:provability-lif},
    prfAX/{fol:axd:ppr:prop:provability-lif}}}{%
  \tagrefs{prfSC/{pl:seq:ppr:prop:provability-lif},
    prfND/{pl:ntd:ppr:prop:provability-lif},
    prfTab/{pl:tab:ppr:prop:provability-lif},
    prfAX/{pl:axd:ppr:prop:provability-lif}}}،
البند (\OLClassicalDigits{2}) كذلك، فنجد ${\OLId{greek-variable}{Gamma}} \Proves !A \lif !B$؛ وتثبت
\olref{prop:ccs-prov-in} أن $!A \lif !B \in {\OLId{greek-variable}{Gamma}}$.}}
\end{enumerate}
\end{proof}
