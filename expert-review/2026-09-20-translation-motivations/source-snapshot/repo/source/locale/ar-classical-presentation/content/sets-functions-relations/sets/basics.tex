\olsection{الامتدادية}

\emph{المجموعة} أشياء مجتمعة يؤخذ مجموعها {على أنه} شيء واحد.
وآحاد تلك الأشياء \emph{عناصر} المجموعة، وتسمى أيضًا \emph{أعضاءها}.
فإن كان ${\OLId{latin-ordinary}{x}}$ هو !!a{element} من مجموعة~${\OLId{latin-looped}{A}}$ كتبنا
${\OLId{latin-ordinary}{x}} \in {\OLId{latin-looped}{A}}$، وإن لم يكن كتبنا ${\OLId{latin-ordinary}{x}} \notin {\OLId{latin-looped}{A}}$. وأما المجموعة التي ليس لها
!!{element}s فتسمى المجموعة \emph{الخالية}، ورمزها «$\emptyset$».

\begin{explain}
والاعتبار في المجموعة إنما هو بما لها من !!{element}s، لا بكيف
\emph{نعيّنها}، ولا بما نجعل لـ!!{element}sها من \emph{ترتيب}، ولا بعدد
\emph{المرات} التي نعد فيها !!{element}sها. ويضبط هذا المعنى المبدأ الآتي.
\end{explain}

\begin{defn}[الامتدادية]
  لتكن ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$ مجموعتين. حينئذ ${\OLId{latin-looped}{A}} = {\OLId{latin-looped}{B}}$ إذا وفقط إذا كان
  كل !!{element} من~${\OLId{latin-looped}{A}}$ أيضًا !!a{element} من~${\OLId{latin-looped}{B}}$، وكذلك بالعكس.
\end{defn}

ومن هذا المبدأ يصح استعمال بعض الرموز. فإذا أعطيت الأشياء
${\OLId{latin-ordinary}{a}}_{{\OLMathNumeral{1}}}$، \dots، ${\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{n}}}$، دل $\{{\OLId{latin-ordinary}{a}}_{{\OLMathNumeral{1}}}, \dots, {\OLId{latin-ordinary}{a}}_{{\OLId{latin-ordinary}{n}}}\}$ على
\emph{المجموعة الوحيدة} التي !!{element}sها ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, \ldots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}$.
وإنما قلنا «\emph{الوحيدة}» لأن الامتدادية تمنع تعدد المجموعات مع اتحاد
العناصر؛ فلا تكون هنا إلا \emph{مجموعة واحدة}. ومنها يصح أيضًا:
  \[
    \{{\OLId{latin-ordinary}{a}}, {\OLId{latin-ordinary}{a}}, {\OLId{latin-ordinary}{b}}\} = \{{\OLId{latin-ordinary}{a}}, {\OLId{latin-ordinary}{b}}\} = \{{\OLId{latin-ordinary}{b}},{\OLId{latin-ordinary}{a}}\}.
  \]
فهذا يبين أن ترتيب !!{element}s المجموعة وتكرار ذكرها لا أثر لهما
في تعيين المجموعة نفسها.

\begin{tagblock}{novice}
\begin{ex}
إذا كانت لك طائفة من الأشياء جاز جمعها في مجموعة. فإخوة ريتشارد
وأخواته يؤلفون، في هذا المثال، مجموعة من شخص واحد، وكتابتها
${\OLId{latin-looped}{S}}=\{\textrm{روث}\}$. وأما الأعداد الصحيحة الموجبة الأصغر من ${\OLMathNumeral{4}}$
فمجموعتها $\{{\OLMathNumeral{1}}, {\OLMathNumeral{2}}, {\OLMathNumeral{3}}\}$؛ ولك أن تكتب $\{{\OLMathNumeral{3}}, {\OLMathNumeral{2}}, {\OLMathNumeral{1}}\}$ أو حتى
$\{{\OLMathNumeral{1}}, {\OLMathNumeral{2}}, {\OLMathNumeral{1}}, {\OLMathNumeral{2}}, {\OLMathNumeral{3}}\}$. وليس في ذلك تعدد مجموعات، بل هي مجموعة واحدة
بالامتدادية: فكل !!{element} من $\{{\OLMathNumeral{1}}, {\OLMathNumeral{2}}, {\OLMathNumeral{3}}\}$ هو !!a{element} من
$\{{\OLMathNumeral{3}}, {\OLMathNumeral{2}}, {\OLMathNumeral{1}}\}$ (ومن $\{{\OLMathNumeral{1}},
{\OLMathNumeral{2}}, {\OLMathNumeral{1}}, {\OLMathNumeral{2}}, {\OLMathNumeral{3}}\}$)، وكذلك بالعكس.
\end{ex}
\end{tagblock}

وقد نعيّن المجموعة بخاصية تجتمع عليها !!{element}sها. ونختصر ذلك
بالرمز $\Setabs{{\OLId{latin-ordinary}{x}}}{\phi({\OLId{latin-ordinary}{x}})}$؛ فـ$\phi({\OLId{latin-ordinary}{x}})$ فيه هي الخاصية التي يشترط
اتصاف~${\OLId{latin-ordinary}{x}}$ بها ليكون من !!{element}s المجموعة.

\begin{tagblock}{novice}
\begin{ex}
ومن ثم يجوز تعيين ${\OLId{latin-looped}{S}}$ في المثال السابق بهذا الوجه أيضًا:
\[
{\OLId{latin-looped}{S}} = \Setabs{{\OLId{latin-ordinary}{x}}}{{\OLId{latin-ordinary}{x}} \text{ من إخوة ريتشارد وأخواته}}.
\]
\end{ex}
\end{tagblock}

\begin{tagblock}{math}
\begin{ex}
العدد \emph{التام} هو المساوي لمجموع قواسمه الفعلية، وهي الأعداد
التي تقسمه بلا باق ولا تساويه هو نفسه. فالعدد ${\OLMathNumeral{6}}$ تام؛ إذ قواسمه
الفعلية ${\OLMathNumeral{1}}$ و${\OLMathNumeral{2}}$ و~${\OLMathNumeral{3}}$، و${\OLMathNumeral{6}} = {\OLMathNumeral{1}} + {\OLMathNumeral{2}} + {\OLMathNumeral{3}}$. وهو ${\OLMathNumeral{6}}$ وحده التام
بين الأعداد الصحيحة الموجبة الأصغر من ${\OLMathNumeral{10}}$. وبالامتدادية إذن:
\[
	\{{\OLMathNumeral{6}}\} = \Setabs{{\OLId{latin-ordinary}{x}}}{{\OLId{latin-ordinary}{x}}\text{ تام و }{\OLMathNumeral{0}} \leq {\OLId{latin-ordinary}{x}} \leq {\OLMathNumeral{10}}}
\]
والطرف الأيمن يقرأ: «مجموعة قيم ${\OLId{latin-ordinary}{x}}$ التي يكون فيها ${\OLId{latin-ordinary}{x}}$ تامًا
و${\OLMathNumeral{0}} \leq {\OLId{latin-ordinary}{x}} \leq {\OLMathNumeral{10}}$». فتساوي الطرفين شاهد على أن اختلاف طريق
التعيين لا يقتضي اختلاف المجموعة. وعلى العموم، إذا عُيّنت مجموعة قيم
${\OLId{latin-ordinary}{x}}$ بالخاصية $\phi({\OLId{latin-ordinary}{x}})$، قضت الامتدادية بوحدتها. ولهذا يصح أن نسمي
$\Setabs{{\OLId{latin-ordinary}{x}}}{\phi({\OLId{latin-ordinary}{x}})}$ \emph{المجموعة الوحيدة} لقيم ${\OLId{latin-ordinary}{x}}$ التي تحقق~$\phi({\OLId{latin-ordinary}{x}})$.
\end{ex}
\end{tagblock}

ومن الامتدادية تؤخذ طريقة البرهان على تساوي مجموعتين: لإثبات
${\OLId{latin-looped}{A}} = {\OLId{latin-looped}{B}}$ بيّن أن ${\OLId{latin-ordinary}{x}} \in {\OLId{latin-looped}{A}}$ يستلزم ${\OLId{latin-ordinary}{x}} \in {\OLId{latin-looped}{B}}$، وأن ${\OLId{latin-ordinary}{y}} \in {\OLId{latin-looped}{B}}$
يستلزم ${\OLId{latin-ordinary}{y}} \in {\OLId{latin-looped}{A}}$.

\begin{prob}
برهن على أن المجموعة الخالية، إن وجدت، واحدة؛ أي إذا كانت ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$
مجموعتين لا !!{element}s لهما، لزم ${\OLId{latin-looped}{A}} = {\OLId{latin-looped}{B}}$.
\end{prob}
