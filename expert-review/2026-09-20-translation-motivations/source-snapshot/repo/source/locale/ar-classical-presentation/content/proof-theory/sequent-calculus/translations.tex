\olsection{في التحويل بين~\Log{G1c} و\Log{G3c}}

قد تقدم أن~\Log{G1c} و\Log{G3c} كليهما سليم وكامل للمنطق الكلاسيكي؛
فكل واحد منهما يبرهن المتتاليات الصادقة في كل~!!{structure} وحدها. ومن
ثم تتحد المتتاليات القابلة للبرهنة فيهما. ونثبت الآن هذا الاتحاد بحجة
من نظرية البرهان نفسها، من غير توسط مبرهنتي السلامة والاكتمال. وهذه الحجة
تعطينا \emph{تحويلات} تنقل~!!{proof}s في~\Log{G1c} إلى~!!{proof}s في
\Log{G3c}، وتنقلها بالعكس.

\begin{prop}
إذا كان~$\Log{G1c}\Proves {\OLId{latin-looped}{S}}$، فإن~$\Log{G3c}\Proves {\OLId{latin-looped}{S}}$.
\end{prop}

\begin{proof}
  نثبت أن كل~$\pi$~!!a{proof} في~\Log{G1c} يقابله~!!a{proof}
  $\pi'$ للمتتالية~${\OLId{latin-looped}{S}}$ في~\Log{G3c}، وذلك بالاستقراء على
  ${\OLId{latin-ordinary}{n}}=\pheight{\pi}$.

  فإن كان~${\OLId{latin-ordinary}{n}}={\OLMathNumeral{0}}$، كان~$\pi$ بديهية. أما~$\lfalse\Sequent\quad$ فهي
  بديهية في~\Log{G3c} أيضًا، إذا أخذنا السياقين~${\OLId{greek-variable}{Gamma}}$ و${\OLId{greek-variable}{Delta}}$
  فارغين. وأما~\Log{G1c} فيجيز~$!A\Sequent !A$ بديهية لأي~$!A$، على
  حين لا تكون بديهيات الهوية في~\Log{G3c} إلا ذرية؛ وقد أثبتنا في
  \olref[ex]{prop:G3c-axioms} أن كل المتتاليات~$!A\Sequent !A$
  !!{provable} في~\Log{G3c}.

  وإن كان~${\OLId{latin-ordinary}{n}}>{\OLMathNumeral{0}}$، ختم~$\pi$ بالقاعدة~${\OLId{latin-looped}{R}}$. ويفيد فرض الاستقراء~!!{proof}s
  لمقدمات القاعدة في~\Log{G3c}؛ أي يفيد تحويلات للـ!!{proof}s الجزئية
  المباشرة إلى~\Log{G3c}. فإن كانت~${\OLId{latin-looped}{R}}$ من قواعد~\Log{G3c} أيضًا، أجريناها
  % REVIEW-0713-M1: الأربع هي القواعد المنطقية؛ ويأتي الإضعاف والتقلص بعد ذلك.
  على~!!{proof}s المقدمات بعد تحويلها. وأما القواعد المنطقية الأربع المختلفة بين
  النظامين فتحاكى هكذا: نستعمل، قبل قاعدة~\Log{G3c}، الإضعاف المقبول في
  \olref[adm]{prop:weak-G3c-adm}؛ فنضيف الصيغة الفعالة المفقودة في حالتي
  \LeftR{\land} و\RightR{\lor}، أو نضيف الوقوع المحفوظ للصيغة الرئيسة
  في حالتي~\LeftR{\lforall} و\RightR{\lexists}. فإن كانت~${\OLId{latin-looped}{R}}$ قاعدة
  إضعاف في~\Log{G1c}، استعملنا نتيجة قبول الإضعاف نفسها؛ وإن كانت قاعدة
  تقلص استعملنا~\olref[inv]{prop:G3c-cont-adm}.
\end{proof}

\begin{prop}
إذا كان~$\Log{G3c}\Proves {\OLId{latin-looped}{S}}$، فإن~$\Log{G1c}\Proves {\OLId{latin-looped}{S}}$.
\end{prop}

\begin{proof}
  نستقرئ مرة أخرى على~!!{height}~!!a{proof}~$\pi$ للمتتالية~${\OLId{latin-looped}{S}}$.
  فكل بديهية من بديهيات~\Log{G3c} يمكن~!!{prove}ها في~\Log{G1c}، بأن
  نبدأ من بديهية ثم نكرر قاعدتي~\LeftR{\Weakening}
  و\RightR{\Weakening}:
  \[
  \Axiom$!C \fCenter !C$
  \RightLabel{\LeftR{\Weakening}}
  \Deduce$!C, {\OLId{greek-variable}{Gamma}} \fCenter !C$
  \RightLabel{\RightR{\Weakening}}
  \Deduce$!C, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !C$
  \DisplayProof
\qquad
  \Axiom$\lfalse \fCenter$
  \RightLabel{\LeftR{\Weakening}}
  \Deduce$\lfalse, {\OLId{greek-variable}{Gamma}} \fCenter$
  \RightLabel{\RightR{\Weakening}}
  \Deduce$\lfalse, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \DisplayProof
  \]
  فإذا ختم~$\pi$ بالقاعدة~${\OLId{latin-looped}{R}}$، أعطانا فرض الاستقراء~!!{proof}s للمقدمات
  في~\Log{G1c}، ثم أجرينا القاعدة~${\OLId{latin-looped}{R}}$ نفسها على تلك~!!{proof}s. ويستثنى من ذلك
  \LeftR{\land} و\RightR{\lor} و\LeftR{\lforall}
  و\RightR{\lexists}، لاختلاف صورها في النظامين. فأما صورتا
  \Log{G3c} الأوليان فتشتقان في~\Log{G1c} هكذا:
  \[
  \Axiom$!A, !B, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \RightLabel{\LeftR{\land}}
  \UnaryInf$!A \land !B, !B, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \RightLabel{\LeftR{\land}}
  \UnaryInf$!A \land !B, !A \land !B, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \RightLabel{\LeftR{\Contraction}}
  \UnaryInf$!A \land !B, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
    \DisplayProof
\qquad
  \Axiom${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A, !B$
  \RightLabel{\RightR{\lor}}
  \UnaryInf${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A \lor !B, !B$
  \RightLabel{\RightR{\lor}}
  \UnaryInf${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A \lor !B, !A \lor !B$
  \RightLabel{\RightR{\Contraction}}
  \UnaryInf${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A \lor !B$
    \DisplayProof
  \]
  وأما صورتا القاعدتين الكميتين فتشتقان على الوجه الآتي:
  \[
  \Axiom$!A({\OLId{latin-ordinary}{t}}), \lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \RightLabel{\LeftR{\lforall}}
  \UnaryInf$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], \lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \RightLabel{\LeftR{\Contraction}}
  \UnaryInf$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
  \DisplayProof
\qquad
  \Axiom${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, \lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], !A({\OLId{latin-ordinary}{t}})$
  \RightLabel{\RightR{\lexists}}
  \UnaryInf${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, \lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})], \lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$
  \RightLabel{\RightR{\Contraction}}
  \UnaryInf${\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, \lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$
  \DisplayProof
  \]
\end{proof}
