\olsection{ال\printtoken{p}{structure} الجزئية}

إذا كان !!{domain}~!!a{structure}~$\Struct{M}$ داخلًا في
!!{domain}~$\Struct{M'}$ أخرى، لم يكف هذا وحده لجعل~$\Struct{M}$
«جزءًا» من~$\Struct{M'}$. فلا بد، مع الشرط $\Domain{{\OLId{latin-looped}{M}}} \subseteq
\Domain{{\OLId{latin-looped}{M}}'}$، من اتفاق $\Struct{M}$ و$\Struct{M'}$ في تأويل
رموز اللغة على الجزء المشترك~$\Domain{{\OLId{latin-looped}{M}}}$ على الأقل.

\begin{defn}
\ollabel{defn:substructure}
لتكن !!{structure}s~$\Struct M$ و$\Struct M'$ للغة واحدة~$\Lang
L$. نقول إن $\Struct M$~\emph{!!{structure} جزئية} من~$\Struct M'$،
وإن $\Struct M'$~\emph{امتداد} لـ~$\Struct M$، وكتبنا $\Struct M
\substruct \Struct M'$، إذا وفقط إذا
\begin{enumerate}
% تصحيح مصطلحي (C185-01): رُدّ n-place إلى «ذو/ذات n مواضع» بدل «n-محلية» المبهمة؛ والاختيار مفتوح لمراجعة خبير في مصطلح arity.
\item $\Domain{{\OLId{latin-looped}{M}}} \subseteq \Domain{{\OLId{latin-looped}{M}}'}$؛
\item لكل ثابت~${\OLId{latin-ordinary}{c}} \in \Lang L$، $\Assign{{\OLId{latin-ordinary}{c}}}{{\OLId{latin-looped}{M}}} =
    \Assign{{\OLId{latin-ordinary}{c}}}{{\OLId{latin-looped}{M}}'}$؛
\item لكل !!{function}~${\OLId{latin-ordinary}{f}} \in \Lang L$ ذات ${\OLId{latin-ordinary}{n}}$ مواضع،
  $\Assign{{\OLId{latin-ordinary}{f}}}{{\OLId{latin-looped}{M}}}({\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, \dots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}) = \Assign{{\OLId{latin-ordinary}{f}}}{{\OLId{latin-looped}{M}}'}({\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, \dots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}})$
  لجميع ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}$، و\dots، ${\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}} \in \Domain{{\OLId{latin-looped}{M}}}$؛
\item لكل !!{predicate}~${\OLId{latin-looped}{R}} \in \Lang L$ ذي ${\OLId{latin-ordinary}{n}}$ مواضع، تكون $\langle
  {\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, \dots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}\rangle \in \Assign{{\OLId{latin-looped}{R}}}{{\OLId{latin-looped}{M}}}$ إذا وفقط إذا كانت
  $\langle {\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}, \dots, {\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}}\rangle \in \Assign{{\OLId{latin-looped}{R}}}{{\OLId{latin-looped}{M}}'}$ لجميع ${\OLId{latin-ordinary}{a}}_{\OLMathNumeral{1}}$،
  و\dots، ${\OLId{latin-ordinary}{a}}_{\OLId{latin-ordinary}{n}} \in \Domain{{\OLId{latin-looped}{M}}}$.
\end{enumerate}
\end{defn}

\begin{rem}
\ollabel{rem:substructure}
إن خلت اللغة من الثوابت ومن !!{function}s، كفت كل مجموعة جزئية غير
خالية ${\OLId{latin-looped}{N}} \subseteq \Domain{{\OLId{latin-looped}{M}}}$ لتعيين !!{structure} جزئية~$\Struct{N}$
من~$\Struct M$ ذات !!{domain}~$\Domain{{\OLId{latin-looped}{N}}} = {\OLId{latin-looped}{N}}$، وذلك بأن نضع
$\Assign{{\OLId{latin-looped}{R}}}{{\OLId{latin-looped}{N}}} = \Assign{{\OLId{latin-looped}{R}}}{{\OLId{latin-looped}{M}}} \cap {\OLId{latin-looped}{N}}^{\OLId{latin-ordinary}{n}}$.
\end{rem}
