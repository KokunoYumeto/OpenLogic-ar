\olsection{\usetoken{S}{derivability} والاتساق}

لعلاقة ال!!{derivability} خواص نثبتها الآن؛ لكل منها فائدة في نفسه، ثم يحتاج إليه برهان مبرهنة الاكتمال.

\begin{prop}\ollabel{prop:provability-contr}
  إذا كان~${\OLId{greek-variable}{Gamma}} \Proves !A$ وكانت~${\OLId{greek-variable}{Gamma}} \cup \{!A\}$ غير متسقة،
  فإن~${\OLId{greek-variable}{Gamma}}$ غير متسقة.
\end{prop}

\begin{proof}
توجد مجموعتان منتهيتان~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}$ و
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$ بحيث $\Log{LK}$ !!{derive}s
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent !A$ و$!A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \Sequent \quad$. وليكن
$\Log{LK}$-!!{derivation} للتتابعية~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent !A$ هو
~$\pi_{\OLMathNumeral{0}}$، وليكن $\Log{LK}$-!!{derivation} للتتابعية
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}}, !A \Sequent \quad$ هو~$\pi_{\OLMathNumeral{1}}$. وعندئذ يمكننا !!{derive}
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_{\OLMathNumeral{0}}$}
\Deduce$ {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \fCenter !A $
\AxiomC{}
\RightLabel{$\pi_{\OLMathNumeral{1}}$}
\Deduce$!A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter $
\RightLabel{\Cut}
\BinaryInf$ {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}},{\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter $
\end{prooftree}

بما أن~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \subseteq {\OLId{greek-variable}{Gamma}}$ و${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$،
فإن~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \cup {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$؛ ومن ثم فإن~${\OLId{greek-variable}{Gamma}}$
غير متسقة.
\end{proof}

\begin{prop}
\ollabel{prop:prov-incons}
${\OLId{greek-variable}{Gamma}} \Proves !A$ إذا وفقط إذا كانت~${\OLId{greek-variable}{Gamma}} \cup \{\lnot !A\}$ غير
متسقة.
\end{prop}

\begin{proof}
أما الاتجاه الأول، فمن~${\OLId{greek-variable}{Gamma}} \Proves !A$ توجد مجموعة منتهية
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}\subseteq{\OLId{greek-variable}{Gamma}}$ و!!a{derivation}~$\pi_{\OLMathNumeral{0}}$ لـ
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent !A$. نُلحق به
$\LeftR{\lnot}$، فينتج !!a{derivation} لـ
~$\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent \quad$. ولأن
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}\cup\{\lnot !A\}\subseteq{\OLId{greek-variable}{Gamma}}\cup\{\lnot !A\}$، فهذا يثبت
عدم اتساق ${\OLId{greek-variable}{Gamma}} \cup \{\lnot !A\}$.

وأما العكس، فعدم اتساق~${\OLId{greek-variable}{Gamma}} \cup \{\lnot !A\}$ يعطي مجموعة منتهية
${\OLId{greek-variable}{Delta}}\subseteq{\OLId{greek-variable}{Gamma}}\cup\{\lnot !A\}$ واشتقاقًا لـ
${\OLId{greek-variable}{Delta}} \Sequent \quad$. ضع ${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}}={\OLId{greek-variable}{Delta}}\setminus\{\lnot !A\}$؛
فهي جزئية منتهية من~${\OLId{greek-variable}{Gamma}}$، وبالإضعاف والتبادل، عند الحاجة،
نحصل على !!a{derivation}~$\pi_{\OLMathNumeral{1}}$ لـ
$\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \Sequent \quad$. وبالقطع نحصل على !!a{derivation}
لـ~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \Sequent !A$ كما يأتي:
\begin{prooftree}
  \Axiom$!A \fCenter !A$
  \RightLabel{\RightR{\lnot}}
  \UnaryInf$\fCenter !A, \lnot !A$
  \AxiomC{}
  \RightLabel{$\pi_{\OLMathNumeral{1}}$}
  \Deduce$\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter$
  \RightLabel{\Cut}
  \BinaryInf${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter !A$
\end{prooftree}
ومن ${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}}\subseteq{\OLId{greek-variable}{Gamma}}$ ينتج ${\OLId{greek-variable}{Gamma}}\Proves !A$.

\textbf{تنبيه تصحيحي على الأصل (078-I1).} صُرّح بالمجموعتين المنتهيتين اللتين
تستعملهما الاشتقاقات؛ فقد تكون ${\OLId{greek-variable}{Gamma}}$ غير منتهية، فلا يصح وضعها كلها
في تتابعية منتهية. ثم رُفعت النتيجة إلى ${\OLId{greek-variable}{Gamma}}$ بالرتابة.
\end{proof}

\begin{prob}
أثبت أن~${\OLId{greek-variable}{Gamma}} \Proves \lnot !A$ إذا وفقط إذا كانت
${\OLId{greek-variable}{Gamma}} \cup \{!A\}$ غير متسقة.
\end{prob}

\begin{prop}\ollabel{prop:explicit-inc}
  إذا كان~${\OLId{greek-variable}{Gamma}} \Proves !A$ وكان~$\lnot !A \in {\OLId{greek-variable}{Gamma}}$، فإن
  ${\OLId{greek-variable}{Gamma}}$ غير متسقة.
\end{prop}

\begin{proof}
  افترض أن~${\OLId{greek-variable}{Gamma}} \Proves !A$ وأن~$\lnot !A \in {\OLId{greek-variable}{Gamma}}$. عندئذ يوجد
  !!a{derivation}~$\pi$ لتتابعية~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent !A$. والتتابعية
  $\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent \quad$ !!{derivable} أيضًا:
  \begin{prooftree}
    \AxiomC{}
    \RightLabel{$\pi$}
    \Deduce${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \fCenter !A$
    \Axiom$!A \fCenter !A$
    \RightLabel{\LeftR{\lnot}}
    \UnaryInf$\lnot !A, !A \fCenter$
    \RightLabel{\LeftR{\Exchange}}
    \UnaryInf$!A, \lnot !A \fCenter$
    \RightLabel{\Cut}
    \BinaryInf${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}, \lnot !A \fCenter$
  \end{prooftree}
  وبما أن~$\lnot !A \in {\OLId{greek-variable}{Gamma}}$ و${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \subseteq {\OLId{greek-variable}{Gamma}}$، فإن
  هذا يبين أن~${\OLId{greek-variable}{Gamma}}$ غير متسقة.
\end{proof}

\begin{prop}\ollabel{prop:provability-exhaustive}
  إذا كانت كل من~${\OLId{greek-variable}{Gamma}} \cup \{!A\}$ و
  ${\OLId{greek-variable}{Gamma}} \cup \{\lnot !A\}$ غير متسقة، فإن~${\OLId{greek-variable}{Gamma}}$ غير متسقة.
\end{prop}

\begin{proof}
يعطينا الفرض مجموعتين منتهيتين~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \subseteq {\OLId{greek-variable}{Gamma}}$ و
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$، ومعهما $\Log{LK}$-!!{derivation}s
$\pi_{\OLMathNumeral{0}}$ و$\pi_{\OLMathNumeral{1}}$ لـ~$!A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Sequent \quad$ و
$\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \Sequent \quad$ على الترتيب.
فنستعمل النفي ثم القطع لنتمكن من !!{derive}
\begin{prooftree}
\AxiomC{}
\RightLabel{$\pi_{\OLMathNumeral{0}}$}
\Deduce$ !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \fCenter $
\RightLabel{\RightR{\lnot}}
\UnaryInf$ {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \fCenter \lnot !A$
\AxiomC{}
\RightLabel{$\pi_{\OLMathNumeral{1}}$}
\Deduce$\lnot !A, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter  $
\RightLabel{\Cut}
\BinaryInf$ {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}, {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \fCenter $
\end{prooftree}
وبما أن~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \subseteq {\OLId{greek-variable}{Gamma}}$ و${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$،
فإن~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \cup {\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{1}} \subseteq {\OLId{greek-variable}{Gamma}}$. ومن ثم فإن~${\OLId{greek-variable}{Gamma}}$
غير متسقة.
\end{proof}
