\olsection{\usetoken{S}{derivability} من مجموعة \usetoken{P}{formula}}

كان النظر في \olref[prf][prs]{sec} في قابلية برهنة !!a{formula}
في النسق~$\Sigma$ من غير فرض مقدمات. وأما الآن فننظر في البرهنة في
$\Sigma$ مع اتخاذ !!{formula}s من مجموعة~${\OLId{greek-variable}{Gamma}}$ مقدمات، على الوجه الآتي.

\begin{defn}\ollabel{defn:Gammaproves}
  نقول إن !!^a{formula}~$!A$ !!{derivable} في النسق~$\Sigma$ من مجموعة
  !!{formula}s ${\OLId{greek-variable}{Gamma}}$، ونرمز إلى ذلك بقولنا ${\OLId{greek-variable}{Gamma}} \Proves[\Sigma] !A$،
  إذا وفقط إذا أمكن اختيار عدد صحيح ${\OLId{latin-ordinary}{n}} \geq {\OLMathNumeral{0}}$ وعناصر من المجموعة،
  $!B_{\OLMathNumeral{1}}$، \dots، $!B_{\OLId{latin-ordinary}{n}} \in {\OLId{greek-variable}{Gamma}}$، بحيث
  $\Sigma \Proves !B_{\OLMathNumeral{1}} \lif (!B_{\OLMathNumeral{2}} \lif \cdots (!B_{\OLId{latin-ordinary}{n}} \lif !A)
  \cdots)$.
  وعند ${\OLId{latin-ordinary}{n}}={\OLMathNumeral{0}}$ تكون قائمة المقدمات خالية، ويؤول شرط البرهنة المعروض إلى
  $\Sigma \Proves !A$؛ ومن ثم يكون
  $\emptyset \Proves[\Sigma] !A$ إذا وفقط إذا كان $\Sigma \Proves !A$.
\end{defn}
