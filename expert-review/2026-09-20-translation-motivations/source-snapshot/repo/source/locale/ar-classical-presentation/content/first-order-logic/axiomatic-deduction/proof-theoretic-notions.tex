\olsection{المفاهيم النظرية البرهانية}

\begin{explain}
قد تقدّم تعريف المعاني الدلالية:
\iftag{FOL}{الصلاحية}{تحصيل الحاصل} والاستلزام وقابلية الإرضاء.
ولها نظائر من \emph{مفاهيم نظرية البرهان} نعرّفها الآن.
فليست هذه النظائر مبنية على إرضاء ال!!{sentence}s في
\iftag{FOL}{ال!!{structure}s}{ال!!{valuation}s}، وإنما هي مبنية على
!!{derivability} بعض الصيغ أو !!{nonderivability}ها.
ومن النتائج المهمة تطابق المعاني الدلالية ونظائرها البرهانية؛
وهذا هو مضمون \emph{مبرهنة السلامة} و\emph{مبرهنة الاكتمال}.
\end{explain}

\begin{defn}[!!^{derivability}]
ال!!^a{formula}~$!A$ \emph{!!{derivable}} من~${\OLId{greek-variable}{Gamma}}$،
ونرمز إلى ذلك بـ~${\OLId{greek-variable}{Gamma}} \Proves !A$، متى وجد !!a{derivation}
من~${\OLId{greek-variable}{Gamma}}$ ينتهي إلى~$!A$.
\end{defn}

\begin{defn}[المبرهنات]
ال!!^a{formula}~$!A$ \emph{مبرهنة} متى وجد !!a{derivation}
لـ~$!A$ من المجموعة الخالية. فنكتب~$\Proves !A$ إذا كانت~$!A$
مبرهنة، ونكتب~$\Proves/ !A$ إذا لم تكن.
\end{defn}

\begin{defn}[الاتساق]
المجموعة~${\OLId{greek-variable}{Gamma}}$ من ال!!{formula}s \emph{متسقة} إذا وفقط إذا
كان~${\OLId{greek-variable}{Gamma}}\Proves/ \lfalse$؛ فإن لم يكن فهي \emph{غير متسقة}.
\end{defn}

\begin{prop}[الانعكاسية]
\ollabel{prop:reflexivity}
إذا كان $!A \in {\OLId{greek-variable}{Gamma}}$، فإن ${\OLId{greek-variable}{Gamma}} \Proves !A$.
\end{prop}

\begin{proof}
  يكفي أخذ ال!!{formula}~$!A$ وحدها؛ فهي !!a{derivation}
  لـ~$!A$ من~${\OLId{greek-variable}{Gamma}}$.
\end{proof}

\begin{prop}[الرتابة]
\ollabel{prop:monotonicity}
إذا كان ${\OLId{greek-variable}{Gamma}} \subseteq {\OLId{greek-variable}{Delta}}$ و${\OLId{greek-variable}{Gamma}} \Proves !A$، فإن ${\OLId{greek-variable}{Delta}}
\Proves !A$.
\end{prop}

\begin{proof}
\iftag{FOL}{لل!!{derivation} لـ~$!A$ من~${\OLId{greek-variable}{Gamma}}$ دعم منتهٍ
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}\subseteq{\OLId{greek-variable}{Gamma}}$. ولما كان~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}\subseteq{\OLId{greek-variable}{Delta}}$، بقيت
المتتالية نفسها !!a{derivation} لـ~$!A$ من~${\OLId{greek-variable}{Delta}}$ بحسب شرط الدعم في
\QR{}.}{فال!!{derivation} لـ~$!A$ من~${\OLId{greek-variable}{Gamma}}$ يصح أن يؤخذ
!!a{derivation} لـ~$!A$ من~${\OLId{greek-variable}{Delta}}$ أيضًا.}
\end{proof}

\begin{prop}[التعدية]
\ollabel{prop:transitivity}
إذا كان ${\OLId{greek-variable}{Gamma}} \Proves !A$ و$\{!A\} \cup {\OLId{greek-variable}{Delta}} \Proves
!B$، فإن ${\OLId{greek-variable}{Gamma}} \cup {\OLId{greek-variable}{Delta}} \Proves !B$.
\end{prop}

\begin{proof}
% تصحيح مصدر (0118-I1): جُدّدت ثوابت QR المميزة قبل وصل الاشتقاقين.
\iftag{FOL}{من~$\{!A\}\cup{\OLId{greek-variable}{Delta}}\Proves !B$ نأخذ دعمًا منتهيًا
${\OLId{greek-variable}{Delta}}_{\OLMathNumeral{0}}\subseteq{\OLId{greek-variable}{Delta}}$ بحيث يكون دعم ال!!{derivation} لـ~$!B$
داخلًا في~$\{!A\}\cup{\OLId{greek-variable}{Delta}}_{\OLMathNumeral{0}}$، ونأخذ دعمًا منتهيًا
${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}\subseteq{\OLId{greek-variable}{Gamma}}$ ل!!{derivation}~$!A$. وشجرتا البرهان
متناهيتان، فطائفة الثوابت الواقعة فيهما متناهية. طبّق
\olref[qua]{lem:qr-freshening} أولًا على شجرة~$!A$ مع تحريم ثوابت الشجرة
الثانية، ثم على الشجرة الثانية مع تحريم ثوابت الأولى بعد تجديدها. فنحصل
على شجرتين تحفظان الخاتمتين ولا يزيد دعمهما، وتتباين جميع ثوابتهما المميزة وتفي
بشروط الجدة بالنسبة إلى الدعم المشترك. ثم نعيد خطيتهما، ولتكن الأولى
$!A_{\OLMathNumeral{1}}$، \dots، $!A_{\OLId{latin-ordinary}{k}} = !A$ والثانية~$!B_{\OLMathNumeral{1}}$، \dots، $!B_{\OLId{latin-ordinary}{l}} = !B$.
فتكون المتتالية
  \[
  !A_{\OLMathNumeral{1}}, \dots, !A_{\OLId{latin-ordinary}{k}} = !A, !B_{\OLMathNumeral{1}}, \dots, !B_{\OLId{latin-ordinary}{l}} = !B.
  \]
!!a{derivation} صحيحة لـ~$!B$ من~${\OLId{greek-variable}{Gamma}}\cup{\OLId{greek-variable}{Delta}}$؛ إذ يبرر كل ورود
$!B_{\OLId{latin-ordinary}{i}}=!A$ ما برّر~$!A_{\OLId{latin-ordinary}{k}}=!A$، وتبقى شروط~\QR{} صحيحة.}{من
$\{!A\}\cup{\OLId{greek-variable}{Delta}}\Proves !B$ نأخذ !!a{derivation}
$!B_{\OLMathNumeral{1}}$، \dots، $!B_{\OLId{latin-ordinary}{l}}=!B$ من~$\{!A\}\cup{\OLId{greek-variable}{Delta}}$. وبالفرض توجد
!!a{derivation}~$!A_{\OLMathNumeral{1}}$، \dots، $!A_{\OLId{latin-ordinary}{k}}=!A$ من~${\OLId{greek-variable}{Gamma}}$. فالمتتالية
\[
!A_{\OLMathNumeral{1}}, \dots, !A_{\OLId{latin-ordinary}{k}} = !A, !B_{\OLMathNumeral{1}}, \dots, !B_{\OLId{latin-ordinary}{l}} = !B
\]
!!a{derivation} لـ~$!B$ من~${\OLId{greek-variable}{Gamma}}\cup{\OLId{greek-variable}{Delta}}$؛ إذ يبرر كل ورود
$!B_{\OLId{latin-ordinary}{i}}=!A$ ما برّر~$!A_{\OLId{latin-ordinary}{k}}=!A$.}
\end{proof}

ومن التعدّي، على الخصوص، أنه إذا كان~${\OLId{greek-variable}{Gamma}} \Proves !A$ و$!A
\Proves !B$ لزم~${\OLId{greek-variable}{Gamma}} \Proves !B$. وبتكراره، إذا كان~$!A_{\OLMathNumeral{1}},
\dots, !A_{\OLId{latin-ordinary}{n}} \Proves !B$ وكان~${\OLId{greek-variable}{Gamma}} \Proves !A_{\OLId{latin-ordinary}{i}}$ لكل~${\OLId{latin-ordinary}{i}}$،
حصل~${\OLId{greek-variable}{Gamma}} \Proves !B$.


\begin{prop}
\ollabel{prop:incons}
تكون ${\OLId{greek-variable}{Gamma}}$ غير متسقة إذا وفقط إذا كان ${\OLId{greek-variable}{Gamma}} \Proves !A$ لكل~$!A$.
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\begin{prop}[التراص]
\ollabel{prop:proves-compact}
  \begin{enumerate}
  \item إذا كان ${\OLId{greek-variable}{Gamma}} \Proves !A$، فثمة مجموعة جزئية منتهية ${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}
    \subseteq {\OLId{greek-variable}{Gamma}}$ بحيث ${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Proves !A$.
  \item إذا كانت كل مجموعة جزئية منتهية من~${\OLId{greek-variable}{Gamma}}$ متسقة، فإن
    ${\OLId{greek-variable}{Gamma}}$ متسقة.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item من~${\OLId{greek-variable}{Gamma}} \Proves !A$ نأخذ متتالية منتهية من
      ال!!{formula}s~$!A_{\OLMathNumeral{1}}$، \dots،~$!A_{\OLId{latin-ordinary}{n}}$ بحيث~$!A \ident !A_{\OLId{latin-ordinary}{n}}$،
      وكل~$!A_{\OLId{latin-ordinary}{i}}$ إما بديهية منطقية، أو !!a{element} من~${\OLId{greek-variable}{Gamma}}$،
      أو مأخوذة من !!{formula}s سابقة بقاعدة استدلال.
      نسمّي~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}$ مجموعة تلك الصيغ~$!A_{\OLId{latin-ordinary}{i}}$ التي تقع
      في~${\OLId{greek-variable}{Gamma}}$؛ فال!!{derivation} نفسه !!a{derivation}
      من~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}}$، وبذلك~${\OLId{greek-variable}{Gamma}}_{\OLMathNumeral{0}} \Proves !A$.
    \item نأخذ المقابل بالنقيض للفقرة~(\OLClassicalDigits{1}) في الحالة الخاصة~$!A
      \ident \lfalse$.

\end{enumerate}
\end{proof}
