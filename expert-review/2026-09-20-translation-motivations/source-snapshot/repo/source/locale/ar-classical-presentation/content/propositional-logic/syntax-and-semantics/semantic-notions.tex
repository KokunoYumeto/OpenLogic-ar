\olsection{المفاهيم الدلالية}

نضبط الآن المعاني الدلالية التي سيجري عليها البحث:

\begin{defn}
\begin{enumerate}
\item !!^a{formula}~$!A$ \emph{قابلة للإرضاء} متى وُجد تقييم~$\pAssign{{\OLId{latin-ordinary}{v}}}$
  بحيث $\pSat{{\OLId{latin-ordinary}{v}}}{!A}$؛ وهي \emph{غير قابلة للإرضاء} متى لم يوجد تقييم~$\pAssign{{\OLId{latin-ordinary}{v}}}$
  بحيث $\pSat{{\OLId{latin-ordinary}{v}}}{!A}$.
\item !!^a{formula}~$!A$ \emph{تحصيل حاصل} متى كان
  $\pSat{{\OLId{latin-ordinary}{v}}}{!A}$ لجميع !!{valuation}s~$\pAssign{{\OLId{latin-ordinary}{v}}}$.
\item !!^a{formula}~$!A$ \emph{عارضة} متى أمكن إرضاؤها ولم تكن تحصيل حاصل.
\item لتكن ${\OLId{greek-variable}{Gamma}}$ مجموعة من !!{formula}s. قولنا ${\OLId{greek-variable}{Gamma}} \Entails !A$،
  وقراءته «${\OLId{greek-variable}{Gamma}}$ تستلزم~$!A$»، معناه أن $\pSat{{\OLId{latin-ordinary}{v}}}{!A}$ لكل
  !!{valuation}~$\pAssign{{\OLId{latin-ordinary}{v}}}$ يكون فيه $\pSat{{\OLId{latin-ordinary}{v}}}{{\OLId{greek-variable}{Gamma}}}$، وهذا شرط لازم وكاف.
\item ولتكن ${\OLId{greek-variable}{Gamma}}$ مجموعة من !!{formula}s. إن وُجد لـ~${\OLId{greek-variable}{Gamma}}$
  !!a{valuation}~$\pAssign{{\OLId{latin-ordinary}{v}}}$ بحيث
  $\pSat{{\OLId{latin-ordinary}{v}}}{{\OLId{greek-variable}{Gamma}}}$، قلنا إنها \emph{قابلة للإرضاء}؛ وإلا كانت ${\OLId{greek-variable}{Gamma}}$
  \emph{غير قابلة للإرضاء}.
\end{enumerate}
\end{defn}

\begin{prob}
 انظر في كل واحدة من !!{formula}s الأربع الآتية، وبيّن هل هي
 (أ)~قابلةٌ للإرضاء، و(ب)~تحصيلُ حاصل، و(ج)~عارضةٌ.
\begin{enumerate}
  \item \( ( \Obj p_{\OLMathNumeral{0}} \lif ( \lnot \Obj p_{\OLMathNumeral{1}} \lif \lnot \Obj p_{\OLMathNumeral{0}} ) ) \).
  \item \( ( ( \Obj p_{\OLMathNumeral{0}} \land \lnot \Obj p_{\OLMathNumeral{1}} ) \lif ( \lnot \Obj p_{\OLMathNumeral{0}} \land \Obj p_{\OLMathNumeral{2}} )) \liff ( ( \Obj p_{\OLMathNumeral{2}} \lif \Obj p_{\OLMathNumeral{0}} ) \lif ( \Obj p_{\OLMathNumeral{0}} \lif \Obj p_{\OLMathNumeral{1}} )) \).
  \item \( ( \Obj p_{\OLMathNumeral{0}} \liff \Obj p_{\OLMathNumeral{1}} ) \lif ( \Obj p_{\OLMathNumeral{2}} \liff \lnot \Obj p_{\OLMathNumeral{1}} ) \).
  \item \( (( \Obj p_{\OLMathNumeral{0}} \liff ( \lnot \Obj p_{\OLMathNumeral{1}} \land \Obj p_{\OLMathNumeral{2}} )) \lor ( \Obj p_{\OLMathNumeral{2}} \lif ( \Obj p_{\OLMathNumeral{0}} \liff \Obj p_{\OLMathNumeral{1}} ))) \).
\end{enumerate}
\end{prob}

\begin{prop}
\ollabel{prop:semanticalfacts}
\begin{enumerate}
\item تكون $!A$ تحصيلَ حاصل إذا وفقط إذا كان
  $\emptyset \Entails !A$؛
\item إذا كان ${\OLId{greek-variable}{Gamma}} \Entails !A$ و${\OLId{greek-variable}{Gamma}} \Entails !A \lif !B$، فإن
  ${\OLId{greek-variable}{Gamma}} \Entails !B$؛
\item إذا كانت ${\OLId{greek-variable}{Gamma}}$ قابلةً للإرضاء، فإن كل مجموعة جزئية منتهية
  من~${\OLId{greek-variable}{Gamma}}$ قابلةٌ للإرضاء أيضًا؛
\item \ollabel{def:monotonicity} الرتابة: إذا كانت ${\OLId{greek-variable}{Gamma}} \subseteq {\OLId{greek-variable}{Delta}}$
  وكان ${\OLId{greek-variable}{Gamma}} \Entails !A$، فإن ${\OLId{greek-variable}{Delta}} \Entails !A$ أيضًا؛
\item \ollabel{def:Cut} التعدي: إذا كان ${\OLId{greek-variable}{Gamma}} \Entails !A$ و
  ${\OLId{greek-variable}{Delta}} \cup \{ !A\} \Entails !B$، فإن ${\OLId{greek-variable}{Gamma}} \cup {\OLId{greek-variable}{Delta}} \Entails
  !B$.
\end{enumerate}
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\end{prob}

\begin{prop}\ollabel{prop:entails-unsat}
  ${\OLId{greek-variable}{Gamma}} \Entails !A$ إذا وفقط إذا كانت ${\OLId{greek-variable}{Gamma}} \cup \{\lnot !A\}$
  غير قابلة للإرضاء.
\end{prop}

\begin{proof}

\end{proof}
