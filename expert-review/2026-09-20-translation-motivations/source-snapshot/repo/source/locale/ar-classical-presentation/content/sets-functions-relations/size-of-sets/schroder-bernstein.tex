\olsection{مفهوم الحجم ومبرهنة شرودر–برنشتاين}

\begin{explain}
من مقتضى تصور الحجم أن ${\OLId{latin-looped}{A}}$ إذا لم تزد حجمًا على ${\OLId{latin-looped}{B}}$، ولم تزد ${\OLId{latin-looped}{B}}$ حجمًا على ${\OLId{latin-looped}{A}}$، وجب أن تتساوى ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$ عددًا. ولو كان هذا \emph{باطلًا} لعسر علينا أن نجد لتعريف التساوي العددي صلة بمقارنة «أحجام» المجموعات!{} غير أن هذا المعنى صحيح، ومبرهنة شرودر–برنشتاين هي التي تقيم الدليل عليه.
\end{explain}

\begin{thm}[شرودر–برنشتاين]
\ollabel{thm:schroder-bernstein}
من $\cardle{{\OLId{latin-looped}{A}}}{{\OLId{latin-looped}{B}}}$ و$\cardle{{\OLId{latin-looped}{B}}}{{\OLId{latin-looped}{A}}}$ يلزم $\cardeq{{\OLId{latin-looped}{A}}}{{\OLId{latin-looped}{B}}}$.
\end{thm}

\begin{explain}
ومعناه أن وجود !!a{injection} من ${\OLId{latin-looped}{A}}$ إلى~${\OLId{latin-looped}{B}}$، و!!a{injection} من ${\OLId{latin-looped}{B}}$ إلى~${\OLId{latin-looped}{A}}$، يستلزم وجود !!a{bijection} من ${\OLId{latin-looped}{A}}$ إلى~${\OLId{latin-looped}{B}}$.

على أن إقامة البرهان على هذه النتيجة ليست هينة، بل هي \emph{صعبة}. وقد صرح كانتور بالحكم، وكان إثباته على أيدي غيره.\footnote{وفي تاريخ هذه المبرهنة يراجع مثلًا \citet[ص~165--6]{Potter2004}.}
\oliflabeldef{sfr:cardinals:card-sb:sec}{ولن تتهيأ لنا وسائل \emph{إثبات} مبرهنة شرودر–برنشتاين حتى نبلغ \olref[sfr][cardinals][card-sb]{sec}.}{}%
فنستعملها الآن مسلّمين بصحتها، ونؤخر برهانها.

فقد وافقت مبرهنة شرودر–برنشتاين ذلك الحدس، وهي \emph{صحيحة}، وبصحتها يستقيم حمل العلاقتين اللتين عرفناهما، أعني $\cardeq{{\OLId{latin-looped}{A}}}{{\OLId{latin-looped}{B}}}$ و$\cardle{{\OLId{latin-looped}{A}}}{{\OLId{latin-looped}{B}}}$، على المقارنة في «الحجم». ولها، وراء هذا، \emph{نفع عظيم} في البرهان: فقد يعسر تعيين !!a{bijection} بين مجموعتين متساويتين عددًا، فتقسم المبرهنة هذا المطلب إلى مطلبين أيسر منه؛ نعين !!a{injection} من الأولى إلى الثانية، وأخرى في الجهة المقابلة.
\end{explain}
