\olsection{المفاهيم الدلالية}

\begin{explain}
تبقى تعريفات \emph{الصحة} و\emph{الاستلزام} و\emph{قابلية الإرضاء}،
وهي أصول البحث المنطقي في الدلالة، على حالها عند الانتقال من الرتبة
الأولى إلى الثانية. وإنما نبنيها الآن على علاقة إرضاء !!{formula}s
الرتبة الثانية. وأما !!{sentence} هذه الرتبة فهي
!!a{formula} قُيّدت جميع متغيراتها، بما فيها متغيرات المحمولات
والدوال.
\end{explain}

\begin{defn}[الصحة]
الجملة~$!A$ \emph{صحيحة منطقيًا}، ورمز ذلك~$\Entails !A$، إذا
وفقط إذا كان~$\Sat{{\OLId{latin-looped}{M}}}{!A}$ في كل~!!{structure}~$\Struct M$.
\end{defn}

\begin{defn}[الاستلزام]
نقول إن مجموعة الجمل~${\OLId{greek-variable}{Gamma}}$ \emph{تستلزم} الجملة~$!A$، ونكتب
${\OLId{greek-variable}{Gamma}} \Entails !A$، إذا وفقط إذا كانت كل~!!{structure}~$\Struct M$
يتحقق فيها~$\Sat{{\OLId{latin-looped}{M}}}{{\OLId{greek-variable}{Gamma}}}$ يتحقق فيها كذلك~$\Sat{{\OLId{latin-looped}{M}}}{!A}$.
\end{defn}

\begin{defn}[قابلية الإرضاء]
مجموعة الجمل~${\OLId{greek-variable}{Gamma}}$ \emph{قابلة للإرضاء} متى كان
$\Sat{{\OLId{latin-looped}{M}}}{{\OLId{greek-variable}{Gamma}}}$ في بعض~!!{structure}~$\Struct M$.
فإن لم تكن~${\OLId{greek-variable}{Gamma}}$ كذلك قلنا إنها \emph{غير قابلة للإرضاء}.
\end{defn}
