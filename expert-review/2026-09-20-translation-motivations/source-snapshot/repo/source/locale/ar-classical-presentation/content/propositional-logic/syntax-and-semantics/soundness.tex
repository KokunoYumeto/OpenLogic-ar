\olsection{في سلامة منطق القضايا}

\begin{thm}[السلامة] 
\ollabel{thm:soundness} 
متى كان ${\OLId{greek-variable}{Gamma}}\Proves!A$، كان
${\OLId{greek-variable}{Gamma}} \Entails !A$. 
\end{thm}

\begin{proof} 
نستقرئ في اشتقاق $!A$. فإن كانت $!A$ مسلّمة منطقية، كان $\Entails
!A$، ومن باب أولى ${\OLId{greek-variable}{Gamma}} \Entails!A$. وكذلك إذا كانت $!A \in {\OLId{greek-variable}{Gamma}}$،
فإن ${\OLId{greek-variable}{Gamma}} \Entails!A$. وأما خطوة الاستقراء، فافترض أن $!A$ نتجت بقاعدة
\emph{الفصل} من $!C$ و$!C\lif!A$؛ وأن فرضي الاستقراء يعطيان
${\OLId{greek-variable}{Gamma}} \Entails !C$ و${\OLId{greek-variable}{Gamma}} \Entails (!C\lif !A)$. فتنتج الدعوى من
البند الثالث في \olref[sem]{prop:semanticalfacts}. 
\end{proof}

\begin{cor} 
كل ${\OLId{greek-variable}{Gamma}}$ قابلة للإرضاء متسقة؛ ومن ثم كان منطق القضايا متسقًا. 
\end{cor}
