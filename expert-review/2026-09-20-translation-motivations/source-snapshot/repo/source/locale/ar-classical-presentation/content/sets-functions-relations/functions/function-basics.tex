\olsection{الأساسيات}

\begin{explain}
\emph{الدالة} تعيين يجعل لكل !!{element} من مجموعة معطاة
!!{element} بعينه من مجموعة معطاة أخرى. فإضافة~${\OLMathNumeral{1}}$ دالة؛ إذ
تقابل كل عدد~${\OLId{latin-ordinary}{n}}$ بالعدد الواحد~${\OLId{latin-ordinary}{n}}+{\OLMathNumeral{1}}$.

وقد تكون المدخلات، على وجه أعم، أزواجًا أو ثلاثيات أو غير ذلك،
وتكون لها مخرجات. ومن مألوف الحساب الجمع والضرب؛ كل منهما دالة
تأخذ عددين وتعطي عددًا ثالثًا.

فالدالة بهذا المعنى الرياضي المجرد \emph{صندوق أسود}:
إنما الاعتبار بما يقابل كل مدخل من مخرج، لا بطريقة حساب ذلك المخرج.
\end{explain}

\begin{defn}[الدالة]
\emph{الدالة} ${\OLId{latin-ordinary}{f}} \colon {\OLId{latin-looped}{A}} \to {\OLId{latin-looped}{B}}$ تعيين يقابل كل !!{element}
من~${\OLId{latin-looped}{A}}$ بـ!!{element} واحد من~${\OLId{latin-looped}{B}}$.

نسمي ${\OLId{latin-looped}{A}}$ \emph{مجال}~${\OLId{latin-ordinary}{f}}$، و${\OLId{latin-looped}{B}}$ \emph{المجال المقابل} لـ~${\OLId{latin-ordinary}{f}}$.
و!!{element}s~${\OLId{latin-looped}{A}}$ مدخلات~${\OLId{latin-ordinary}{f}}$، أو \emph{وسائطها}. أما !!{element}
من~${\OLId{latin-looped}{B}}$ المقابل للوسيطة~${\OLId{latin-ordinary}{x}}$ بواسطة~${\OLId{latin-ordinary}{f}}$ فهو \emph{قيمة~${\OLId{latin-ordinary}{f}}$} عند~${\OLId{latin-ordinary}{x}}$،
وكتابتها~${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}})$.

و\emph{المدى} $\ran{{\OLId{latin-ordinary}{f}}}$ للدالة~${\OLId{latin-ordinary}{f}}$ جزئية المجال المقابل
الجامعة لقيم~${\OLId{latin-ordinary}{f}}$ التي تحصل عند وسيطة ما؛ أي $\ran{{\OLId{latin-ordinary}{f}}} =
\Setabs{{\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}})}{{\OLId{latin-ordinary}{x}} \in {\OLId{latin-looped}{A}}}$.
\end{defn}

ولتقريب التصور انظر \olref{fig:function}: فالقطع الناقص الأيسر
\emph{مجال} الدالة، والأيمن \emph{مجالها المقابل}؛ والسهم ينتقل من
\emph{وسيطة} في المجال إلى \emph{قيمتها} الموافقة في المجال المقابل.

\begin{ex}
الضرب يأخذ زوجًا من الطبيعيّات ويعطي طبيعيًا؛ فمجاله
$\Nat \times \Nat$، ومجاله المقابل $\Nat$. ومداه أيضًا $\Nat$،
إذ كل ${\OLId{latin-ordinary}{n}} \in \Nat$ حاصل ضرب ${\OLId{latin-ordinary}{n}} \times {\OLMathNumeral{1}}$.
\end{ex}

\begin{ex}
وإنما كان الضرب دالة لأنه يجعل للمدخل، وهو زوج الطبيعيّات،
مخرجًا واحدًا: $\times \colon \Nat^{\OLMathNumeral{2}} \to \Nat$.
وأما مقابلة الطبيعي~${\OLId{latin-ordinary}{n}}$ بحقيقي~${\OLId{latin-ordinary}{x}}$ يحقق ${\OLId{latin-ordinary}{x}}^{\OLMathNumeral{2}} = {\OLId{latin-ordinary}{n}}$ فليست دالة؛
فلكل صحيح موجب~${\OLId{latin-ordinary}{n}}$ جذران: $\sqrt{{\OLId{latin-ordinary}{n}}}$ و~$-\sqrt{{\OLId{latin-ordinary}{n}}}$.
فإن اقتصرنا على الجذر غير السالب، والموجب عند كون المدخل موجبًا،
حصلت الدالة $\sqrt{\phantom{{\OLId{latin-looped}{X}}}} \colon \Nat \to \Real$.
\end{ex}

\begin{explain}
\textbf{تنبيه على لفظ الأصل.} وصف الأصل الإنجليزي المختار على~$\Nat$
بالموجب، ولما كان~${\OLMathNumeral{0}}\in\Nat$ و$\sqrt{{\OLMathNumeral{0}}}={\OLMathNumeral{0}}$ لم يكن موجبًا؛ فلذلك قلنا غير السالب، وهو الجذر
الرئيسي. وأما كون العدد الصحيح الموجب ذا جذرين فباق على حاله.
\end{explain}

\begin{ex}
ربط كل طالب في مقرر بدرجته النهائية دالة؛ إذ لا يكون له في المقرر
نفسه درجتان نهائيتان مختلفتان. وأما ربطه بمن يعد والدًا أو والدة
له فليس دالة: فقد لا يوجد أحد، وقد يوجد اثنان أو أكثر.
\end{ex}

\begin{explain}
لتعريف الدالة نضبط قيمتها عند كل وسيطة ممكنة. وقد يكون الضبط بصيغة،
أو بوصف طريقة الحساب، أو بذكر قيمة كل وسيطة. وأيًا كانت الطريقة،
فالشرط أن تعيّن لكل وسيطة قيمة واحدة لا تزيد ولا تنقص.
\end{explain}

\begin{ex}
عرّف ${\OLId{latin-ordinary}{f}} \colon \Nat \to \Nat$ بالشرط ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}) = {\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}}$.
فقد جعلنا ${\OLId{latin-ordinary}{f}}$ تأخذ الطبيعيّات وتعطي الطبيعيّات: إذا أعطيت~${\OLId{latin-ordinary}{x}}$
أخرجت~${\OLId{latin-ordinary}{f}}$ لاحقه~${\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}}$. غير أن المجال المقابل $\Nat$ ليس
مدى~${\OLId{latin-ordinary}{f}}$؛ فالعدد الطبيعي~${\OLMathNumeral{0}}$ ليس لاحقًا لطبيعي. وإنما مدى~${\OLId{latin-ordinary}{f}}$
جميع الصحيحات الموجبة $\Int^{+}$.
\end{ex}

\begin{ex}\ollabel{examplefunext}
وعرّف ${\OLId{latin-ordinary}{g}} \colon \Nat \to \Nat$ بالشرط ${\OLId{latin-ordinary}{g}}({\OLId{latin-ordinary}{x}}) = {\OLId{latin-ordinary}{x}}+{\OLMathNumeral{2}}-{\OLMathNumeral{1}}$.
فتأخذ ${\OLId{latin-ordinary}{g}}$ الطبيعيّات وتعطي الطبيعيّات. وإذا أعطي طبيعي~${\OLId{latin-ordinary}{x}}$،
تحسب~${\OLId{latin-ordinary}{g}}$ العدد السابق للعدد اللاحق للعدد اللاحق لـ~${\OLId{latin-ordinary}{x}}$،
فيحصل ${\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}}$.
\end{ex}

\begin{explain}
\textbf{تنبيه على لفظ الأصل.} سمى الأصل الوسيطة~${\OLId{latin-ordinary}{n}}$ ثم عاد فسماها~${\OLId{latin-ordinary}{x}}$؛
فوحدنا الاسم على~${\OLId{latin-ordinary}{x}}$، وهو اسمها في التعريف، ولم نغيّر الدالة.
\end{explain}

\begin{explain}
فـ${\OLId{latin-ordinary}{f}}$ و${\OLId{latin-ordinary}{g}}$ مختلفتان في \emph{التعريف}، ولكنهما \emph{دالة واحدة}.
إذ لكل طبيعي~${\OLId{latin-ordinary}{n}}$ يكون ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{n}}) = {\OLId{latin-ordinary}{n}}+{\OLMathNumeral{1}} = {\OLId{latin-ordinary}{n}}+{\OLMathNumeral{2}}-{\OLMathNumeral{1}} =
{\OLId{latin-ordinary}{g}}({\OLId{latin-ordinary}{n}})$. فتعريف~${\OLId{latin-ordinary}{f}}$ وتعريف~${\OLId{latin-ordinary}{g}}$ يضبطان التعيين نفسه بمعادلتين
مختلفتين. وهذا مبني ضمنًا على امتدادية الدوال:
\[
  \text{إذا كان }\forall {\OLId{latin-ordinary}{x}}\, {\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}) = {\OLId{latin-ordinary}{g}}({\OLId{latin-ordinary}{x}})\text{، فإن }{\OLId{latin-ordinary}{f}} = {\OLId{latin-ordinary}{g}}
\]
وشرطه أن يتحد مجال ${\OLId{latin-ordinary}{f}}$ و~${\OLId{latin-ordinary}{g}}$ ومجالهما المقابل.
\end{explain}

\begin{ex}
ومن طرق التعريف التفصيل بحسب الحالات. فالدالة
${\OLId{latin-ordinary}{h}} \colon \Nat \to \Nat$ يمكن وضعها هكذا:
\[
{\OLId{latin-ordinary}{h}}({\OLId{latin-ordinary}{x}}) =
\begin{cases}
  \frac{{\OLId{latin-ordinary}{x}}}{{\OLMathNumeral{2}}} & \text{إذا كان ${\OLId{latin-ordinary}{x}}$ زوجيًا} \\
  \frac{{\OLId{latin-ordinary}{x}}+{\OLMathNumeral{1}}}{{\OLMathNumeral{2}}} & \text{إذا كان ${\OLId{latin-ordinary}{x}}$ فرديًا.}
\end{cases}
\]
وكل طبيعي إما زوجي وإما فردي، فتكون القيمة في الحالين طبيعية.
والمعتبر في هذا الطريق أن يستوعب التفصيل كل مدخل، وألا يدخل
المدخل في أكثر من حالة واحدة. وقد يحتاج استيفاء الحالات
وتنافيها إلى برهان.
\end{ex}
