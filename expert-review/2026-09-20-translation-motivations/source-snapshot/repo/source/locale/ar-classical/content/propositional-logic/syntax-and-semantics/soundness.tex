\olsection{في سلامة منطق القضايا}

\begin{thm}[السلامة] 
\ollabel{thm:soundness} 
متى كان $\Gamma\Proves!A$، كان
$\Gamma \Entails !A$. 
\end{thm}

\begin{proof} 
نستقرئ في اشتقاق $!A$. فإن كانت $!A$ مسلّمة منطقية، كان $\Entails
!A$، ومن باب أولى $\Gamma \Entails!A$. وكذلك إذا كانت $!A \in \Gamma$،
فإن $\Gamma \Entails!A$. وأما خطوة الاستقراء، فافترض أن $!A$ نتجت بقاعدة
\emph{الفصل} من $!C$ و$!C\lif!A$؛ وأن فرضي الاستقراء يعطيان
$\Gamma \Entails !C$ و$\Gamma \Entails (!C\lif !A)$. فتنتج الدعوى من
البند الثالث في \olref[sem]{prop:semanticalfacts}. 
\end{proof}

\begin{cor} 
كل $\Gamma$ قابلة للإرضاء متسقة؛ ومن ثم كان منطق القضايا متسقًا. 
\end{cor}
