\olsection{\usetoken{P}{derivation} مع \usetoken{S}{identity}}

إذا أُدخلت !!{identity} في !!^{derivation}s، لزم أن نزيد تتابعيات ابتدائية وقواعد استدلال.

\begin{defn}[التتابعيات الابتدائية لـ~$\eq$]
إذا كان $t$ حدًا مغلقًا، فإن ${} \Sequent \eq[t][t]$ تتابعية ابتدائية.
\end{defn}

قواعد~$\eq$ هي الآتية ($t_1$ و$t_2$ حدان مغلقان):

\begin{defish}
\Axiom$ \eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_1) $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_2)$
\DisplayProof
\hfill
\Axiom$\eq[t_1][t_2], \Gamma \fCenter \Delta, !A(t_2) $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_1][t_2], \Gamma  \fCenter \Delta, !A(t_1)$
\DisplayProof
\end{defish}

\begin{ex}
إذا كان $s$ و$t$ حدين مغلقين، فإن $\eq[s][t], !A(s)
\Proves !A(t)$:
\begin{prooftree}
\Axiom$ !A(s) \fCenter !A(s)$
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$\eq[s][t], !A(s)  \fCenter !A(s)$
\RightLabel{$\eq$}
\UnaryInf$\eq[s][t], !A(s)  \fCenter !A(t)$
\end{prooftree}
وهذا هو المبدأ المعروف باستبدال المتطابقات، ويسمى أيضًا قانون لايبنتس.

يثبت~$\Log{LK}$ أن~$\eq$ تناظرية ومتعدية:
\begin{prooftree}
\Axiom$ \fCenter \eq[t_1][t_1] $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$ \eq[t_1][t_2] \fCenter \eq[t_1][t_1] $
\RightLabel{$\eq$}
\UnaryInf$ \eq[t_1][t_2] \fCenter \eq[t_2][t_1]$
\DisplayProof\qquad\bottomAlignProof
\Axiom$ \eq[t_1][t_2] \fCenter \eq[t_1][t_2] $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$\eq[t_2][t_3], \eq[t_1][t_2]  \fCenter \eq[t_1][t_2] $
\RightLabel{$\eq$}
\UnaryInf$\eq[t_2][t_3], \eq[t_1][t_2]  \fCenter \eq[t_1][t_3]$
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$\eq[t_1][t_2], \eq[t_2][t_3]  \fCenter \eq[t_1][t_3]$
\end{prooftree}
أما في ال!!{derivation} الأيسر فال!!{formula}~$\eq[x][t_1]$ هي المختارة لـ~$!A(x)$.
وأما في الأيمن فنختار لـ~$!A(x)$ الصيغة~$\eq[t_1][x]$.
\end{ex}

\begin{prob}
أعط !!{derivation}s للتتابعيات الآتية:
\begin{enumerate}
\item $\Sequent \lforall[x][\lforall[y][((x = y \land !A(x)) \lif !A(y))]]$
\item $\lexists[x][!A(x)] \land \lforall[y][\lforall[z][((!A(y) \land
    !A(z)) \lif y = z)]] \Sequent 
\lexists[x][(!A(x) \land \lforall[y][(!A(y) \lif y = x)])]$
\end{enumerate}
\end{prob}
