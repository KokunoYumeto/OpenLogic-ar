\olsection{خواص الإطار الزمني}

لما لم نفرض في نماذجنا الزمنية شرطًا على العلاقة~$\prec$، لم يصح في جميعها
من البديهيات المألوفة إلا~$K$، أو نظيراها $K_{\Gtemp}$ و$K_{\Htemp}$:
\begin{align*}
\tag{$K_{\Gtemp}$}  \Gtemp (p \to q) & \to (\Gtemp p \to \Gtemp q)\\
\tag{$K_{\Htemp}$}  \Htemp (p \to q) & \to (\Htemp p \to \Htemp q)
\end{align*}

فإن أردنا تمثيل الزمن على وجه أخص، ففرضنا مثلًا أن~$\prec$ ترتيب خطي، لزمت
في النماذج صلاحيات أخرى.

\begin{table}[t]
  \begin{tabular}{| p{.48\textwidth} || p{.48\textwidth} |}
    \hline
    {\emph{إذا كانت~$\prec$ \dots}} & {\emph{فإن~\dots تصدق في~$\mModel{M}$:}} \\
    \hline \hline
    \emph{متعدية}: \newline
    $\forall u \forall v \forall w ((u \prec v \land v \prec w) \lif u \prec w)$ &
    $\Ftemp \Ftemp p \lif \Ftemp p$  \\
    \hline
    \emph{خطية}: \newline
    $\forall w \forall v (w \prec v \lor w = v \lor v \prec w)$ &
    $(\Ftemp \Ptemp p \lor \Ptemp \Ftemp p) \lif (\Ptemp p \lor p \lor \Ftemp p)$\\
    \hline
    \emph{كثيفة}: \newline
    $\forall w \forall v (w \prec v \to \exists u(w \prec u \land u \prec v))$ &
    $\Ftemp p \lif \Ftemp \Ftemp p$ \\
    \hline
    \emph{غير محدودة (في الماضي)}: \newline
    $\forall w \exists v( v \prec w)$ &
    $\Htemp p \to \Ptemp p$ \\
    \hline
    \emph{غير محدودة (في المستقبل)}: \newline
    $\forall w \exists v( w \prec v)$ &
    $\Gtemp p \to \Ftemp p$ \\
    \hline
  \end{tabular}
  \caption{بعض الخواص المتقابلة بين الأطر والصيغ الزمنية.}
  \ollabel{tab:correspondence}
\end{table}

وكثير من خواص~\olref{tab:correspondence} مما يلائم نموذجًا يراد به تمثيل
الزمن. غير أن الشروط التي نستطيع فرضها على~$\prec$ أوسع من الشروط التي
تقابلها !!{formula}s في لغة المنطق الزمني. فاللاانعكاسية، مثلًا، ومعناها
$\forall w \lnot (w \prec w)$، لا تقابلها صيغة في هذه اللغة.
