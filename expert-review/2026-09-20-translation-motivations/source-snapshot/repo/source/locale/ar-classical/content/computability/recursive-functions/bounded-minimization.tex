\olsection{التصغير المحدود}

\begin{explain}
قد تدعو الحاجة إلى تعريف دالة بأصغر عدد يحقق خاصية أو
علاقة~$P$. ومتى كانت $P$ قابلة للقرار، أمكن البحث عن ذلك العدد
بتجريب $0$ و$1$ و$2$ و… على الترتيب، حتى نجد أول ما
يحقق~$P$. وهذا بحث غير محدود، فلا يلزم بقاؤه في جملة الدوال
العودية البدائية. وأما إذا طلبنا أصغر عدد
\emph{أدنى من حد معطى مستقلًا}، بقي الأمر في نطاق العودية البدائية.
وبيانه على وجه أعم: لنأخذ علاقة عودية بدائية~$R(x,z)$،
ولنطلب الدالة التي تسند إلى كل $x$ و~$y$ أصغر $z<y$ تصدق معه
$R(x,z)$. يتأتى حسابها بفحص $R(x,0)$، ثم $R(x,1)$، وهكذا إلى
$R(x,y-1)$. فما الدليل على عوديتها البدائية؟
\end{explain}

\begin{prop}
متى كانت $R(\vec x,z)$ عودية بدائية، كانت $m_R(\vec{x},y)$
عودية بدائية كذلك. وقيمتها أصغر~$z$ أدنى من~$y$ يحقق
$R(\vec x,z)$ إن وجد، وإلا فقيمتها $y$. ونرمز إلى هذه الدالة~$m_R$ بـ
\[
\bmin{z < y}{R(\vec{x}, z)}.
\]
\end{prop}

\begin{proof}
أما حد البدء فلا يوجد~$z<0$ يحقق $R(\vec{x},z)$، لامتناع
$z<0$ أصلًا؛ فلا بد من $m_R(\vec x,0)=0$.

وأما إذا صار الحد $y+1$، فلا يخلو الأمر من ثلاث حالات:
\begin{enumerate}
\item يوجد $z<y$ بحيث تصدق $R(\vec{x},z)$، وعندئذ
$m_R(\vec{x},y+1)=m_R(\vec{x},y)$.
\item لا يوجد $z$ كهذا أدنى من $y$، لكن $R(\vec{x},y)$ تصدق، وعندئذ
$m_R(\vec{x},y+1)=y$.
\item لا يوجد $z<y+1$ بحيث تصدق $R(\vec{x},z)$، وعندئذ
$m_R(\vec{x},y+1)=y+1$.
\end{enumerate}
فبدءًا بالقيمة $m_R(\vec x,0)$ نعرّف الدالة بالعودية البدائية كما يأتي:
\begin{align*}
m_R(\vec{x}, 0) & = 0\\
m_R(\vec{x}, y+1) & =
\begin{cases}
m_R(\vec{x}, y) & \text{إذا كان $m_R(\vec x, y) \neq y$}\\
y & \text{إذا كان $m_R(\vec x, y) = y$ وصدقت $R(\vec{x}, y)$}\\
y+1 & \text{في غير ذلك.}
\end{cases}
\end{align*}
ووجه الاستدلال في شرط الفرع الأول أن وجود $z<y$ يحقق $R(\vec x,z)$ يكافئ
$m_R(\vec x,y)\neq y$.
\end{proof}

\begin{prob}
لتكن $R(\vec x,z)$ عودية بدائية. استخرج بالعودية البدائية من
~$\Char{R}$ الدالة $m'_R(\vec{x},y)$، جاعلًا قيمتها أصغر~$z$ أدنى من~$y$
يحقق $R(\vec{x},z)$ إن وجد، و$0$ إن لم يوجد.
\end{prob}
