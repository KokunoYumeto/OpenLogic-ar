\olsection{\usetoken{P}{derivation} مع \usetoken{S}{identity}}

إذا دخلت !!{identity} في !!^{derivation}s، لزم أن نضم إلى القواعد
المتقدمة القواعد الآتية.

\begin{defish}
\AxiomC{}
\RightLabel{\Intro{\eq}}
\UnaryInfC{$\eq[t][t]$}
\DisplayProof
\hfill
\begin{tabular}{r}
\AxiomC{$\eq[t_1][t_2]$}
\AxiomC{$!A(t_1)$}
\RightLabel{\Elim{\eq}}
\BinaryInfC{$!A(t_2)$}
\DisplayProof
\\[3ex]
\AxiomC{$\eq[t_1][t_2]$}
\AxiomC{$!A(t_2)$}
\RightLabel{\Elim{\eq}}
\BinaryInfC{$!A(t_1)$}
\DisplayProof
\end{tabular}
\end{defish}

والحدود~$t$ و$t_1$ و$t_2$ في هذه القواعد مغلقة. أما
قاعدة~\Intro{\eq} فتجيز !!{derive} كل عبارة هوية صورتها
$\eq[t][t]$ ابتداءً، من غير أن نفتقر إلى افتراض.

\begin{ex}
إذا كان~$s$ و$t$ حدين مغلقين، فإن~$!A(s), \eq[s][t] \Proves !A(t)$:
\begin{prooftree}
\AxiomC{$\eq[s][t]$}
\AxiomC{$!A(s)$}
\RightLabel{$\Elim{\eq}$}
\BinaryInfC{$!A(t)$}
\end{prooftree}
وهذا هو المعروف بـ``مبدأ استبدال المتطابقات''، ويسمّى أيضًا
قانون لايبنتس.
\end{ex}

\begin{prob}
أثبت أن~$=$ تناظرية ومتعدية معًا، أي أعط !!{derivation}s لـ
$\lforall[x][\lforall[y][(\eq[x][y] \lif \eq[y][x])]]$ و
$\lforall[x][\lforall[y][\lforall[z]((\eq[x][y]
    \land \eq[y][z]) \lif \eq[x][z])]]$.
\end{prob}

\begin{ex}
لنبيّن كيف يمكن !!{derive} ال!!{sentence}
\begin{align*}
& \lforall[x][\lforall[y][((!A(x) \land !A(y)) \lif \eq[x][y])]]
\intertext{من ال!!{sentence}}
& \lexists[x][\lforall[y][(!A(y) \lif \eq[y][x])]]
\end{align*}
نبدأ من المطلوب ونرجع إلى ما يكفي له، فتكون خاتمة ال!!{derivation}:
\begin{prooftree}
\AxiomC{$\lexists[x][\lforall[y][(!A(y) \lif \eq[y][x])]]
\quad \Discharge{!A(a) \land !A(b)}{1}$}
\DeduceC{$\eq[a][b]$}
\DischargeRule{\Intro{\lif}}{1}
\UnaryInfC{$((!A(a) \land !A(b)) \lif \eq[a][b])$}
\RightLabel{\Intro{\lforall}}
\UnaryInfC{$\lforall[y][((!A(a) \land !A(y)) \lif \eq[a][y])]$}
\RightLabel{\Intro{\lforall}}
\UnaryInfC{$\lforall[x][\lforall[y][((!A(x) \land !A(y)) \lif \eq[x][y])]]$}
\end{prooftree}
وبقي الانتفاع بالافتراض الرئيس. وهو !!{formula} وجودية، فالقاعدة
المناسبة له~\Elim{\lexists}؛ وبها يمكن !!{derive} النتيجة المتوسطة~$\eq[a][b]$.
\begin{prooftree}
\AxiomC{$\lexists[x][\lforall[y][(!A(y) \lif \eq[y][x])]]$}
\AxiomC{$\Discharge{\lforall[y][(!A(y) \lif \eq[y][c]])}{2}$}
\noLine
\UnaryInfC{$\Discharge{!A(a) \land !A(b)}{1}$}
\DeduceC{$\eq[a][b]$}
\DischargeRule{\Elim{\lexists}}{2}
\BinaryInfC{$\eq[a][b]$}
\DischargeRule{\Intro{\lif}}{1}
\UnaryInfC{$((!A(a) \land !A(b)) \lif \eq[a][b])$}
\RightLabel{\Intro{\lforall}}
\UnaryInfC{$\lforall[y][((!A(a) \land !A(y)) \lif \eq[a][y])]$}
\RightLabel{\Intro{\lforall}}
\UnaryInfC{$\lforall[x][\lforall[y][((!A(x) \land !A(y)) \lif \eq[x][y])]]$}
\end{prooftree}
أما ال!!{derivation} الجزئي في أعلى اليمين، فيتم إذا استخرجنا من
افتراضاته~$\eq[a][c]$ و$\eq[b][c]$. ولكل منهما طريق، فنحتاج إلى
!!{derivation}s منفصلة. وهذا ال!!{derivation} لـ~$\eq[a][c]$:
\begin{prooftree}
\AxiomC{$\Discharge{\lforall[y][(!A(y) \lif \eq[y][c]])}{2}$}
\RightLabel{\Elim{\lforall}}
\UnaryInfC{$!A(a) \lif \eq[a][c]$}
\AxiomC{$\Discharge{!A(a) \land !A(b)}{1}$}
\RightLabel{\Elim{\land}}
\UnaryInfC{$!A(a)$}
\RightLabel{\Elim{\lif}}
\BinaryInfC{$\eq[a][c]$}
\end{prooftree}
فإذا حصل~$\eq[a][c]$ و$\eq[b][c]$ أمكن !!{derive}~$\eq[a][b]$
بقاعدة~\Elim{\eq}، وتم المطلوب.
\end{ex}

\begin{prob}
أعط !!{derivation}s لل!!{formula}s الآتية:
\begin{enumerate}
\item $\lforall[x][\lforall[y][((\eq[x][y] \land !A(x)) \lif !A(y))]]$
\item $\lexists[x][!A(x)] \land \lforall[y][\lforall[z][((!A(y) \land
    !A(z)) \lif \eq[y][z])]] \lif \lexists[x][(!A(x) \land
  \lforall[y][(!A(y) \lif \eq[y][x])])]$
\end{enumerate}
\end{prob}
