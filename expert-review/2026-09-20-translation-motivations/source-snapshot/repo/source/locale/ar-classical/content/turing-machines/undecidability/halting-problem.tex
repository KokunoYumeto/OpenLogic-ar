\olsection{مسألة التوقف}

\begin{explain}
لنثبت تعدادًا لأوصاف آلات تورنغ. فلكل آلة \emph{فهرس} يدل على
موضعها في هذا التعداد: $M_1$، و$M_2$، و$M_3$،~\dots.

وقد علمنا وجود دوال لا تقبل الحساب بآلة تورنغ؛ لأن مجموعة أوصاف
الآلات، ومن ثم الآلات نفسها بالمعنى المتقدم، !!{enumerable}،
ومجموعة الدوال كلها من $\Nat$ إلى~$\Nat$ ليست كذلك. إلا أن الحجة
لا تقف عند إثبات الوجود المجرد؛ بل يمكن تعيين دوال بهذه الصفة.
ومنها دالة التوقف.
\end{explain}

\begin{defn}[دالة التوقف]
 \emph{دالة التوقف}~$h$ هي الدالة التي حدها:
\[
h(e,n) =
\begin{cases}
  \text{0} & \text{إذا لم تتوقف الآلة~$M_e$ عند الدخل $n$} \\
  \text{1} & \text{إذا توقفت الآلة~$M_e$ عند الدخل $n$}
\end{cases}
\]
\end{defn}

\begin{defn}[مسألة التوقف]
\emph{مسألة التوقف} طلب تقرير توقف آلة تورنغ~$M_e$ أو عدم توقفها
عند دخل من~$n$ علامات، وذلك لكل اختيار لـ$e$ و~$n$.
\end{defn}

\begin{explain}
ولإثبات أن~$h$ لا تقبل الحساب بآلة تورنغ، نبدأ بدالة متصلة بها
هي $s$، فنثبت امتناع حسابها. ونعتمد في البرهان على أمرين: أن كل
ما تحسبه آلة تورنغ تحسبه آلة منضبطة
(\olref[mac][dis]{sec})، وأن وصل آلتين ينتج آلة تورنغ واحدة
(\olref[mac][cmb]{sec}).
\end{explain}

\begin{defn} لنجعل الدالة~$s$ محددة بالشرط الآتي:
\[
s(e) =
\begin{cases}
  \text{0} & \text{إذا لم تتوقف الآلة~$M_e$ عند الدخل $e$} \\
  \text{1} & \text{إذا توقفت الآلة~$M_e$ عند الدخل $e$}
\end{cases}
\]
\end{defn}

\begin{lem}
لا تقبل الدالة~$s$ الحساب بآلة تورنغ.
\end{lem}

\begin{proof}
لنفرض خلاف المطلوب، وهو أن الدالة~$s$ قابلة للحساب بآلة تورنغ.
فتوجد آلة~$S$ تحسب~$s$. ولنا، من غير نقصان في العموم، أن نجعل
$S$ منضبطة، أي تتوقف ورأسها على الخانة الأولى. ونصلها بآلة~$J$
تتوقف إذا ابتدأت بالدخل~$0$، أي إذا قرأت $\TMblank$ في الحالة
الابتدائية على الخانة التي تلي علامة نهاية الشريط؛ وأما في غير
ذلك فتسير إلى اليمين أبدًا ولا تتوقف. فالآلة $S\concat J$،
المؤلفة بوصل $S$ بـ~$J$، آلة تورنغ، فلا بد أن تكون $M_e$
لفهرس ما~$e$؛ أي يقع وصفها في التعداد. ولنبدأ تشغيل $M_e$
على دخل من $e$ علامات~$\TMstroke$. ولا تخلو $M_e$ من التوقف
أو عدمه:
\begin{enumerate}
\item إن توقفت $M_e$ على دخل من $e$ علامات~$\TMstroke$،
  كان $s(e)=1$. فتتوقف $S$ عند الدخل~$e$ وقد أخرجت
  علامة~$\TMstroke$ واحدة. وتبدأ $J$ وفي الشريط علامة
  $\TMstroke$، فلا تتوقف $J$ حينئذ. لكن $M_e$ هي
  $S\concat J$، وعملها عمل $S$ تتلوها~$J$؛ فيلزم أن تسير
  إلى اليمين بلا توقف. وهذا يمنع توقف $M_e$ عند الدخل المؤلف
  من $e$ علامات~$\TMstroke$.

\item وإن لم تتوقف $M_e$ على دخل من $e$ علامات
  $\TMstroke$، كان $s(e)=0$. فتتوقف $S$ عند الدخل~$e$
  وقد أخلت الشريط من علامات العدد. وتبدأ $J$ بهذا الشريط فتتوقف
  فورًا. ولما كانت $M_e$ تعمل عمل $S$ تتلوها~$J$، لزم أن
  تتوقف $M_e$ عند دخل من $e$ علامات~$\TMstroke$.
\end{enumerate}
فكلا الفرضين يؤدي إلى نقيضه. إذن لا توجد الآلة~$S$ المفروضة،
ولا تقبل الدالة~$s$ الحساب بآلة تورنغ.
\end{proof}

\begin{thm}[استحالة حل مسألة التوقف]
\ollabel{thm:halting-problem}
لا حل لمسألة التوقف؛ وبعبارة أخرى، لا تقبل الدالة~$h$ الحساب
بآلة تورنغ.
\end{thm}

\begin{proof}
لو كانت $h$ قابلة للحساب بآلة تورنغ~$H$، لأمكن اتخاذ $H$
في بناء آلة تحسب~$s$. ننسخ الدخل أولًا، ونجعل بينه وبين نسخته
علامة~$\TMblank$، ثم نعيد الرأس إلى البداية ونشغّل~$H$.
وصنع آلة للنسخ ممكن كما في \cref{tur:mac:dis:prob:copier}.
فلو وجدت $H$، لوصلناها بالناسخة، فحصلت آلة تقرر توقف $M_e$
عند الدخل~$e$، أي تحسب~$s$. وقد ثبت أن هذه الآلة ممتنعة
الوجود. فلا تكون $h$ قابلة للحساب بآلة تورنغ.
\end{proof}

\begin{prob}
يراد بمسألة التوقف الثلاثي (3-Halt) إيجاد إجراء قرار يبين، لآلة
تورنغ مختارة كيفما اتفق، أتتوقف عند دخل من ثلاث علامات
$\TMstroke$ على شريط خال فيما عداها أم لا. برهن أن هذه المسألة
لا تقبل الحل.
\end{prob}

\begin{prob}
لتكن مسألة التوقف قابلة للحل على أزواج الآلة والدخل
$M_e$ و~$n$ التي فيها $e\neq n$. بيّن أنها تقبل الحل حينئذ
على الأزواج التي فيها $e=n$ أيضًا.
\end{prob}

\begin{prob}
ثبت امتناع حل مسألة التوقف حين يعطى عدد~$e$ يعيّن الآلة~$M_e$
بموضعها في تعداد الآلات. فانظر الآن في إعطاء وصف الآلة نفسه،
على الوجه المبين في \olref[tur][und][enu]{sec}، دخلًا مباشرًا.
أيصح أن توجد آلة تقرر مسألة التوقف إذا أخذت الأوصاف بدل الفهارس؟
أجب مع بيان الحجة.
\end{prob}

\begin{prob} لتكن الدالة \emph{الجزئية}~$s'$ معرفة بالشرط:
  \[
  s'(e) =
  \begin{cases}
    \text{1} & \text{إذا توقفت الآلة~$M_e$ عند الدخل $e$}\\
    \text{غير معرّفة} & \text{إذا لم تتوقف الآلة~$M_e$ عند الدخل $e$}
  \end{cases}
  \]
  برهن أنها \emph{قابلة} للحساب بآلة تورنغ.
\end{prob}
