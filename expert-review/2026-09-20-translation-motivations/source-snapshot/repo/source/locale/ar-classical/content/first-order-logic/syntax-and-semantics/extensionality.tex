\olsection{الامتدادية}

\begin{explain}
الامتدادية، ويقال لها أحيانًا وثاقة الصلة، مؤداها في العبارة غير
الصورية أن إرضاء !!{formula}~$!A$ في
!!a{structure}~$\Struct M$ بحسب إسناد ل!!a{variable}~$s$
لا يتأثر إلا بحجم ال!!{domain}، وبما تعيّنه
$\Struct M$ و$s$ لرموز اللغة التي وردت بالفعل في~$!A$.

ومن ثمرتها أن ال!!{structure}s
~$\Struct M$ و$\Struct M'$ إذا اتفقتا في تأويل رموز جملة~$!A$
وكان مجالهما واحدًا، اتفقت~$\Struct M$
و$\Struct M'$ في الحكم على صدق~$!A$ أو عدم صدقها.
\end{explain}

\begin{prop}[الامتدادية]
  \ollabel{prop:extensionality}
  لتكن $!A$ !!a{formula}، ولتكن $\Struct M_1$ و$\Struct M_2$
  !!{structure}s بحيث $\Domain{M_1} = \Domain{M_2}$، وليكن $s$
  إسناد متغيرات على~$\Domain{M_1} = \Domain{M_2}$. إذا كان
  $\Assign{c}{M_1} = \Assign{c}{M_2}$،
  و$\Assign{R}{M_1}=\Assign{R}{M_2}$، و
  $\Assign{f}{M_1} = \Assign{f}{M_2}$ لكل !!{constant}~$c$،
  ورمز علاقة~$R$، و!!{function}~$f$ ترد في~$!A$، فإن
  $\Sat{M_1}{!A}[s]$ إذا وفقط إذا كان $\Sat{M_2}{!A}[s]$.
\end{prop}

\begin{proof}
  % تصحيح مصدر (0165-I1): قُيّد استقراء الحدود بما يرد في A؛ فالفرض لا يسوّي تأويل الرموز التي لا ترد فيها.
  ابدأ بالاستقراء على بنية كل حد~$t$ يرد في~$!A$ لإثبات أن قيمته
  واحدة في البنيتين، أي $\Value{t}{M_1}[s] = \Value{t}{M_2}[s]$. ثم أجرِ الاستقراء
  على~$!A$ لإثبات القضية، مستعينًا بهذه النتيجة في أساس الاستقراء،
  حيث تكون $!A$ ذرية.
\end{proof}

\begin{prob}
فصّل برهان \olref[fol][syn][ext]{prop:extensionality} بالتفصيل.
\end{prob}

\begin{cor}[امتدادية ال!!^{sentence}s]
  \ollabel{cor:extensionality-sent}
  لتكن $!A$ !!a{sentence}، ولتكن $\Struct{M_1}$ و$\Struct{M_2}$ كما
  في~\olref{prop:extensionality}. عندئذ
  $\Sat{M_1}{!A}$ إذا وفقط إذا كان $\Sat{M_2}{!A}$.
\end{cor}

\begin{proof}
تلزم من \olref{prop:extensionality} باستعمال
\olref[ass]{cor:sat-sentence}.
\end{proof}

وكذلك لا تتوقف قيمة الحد، ولا إرضاء !!a{structure} ل!!a{formula}
أو عدمه، إلا على قيم الحدود الجزئية المعنية.

\begin{prop}\ollabel{prop:ext-terms}
لتكن $\Struct M$ !!a{structure}، وليكن $t$ و$t'$ حدين، وليكن $s$
إسناد متغيرات. عندئذ $\Value{\Subst{t}{t'}{x}}{M}[s] =
\Value{t}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]$.
\end{prop}

\begin{proof}
بالاستقراء على~$t$.
\begin{enumerate}
\item أما إذا كان $t$ رمزًا ثابتًا، وفرضنا $t\ident c$، فإن
  $\Subst{t}{t'}{x} = c$، و$\Value{c}{M}[s] = \Assign{c}{M} =
  \Value{c}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]$.

\item وأما إذا كان $t$ متغيرًا غير~$x$، وفرضنا $t \ident y$، فإن
  $\Subst{t}{t'}{x} = y$، و$\Value{y}{M}[s] =
  \Value{y}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]$ لأن
  $\varAssign{s}{\Subst{s}{\Value{t'}{M}[s]}{x}}{x}$.

\item وأما إذا كان $t \ident x$، فإن $\Subst{t}{t'}{x} = t'$. لكن
  $\Value{x}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}] = \Value{t'}{M}[s]$
  بحسب تعريف~$\Subst{s}{\Value{t'}{M}[s]}{x}$.

\item ويبقى أن يكون $t \ident \Atom{f}{t_1,\dots,t_n}$، وعندئذ:
\begin{multline*}
  \Value{\Subst{t}{t'}{x}}{M}[s]  = \\
  \begin{aligned}[b]
& \Value{\Atom{f}{\Subst{t_1}{t'}{x}, \dots, \Subst{t_n}{t'}{x}}}{M}[s]\\
& \qquad    \text{ بحسب تعريف $\Subst{t}{t'}{x}$}\\
& = \Assign{f}{M}(\Value{\Subst{t_1}{t'}{x}}{M}[s], \dots,
    \Value{\Subst{t_n}{t'}{x}}{M}[s])\\
    & \qquad  \text{ بحسب تعريف $\Value{\Atom{f}{\dots}}{M}[s]$}\\
& = \Assign{f}{M}(\Value{t_1}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}], \dots,
   \Value{t_n}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}])\\
& \qquad    \text{ بفرضية الاستقراء}\\
& = \Value{t}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]
    \text{ بحسب تعريف $\Value{\Atom{f}{\dots}}{M}[\Subst{s}{\Value{t'}{M}[s]}{x}]$}
  \end{aligned}
\end{multline*}
\end{enumerate}
\end{proof}

\begin{prop}\ollabel{prop:ext-formulas} لتكن $\Struct M$
!!a{structure}، ولتكن $!A$ !!a{formula}، وليكن $t'$~حدًا، وليكن
$s$~إسناد متغيرات. عندئذ
$\Sat{M}{\Subst{!A}{t'}{x}}[s]$ إذا وفقط إذا كان
$\Sat{M}{!A}[\Subst{s}{\Value{t'}{M}[s]}{x}]$.
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\begin{prob}
أثبت \olref[fol][syn][ext]{prop:ext-formulas}.
\end{prob}

\begin{explain}
  وبيان المراد من
  \cref{fol:syn:ext:prop:ext-terms,fol:syn:ext:prop:ext-formulas}
  أن يكون معنا حد~$t$ أو !!a{formula}~$!A$، وحد آخر~$t'$،
  ونريد تعيين قيمة $\Subst{t}{t'}{x}$، أو الحكم على إرضاء
  $\Subst{!A}{t'}{x}$ في !!a{structure}~$\Struct M$
  بالنسبة إلى إسناد ل!!a{variable}~$s$. ولنا في ذلك طريقان:
  أن نقدّم الإحلال ثم ننظر في القيمة أو الإرضاء بالنسبة إلى
  $\Struct{M}$ و$s$؛ أو أن نقدّم حساب
  $m = \Value{t'}{M}[s]$، وهي قيمة الحد~$t'$ في
  $\Struct{M}$ بحسب~$s$، ثم نبدّل إسناد ال!!{variable}s
  إلى~$\Subst{s}{m}{x}$، وننظر في قيمة~$t$ في~$\Struct{M}$
  مع~$\Subst{s}{m}{x}$، أو في صدق
  $\Sat{M}{!A}[\Subst{s}{m}{x}]$.
  وبمقتضى \Cref{fol:syn:ext:prop:ext-terms,fol:syn:ext:prop:ext-formulas}
  لا يختلف الجواب باختلاف الطريقين.
\end{explain}
