\olsection{المجموعات الجزئية ومجموعات القوى}

\begin{explain}
من وجوه النظر في مجموعتين أن نسأل: أكل ما في إحداهما في الأخرى؟
وهذا وجه كثير الورود في المقارنة بين المجموعات. فلأهميته نخص الحالة
التي يكون فيها \emph{كل ما في مجموعة موجودًا في الأخرى أيضًا} برمز.
\end{explain}

\begin{defn}[المجموعة الجزئية]
المجموعة $A$ \emph{مجموعة جزئية} من~$B$ متى كان كل !!{element} من $A$
أيضًا !!a{element} من~$B$، ونكتب لذلك $A \subseteq B$.
فإن لم تكن $A$ مجموعة جزئية من~$B$ كتبنا $A \not\subseteq B$.
وإن كان $A \subseteq B$ مع $A \neq B$، فكتابته $A \subsetneq B$،
وتسمى $A$ حينئذ \emph{مجموعة جزئية فعلية} من $B$.
\end{defn}

\begin{ex}
كل مجموعة جزئية من نفسها، والخالية $\emptyset$ جزئية من كل مجموعة.
ومن أمثلة ذلك أن الأعداد الطبيعية الزوجية مجموعة جزئية من الأعداد
الطبيعية، وأن $\{ a, b \} \subseteq \{ a, b, c \}$.
وأما $\{ a, b, e
\}$ فليست مجموعة جزئية من $\{ a, b, c \}$.
\end{ex}

\begin{ex}
ليست العضوية هي الجزئية: فالعدد $2$ هو !!{element} في مجموعة الأعداد
الصحيحة، وأما مجموعة الأعداد الزوجية فمجموعة جزئية منها. وقد يجتمع
الوصفان لمجموعة بالنسبة إلى أخرى، فيصدق عليها \emph{كلا الأمرين}: كونها
!!a{element} وكونها مجموعة جزئية. ومثاله $\{0\} \in \{0, \{0\}\}$ مع $\{0\} \subseteq \{0,
\{0\}\}$.
\end{ex}

معيار الامتدادية لتساوي المجموعات هو أن $A = B$ إذا وفقط إذا كان
كل !!{element} من~$A$ أيضًا !!a{element} من~$B$، وبالعكس.
فأما شطره الأول فهو بعينه معنى $A \subseteq B$: كل !!{element}
من~$A$ هو !!a{element} من~$B$. وأما إذا بدّلنا $A$ و$B$، كان
$B \subseteq A$ معناه أن كل !!{element} من~$B$ هو !!a{element}
من~$A$؛ وهذا هو شطر «بالعكس». فحاصل الامتدادية أن المجموعتين تتساويان
إذا وفقط إذا كانت كل واحدة منهما جزئية من صاحبتها.

\begin{prop}
شرط $A = B$ اللازم والكافي اجتماع $A \subseteq B$ و$B \subseteq A$.
\end{prop}

ولنضع هنا رمزًا آخر ينفع فيما يأتي. فقد قلنا في تعريف كون $A$ جزئية
من~$B$: «كل !!{element} من~$A$ هو \dots»، وجعلنا موضع «$\dots$»
قولنا «!!a{element} من $B$». وهذا \emph{النمط} من القول كثير الدوران؛
فحسن أن نجعل له اختصارًا صوريًا.

\begin{defn}\ollabel{forallxina}
الرمز $(\forall x \in A)\phi$ اختصار لـ$\forall x(x \in A \lif
\phi)$، وكذلك $(\exists x \in A)\phi$ اختصار لـ$\exists x(x
\in A \land \phi)$.
\end{defn}

وعلى هذا الاصطلاح يكون $A \subseteq B$ إذا وفقط إذا كان $(\forall
x \in A)x \in B$.

ولننظر الآن في جمع المجموعات الجزئية كلها من مجموعة معطاة في مجموعة واحدة.

\begin{defn}[مجموعة القوى]
\emph{مجموعة قوى}~$A$ هي المجموعة الجامعة لجميع المجموعات الجزئية
من~$A$، ورمزها $\Pow{A}$:
  \[
    \Pow{A} = \Setabs{B}{B \subseteq A}
  \]
\end{defn}

\begin{ex}
إذا طلبنا جميع جزئيات $\{ a, b, c \}$ وجدناها:
$\emptyset$، و$\{a \}$، و$\{b\}$، و$\{c\}$، و$\{a, b\}$، و$\{a, c\}$، و$\{b,
c\}$، و$\{a, b, c\}$. فالمجموعة التي تجمع هذه كلها هي
$\Pow{\{a,b,c\}}$، أي:
\[
\Pow{\{ a, b, c \}} = \{\emptyset, \{a \}, \{b\}, \{c\}, \{a, b\},
\{b, c\}, \{a, c\}, \{a, b, c\}\}
\]
\end{ex}

\begin{prob}
اذكر جزئيات $\{a, b, c, d\}$ جميعًا.
\end{prob}

\begin{prob}
بيّن أن $A$ إذا كان عدد !!{element}sها $n$، كان عدد !!{element}s
المجموعة $\Pow{A}$ هو $2^n$.
\end{prob}
