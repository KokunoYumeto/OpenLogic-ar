  {Duplicate~typed~letter~entry~'#1'.}

% Cache one immutable pinned glyph font per point size. No mathcode/family
% changes; the glyph is a single nonjoining TLT box inside an ordinary atom.
\cs_new_protected:Npn \ol_classical_glyph_box:nn #1#2
  {
    \hbox
      {
        \textdir TLT
        \cs_if_exist:cF {g_ol_classical_letter_font_#1}
          {
            \tex_global:D \exp_after:wN \tex_font:D
              \cs:w g_ol_classical_letter_font_#1\cs_end:
              = "[\OLClassicalLetterMathFontPath]:mode=base"~at~#1pt\scan_stop:
          }
        \cs_set_eq:Nc \OLClassicalSelectedGlyphFont {g_ol_classical_letter_font_#1}
        \OLClassicalSelectedGlyphFont
        \iffontchar\OLClassicalSelectedGlyphFont "#2\scan_stop:
          \char"#2\scan_stop:
        \else
          \msg_fatal:nnn {ol-classical-letters}{glyph}{#2}
        \fi
      }
  }
\cs_new_protected:Npn \ol_classical_glyph:n #1
  {
    \check@mathfonts
    \mathord{\mathchoice
      {\ol_classical_glyph_box:nn{\tf@size}{#1}}
      {\ol_classical_glyph_box:nn{\tf@size}{#1}}
      {\ol_classical_glyph_box:nn{\sf@size}{#1}}
      {\ol_classical_glyph_box:nn{\ssf@size}{#1}}}
  }
% An Arabic name is one shaped RTL label, not a product, operator, index,
% or exponent. Source scripts attach to the resulting outer mathord.
\cs_new_protected:Npn \ol_classical_label_box:nn #1#2
  {
    \hbox{\textdir TRT
      \OLClassicalLetterArabicFont
      \fontsize{#1}{#1}\selectfont
      #2}
  }
\cs_new_protected:Npn \ol_classical_label:n #1
  {
    \check@mathfonts
    \mathord{\mathchoice
      {\ol_classical_label_box:nn{\tf@size}{#1}}
      {\ol_classical_label_box:nn{\tf@size}{#1}}
      {\ol_classical_label_box:nn{\sf@size}{#1}}
      {\ol_classical_label_box:nn{\ssf@size}{#1}}}
  }

% Keys are literal property-list keys, not expanded control-sequence guesses.
\NewDocumentCommand \OLDeclareClassicalLetter {m m m m m}
  {
    \prop_if_in:NnT \g_ol_classical_letter_render_prop {#1/#2}
      {\msg_fatal:nnn {ol-classical-letters}{duplicate}{#1/#2}}
    \str_case:nnF {#3}
      {
        {glyph}{\prop_gput:Nnn \g_ol_classical_letter_render_prop
          {#1/#2}{\ol_classical_glyph:n{#4}}}
        {label}{\prop_gput:Nnn \g_ol_classical_letter_render_prop
          {#1/#2}{\ol_classical_label:n{#4}}}
      }
      {\msg_fatal:nnn {ol-classical-letters}{kind}{#3}}
    \prop_gput:Nnn \g_ol_classical_letter_source_prop {#1/#2}{#5}
  }
\NewDocumentCommand \OLId {m m}
  {
    \prop_get:NnNTF \g_ol_classical_letter_render_prop {#1/#2} \l_tmpa_tl
      {\ensuremath{\tl_use:N \l_tmpa_tl}}
      {\msg_fatal:nnn {ol-classical-letters}{unknown}{#1/#2}}
  }

% Preserve exact original international specimen behavior locally, including
% the source phi/varphi and epsilon/varepsilon aliases. No Greek is rebound.
\NewDocumentCommand \OLInternationalMath {m}
  {
    \ensuremath{\group_begin:
      \bool_if:NT \g_ol_classical_letters_installed_bool
        {\cs_set_eq:NN \formula \OLClassicalSavedFormula}
      #1
    \group_end:}
  }
\NewDocumentCommand \OLIdSource {m m}
  {
    \prop_get:NnNTF \g_ol_classical_letter_source_prop {#1/#2} \l_tmpa_tl
      {\OLInternationalMath{\tl_use:N \l_tmpa_tl}}
      {\msg_fatal:nnn {ol-classical-letters}{unknown}{#1/#2}}
  }
% Same one-argument contract as the original lookup-driven formula macro:
% !A and !{A} consume A only; following scripts/primes/groups are untouched.
\cs_new_protected:Npn \OLClassicalFormula #1 {\OLId{formula}{#1}}
\NewDocumentCommand \OLInstallClassicalLetters {}
  {
    \bool_if:NF \g_ol_classical_letters_installed_bool
      {
        \cs_if_exist:NF \formula
          {\msg_fatal:nn {ol-classical-letters}{installation}}
        \cs_gset_eq:NN \OLClassicalSavedFormula \formula
        \cs_gset_eq:NN \formula \OLClassicalFormula
        \bool_gset_true:N \g_ol_classical_letters_installed_bool
        \iow_term:n {OL-CLASSICAL-LETTERS=typed-opt-in-nucleus-v1}
        \iow_term:n {OL-CLASSICAL-LETTER-UNKNOWN=fatal}
      }
  }
\NewDocumentCommand \OLCheckClassicalLetters {}
  {
    \bool_if:NF \g_ol_classical_letters_installed_bool
      {\msg_fatal:nn {ol-classical-letters}{installation}}
    \cs_if_eq:NNF \formula \OLClassicalFormula
      {\msg_fatal:nn {ol-classical-letters}{installation}}
  }
