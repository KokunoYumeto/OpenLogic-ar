\olsection{القواعد البنيوية}

لا غنى كذلك عن قواعد لتغيير مواضع ال!!{sentence}s في جانبي التتابعية. فإن القاعدة المنطقية لا تعمل إلا إذا كانت ال!!{sentence}s المقصودة في أقصى اليمين أو أقصى اليسار؛ فلا بد من «التبادل» لنقل ال!!{sentence}s إلى مواضع العمل.
وقد نحتاج أيضًا إلى جمع !!{sentence}s متطابقتين في واحدة، أو زيادة
!!a{sentence} في أحد الجانبين.

\subsection{الإضعاف}

\begin{defish}
\Axiom$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}} $
\RightLabel{\LeftR{\Weakening}}
\UnaryInf$ !A, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
\DisplayProof
\hfill
\Axiom$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
\RightLabel{\RightR{\Weakening}}
\UnaryInf$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A$
\DisplayProof
\end{defish}

\subsection{الانكماش}

\begin{defish}
\Axiom$ !A, !A, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}} $
\RightLabel{\LeftR{\Contraction}}
\UnaryInf$ !A, {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}$
\DisplayProof
\hfill
\Axiom$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A, !A$
\RightLabel{\RightR{\Contraction}}
\UnaryInf$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A$
\DisplayProof
\end{defish}

\subsection{التبادل}

\begin{defish}
\Axiom$ {\OLId{greek-variable}{Gamma}}, !A, !B, \Pi \fCenter {\OLId{greek-variable}{Delta}} $
\RightLabel{\LeftR{\Exchange}}
\UnaryInf$ {\OLId{greek-variable}{Gamma}}, !B, !A, \Pi \fCenter {\OLId{greek-variable}{Delta}}$
\DisplayProof
\hfill
\Axiom$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A, !B, \Lambda$
\RightLabel{\RightR{\Exchange}}
\UnaryInf$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !B, !A, \Lambda$
\DisplayProof
\end{defish}

وقد يُكتفى بخطَّي استدلال للدلالة على سلسلة من تطبيقات الإضعاف والانكماش والتبادل.

وأما «القطع»، وهو القاعدة الآتية، فليس مما يتوقف عليه الحساب في المعنى الدقيق؛ غير أنه ييسّر إعادة استعمال ال!!{derivation}s وضم بعضها إلى بعض تيسيرًا كبيرًا.
\begin{defish}
\[
\Axiom$ {\OLId{greek-variable}{Gamma}} \fCenter {\OLId{greek-variable}{Delta}}, !A$
\Axiom$ !A, \Pi \fCenter \Lambda $
\RightLabel{\Cut}
\BinaryInf$ {\OLId{greek-variable}{Gamma}}, \Pi \fCenter {\OLId{greek-variable}{Delta}}, \Lambda$
\DisplayProof
\]
\end{defish}
