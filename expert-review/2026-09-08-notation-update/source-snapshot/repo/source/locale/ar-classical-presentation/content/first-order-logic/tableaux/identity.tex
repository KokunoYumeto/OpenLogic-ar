\olsection{\usetoken{P}{tableau} مع \usetoken{S}{identity}}

إذا دخلت ال!!{identity} في !!^{tableau}s لزم إلحاق قواعد بها.
فقواعد~$\eq$ هي الآتية، على أن تكون~${\OLId{latin-ordinary}{t}}$ و${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$ و${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}$ حدودًا مغلقة:

\begin{defish}
\AxiomC{}
\RightLabel{$\eq$}
\UnaryInfC{\sFmla{\True}{\eq[t][t]}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\True}{\eq[t_1][t_2]}}
\noLine
\UnaryInfC{\sFmla{\True}{!A(t_1)}}
\RightLabel{$\TRule{\True}{\eq}$}
\UnaryInfC{\sFmla{\True}{!A(t_2)}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\True}{\eq[t_1][t_2]}}
\noLine
\UnaryInfC{\sFmla{\False}{!A(t_1)}}
\RightLabel{$\TRule{\False}{\eq}$}
\UnaryInfC{\sFmla{\False}{!A(t_2)}}
\DisplayProof
\end{defish}
وتنفرد~$\TRule{\True}{\eq}$ و$\TRule{\False}{\eq}$ عن القواعد
المتقدمة بأنهما تحتاجان إلى !!{formula}s موقّعتين \emph{اثنتين}
سبقتا على الفرع:~$\sFmla{\True}{\eq[{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}]}$ و
$\sFmla{{\OLId{latin-looped}{S}}}{!A({\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}})}$.

\begin{ex}
إذا كان ${\OLId{latin-ordinary}{s}}$ و${\OLId{latin-ordinary}{t}}$ حدين مغلقين، فإن $\eq[{\OLId{latin-ordinary}{s}}][{\OLId{latin-ordinary}{t}}], !A({\OLId{latin-ordinary}{s}})
\Proves !A({\OLId{latin-ordinary}{t}})$:
\begin{oltableau}
  [\sFmla{\False}{\formula{A}(t)}, just = \TAss
    [\sFmla{\True}{\eq[s][t]}, just = \TAss
      [\sFmla{\True}{\formula{A}(s)}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just={\TRule{\True}{\eq}[2, 3]}, close]
      ]
    ]
  ]
\end{oltableau}
وهذا هو المعروف بمبدأ إحلال المتطابقات، أو قانون لايبنتس.

وبال!!^{tableau}s نثبت تناظر~$\eq$؛ أي
$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}]
\Proves \eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}]$، كما يأتي:
\begin{oltableau}
  [\sFmla{\False}{\eq[s_2][s_1]}, just = \TAss
    [\sFmla{\True}{\eq[s_1][s_2]}, just = \TAss
      [\sFmla{\True}{\eq[s_1][s_1]}, just = {$\eq$}
        [\sFmla{\True}{\eq[s_2][s_1]}, just = {\TRule{\True}{\eq}[2, 3]}, close]
      ]
    ]
  ]
\end{oltableau}
أما السطر~${\OLMathNumeral{2}}$ ففيه ال!!{formula} الأولى المطلوبة،
وهي~$\sFmla{\True}{\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}]}$، لقاعدة~$\TRule{\True}{\eq}$.
وأما السطر~${\OLMathNumeral{3}}$ ففيه الثانية، وصورتها~$\sFmla{\True}{!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}})}$.
وبيان المطابقة أن نأخذ~$!A({\OLId{latin-ordinary}{x}})$ هي~$\eq[{\OLId{latin-ordinary}{x}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}]$؛
فيكون~$!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}})$ هو~$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}]$،
ويكون~$!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}})$ هو~$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}]$.

ونثبت كذلك تعدّي~$\eq$؛ أي~$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}],
\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}}] \Proves \eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}}]$:
\begin{oltableau}
  [\sFmla{\False}{\eq[s_1][s_3]}, just = \TAss
    [\sFmla{\True}{\eq[s_1][s_2]}, just = \TAss
      [\sFmla{\True}{\eq[s_2][s_3]}, just = \TAss
        [\sFmla{\True}{\eq[s_1][s_3]}, just = {\TRule{\True}{\eq}[3, 2]}, close]
      ]
    ]
  ]
\end{oltableau}
في هذا ال!!{tableau} تؤخذ ال!!{formula} الأولى لقاعدة
$\TRule{\True}{\eq}$ من السطر~${\OLMathNumeral{3}}$، وهي~$\sFmla{\True}{\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}}]}$؛
فيقوم~${\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}$ مقام~${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{1}}$، و${\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}}$ مقام~${\OLId{latin-ordinary}{t}}_{\OLMathNumeral{2}}$.
وأما الثانية، ذات الصورة~$\sFmla{\True}{!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}})}$، ففي السطر~${\OLMathNumeral{2}}$.
خذ~$!A({\OLId{latin-ordinary}{x}})$ هي~$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{x}}]$؛ فيصير~$!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}})$
$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{2}}]$، وهو السطر~${\OLMathNumeral{2}}$، ويصير~$!A({\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}})$
ال!!{formula}~$\eq[{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{1}}][{\OLId{latin-ordinary}{s}}_{\OLMathNumeral{3}}]$ التي انتهينا إليها.
\end{ex}

\begin{prob}
أنشئ !!{tableau}s مغلقة لما يأتي:
\begin{enumerate}
\item $\sFmla{\False}{\lforall[{\OLId{latin-ordinary}{x}}][\lforall[{\OLId{latin-ordinary}{y}}][(({\OLId{latin-ordinary}{x}} = {\OLId{latin-ordinary}{y}} \land !A({\OLId{latin-ordinary}{x}}))
      \lif !A({\OLId{latin-ordinary}{y}}))]]}$
\item $\sFmla{\False}{\lexists[{\OLId{latin-ordinary}{x}}][(!A({\OLId{latin-ordinary}{x}}) \land
    \lforall[{\OLId{latin-ordinary}{y}}][(!A({\OLId{latin-ordinary}{y}}) \lif {\OLId{latin-ordinary}{y}} = {\OLId{latin-ordinary}{x}})])]}$,\\
  $\sFmla{\True}{\lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})] \land
    \lforall[{\OLId{latin-ordinary}{y}}][\lforall[{\OLId{latin-ordinary}{z}}][((!A({\OLId{latin-ordinary}{y}}) \land !A({\OLId{latin-ordinary}{z}})) \lif {\OLId{latin-ordinary}{y}} = {\OLId{latin-ordinary}{z}})]]}$
\end{enumerate}
\end{prob}
