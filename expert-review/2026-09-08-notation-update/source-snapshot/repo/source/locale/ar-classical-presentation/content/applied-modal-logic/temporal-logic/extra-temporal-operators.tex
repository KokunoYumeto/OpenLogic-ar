\olsection{مؤثرات زمنية زائدة}

قد يزاد على مؤثرات الماضي والمستقبل الأحادية مؤثران ثنائيان، هما~$\Since$
و$\Until$، للدلالة على «منذ» و«حتى». وعندئذ يضاف~$\Since$ و$\Until$ إلى
لغة المنطق الزمني، ويزاد في حد ال!!{formula} الزمنية الشرط الآتي:

\begin{itemize}
\item[] إذا كانت~$!A$ و$!B$ !!{formula}s، كان كل من
  $(\Since !A !B)$ و$(\Until !A !B)$ !!{formula}s.
\end{itemize}

وأما دلالتهما فهي كما يأتي:

\begin{defn}\ollabel{defn:since-until}
  \emph{صدق !!a{formula}~$!A$ عند~${\OLId{latin-ordinary}{t}}$} في~$\mModel M$:
  \begin{enumerate}
  \item\ollabel{defn:sub:mmodels-since} \indcase{!A}{\Since !B
    !C}{$\mSat{{\OLId{latin-looped}{M}}}{\indfrm}[{\OLId{latin-ordinary}{t}}]$ إذا وفقط إذا كان~$\mSat{{\OLId{latin-looped}{M}}}{!B}[{\OLId{latin-ordinary}{t}}']$
    لبعض~${\OLId{latin-ordinary}{t}}' \in {\OLId{latin-looped}{T}}$ بحيث~${\OLId{latin-ordinary}{t}}' \prec {\OLId{latin-ordinary}{t}}$، وكان لكل~${\OLId{latin-ordinary}{s}}$ بحيث
    ${\OLId{latin-ordinary}{t}}' \prec {\OLId{latin-ordinary}{s}} \prec {\OLId{latin-ordinary}{t}}$، $\mSat{{\OLId{latin-looped}{M}}}{!C}[{\OLId{latin-ordinary}{s}}]$}
  \item\ollabel{defn:sub:mmodels-until} \indcase{!A}{\Until !B
    !C}{$\mSat{{\OLId{latin-looped}{M}}}{\indfrm}[{\OLId{latin-ordinary}{t}}]$ إذا وفقط إذا كان~$\mSat{{\OLId{latin-looped}{M}}}{!B}[{\OLId{latin-ordinary}{t}}']$
    لبعض~${\OLId{latin-ordinary}{t}}' \in {\OLId{latin-looped}{T}}$ بحيث~${\OLId{latin-ordinary}{t}} \prec {\OLId{latin-ordinary}{t}}'$، وكان لكل~${\OLId{latin-ordinary}{s}}$ بحيث
    ${\OLId{latin-ordinary}{t}} \prec {\OLId{latin-ordinary}{s}} \prec {\OLId{latin-ordinary}{t}}'$، $\mSat{{\OLId{latin-looped}{M}}}{!C}[{\OLId{latin-ordinary}{s}}]$}
  \end{enumerate}
\end{defn}

فالمعنى المقصود من~$\Since !B !C$ هو: «منذ أن كانت~$!B$ واقعة، ظلت
$!C$ واقعة»؛ والمقصود من~$\Until !B !C$ هو: «إلى أن تصبح~$!B$ واقعة،
ستظل~$!C$ واقعة».
