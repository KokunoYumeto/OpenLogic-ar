\olsection{المخططات والصلاحية}

\begin{defn}
  \emph{المخطط} مجموعة من ال!!{formula}s تضم جميع حالات الإحلال
  ل!!{formula} موجهية ما~$!C$، ولا تضم سواها، أي
  \[
  \Setabs{!B}{\lexists[!D_{\OLMathNumeral{1}}], \dots, \lexists[!D_{\OLId{latin-ordinary}{n}}] \left(!B =
  \SSubst{!C}{\subst{!D_{\OLMathNumeral{1}}}{{\OLId{latin-ordinary}{p}}_{\OLMathNumeral{1}}}, \dots, \subst{!D_{\OLId{latin-ordinary}{n}}}{{\OLId{latin-ordinary}{p}}_{\OLId{latin-ordinary}{n}}}} \right) }.
  \]
  وتسمى ال!!{formula}~$!C$ ال!!{formula} \emph{المميزة} للمخطط، وهي
  وحيدة حتى إعادة تسمية ال!!{propositional variable}s. وتكون
  !!^a{formula}~$!A$ \emph{حالة} من مخطط إذا كانت عنصرًا في المجموعة.
\end{defn}

من الملائم أن نرمز إلى المخطط بالتعبير الميتالغوي الناتج من إحلال
`$!A$'، و`$!B$'، و\dots، محل المكونات الذرية لـ$!C$. فمثلًا، تدل
التعبيرات الآتية على مخططات: `$!A$'، و`$!A \lif \Box!A$'، و
`$!A \lif (!B \lif !A)$'. وهي تقابل ال!!{formula}s المميزة~${\OLId{latin-ordinary}{p}}$، و
${\OLId{latin-ordinary}{p}} \lif \Box {\OLId{latin-ordinary}{p}}$، و${\OLId{latin-ordinary}{p}} \lif ({\OLId{latin-ordinary}{q}} \lif {\OLId{latin-ordinary}{p}})$. ويدل المخطط~`$!A$' على
مجموعة \emph{جميع} ال!!{formula}s.

\begin{defn}
  يكون المخطط \emph{صادقًا} في نموذج إذا وفقط إذا كانت جميع حالاته
  صادقة فيه؛ ويكون المخطط \emph{صالحًا} إذا وفقط إذا كان صادقًا في كل
  نموذج.
\end{defn}

\begin{prop}\ollabel{prop:Kvalid}
  المخطط الآتي~\Ax{K} صالح:
  \begin{equation*}
  \Box(!A \lif !B) \lif (\Box !A \lif \Box !B). \tag{\Ax{K}}
  \end{equation*}
\end{prop}

\begin{proof}
  المطلوب تعميم الصدق على كل حالة من المخطط، في كل عالم من كل
  نموذج. فخذ~$\mModel{M} = \tuple{{\OLId{latin-looped}{W}},{\OLId{latin-looped}{R}},{\OLId{latin-looped}{V}}}$ و${\OLId{latin-ordinary}{w}} \in {\OLId{latin-looped}{W}}$ كيف اتفق.
  وسبيل إثبات الشرط أن يُفرض مقدمه ويُستخرج تاليه؛ فلنفرض هنا
  $\mSat{{\OLId{latin-looped}{M}}}{\Box(!A \lif !B)}[{\OLId{latin-ordinary}{w}}]$ و$\mSat{{\OLId{latin-looped}{M}}}{\Box !A}[{\OLId{latin-ordinary}{w}}]$،
  ولنثبت~$\mSat{{\OLId{latin-looped}{M}}}{\Box !B}[{\OLId{latin-ordinary}{w}}]$. خذ~${\OLId{latin-ordinary}{w}}'$ بحيث~$Rww'$.
  يفيد الفرض الأول~$\mSat{{\OLId{latin-looped}{M}}}{!A \lif !B}[{\OLId{latin-ordinary}{w}}']$،
  ويفيد الثاني~$\mSat{{\OLId{latin-looped}{M}}}{!A}[{\OLId{latin-ordinary}{w}}']$؛ فينتج~$\mSat{{\OLId{latin-looped}{M}}}{!B}[{\OLId{latin-ordinary}{w}}']$.
  ولما لم نخصّ~${\OLId{latin-ordinary}{w}}'$ بشيء، ثبت~$\mSat{{\OLId{latin-looped}{M}}}{\Box !B}[{\OLId{latin-ordinary}{w}}]$.
\end{proof}

\begin{prop}\ollabel{prop:Dual-valid}
  مخطط الثنائية الآتي~\Dual{} صالح:
  \begin{equation*}
  \Diamond !A \liff \lnot\Box\lnot !A. \tag{\Dual}
  \end{equation*}
\end{prop}

\begin{proof}
  تمرين.
\end{proof}

\begin{prob}
  أثبت~\olref[nml][syn][sch]{prop:Dual-valid}.
\end{prob}

\begin{prop}\ollabel{prop:soundMP}
  إذا كانت~$!A$ و$!A \lif !B$ صادقتين في عالم من نموذج، كانت~$!B$
  صادقة فيه أيضًا. ومن ثم فإن ال!!{formula}s الصالحة مغلقة تحت قاعدة
  إثبات المقدَّم.
\end{prop}

\begin{prop}\ollabel{prop:valid-instances}
  تكون !!^a{formula}~$!A$ صالحة إذا وفقط إذا كانت جميع حالات إحلالها
  صالحة. وبعبارة أخرى، يكون المخطط صالحًا إذا وفقط إذا كانت
  !!{formula}ه المميزة صالحة.
\end{prop}

\begin{proof}
  اتجاه ``إذا'' واضح، لأن~$!A$ حالة إحلال من نفسها.

  وأما الاتجاه الآخر فمداره على دعوى نذكرها أولًا. إذا كان
  $\mModel{M} = \tuple{{\OLId{latin-looped}{W}}, {\OLId{latin-looped}{R}}, {\OLId{latin-looped}{V}}}$ نموذجًا موجهيًا، وكانت
  $!B \ident \SSubst{!A}{\subst{!D_{\OLMathNumeral{1}}}{{\OLId{latin-ordinary}{p}}_{\OLMathNumeral{1}}},\dots,\subst{!D_{\OLId{latin-ordinary}{n}}}{{\OLId{latin-ordinary}{p}}_{\OLId{latin-ordinary}{n}}}}$
  حالة إحلال من~$!A$، فعرّف~$\mModel{M'} = \tuple{{\OLId{latin-looped}{W}}, {\OLId{latin-looped}{R}}, {\OLId{latin-looped}{V}}'}$
  بجعل~${\OLId{latin-looped}{V}}'({\OLId{latin-ordinary}{p}}_{\OLId{latin-ordinary}{i}}) = \Setabs{{\OLId{latin-ordinary}{w}}}{\mSat{{\OLId{latin-looped}{M}}}{!D_{\OLId{latin-ordinary}{i}}}[{\OLId{latin-ordinary}{w}}]}$، وأبقِ تعيين
  سائر المتغيرات على حاله. عندئذ~$\mSat{{\OLId{latin-looped}{M}}}{!B}[{\OLId{latin-ordinary}{w}}]$ متى وفقط متى
  كان~$\mSat{{\OLId{latin-looped}{M}}'}{!A}[{\OLId{latin-ordinary}{w}}]$، لكل~${\OLId{latin-ordinary}{w}} \in {\OLId{latin-looped}{W}}$. وبرهان هذه الدعوى تمرين.
  فلو كانت~$!A$ صالحة، وكانت حالة منها~$!B$ غير صالحة، لوجدنا
  نموذجًا~$\mModel{M} = \tuple{{\OLId{latin-looped}{W}}, {\OLId{latin-looped}{R}}, {\OLId{latin-looped}{V}}}$ وعالمًا~${\OLId{latin-ordinary}{w}} \in {\OLId{latin-looped}{W}}$ فيه
  $\mSat/{{\OLId{latin-looped}{M}}}{!B}[{\OLId{latin-ordinary}{w}}]$. ومن الدعوى يلزم~$\mSat/{{\OLId{latin-looped}{M}}'}{!A}[{\OLId{latin-ordinary}{w}}]$،
  فيخالف ذلك صلاحية~$!A$.
\end{proof}

\begin{prob}
  أثبت الدعوى في جزء ``فقط إذا'' من برهان
  \olref[nml][syn][sch]{prop:valid-instances}. (تلميح: استعمل
  الاستقراء على~$!A$.)
\end{prob}

ولا يحملنّ هذا التكافؤ في الصلاحية على توهّم مثله في نموذج واحد.
فإن صدق المخطط في نموذج يقتضي صدق صيغته المميزة فيه: إذا صدقت كل
حالة من~$!A$ في~$\mModel{M}$، صدقت~$!A$ نفسها في~$\mModel{M}$.
وأما العكس فلا يلزم؛ فقد تصدق~$!A$ في~$\mModel{M}$، وتكذب
إحدى حالاتها في عالم من~$\mModel{M}$. وأبسط شاهد هو~${\OLId{latin-ordinary}{p}}$ في نموذج
عالمه الوحيد~${\OLId{latin-ordinary}{w}}$، وتعيينه~${\OLId{latin-looped}{V}}({\OLId{latin-ordinary}{p}}) = \{{\OLId{latin-ordinary}{w}}\}$. فهنا~${\OLId{latin-ordinary}{p}}$ صادقة
في~${\OLId{latin-ordinary}{w}}$، مع أن~$\lfalse$ حالة من~${\OLId{latin-ordinary}{p}}$ وليست صادقة في~${\OLId{latin-ordinary}{w}}$.

\begin{prob}
  أثبت أن أيًا من ال!!{formula}s الآتية ليس صالحًا:
  \begin{enumerate}
    \item[\Ax{D}:] \quad $\Box {\OLId{latin-ordinary}{p}} \lif \Diamond {\OLId{latin-ordinary}{p}}$؛
    \item[\Ax{T}:] \quad $\Box {\OLId{latin-ordinary}{p}} \lif {\OLId{latin-ordinary}{p}}$؛
    \item[\Ax{B}:] \quad ${\OLId{latin-ordinary}{p}} \lif \Box\Diamond {\OLId{latin-ordinary}{p}}$؛
    \item[\Ax{4}:] \quad $\Box {\OLId{latin-ordinary}{p}} \lif \Box \Box {\OLId{latin-ordinary}{p}}$؛
    \item[\Ax{5}:] \quad $\Diamond {\OLId{latin-ordinary}{p}} \lif \Box \Diamond {\OLId{latin-ordinary}{p}}$.
  \end{enumerate}
\end{prob}

\begin{prob}%\ollabel{ex:in/validSchemas}
  أثبت أن المخططات في العمود الأول من
  \olref[nml][syn][sch]{tab:valid-invalidSchemas} صالحة، وأن التي في
  العمود الثاني غير صالحة.
\end{prob}

\begin{prob}
  قرر هل المخططان الآتيان صالحان أم غير صالحين:
  \begin{enumerate}
  \item $(\Diamond !A \lif \Box !B) \lif (\Box !A \lif \Box
    !B)$؛
  \item $\Diamond(!A \lif !B) \lor \Box(!B \lif !A)$.
  \end{enumerate}
\end{prob}

\begin{prob}
  أوجد لكل واحد من المخططين الآتيين نموذجًا~$\mModel{M}$ تكون فيه كل
  حالة من ال!!{formula} صادقة:
  \begin{enumerate}
  \item ${\OLId{latin-ordinary}{p}} \lif \Diamond\Diamond {\OLId{latin-ordinary}{p}}$؛
  \item $\Diamond {\OLId{latin-ordinary}{p}} \lif \Box {\OLId{latin-ordinary}{p}}$.
  \end{enumerate}
\end{prob}
