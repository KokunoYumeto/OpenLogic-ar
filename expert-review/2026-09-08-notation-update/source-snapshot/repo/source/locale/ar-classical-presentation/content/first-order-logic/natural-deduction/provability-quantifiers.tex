\olsection{\usetoken{S}{derivability} والمكممات}

\begin{explain}
  ومن مقدمات برهان الاكتمال أن تثبت بقواعد الاستنتاج الطبيعي
  النتائج الآتية في علاقة~$\Proves$؛ فلذلك نفردها بالبيان هنا.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} إذا كان~${\OLId{latin-ordinary}{c}}$ رمزًا ثابتًا لا يرد
في~${\OLId{greek-variable}{Gamma}}$ ولا في~$!A({\OLId{latin-ordinary}{x}})$، وكان~${\OLId{greek-variable}{Gamma}} \Proves !A({\OLId{latin-ordinary}{c}})$، فإن
${\OLId{greek-variable}{Gamma}} \Proves \lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$.
\end{thm}

\begin{proof}
نأخذ~$\delta$، وهو !!a{derivation} لـ~$!A({\OLId{latin-ordinary}{c}})$ من~${\OLId{greek-variable}{Gamma}}$، ونختمه
باستدلال \Intro{\lforall}؛ فيتم بذلك !!a{derivation} لـ
$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$. وهذا جائز، إذ~${\OLId{latin-ordinary}{c}}$ لا يرد في~${\OLId{greek-variable}{Gamma}}$ ولا
في~$!A({\OLId{latin-ordinary}{x}})$؛ فلا يختل شرط المتغير المميَّز.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A({\OLId{latin-ordinary}{t}}) \Proves \lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$.}{}

\tagitem{prvAll}{$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})] \Proves !A({\OLId{latin-ordinary}{t}})$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{يكفي !!a{derivation} لـ~$\lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$ من~$!A({\OLId{latin-ordinary}{t}})$
  بإدخال المكمّم الوجودي مرة واحدة:
\begin{prooftree}
\AxiomC{$!A({\OLId{latin-ordinary}{t}})$}
\RightLabel{\Intro{\lexists}}
\UnaryInfC{$\lexists[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$}
\end{prooftree}}{}

\tagitem{prvAll}{وأما !!a{derivation} لـ~$!A({\OLId{latin-ordinary}{t}})$ من
  $\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$ فيكفي فيه حذف المكمّم الكلي مرة واحدة:
\begin{prooftree}
\AxiomC{$\lforall[{\OLId{latin-ordinary}{x}}][!A({\OLId{latin-ordinary}{x}})]$}
\RightLabel{\Elim{\lforall}}
\UnaryInfC{$!A({\OLId{latin-ordinary}{t}})$}
\end{prooftree}}{}
\end{tagenumerate}
\end{proof}
