\olsection{مثال}

نبدأ بحقيقة يسيرة في اتحاد المجموعات وتقاطعها؛ ففي برهانها تتجلى
كيفية بسط التعريفات، وإثبات الاقتران والدعوى الكلية، وتقسيم النظر إلى
حالات.

\begin{prop}
لكل مجموعات ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$ و${\OLId{latin-looped}{C}}$،
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}) = ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\end{prop}

فلنقم برهانها!{}

\begin{proof}
المطلوب أن نبين، لكل مجموعات ${\OLId{latin-looped}{A}}$ و${\OLId{latin-looped}{B}}$ و${\OLId{latin-looped}{C}}$، أن
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}) = ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\begin{quote}
أول العمل بسط «$=$» الواردة في القضية. فتطابق مجموعتين معناه اتحادهما
في !!{element} أنفسها: أي إن كل !!{element} للمجموعة
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$ هو أيضاً !!{element} للمجموعة
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$، وأن الأمر بالعكس. ومراد العكس أن كل
!!{element} في $({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$ يكون كذلك
!!a{element} في ${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$. فبسط التعريف يكشف أن المطلوب
اقتران؛ ولنكتبه صريحاً:
\end{quote}
وبالتعريف، يصدق
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}) = ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$ إذا وفقط إذا كان كل
!!{element} في ${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$ !!a{element} أيضاً في
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$، وكان كل !!{element} في
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$ !!a{element} أيضاً في
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\begin{quote}
ولما كان هذا اقتراناً وجب إثبات طرفيه منفردين. فنبدأ بإثبات أن كل
!!{element} في ${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$ هو أيضاً !!a{element} في
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.

وهذه دعوى كلية؛ فنأخذ !!{element} كيف اتفق من
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$، ثم نبين أنه !!a{element} لا محالة في
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. وللدلالة على هذا الـ!!{element}
الاعتباطي نضع المتغير~${\OLId{latin-ordinary}{z}}$، ثم نمضي هكذا:
\end{quote}
أولاً، نثبت أن كل !!{element} في ${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$ هو
!!a{element} أيضاً في $({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. ولتكن
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$؛ فنريد أن نبين أن
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\begin{quote}
وهنا ينبغي بسط تعريفي $\cup$ و~$\cap$. فـ$\cup$، مثلاً، معرّف بـ:
${\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}} = \Setabs{{\OLId{latin-ordinary}{z}}}{{\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \text{ أو } {\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}}$. وإذا أُجري
التعريف على «${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$» قام «${\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$» مقام «${\OLId{latin-looped}{B}}$» في
صورة التعريف؛ ومن ثم
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}) = \Setabs{{\OLId{latin-ordinary}{z}}}{{\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \text{ أو }
{\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}}$. وعلى ذلك يكافئ فرضنا
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$ القول:
${\OLId{latin-ordinary}{z}} \in \Setabs{{\OLId{latin-ordinary}{z}}}{{\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \text{ أو } {\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}}$.
ثم إن ${\OLId{latin-ordinary}{z}} \in \Setabs{{\OLId{latin-ordinary}{z}}}{\dots {\OLId{latin-ordinary}{z}}\dots}$ إذا وفقط إذا صدقت
\dots ${\OLId{latin-ordinary}{z}}$ \dots؛ أي إذا صدق في هذا الموضع ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$.
\end{quote}
فتعريف $\cup$ يعطينا: إما أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ وإما أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$.
\begin{quote}
وهذا فصل، فالأليق أن نقسم البرهان إلى حالتين، ونثبت في كل واحدة
النتيجة المقصودة، وهي «${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$».
\end{quote}
الحالة \OLClassicalDigits{1}: لنفرض أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$.
\begin{quote}
وليس في الفرض مادة كثيرة، فنرجع إلى صورة المطلوب. نريد إثبات
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. وبحسب تعريف $\cap$ يكفينا أن
نبيّن انتماء ${\OLId{latin-ordinary}{z}}$ إلى $({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}})$ وإلى $({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. غير أن
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}$ إذا وفقط إذا كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$، وقد فرضنا
في الحالة~\OLClassicalDigits{1} أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$. وبعين الاستدلال، بعد إحلال ${\OLId{latin-looped}{C}}$ محل ${\OLId{latin-looped}{B}}$،
ينتج ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$. وقد رجعنا في هذا البيان من النتيجة إلى
مقدماتها، فلنكتبه الآن مرتباً من المقدمات إلى النتيجة.
\end{quote}
لما كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$، صدق أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$؛ ومن ثم يكون،
بتعريف $\cup$،~${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}$. وعلى القياس نفسه
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$. فيلزم، بتعريف $\cap$، أن
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\begin{quote}
تمت بهذه الحالة الأولى، وبقي أن نستخرج النتيجة في الحالة الثانية،
حيث ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$.
\end{quote}
الحالة \OLClassicalDigits{2}: لنفرض أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$.
\begin{quote}
عاد الكلام إلى تقاطع مجموعتين، فنُعمل تعريف~$\cap$:
\end{quote}
لما كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$، كان ${\OLId{latin-ordinary}{z}}$ !!a{element} في كل من ${\OLId{latin-looped}{B}}$ و${\OLId{latin-looped}{C}}$،
وذلك بتعريف~$\cap$.
\begin{quote}
ونردّ النظر إلى النتيجة: يلزم بيان انتماء ${\OLId{latin-ordinary}{z}}$ إلى كل من
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}})$ و$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$، وهو الآن بيّن قريب.
\end{quote}
إذ كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$، كان ${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}})$. وإذ كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{C}}$، كان
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$ أيضاً. ومن ثم
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\begin{quote}
أجرينا هنا تعريفي $\cup$ و$\cap$ مرة أخرى؛ ولما سبق بيانهما، وسبق
إثبات أن انتماء ${\OLId{latin-ordinary}{z}}$ إلى إحدى مجموعتين يوجب انتماءه إلى اتحادهما، جاز
طيّ التفصيل.

وبتمام الحالة الثانية يتم هذا البرهان بالحالات، فنقرر النتيجة الأولى.
\end{quote}
فإذا كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$، لزم أن
${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$.
\begin{quote}
وبقي الاتجاه المقابل: أن كل !!{element} في
$({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$ هو !!a{element} في
${\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$. ونسلك فيه ما سلكناه أولاً: نأخذ !!{element}
اعتباطياً من المجموعة الأولى، ثم نثبت لزوم انتمائه إلى الثانية.
فلنصرح بذلك.
\end{quote}
لنفرض الآن أن ${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. ومرادنا أن نبين
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\begin{quote}
مبدأ هذا القسم هو الفرض ${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$. ولا
ينبغي أن يوقع استعمال ${\OLId{latin-ordinary}{z}}$ هنا أيضاً في لبس؛ فإن فروض القسم الأول قد
انقضت بانقضائه، فجاز أن نضع الآن فروضاً جديدة في ${\OLId{latin-ordinary}{z}}$. ومن وجد في ذلك
مشقة فليجعل مكان ${\OLId{latin-ordinary}{z}}$ متغيراً آخر فيما يأتي.

وبتعريف $\cap$ نعلم أن ${\OLId{latin-ordinary}{z}}$ ينتمي إلى ${\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}$ وإلى
${\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$. ثم يبسط تعريف $\cup$ القول الأول إلى: ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$، ويبسط الثاني إلى: ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{C}}$.
وهذا يدعو إلى برهان بالحالات، غير أن اجتماعهما بـ«و» يوجب الاحتراز.
فليس معناه أن الاحتمالات ثلاثة، أي كون ${\OLId{latin-ordinary}{z}}$ في ${\OLId{latin-looped}{A}}$ أو ${\OLId{latin-looped}{B}}$ أو ${\OLId{latin-looped}{C}}$؛ بل
لا بد من تناول كل فصل على حدته.
\end{quote}
فبتعريف $\cap$، لدينا ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}$ و${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$. ومن
تعريف $\cup$، إما أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$؛ فنميز حالتين.
\begin{quote}
لأن النظر مقصور الآن على الفصل الأول، لم نستخرج بعد الفصل الثاني من
بسط ${\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$؛ ولا حاجة إليه في هذا الموضع. والحالة الأولى
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$، وكل !!a{element} لمجموعة هو !!a{element} أيضاً لاتحادها
بأية مجموعة أخرى؛ فالحالة~\OLClassicalDigits{1} قريبة:
\end{quote}
الحالة \OLClassicalDigits{1}: لنفرض أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$. فينتج ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\begin{quote}
وأما الحالة الثانية، وهي ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$، فنفك فيها $\cup$ الثانية ونقسم
إلى حالتين أخريين:
\end{quote}
الحالة \OLClassicalDigits{2}: لنفرض أن ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$. ولما كان ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}}$، فإما أن
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$ أو ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{C}}$. ونميز حالتين فرعيتين:

الحالة \OLClassicalDigits{2}أ: ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}}$. وحينئذ، كما تقدم،
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\begin{quote}
وفي هذا شيء من الغرابة، إذ لم نحتج في هذه الحالة إلى الفرض
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$؛ ولا يقدح ذلك في البرهان.
\end{quote}
الحالة \OLClassicalDigits{2}ب: ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{C}}$. فحينئذ ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}}$ و${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{C}}$، ومن ثم
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}}$، ويلزم عنه ${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\begin{quote}
وبذلك يتم البرهانان بالحالات، ويتم معهما النصف الثاني.
\end{quote}
فإذا كان ${\OLId{latin-ordinary}{z}} \in ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{B}}) \cap ({\OLId{latin-looped}{A}} \cup {\OLId{latin-looped}{C}})$، لزم أن
${\OLId{latin-ordinary}{z}} \in {\OLId{latin-looped}{A}} \cup ({\OLId{latin-looped}{B}} \cap {\OLId{latin-looped}{C}})$.
\end{proof}
