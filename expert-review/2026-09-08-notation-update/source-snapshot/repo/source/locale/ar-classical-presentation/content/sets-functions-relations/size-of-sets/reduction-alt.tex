\olsection{الاختزال}

\begin{editorial}
نثبت في هذا القسم عدم القابلية للتعداد بالاختزال، ونحصل على نتائج \olref[nen-alt]{sec}. وأما العرض البديل الذي يزيد شيئًا من التفصيل ويوافق نتائج \olref[nen]{sec}، ففي \olref[red]{sec}.
\end{editorial}

ثبت بالحجة القطرية أن $\Bin^\omega$ !!{nonenumerable}، وثبت بحجة من جنسها أن $\Pow{\Nat}$ !!{nonenumerable}. غير أن لنا إلى إثبات كون $\Pow{\Nat}$ !!{nonenumerable} سبيلًا آخر: نبين أنه \emph{لو كانت $\Pow{\Nat}$ !!{enumerable} لكانت $\Bin^\omega$ !!{enumerable} أيضًا}. وقد علمنا أن $\Bin^\omega$ !!{nonenumerable}، فيلزم مثل ذلك في $\Pow{\Nat}$.

وهذا \emph{اختزال} مسألة إلى أخرى؛ فقد رددنا تعداد $\Bin^\omega$ إلى تعداد $\Pow{\Nat}$. فحل المسألة الأخيرة، أعني تعداد $\Pow{\Nat}$، لو وجد لأمكن أن نستخرج منه حل الأولى، أعني تعداد $\Bin^\omega$.

ومتى أردنا رد تعداد مجموعة~${\OLId{latin-looped}{B}}$ إلى تعداد مجموعة~${\OLId{latin-looped}{A}}$، وجب أن نعين طريقًا يحول تعداد~${\OLId{latin-looped}{A}}$ إلى تعداد~${\OLId{latin-looped}{B}}$. وأيسر ذلك أن نضع !!a{surjection} ${\OLId{latin-ordinary}{f}}\colon {\OLId{latin-looped}{A}} \to {\OLId{latin-looped}{B}}$. فلو كانت ${\OLId{latin-ordinary}{x}}_{\OLMathNumeral{1}}$، ${\OLId{latin-ordinary}{x}}_{\OLMathNumeral{2}}$، \dots{} تعدد~${\OLId{latin-looped}{A}}$، لكانت ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}_{\OLMathNumeral{1}})$، ${\OLId{latin-ordinary}{f}}({\OLId{latin-ordinary}{x}}_{\OLMathNumeral{2}})$، \dots{} بعد حذف التكرار وإعادة ترقيم الباقي تعداد~${\OLId{latin-looped}{B}}$. فمطلبنا هنا !!a{surjection} ${\OLId{latin-ordinary}{f}}\colon \Pow{\Nat}
\to \Bin^\omega$.

\textbf{تنبيه تصحيحي على الأصل (C040-01).} قد تتكرر قيم الدالة الشاملة؛ فلا تكون
قائمة الصور تعدادًا تقابليًا على التعريف البديل إلا بعد حذف المكرر وإعادة
فهرسة ما بقي.

\begin{prob}
إذا وجدت دالة~!!{injective} ${\OLId{latin-ordinary}{g}}\colon {\OLId{latin-looped}{B}} \to {\OLId{latin-looped}{A}}$، وكانت ${\OLId{latin-looped}{B}}$~!!{nonenumerable}، فبين أن ${\OLId{latin-looped}{A}}$ كذلك. وذلك بأن تستعمل~${\OLId{latin-ordinary}{g}}$ لتحويل تعداد~${\OLId{latin-looped}{A}}$ المفروض إلى تعداد~${\OLId{latin-looped}{B}}$.
\end{prob}

\begin{proof}[برهان {\olref[nen-alt]{thm:nonenum-pownat}} بالاختزال]
لنفرض أن $\Pow{\Nat}$ !!{enumerable}، ونأخذ لها تعدادًا ${\OLId{latin-looped}{N}}_{{\OLMathNumeral{1}}}$، ${\OLId{latin-looped}{N}}_{{\OLMathNumeral{2}}}$، ${\OLId{latin-looped}{N}}_{{\OLMathNumeral{3}}}$، \dots

ونضع الدالة ${\OLId{latin-ordinary}{f}} \colon \Pow{\Nat} \to \Bin^\omega$ بحيث تكون ${\OLId{latin-ordinary}{f}}({\OLId{latin-looped}{N}})$ السلسلة ${\OLId{latin-ordinary}{s}}$ المعينة بالشرط ${\OLId{latin-ordinary}{s}}({\OLId{latin-ordinary}{n}}) = {\OLMathNumeral{1}}$ إذا وفقط إذا كان ${\OLId{latin-ordinary}{n}} \in {\OLId{latin-looped}{N}}$، وبالشرط ${\OLId{latin-ordinary}{s}}({\OLId{latin-ordinary}{n}}) = {\OLMathNumeral{0}}$ فيما عدا ذلك.

وهذا تعيين صحيح لدالة؛ إذ متى كان ${\OLId{latin-looped}{N}} \subseteq \Nat$، فكل ${\OLId{latin-ordinary}{n}} \in \Nat$ إما !!a{element} من~${\OLId{latin-looped}{N}}$ وإما خارج عنها. فمثلًا مجموعة الأعداد الطبيعية الزوجية ${\OLMathNumeral{2}}\Nat = \Setabs{{\OLMathNumeral{2}}{\OLId{latin-ordinary}{n}}}{{\OLId{latin-ordinary}{n}} \in \Nat} = \{{\OLMathNumeral{0}},{\OLMathNumeral{2}}, {\OLMathNumeral{4}}, {\OLMathNumeral{6}},
\dots\}$ ترسل إلى ${\OLMathNumeral{1010101}}\dots$؛ والمجموعة $\emptyset$ ترسل إلى ${\OLMathNumeral{0000}}\dots$؛ والمجموعة $\Nat$ ترسل إلى ${\OLMathNumeral{1111}}\dots$.

والدالة !!{surjective} أيضًا، لأن كل سلسلة من~${\OLMathNumeral{0}}$ و~${\OLMathNumeral{1}}$ تعين مجموعة الأعداد الطبيعية التي يشغل~${\OLMathNumeral{1}}$ مواضعها، فتكون السلسلة صورة تلك المجموعة. وللتفصيل، إذا كان ${\OLId{latin-ordinary}{s}} \in \Bin^\omega$ نعين ${\OLId{latin-looped}{N}}
\subseteq \Nat$ بوضع
\[
{\OLId{latin-looped}{N}} = \Setabs{{\OLId{latin-ordinary}{n}} \in \Nat}{{\OLId{latin-ordinary}{s}}({\OLId{latin-ordinary}{n}}) = {\OLMathNumeral{1}}}
\]
فيكون ${\OLId{latin-ordinary}{f}}({\OLId{latin-looped}{N}}) = {\OLId{latin-ordinary}{s}}$، كما يبينه تعريف~${\OLId{latin-ordinary}{f}}$.

ولننظر في القائمة
\[
{\OLId{latin-ordinary}{f}}({\OLId{latin-looped}{N}}_{\OLMathNumeral{1}}), {\OLId{latin-ordinary}{f}}({\OLId{latin-looped}{N}}_{\OLMathNumeral{2}}), {\OLId{latin-ordinary}{f}}({\OLId{latin-looped}{N}}_{\OLMathNumeral{3}}), \dots
\]
إن ${\OLId{latin-ordinary}{f}}$ !!{surjective}، فكل عنصر من $\Bin^\omega$ قيمة لـ~${\OLId{latin-ordinary}{f}}$ عند وسيط ما؛ وهذا الوسيط وارد في التعداد، فترد قيمته في القائمة الجديدة. فهي إذن تستوفي~$\Bin^\omega$ كلها.

فقد تبين أن كون $\Pow{\Nat}$ !!{enumerable} يستلزم كون $\Bin^\omega$ !!{enumerable}. لكن $\Bin^\omega$ !!{nonenumerable} (\olref[nen-alt]{thm:nonenum-bin-omega})، فلزم أن $\Pow{\Nat}$ !!{nonenumerable}.
\end{proof}

\begin{editorial}
\textbf{تنبيه تصحيحي على الأصل (OLSIZ-010).}
سمّى الأصل المتتالية ${\OLId{latin-ordinary}{s}}_{\OLId{latin-ordinary}{k}}$ من غير أن يربط ${\OLId{latin-ordinary}{k}}$؛ فاستُعمل الاسم
${\OLId{latin-ordinary}{s}}$ في التعريف كله، إذ تحدد المجموعة ${\OLId{latin-looped}{N}}$ متتاليتها المميزة وحدها.
\end{editorial}

\begin{prob}\label{sfr:siz:red-alt:prob:nat-nat}
أقم برهانًا بالاختزال على أن المجموعة~${\OLId{latin-looped}{X}}$ لجميع الدوال ${\OLId{latin-ordinary}{f}}\colon \Nat \to \Nat$ !!{nonenumerable}. وتلميحه إيجاد دالة شاملة من ${\OLId{latin-looped}{X}}$ إلى~$\Bin^\omega$.
\end{prob}

\begin{prob}
بين بالاختزال أن مجموعة جميع \emph{المجموعات المؤلفة من} أزواج الأعداد الطبيعية، أي $\Pow{\Nat \times \Nat}$، !!{nonenumerable}.
\end{prob}

\begin{prob}
أقم حجة اختزال على أن $\Nat^\omega$ !!{nonenumerable}؛ والمراد بها مجموعة جميع المتتاليات اللانهائية من الأعداد الطبيعية.
\end{prob}
