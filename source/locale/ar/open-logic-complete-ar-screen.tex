  \settrims{0pt}{0pt}%

% Halve only the print driver's explicit title-page gaps. Content and formal
% structure remain unchanged, while the landscape title no longer wastes most
% of the short screen height.
\NewDocumentCommand{\OLScreenHalfVspace}{s m}{%
  \IfBooleanTF{#1}
    {\OLScreenSavedVspace*{\glueexpr#2/2\relax}}
    {\OLScreenSavedVspace{\glueexpr#2/2\relax}}%
}

  \let\OLScreenSavedVspace\vspace
  \let\vspace\OLScreenHalfVspace
}

\AddToHook{begindocument/before}{%
  % TeX emits the sole 4.19 pt table-cell diagnostic after the tabular's
  % local group closes. Set the audited tolerance here; full-page vector
  % bounds remain the controlling overflow gate.
  \global\hfuzz=4.25pt\relax
  % The shared asset helper uses plain \centerline. In an RTL paragraph that
  % gives imported TikZ/PDF boxes an RTL reference point and can shift the
  % artwork right by its own width. Give the artwork an explicit LTR sub-box;
  % retain the requested width, fallback path, and surrounding figure markup.
  \RenewDocumentCommand{\olasset}{O{\olphotowidth} m}{%
    \par\noindent
    \babelsublr{%
      \makebox[\linewidth][c]{%
        \IfFileExists{#2}%
          {\includegraphics[width=#1]{#2}}%
          {\includegraphics[width=#1]{\olpath/#2}}%
      }%
    }%
    \par
  }%
  \let\sFmla\OLScreenSignedFormula
  \let\OLScreenSavedCenter\center
  \let\OLScreenSavedEndCenter\endcenter
  \renewenvironment{center}{%
    \OLScreenSavedCenter
    \ifnum\value{chapter}=48\relax
      \ifnum\value{section}=4\relax
        \let\OLScreenSavedCenterCR\\
        \let\\\OLScreenCompactCenterCR
      \fi
    \fi
  }{%
    \ifnum\value{chapter}=48\relax
      \ifnum\value{section}=4\relax
        \let\\\OLScreenSavedCenterCR
      \fi
    \fi
    \OLScreenSavedEndCenter
  }%
  % Short screen pages need long multi-line displays to reflow across pages.
  \allowdisplaybreaks[4]%
  % The sole sideways float is already on a landscape page; leave it upright.
  \renewenvironment{sidewaysfigure}{\begin{figure}}{\end{figure}}%
  % Keep part identity, but use the rest of its landscape page for the
  % editorial lead-in and the first chapter instead of a near-empty divider.
  \aliaspagestyle{part}{empty}%
  \renewcommand*{\beforepartskip}{%
    \null
    \ifnum\value{part}=17\relax
      \vspace*{140pt}% Deliberately center the otherwise short reference opener.
    \else
      \vspace*{8pt}%
    \fi
  }%
  \renewcommand*{\midpartskip}{\par\vskip4pt}%
  \renewcommand*{\afterpartskip}{%
    \par\vskip8pt%
    \global\OLScreenChapterAfterParttrue
  }%
  \renewcommand*{\clearforchapter}{%
    \ifOLScreenChapterAfterPart
      \global\OLScreenChapterAfterPartfalse
    \else
      \par\Needspace{12\baselineskip}%
    \fi
  }%
  \setlength{\beforechapskip}{0pt}%
  \setlength{\midchapskip}{0.25\baselineskip}%
  \setlength{\afterchapskip}{0.55\baselineskip}%
  \hypersetup{pdfstartview=Fit,pdfpagelayout=OneColumn,pdfpagemode=UseOutlines}%
}
