\olsection{مبرهنة $s$-$m$-$n$}

\begin{explain}
تعرف المبرهنة التالية باسم «مبرهنة $s$-$m$-$n$»، لسبب سيتضح بعد
قليل. والجانب الصعب هو فهم ما تقوله المبرهنة بالضبط؛ فإذا فهمت عبارتها
بدت واضحة إلى حد بعيد.
\end{explain}

\begin{thm}
\ollabel{thm:s-m-n}
  لكل زوج من العددين الطبيعيين $n$ و~$m$ دالة عودية بدائية~$s^m_n$
  بحيث يكون لدينا، لكل متتالية $e$، $a_0$، …، $a_{m-1}$، $y_0$،
  …،~$y_{n-1}$،
  \[
  \cfind{s^m_n(e, a_0, \dots, a_{m-1})}[n](y_0, \dots, y_{n-1}) \simeq
  \cfind{e}[m+n](a_0, \dots, a_{m-1}, y_0, \dots, y_{n-1}).
\]
\end{thm}

\begin{explain}
يفيد تصور $s^m_n$ على أنها تعمل في \emph{البرامج}. أي إن $s^m_n$
تأخذ برنامجًا~$e$ لدالة ذات $m+n$ حجة، وكذلك المداخل الثابتة $a_0$،
…،~$a_{m-1}$؛ وتعيد برنامجًا
$s^m_n(e,a_0,\dots,a_{m-1})$ للدالة ذات $n$ حجة في الحجج الباقية.
\iftag{TMs}{إذا تصورت $e$ وصف آلة تورنغ، فإن
$s^m_n(e,a_0,\dots,a_{m-1})$ هي آلة تورنغ التي تسبق سلسلة الدخل
$y_0$، …،~$y_{n-1}$ بـ$a_0$، …،~$a_{m-1}$، ثم تشغل~$e$.
وعندئذ لا تكون كل $s^m_n$ إلا دالة عودية بدائية تجد رمز آلة تورنغ
المناسبة.}{}
\end{explain}
