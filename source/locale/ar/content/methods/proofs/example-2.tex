\olsection{مثال آخر}

\begin{prop}
إذا كان $A \subseteq C$، فإن $A \cup (C \setminus A) = C$.
\end{prop}

\begin{proof}
  افترض أن $A \subseteq C$. نريد أن نبين أن
  $A \cup (C \setminus A) = C$.
  \begin{quote}
    نبدأ بملاحظة أن هذه عبارة شرطية. وهي مكممة كلياً ضمناً: فالقضية تصح
    لجميع المجموعات $A$ و$C$. ولذلك فـ$A$ و$C$ متغيران لمجموعتين
    اعتباطيتين. ولإثبات مثل هذه العبارة نفترض المقدم ونثبت التالي.

    ثم نستعمل الفرض $A \subseteq C$. لنفك تعريف~$\subseteq$: يعني
    الفرض أن جميع !!{element} المجموعة~$A$ هي أيضاً !!{element}
    المجموعة~$C$. فلنسجل ذلك، فهو حقيقة مهمة سنستعملها طوال البرهان.
  \end{quote}
  بحسب تعريف~$\subseteq$، وبما أن $A \subseteq C$، فلكل $z$، إذا كان
  $z \in A$ فإن $z \in C$.
  \begin{quote}
    لقد فككنا جميع التعريفات المعطاة لنا في الفرض. وننتقل الآن إلى
    النتيجة. نريد أن نبين أن $A \cup (C \setminus A) = C$، فنرتب
    البرهان على نحو مشابه للمثال السابق: نبين أن كل !!{element} في
    $A \cup (C \setminus A)$ هو أيضاً !!a{element} في~$C$، وأن كل
    !!{element} في $C$ هو، بالعكس، !!a{element} في
    $A \cup (C \setminus A)$. ويمكن اختصار ذلك إلى:
    $A \cup (C \setminus A) \subseteq C$ و
    $C \subseteq A \cup (C \setminus A)$. (نحن هنا نعكس عملية فك
    التعريف، لكن ذلك يجعل البرهان أيسر قراءة.) وبما أن هذا اقتران، فعلينا
    إثبات كلا جزأيه. ولإثبات الجزء الأول، أي إن كل !!{element} في
    $A \cup (C \setminus A)$ هو أيضاً !!a{element} في~$C$، نفترض
    $z \in A \cup (C \setminus A)$ لـ$z$ اعتباطي، ونبين أن
    $z \in C$. ومن $z \in A \cup (C \setminus A)$ نستنتج، بتعريف
    $\cup$، أن $z \in A$ أو $z \in C \setminus A$. ولعلك بدأت الآن
    تألف هذا الأسلوب.
    \end{quote}
  لدينا $A \cup (C \setminus A) = C$ إذا وفقط إذا كان
  $A \cup (C \setminus A) \subseteq C$ و
  $C \subseteq (A \cup (C \setminus A))$. نثبت أولاً أن
  $A \cup (C \setminus A) \subseteq C$. لتكن
  $z \in A \cup (C \setminus A)$. إذن إما أن $z \in A$ وإما أن
  $z \in (C \setminus A)$.
  \begin{quote}
    وصلنا إلى فصل ونريد أن نثبت منه $z \in C$. ونفعل ذلك ببرهان
    بالحالات.
  \end{quote}
  الحالة 1: $z \in A$. وبما أنه لكل $z$، إذا كان $z \in A$ فإن
  $z \in C$، نحصل على $z \in C$.
  \begin{quote}
    استعملنا هنا الحقيقة التي سجلناها من قبل، والتي نتجت من فرض القضية
    $A \subseteq C$. اكتملت الحالة الأولى، فننتقل إلى الحالة الثانية،
    $z \in (C \setminus A)$. تذكر أن $C \setminus A$ يدل على
    \emph{فرق} المجموعتين، أي مجموعة جميع !!{element} في~$C$ التي ليست
    !!{element} في~$A$. لكن أي !!{element} في $C$ لا ينتمي إلى~$A$ هو،
    بوجه خاص، !!a{element} في~$C$.
  \end{quote}
  الحالة 2: $z \in (C \setminus A)$. وهذا يعني أن $z \in C$ و
  $z \notin A$. ومن ثم، بوجه خاص، $z \in C$.
  \begin{quote}
    جيد، لقد أثبتنا الاتجاه الأول. والآن إلى الاتجاه الثاني، حيث نثبت أن
    $C \subseteq A \cup (C \setminus A)$. فنفترض $z \in C$ ونثبت أن
    $z \in A \cup (C \setminus A)$.
  \end{quote}
  لتكن الآن $z \in C$. نريد أن نبين أن $z \in A$ أو
  $z \in C \setminus A$.
  \begin{quote}
    بما أن جميع !!{element} في $A$ هي أيضاً !!{element} في $C$، وأن
    $C \setminus A$ هي مجموعة جميع الأشياء التي هي !!{element} في $C$
    وليست في $A$، ينتج أن $z$ إما في $A$ وإما في $C \setminus A$.
    وقد لا يكون هذا واضحاً تماماً إن لم تكن تعرف مسبقاً لماذا تصح
    النتيجة. والأفضل أن نثبتها خطوة خطوة. وستفيدنا حقيقة بسيطة نستطيع
    ذكرها بلا برهان: $z \in A$ أو $z \notin A$. ويسمى هذا «مبدأ الثالث
    المرفوع»: لكل عبارة~$p$، إما أن تكون $p$ صادقة وإما أن يكون نفيها
    صادقاً. (هنا $p$ هي العبارة $z \in A$.) ولما كان هذا فصلاً، نستطيع
    استعمال البرهان بالحالات مرة أخرى.
  \end{quote}
  إما أن $z \in A$ أو $z \notin A$. في الحالة الأولى
  $z \in A \cup (C \setminus A)$. وفي الثانية $z \in C$ و
  $z \notin A$، ومن ثم $z \in C \setminus A$. وعندئذ
  $z \in A \cup (C \setminus A)$.
  \begin{quote}
    اكتمل برهاننا: لقد بينا أن $A \cup (C \setminus A) = C$.
  \end{quote}
\end{proof}
