\olsection{المجموعات الجزئية ومجموعات القوى}

\begin{explain}
سنحتاج كثيرًا إلى مقارنة المجموعات. ومن أوجه المقارنة البديهية
التي يمكن إجراؤها ما يأتي: \emph{كل ما في إحدى المجموعتين موجود في
الأخرى أيضًا}. وهذه الحالة مهمة بما يكفي لكي
نستحدث لها بعض الرموز.
\end{explain}

\begin{defn}[المجموعة الجزئية]
إذا كان كل !!{element} من مجموعة $A$ أيضًا !!a{element} من~$B$، فإننا
نقول إن $A$ \emph{مجموعة جزئية} من~$B$، ونكتب $A \subseteq B$. وإذا
لم تكن $A$ مجموعة جزئية من~$B$، نكتب $A \not\subseteq B$.
وإذا كان $A \subseteq B$ لكن $A \neq B$، نكتب $A \subsetneq B$، ونقول
إن $A$ \emph{مجموعة جزئية فعلية} من $B$.
\end{defn}

\begin{ex}
كل مجموعة مجموعة جزئية من نفسها، و$\emptyset$ مجموعة جزئية من كل
مجموعة. ومجموعة الأعداد الطبيعية الزوجية مجموعة جزئية من مجموعة الأعداد الطبيعية.
وكذلك $\{ a, b \} \subseteq \{ a, b, c \}$. لكن $\{ a, b, e
\}$ ليست مجموعة جزئية من $\{ a, b, c \}$.
\end{ex}

\begin{ex}
العدد $2$ هو !!{element} في مجموعة الأعداد الصحيحة، في حين أن
مجموعة الأعداد الزوجية مجموعة جزئية من مجموعة الأعداد الصحيحة. ومع ذلك، قد
يحدث \emph{كلا الأمرين} لمجموعة ما: أن تكون !!a{element} وأن تكون مجموعة جزئية من
مجموعة أخرى؛ فمثلًا، $\{0\} \in \{0, \{0\}\}$ وكذلك $\{0\} \subseteq \{0,
\{0\}\}$.
\end{ex}

تمنحنا الامتدادية معيارًا لتطابق المجموعات: $A = B$ إذا وفقط إذا كان
كل !!{element} من~$A$ أيضًا !!a{element} من~$B$، والعكس صحيح.
ويعرّف تعريف «المجموعة الجزئية» $A \subseteq B$ على وجه الدقة بأنه
النصف الأول من هذا المعيار: كل !!{element} من~$A$ هو أيضًا
!!a{element} من~$B$. وبالطبع ينطبق التعريف أيضًا إذا
أبدلنا $A$ و$B$؛ أي إن $B \subseteq A$ إذا وفقط إذا كان كل !!{element}
من~$B$ أيضًا !!a{element} من~$A$. وهذا بدوره هو تمامًا
جزء «العكس صحيح» من الامتدادية. وبعبارة أخرى، تستلزم الامتدادية
أن المجموعات متساوية إذا وفقط إذا كانت كل منها مجموعة جزئية من الأخرى.

\begin{prop}
$A = B$ إذا وفقط إذا تحقق كل من $A \subseteq B$ و$B \subseteq A$.
\end{prop}

وهذه أيضًا فرصة مناسبة لاستحداث بعض الرموز الإضافية
المفيدة. فعند تعريف متى تكون $A$ مجموعة جزئية من~$B$، قلنا
«كل !!{element} من~$A$ هو \dots»، وملأنا موضع «$\dots$» بقولنا
«!!a{element} من $B$». لكن هذا \emph{النمط} من
العبارات شائع إلى حد يجعل من المفيد استحداث ترميز صوري
له.

\begin{defn}\ollabel{forallxina}
تختصر $(\forall x \in A)\phi$ العبارة $\forall x(x \in A \lif
\phi)$. وبالمثل، تختصر $(\exists x \in A)\phi$ العبارة $\exists x(x
\in A \land \phi)$.
\end{defn}

وباستخدام هذا الترميز، نقول إن $A \subseteq B$ إذا وفقط إذا كان $(\forall
x \in A)x \in B$.

وننتقل الآن إلى نوع معين من المجموعات: المجموعة التي تضم جميع
المجموعات الجزئية من مجموعة معطاة.

\begin{defn}[مجموعة القوى]
تُسمى المجموعة المكوّنة من جميع المجموعات الجزئية من مجموعة~$A$
\emph{مجموعة قوى}~$A$، ويرمز إليها بـ$\Pow{A}$.
  \[
    \Pow{A} = \Setabs{B}{B \subseteq A}
  \]
\end{defn}

\begin{ex}
ما جميع المجموعات الجزئية لـ $\{ a, b, c \}$؟ إنها:
$\emptyset$، و$\{a \}$، و$\{b\}$، و$\{c\}$، و$\{a, b\}$، و$\{a, c\}$، و$\{b,
c\}$، و$\{a, b, c\}$. ومجموعة هذه المجموعات الجزئية كلها هي
$\Pow{\{a,b,c\}}$:
\[
\Pow{\{ a, b, c \}} = \{\emptyset, \{a \}, \{b\}, \{c\}, \{a, b\},
\{b, c\}, \{a, c\}, \{a, b, c\}\}
\]
\end{ex}

\begin{prob}
اسرد جميع المجموعات الجزئية من $\{a, b, c, d\}$.
\end{prob}

\begin{prob}
أثبت أنه إذا كان للمجموعة $A$ عدد $n$ من !!{element}s، فإن للمجموعة $\Pow{A}$ عدد $2^n$
من !!{element}s.
\end{prob}
