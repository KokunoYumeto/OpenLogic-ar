\olsection{الدوال بوصفها علاقات}

\begin{explain}
إن الدالة التي ترسل !!{element}s المجموعة~$A$ إلى !!{element}s المجموعة~$B$
تحدد بوضوح علاقة بين $A$ و~$B$، هي العلاقة التي تقوم
بين $x$ و$y$ إذا وفقط إذا كان $f(x) = y$. بل قد يصل بنا الأمر---
إذا كنا مهتمين، مثلًا، بردّ اللبنات الأساسية التي تتألف منها
الرياضيات---إلى \emph{مماهاة} الدالة~$f$ بهذه
العلاقة، أي بمجموعة من الأزواج. وهذا يثير السؤال الآتي:
أي العلاقات تعرّف دوال بهذه الطريقة؟
\end{explain}

\begin{defn}[الرسم البياني لدالة] لتكن $f\colon A \to B$ دالة.
إن \emph{الرسم البياني} للدالة~$f$ هو العلاقة $R_f \subseteq A \times B$
المعرّفة بـ
\[
R_f = \Setabs{\tuple{x,y}}{f(x) = y}.
\]
\end{defn}

\begin{explain}
يتحدد الرسم البياني للدالة على نحو وحيد بحكم الامتدادية.
وعلاوة على ذلك، فإن الامتدادية (للمجموعات) تسوّغ فورًا
مبدأ الامتدادية الضمني للدوال، الذي يقضي بأنه إذا اشتركت
$f$ و~$g$ في المجال والمجال المقابل، فإنهما
متطابقتان متى اتفقتا في جميع القيم.

وبالمثل، إذا كانت العلاقة «دالية»، فهي الرسم البياني لدالة.
\end{explain}

\begin{prop}\ollabel{prop:graph-function}
لتكن $R \subseteq A \times B$ بحيث:
\begin{enumerate}
\item إذا كان $Rxy$ و$Rxz$ فإن $y = z$؛ و
\item لكل $x \in A$ يوجد $y \in B$ بحيث $\tuple{x,
y} \in R$.
\end{enumerate}
عندئذ تكون $R$ الرسم البياني للدالة $f\colon A \to B$ المعرّفة
بالشرط $f(x) = y$ إذا وفقط إذا كان $Rxy$.
\end{prop}

\begin{proof}
افترض أن ثمة $y$ بحيث $Rxy$. ولو وُجد $z \neq
y$ آخر بحيث $Rxz$، لانتُهك الشرط المفروض على~$R$. ومن ثم، إذا
وُجد $y$ بحيث $Rxy$، كان هذا $y$ وحيدًا، ولذلك تكون $f$
معرّفة تعريفًا سليمًا. ومن الواضح أن $R_f = R$.
\end{proof}

\begin{explain}
لكل دالة $f\colon A \to B$ رسم بياني، أي علاقة على $A
\times B$ يعرّفها الشرط $f(x) = y$. ومن جهة أخرى، فإن كل علاقة~$R
\subseteq A \times B$ تتصف بالخاصيتين الواردتين في
\olref{prop:graph-function} هي الرسم البياني لدالة~$f \colon A \to
B$. وبسبب هذه الصلة الوثيقة بين الدوال ورسومها البيانية، يمكننا
النظر إلى الدالة ببساطة بوصفها رسمها البياني. وبعبارة أخرى، يمكن
أن نعامل الدوال بوصفها علاقات معينة، أي مجموعات معينة من الأزواج المرتبة.

\begin{defn}\ollabel{defn:funimage}
لتكن $f \colon A \to B$ دالة، ولتكن $C\subseteq A$.

إن \emph{تقييد}~$f$ على~$C$ هو
الدالة~$\funrestrictionto{f}{C}\colon C \to B$ المعرّفة بـ
$(\funrestrictionto{f}{C})(x) = f(x)$ لكل $x \in C$. وبعبارة
أخرى، $\funrestrictionto{f}{C} = \Setabs{\tuple{x, y} \in R_f}{x \in
C}$.

إن \emph{تطبيق}~$f$ على~$C$ هو $\funimage{f}{C} =
\Setabs{f(x)}{x \in C}$. ونسميه أيضًا \emph{صورة}~$C$
بالدالة~$f$.
\end{defn}

\begin{explain}
يترتب على هذين التعريفين أن $\ran{f} =
\funimage{f}{\dom{f}}$، لأي دالة~$f$.
\oliflabeldef{sfr:rel:ops:sec}{وهذان المفهومان هما بالضبط ما يُتوقع
في ضوء التعريفات الواردة في \olref[sfr][rel][ops]{sec} ومماهاتنا
الدوال بالعلاقات. لكن عمليتين أخريين---هما المعكوسات والضروب
النسبية---تحتاجان إلى شيء من التفصيل. وسنقدمه في
\olref[inv]{sec} و\olref[cmp]{sec}.}{}
\end{explain}
