\olsection{معكوسات الدوال}

\begin{explain}
ننظر إلى الدوال بوصفها تعيينات. ومن ثم فإن أحد الأسئلة البديهية عن
الدوال هو: هل يمكن «عكس» التعيين؟ فمثلًا، يمكن عكس دالة اللاحق
$f(x) = x + 1$، بمعنى أن الدالة $g(y) = y - 1$ «تبطل» ما تفعله~$f$.

لكن لا بد من توخي الحذر. فعلى الرغم من أن تعريف~$g$ يعرّف
دالة $\Int \to \Int$، فإنه لا يعرّف \emph{دالة} $\Nat
\to \Nat$، لأن $g(0) \notin \Nat$. وهكذا، فحتى في الحالات البسيطة
ليس من الواضح تمامًا ما إذا كان يمكن عكس دالة؛ إذ قد يتوقف ذلك على
المجال والمجال المقابل.

ويجعل مفهوم معكوس الدالة هذا الأمر أدق.
\end{explain}

\begin{defn}
تكون الدالة $g \colon B \to A$ \emph{معكوسًا} للدالة $f
\colon A \to B$ إذا كان $f(g(y)) = y$ و$g(f(x)) = x$ لكل $x \in A$
ولكل $y \in B$.
\end{defn}

إذا كان للدالة $f$ معكوس~$g$، فغالبًا ما نكتب $f^{-1}$ بدلًا من~$g$.

\begin{explain}
سنحدد الآن متى يكون للدوال معكوسات. ومن المرشحات المناسبة لأن تكون
معكوسًا للدالة $f\colon A \to B$ الدالة $g\colon B \to A$ «المعرّفة
بـ»
\[
g(y) = \text{«ذلك» $x$ الذي يحقق $f(x) = y$.}
\]
لكن علامتي التنصيص حول «المعرّفة بـ» (وحول «ذلك») توحيان بأن
هذا ليس تعريفًا. فعلى الأقل لن ينجح دائمًا على هذا القدر من العموم.
ذلك أنه لكي يحدد هذا التعريف دالة، لا بد أن يوجد~$x$ واحد لا غير
بحيث $f(x) =
y$---إذ يجب أن يتحدد مخرج~$g$ على نحو وحيد. وعلاوة على ذلك، يجب
أن يتحدد لكل $y \in B$. فإذا وُجد $x_1$ و$x_2 \in
A$ بحيث $x_1 \neq x_2$ ولكن $f(x_1) = f(x_2)$، فلن تتحدد $g(y)$
على نحو وحيد عندما $y = f(x_1) = f(x_2)$. وإذا لم يوجد أي~$x$
بحيث $f(x) = y$، فلن تتحدد $g(y)$ أصلًا. وبعبارة أخرى، لكي
تُعرّف $g$، يجب أن تكون $f$ !!{injective} و!!{surjective} معًا.

ولنمض بتؤدة، فنقسم السؤال قسمين. إذا كانت لدينا
دالة~$f\colon A \to B$، فمتى توجد دالة $g\colon B \to A$
بحيث $g(f(x)) = x$؟ إن $g$ من هذا القبيل «تبطل» ما تفعله~$f$، وتسمى
\emph{معكوسًا أيسر} لـ~$f$. ثم متى توجد
دالة $h\colon B \to A$ بحيث $f(h(y)) = y$؟ إن $h$ من هذا القبيل
تسمى \emph{معكوسًا أيمن} لـ~$f$---فـ$f$ «تبطل» ما تفعله~$h$.
\end{explain}

\begin{prop}
إذا كانت $f\colon A \to B$ !!{injective}، فثمة \emph{معكوس
أيسر}~$g\colon B \to A$ لـ~$f$ بحيث $g(f(x)) = x$ لكل $x
\in A$.
\end{prop}

\begin{proof}
افترض أن $f\colon A \to B$ !!{injective}. ولنتأمل $y \in B$.
إذا كان $y \in \ran{f}$، وُجد $x \in A$ بحيث $f(x) = y$. ولأن
$f$ !!{injective}، فلا يوجد إلا~$x \in A$ واحد من هذا القبيل. ويمكننا
عندئذ أن نعرّف: $g(y) = x$، أي إن $g(y)$ هو «ذلك» $x \in A$ الذي يحقق $f(x)
= y$. وإذا كان $y \notin \ran{f}$، أمكننا أن نقرنه بأي~$a \in A$. ولذلك
يمكننا اختيار $a \in A$ وتعريف $g\colon B \to A$ كما يأتي:
\[
g(y) = \begin{cases}
    x & \text{إذا كان $f(x) = y$}\\
    a & \text{إذا كان $y \notin \ran{f}$.}
\end{cases}
\]
وهي معرّفة لكل $y \in B$؛ إذ إنه متى كان $y \in \ran{f}$
وُجد بالضبط عنصر واحد $x \in A$ بحيث $f(x) = y$. وبحسب التعريف، إذا كان
$y = f(x)$، فإن $g(y) = x$، أي إن $g(f(x)) = x$.
\end{proof}

\begin{prob}
بيّن أنه إذا كان للدالة $f\colon A \to B$ معكوس أيسر~$g$، فإن $f$
!!{injective}.
\end{prob}

\begin{prop}
    إذا كانت $f\colon A \to B$ !!{surjective}، فثمة
    \emph{معكوس أيمن}~$h\colon B \to A$ لـ~$f$ بحيث $f(h(y)) =
    y$ لكل~$y \in B$.
\end{prop}

\begin{proof}
افترض أن $f\colon A \to B$ !!{surjective}. ولنتأمل $y \in
B$. بما أن $f$ !!{surjective}، فثمة $x_y \in A$ يحقق $f(x_y)
= y$. ويمكننا عندئذ أن نعرّف: $h(y) = x_y$، أي نختار لكل $y \in B$
عنصرًا ما $x \in A$ بحيث $f(x) = y$؛ وبما أن $f$ !!{surjective}،
فيوجد دائمًا عنصر واحد على الأقل يمكن اختياره.\footnote{بما أن $f$
!!{surjective}، فلكل~$y \in B$ تكون المجموعة
$\Setabs{x}{f(x) = y}$ غير خالية. ويقتضي تعريفنا لـ~$h$ أن نختار عنصرًا واحدًا $x$
من كل مجموعة من هذه المجموعات. وليس إمكان ذلك دائمًا أمرًا
بديهيًا في الواقع---بل يُفترض إمكان إجراء هذه الاختيارات بوصفه بديهيةً.
وبعبارة أخرى، تفترض هذه القضية ما يسمى بديهية
الاختيار، وهي مسألة سوف \oliflabeldef{sth:choice::chap}{نعود إليها في \olref[sth][choice][]{chap}}{نتجاوزها هنا}.
غير أن بديهية الاختيار لا تلزم في حالات خاصة كثيرة، مثلًا عندما يكون $A = \Nat$ أو تكون منتهيةً، أو عندما تكون $f$ !!{bijective}.
(وفي الحالة الخاصة التي تكون فيها $f$ !!{bijective}، فلكل $y \in B$
لا تضم المجموعة $\Setabs{x}{f(x) = y}$ إلا !!{element} واحدًا،
ومن ثم لا يكون هناك اختيار يتعين إجراؤه.)}
وبحسب التعريف، إذا كان $x = h(y)$،
فإن $f(x) = y$، أي إنه لكل $y \in B$ يكون $f(h(y)) = y$.
\end{proof}

\begin{prob}
بيّن أنه إذا كان للدالة $f\colon A \to B$ معكوس أيمن~$h$، فإن $f$
!!{surjective}.
\end{prob}

\begin{explain}
  بضم الأفكار الواردة في البرهان السابق، نحصل الآن على أن لكل
  !!{bijection} معكوسًا، أي إن ثمة دالة واحدة
  هي معكوس أيسر وأيمن لـ~$f$ في آن واحد.
\end{explain}

\begin{prop}\ollabel{prop:bijection-inverse}
إذا كانت $f\colon A \to B$ !!{bijective}، فثمة
دالة~$f^{-1}\colon B \to A$ بحيث لكل $x \in A$،
$f^{-1}(f(x)) = x$، ولكل $y \in B$، يكون $f(f^{-1}(y)) = y$.
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\begin{prob}
أثبت \olref[sfr][fun][inv]{prop:bijection-inverse}. وعليك أن
تعرّف~$f^{-1}$، وتبين أنها دالة، وتبين أنها
معكوسٌ لـ~$f$، أي إن $f^{-1}(f(x)) = x$ و$f(f^{-1}(y)) = y$ لكل
$x \in A$ ولكل $y \in B$.
\end{prob}

\begin{explain}
ثمة طريقة أعم قليلًا لاستخراج المعكوسات. فقد رأينا في
\olref[kin]{sec} أن كل دالة $f$ تستحث !!a{surjection} $f'
\colon A \to \ran{f}$ بتعريف $f'(x) = f(x)$ لكل $x \in A$.
ومن الواضح أنه إذا كانت $f$ !!{injective}، فإن $f'$ !!{bijective}، ومن ثم
يكون لها معكوس وحيد بموجب \olref{prop:bijection-inverse}. وبتسامح طفيف جدًا
في الترميز، نسمي أحيانًا معكوس $f'$ ببساطة
«معكوس~$f$».
\end{explain}

\begin{prop}\ollabel{prop:left-right}%
  بيّن أنه إذا كان للدالة $f\colon A \to B$ معكوس أيسر~$g$ ومعكوس
  أيمن~$h$، فإن $h = g$.
\end{prop}

\begin{proof}
  تمرين.
\end{proof}

\begin{prob}
  أثبت \olref[sfr][fun][inv]{prop:left-right}.
\end{prob}

\begin{prop}\ollabel{prop:inverse-unique}
لكل دالة~$f$ معكوس واحد على الأكثر.
\end{prop}

\begin{proof}
  افترض أن $g$ و$h$ كلاهما معكوسان لـ~$f$. وعندئذ تكون
  $g$ معكوسًا أيسر لـ~$f$، وتكون $h$ معكوسًا أيمن لها. وبموجب
  \olref{prop:left-right}، يكون $g = h$.
\end{proof}
