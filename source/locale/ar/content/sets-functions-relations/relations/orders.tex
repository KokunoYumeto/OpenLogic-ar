\olsection{الترتيبات}

\begin{explain}
ينطوي كثير من مقارناتنا على وصف بعض الكائنات بأنها «أصغر من» كائنات أخرى،
أو «مساوية لها»، أو «أكبر منها»، من وجه معين. وتتعلق هذه المقارنات بعلاقات
\emph{الترتيب}. لكن علاقات الترتيب أنواع مختلفة؛ فمنها مثلًا ما يشترط أن
يكون كل كائنين قابلين للمقارنة، ومنها ما لا يشترط ذلك. ومنها ما يتضمن
الهوية (مثل~$\le$)، ومنها ما يستبعدها (مثل~$<$). وسيفيدنا أن نضع تصنيفًا
لهذه العلاقات.
\end{explain}

\begin{defn}[الترتيب المسبق]
تسمى العلاقة التي تكون انعكاسية ومتعدية معًا
\emph{ترتيبًا مسبقًا.}
\end{defn}

\begin{defn}[الترتيب الجزئي]
يسمى الترتيب المسبق الذي يكون مضادًا للتناظر أيضًا
\emph{ترتيبًا جزئيًا}.
\end{defn}

\begin{defn}[الترتيب الخطي]\ollabel{def:linearorder}
يسمى الترتيب الجزئي الذي يكون متصلًا أيضًا
\emph{ترتيبًا كليًا} أو \emph{ترتيبًا خطيًا.}
\end{defn}

\begin{ex}
كل ترتيب خطي ترتيب جزئي أيضًا، وكل ترتيب جزئي ترتيب مسبق أيضًا، لكن العكوس
لا تصح. والعلاقة الشاملة على~$A$ ترتيب مسبق، لأنها انعكاسية ومتعدية. ولكن
إذا كان لـ$A$ أكثر من !!{element} واحد، فإن العلاقة الشاملة ليست مضادّة
للتناظر، ومن ثم ليست ترتيبًا جزئيًا.
\end{ex}

\begin{ex}
اعتبر علاقة \emph{لا يزيد طولًا على} $\preccurlyeq$
على~$\Bin^*$: يكون $x \preccurlyeq y$ إذا وفقط إذا كان
$\len{x} \le \len{y}$. وهذه علاقة ترتيب مسبق (انعكاسية ومتعدية)، بل إنها
متصلة أيضًا، لكنها ليست ترتيبًا جزئيًا لأنها ليست مضادّة للتناظر. فمثلًا،
$01 \preccurlyeq 10$ و$10 \preccurlyeq 01$، لكن $01 \neq 10$.
\end{ex}

\begin{ex}
من الترتيبات الجزئية المهمة العلاقة $\subseteq$ على مجموعة من المجموعات.
وهذه ليست ترتيبًا خطيًا بوجه عام؛ لأنه إذا كان $a \neq b$ ونظرنا في
$\Pow{\{a, b\}} = \{\emptyset, \{a\}, \{b\}, \{a,b\}\}$، رأينا أن
$\{a\} \nsubseteq \{b\}$ و$\{a\} \neq \{b\}$ و$\{b\}
\nsubseteq \{a\}$.
\end{ex}

\begin{ex}
تعطينا علاقة \emph{القابلية للقسمة دون باق} ترتيبًا جزئيًا ليس ترتيبًا
خطيًا. فللعددين الصحيحين $n$ و~$m$، نكتب $n \mid m$ لنعني أن $n$ يقسم
$m$ (دون باق)، أي إذا وفقط إذا وجد عدد صحيح~$k$ بحيث $m = kn$. وعلى~$\Nat$
تكون هذه العلاقة ترتيبًا جزئيًا، لكنها ليست ترتيبًا خطيًا؛ فمثلًا،
$2 \nmid 3$ وكذلك $3 \nmid 2$. وإذا اعتبرنا قابلية القسمة علاقة على
$\Int$، لم تكن إلا ترتيبًا مسبقًا لأنها ليست مضادّة للتناظر: إذ
$1 \mid -1$ و$-1 \mid 1$، لكن $1 \neq -1$.
\end{ex}

\begin{ex}
تُعرَّف علاقة \emph{الامتداد} على مجموعة المتتاليات~$A^*$ كما يأتي:
يكون $s \sqsubseteq s'$ إذا وفقط إذا كان $s = \emptyseq$ (أي المتتالية
الخالية)، أو كان $s = s'$، أو كان $s = \tuple{s_1, \dots, s_n}$ وكان
$s' = \tuple{s_1, \dots, s_n, s_{n+1}, \dots, s_m}$. وإذا كان
$s \sqsubseteq s'$ قلنا أيضًا إن $s$ \emph{مقطع ابتدائي} من~$s'$.
وعلاقة الامتداد على $A^*$ ترتيب جزئي، لكنها ليست ترتيبًا خطيًا؛ فمثلًا،
إذا كان $a \neq b$، فإن $ab \not\sqsubseteq ba$ و$ba \not\sqsubseteq ab$.
\end{ex}

\begin{defn}[الترتيب الصارم]
\emph{الترتيب الصارم} علاقة تتصف باللاانعكاسية واللاتناظرية والتعدية.
\end{defn}

\begin{defn}[الترتيب الخطي الصارم]\ollabel{def:strictlinearorder}
يسمى الترتيب الصارم الذي يكون متصلًا أيضًا
\emph{ترتيبًا كليًا صارمًا} أو \emph{ترتيبًا خطيًا صارمًا.}
\end{defn}

\begin{ex}
$\le$ هو الترتيب الخطي المقابل للترتيب الخطي الصارم~$<$.
$\subseteq$ هو الترتيب الجزئي المقابل للترتيب الصارم~$\subsetneq$.
\end{ex}

يمكن تحويل أي ترتيب صارم $R$ على~$A$ إلى ترتيب جزئي بإضافة القطر
$\Id{A}$، أي بإضافة جميع الأزواج~$\tuple{x, x}$. (وتسمى هذه العملية
\emph{الإغلاق الانعكاسي} لـ~$R$.) وبالعكس، يمكن البدء بترتيب جزئي والحصول
على ترتيب صارم بحذف~$\Id{A}$. وتضبط النتيجتان الآتيتان ذلك بدقة.

\begin{prop}\ollabel{prop:stricttopartial}
إذا كان $R$ ترتيبًا صارمًا على~$A$، فإن $R^+ = R \cup \Id{A}$ ترتيب
جزئي. وعلاوة على ذلك، إذا كان $R$ ترتيبًا خطيًا صارمًا، فإن $R^+$ ترتيب
خطي.
\end{prop}

\begin{proof}
افترض أن $R$ ترتيبًا صارمًا، أي إن $R \subseteq A^2$ وإن $R$ تتصف
باللاانعكاسية واللاتناظرية والتعدية. ولتكن $R^+ = R \cup \Id{A}$. علينا أن نثبت أن $R^+$
انعكاسية ومضادّة للتناظر ومتعدية.

من الواضح أن $R^+$ انعكاسية، لأن $\tuple{x, x} \in \Id{A} \subseteq
R^+$ لكل $x \in A$.

ولإثبات أن $R^+$ مضادّة للتناظر، افترض برهانًا بالخلف أن $R^+xy$ و
$R^+yx$، لكن $x \neq y$. وبما أن $\tuple{x,y} \in R \cup \Id{A}$، لكن
$\tuple{x, y} \notin \Id{A}$، فلا بد أن $\tuple{x, y} \in R$، أي إن
$Rxy$. وبالمثل،~$Ryx$. لكن هذا يناقض افتراض أن $R$ لا تناظرية.

ولإثبات التعدية، افترض أن $R^+xy$ و$R^+yz$. فإذا كان كل من
$\tuple{x, y} \in R$ و$\tuple{y,z} \in R$، فإن $\tuple{x, z} \in
R$ لأن $R$~متعدية. وإلا، فيصدق واحد على الأقل من الشرطين الآتيين:
$\tuple{x, y} \in \Id{A}$، أي $x = y$، أو $\tuple{y, z} \in \Id{A}$،
أي $y = z$.
في الحالة الأولى، لدينا $R^+yz$ بالفرض، و$x = y$، ومن ثم $R^+xz$.
وبالمثل في الحالة الثانية. وفي كلتا الحالتين، $R^+xz$؛ ومن ثم فإن
$R^+$ متعدية أيضًا.

أما فقرة «علاوة على ذلك»، فافترض أن $R$ متصلة أيضًا. وعندئذ، لكل
$x \neq y$، يصدق واحد على الأقل من $Rxy$ و~$Ryx$، أي يصدق واحد على الأقل
من $\tuple{x, y} \in R$ و$\tuple{y, x} \in R$. وبما أن $R \subseteq R^+$،
يبقى هذا صحيحًا في $R^+$، فتكون $R^+$ متصلة أيضًا.
\end{proof}

\begin{prop}\ollabel{prop:partialtostrict}
إذا كان $R$ ترتيبًا جزئيًا على~$A$، فإن $R^- = R \setminus \Id{A}$ ترتيب
صارم. وعلاوة على ذلك، إذا كان $R$ ترتيبًا خطيًا، فإن $R^-$ ترتيب خطي
صارم.
\end{prop}

\begin{proof}
يُترك إثبات ذلك تمرينًا.
\end{proof}

\begin{prob}
أعط برهانًا على \olref[sfr][rel][ord]{prop:partialtostrict}.
\end{prob}

تثبت النتيجة البسيطة الآتية أن الترتيبات الخطية الصارمة تحقق خاصية شبيهة
بالامتدادية:

\begin{prop}\ollabel{prop:extensionality-strictlinearorders}
إذا كان $<$ ترتيبًا خطيًا صارمًا على $A$، فإن:
\[
  (\forall a, b \in A)((\forall x \in A)(x < a \liff x < b) \lif a = b).
\]
\end{prop}

\begin{proof}
افترض $(\forall x \in A)(x < a \liff x < b)$. فإذا كان $a < b$، لزم أن
$a < a$، وهذا يناقض كون $<$ لا انعكاسيًا؛ ومن ثم $a \nless b$. وبالطريقة
نفسها تمامًا، $b \nless a$. ولذلك $a = b$، لأن $<$ متصل.
\end{proof}
