\olsection{المصفوفات}

يعرّف المنطق المتعدد القيم بلغته ومجموعة قيم صدقه~$V$ ومجموعة جزئية
من قيم الصدق المميزة ودوال الصدق لروابطه. وتسمى هذه العناصر مجتمعة
\emph{مصفوفة}.

\begin{defn}[المصفوفة]
\ollabel{defn:matrix}
تتألف \emph{مصفوفة} المنطق~$\Log L$ من:
\begin{enumerate}
\item مجموعة من الروابط تؤلف لغة~$\Lang L$؛
\item مجموعة غير خالية~$V \neq \emptyset$ من قيم الصدق؛
\item مجموعة~$V^+ \subseteq V$ من قيم الصدق المميزة؛
\item لكل رابط~$\star$ ذي~$n$ من المواضع في~$\Lang L$ دالة صدق
$\tf{\star} : V^n \to V$. وإذا كان~$n = 0$، كانت~$\tf{\star}$ مجرد
عنصر من~$V$.
\end{enumerate}
\end{defn}

\begin{ex}
تتألف مصفوفة المنطق الكلاسيكي~\LogCL{} من:
\begin{enumerate}
  \item لغة القضايا القياسية~$\Lang L_0$ بروابطها
  $\lfalse$ و$\lnot$ و$\land$ و$\lor$ و$\lif$.
  \item مجموعة قيم الصدق~$V = \{\True, \False\}$.
  \item $\True$ هي القيمة المميزة الوحيدة؛ أي~$V^+ = \{\True\}$.
  \item أما~$\lfalse$، فلدينا~$\tf{\lfalse} = \False$. وأما دوال
  الصدق الأخرى فتعطيها جداول الصدق المعتادة (انظر
  \olref{fig:tf-CL}).
\end{enumerate}
\begin{figure}
  \begin{center}
      \begin{tabular}{c|c}
        $\tf{\lnot}$ & \\
        \hline
        $\True$ & $\False$ \\
        $\False$ & $\True$
      \end{tabular}
      \quad
      \begin{tabular}{c|cc}
        $\tf{\land}$ & $\True$ & $\False$ \\
        \hline
        $\True$ & $\True$ & $\False$ \\
        $\False$ & $\False$ & $\False$
      \end{tabular}
      \quad
      \begin{tabular}{c|cc}
        $\tf{\lor}$ & $\True$ & $\False$ \\
        \hline
        $\True$ & $\True$ & $\True$ \\
        $\False$ & $\True$ & $\False$
      \end{tabular}
      \quad
      \begin{tabular}{c|cc}
        $\tf{\lif}$ & $\True$ & $\False$ \\
        \hline
        $\True$ & $\True$ & $\False$ \\
        $\False$ & $\True$ & $\True$
      \end{tabular}
    \end{center}
    \caption{دوال الصدق للمنطق الكلاسيكي~$\LogCL$.}
    \ollabel{fig:tf-CL}
  \end{figure}
  \end{ex}
