\olsection{وحدانية القراءة}

قد نتساءل هل لكل حد طريقة وحيدة لتكوينه، والجواب نعم. فلكل حد لامبدا
طريقة واحدة فقط لبنائه وتأويله. وفي النقاش الآتي، نعني بـ\emph{التكوين}
إجراء بناء حد باستعمال قواعد التكوين (مرة واحدة أو مرات) في
\olref[trm]{defn:term}.

\begin{lem}\ollabel{lem:term-start}
يبدأ الحد إما بمتغير وإما بقوس.
\end{lem}

\begin{proof}
لا يعد الشيء حدًا إلا إذا بني وفق~\olref[trm]{defn:term}. فإذا كان ناتج
\olref[trm]{defn:term-var}، فلا بد أن يكون متغيرًا. وإذا كان ناتج
\olref[trm]{defn:term-abs} أو~\olref[trm]{defn:term-app}، فإنه يبدأ بقوس.
\end{proof}

\begin{lem}\ollabel{lem:app-start}
يبدأ ناتج التطبيق إما بقوسين وإما بقوس ومتغير.
\end{lem}

\begin{proof}
إذا كان~$M$ ناتج تطبيق، فهو من الصورة~$(PQ)$، ولذلك يبدأ بقوس. ولما كان
$P$ حدًا، فإنه يبدأ، وفق~\olref{lem:term-start}، إما بقوس وإما بمتغير.
\end{proof}

\begin{lem}\ollabel{lem:initial}
لا يكون أي جزء ابتدائي حقيقي من حد حدًا في ذاته.
\end{lem}

\begin{prob}
أثبت~\olref[lam][syn][unq]{lem:initial} بالاستقراء على طول الحدود.
\end{prob}

\begin{prop}[وحدانية القراءة] \ollabel{prop:unq}
لكل حد تكوين وحيد. وبعبارة أخرى، إذا كوّن تكوينٌ ما حدًا~$M$، كان هو
التكوين الوحيد الذي يستطيع تكوين هذا الحد.
\end{prop}

\begin{proof}
  نثبت ذلك بالاستقراء على تكوين الحدود.
  \begin{enumerate}
    \item يكون~$M$ من الصورة~$x$، حيث~$x$ متغير ما. ولأن نواتج التجريد
      والتطبيق تبدأ دائمًا بأقواس، فلا يمكن أن يكون أي منهما قد استعمل
      لبناء~$M$؛ ومن ثم لا بد أن يكون تكوين~$M$ خطوة واحدة من
      \olref[trm]{defn:term}\olref[trm]{defn:term-var}.
    \item يكون~$M$ من الصورة~$(\lambd[x][N])$، حيث~$x$ متغير و$N$ حد.
      ولا يمكن أن يكون قد بني وفق
      \olref[trm]{defn:term}\olref[trm]{defn:term-var}، لأنه ليس متغيرًا
      واحدًا. وليس هو ناتج تطبيق بموجب~\olref{lem:app-start}. لذلك لا
      يمكن أن يكون~$M$ إلا ناتج تجريد على~$N$. وبفرض الاستقراء نعلم أن
      تكوين~$N$ نفسه وحيد.
    \item يكون~$M$ من الصورة~$(PQ)$، حيث~$P$ و$Q$ حدان. ولأنه يبدأ بقوس،
      فلا يمكن أن يكون قد بني أيضًا بواسطة
      \olref[trm]{defn:term}\olref[trm]{defn:term-var}. وبموجب
      \olref{lem:term-start}، لا يمكن أن يبدأ~$P$ بـ$\lambd$، ولذلك لا
      يمكن أن يكون~$(PQ)$ ناتج تجريد. والآن افترض وجود طريقة أخرى لبناء
      $M$ بالتطبيق، كأن يكون أيضًا من الصورة~$(P'Q')$. فإذا كان~$P \neq
      P'$، كان أحد~$P$ و$P'$ جزءًا ابتدائيًا حقيقيًا من الآخر، وهذا
      مستحيل بموجب~\olref{lem:initial}. ومن ثم~$P=P'$، ويلزم بعد ذلك
      $Q=Q'$. وهكذا يتعين~$P$ و$Q$ تعينًا وحيدًا، وبفرض الاستقراء نعلم
      أن تكويني~$P$ و$Q$ وحيدان.
  \end{enumerate}
\end{proof}

وفيما يأتي إعادة صياغة أيسر قراءةً للقضية السابقة:
\begin{prop}
  لا يمكن أن يكون الحد~$M$ إلا من إحدى الصور الآتية:
  \begin{enumerate}
    \item $x$، حيث~$x$ متغير يعيّنه~$M$ تعيينًا وحيدًا.
    \item $(\lambd[x][N])$، حيث~$x$ متغير و$N$ حد آخر، وكلاهما يعيّنه
      $M$ تعيينًا وحيدًا.
    \item $(PQ)$، حيث~$P$ و$Q$ حدان يعيّنهما~$M$ تعيينًا وحيدًا.
  \end{enumerate}
\end{prop}
