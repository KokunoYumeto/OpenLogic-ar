\olsection{بيان تمايز النظم}

رأينا في \olref[prs]{sec} كيف نثبت أن نسقين من المنطق الجهوي هما في
الواقع النسق نفسه. وتتيح لنا \olref[snd]{thm:soundness} أن نبين تمايز
نسقين جهويين $\Sigma$ و$\Sigma'$، وذلك بإيجاد !!a{formula}~$!A$ بحيث
$\Sigma' \Proves !A$ وتخفق هذه الصيغة في نموذج لـ~$\Sigma$.

\begin{prop}
  $\Log{KD} \subsetneq \Log{KT}$
\end{prop}

\begin{proof} هذه هي النظيرة التركيبية للحقيقة الدلالية القائلة إن كل
  علاقة انعكاسية تسلسلية. ولكي نبين $\Log{KD} \subseteq \Log{KT}$،
  علينا أن نرى أن $\Log{KD} \Proves !B$ يستلزم $\Log{KT} \Proves
  !B$، وهذا ينتج من $\Log{KT} \Proves \Ax{D}$، كما بينا في
  \olref[prs]{prop:S5facts}\olref[prs]{prop:S5facts-KT-D}. ولكي نبين
  أن الاحتواء صارم، يكفي، بمقتضى السلامة
  (\olref[snd]{thm:soundness})، أن نبرز نموذجا لـ\Log{KD} تخفق فيه
  \Ax{T}، أي $\Box p \lif p$ (وهي مهمة سهلة تترك تمرينا)، إذ عندئذ
  يترتب بالسلامة أن $\Log{KD} \Proves/ \Box p \lif p$.
\end{proof}

\begin{prop}
  $\Log{KB} \neq \Log{K4}$.
\end{prop}

\begin{proof}
  نبني نموذجا تناظريا تخفق فيه بعض أمثلة \Ax{4}؛ ولما كان المثال
  !!{derivable} بوضوح في \Log{K4} لكنه ليس كذلك في \Log{KB}، ينتج
  أن $\Log{K4} \nsubseteq \Log{KB}$. انظر النموذج التناظري
  $\mModel{M}$ في \olref{fig:Bnot4}. ولأن النموذج تناظري، فإن
  \Ax{K} و~\Ax{B} صادقتان في~$\mModel{M}$ (بحسب
  \olref[syn][sch]{prop:Kvalid} و
  \olref[frd][acc]{thm:soundschemas}، على الترتيب). ومع ذلك،
  $\mSat/{M}{\Box p \lif \Box\Box p}[w_1]$.
\end{proof}

\begin{thm}\ollabel{thm:KTBnot45}
  $\Log{KTB} \Proves/ \Ax{4}$ و$\Log{KTB} \Proves/ \Ax{5}$.
\end{thm}

\begin{proof}
  نعلم من \olref[frd][acc]{thm:soundschemas} أن جميع أمثلة \Ax{T}
  و\Ax{B} صادقة في كل نموذج انعكاسي وتناظري (على الترتيب). ومن ثم
  يكفي بالسلامة أن نجد نموذجا انعكاسيا وتناظريا يشتمل على عالم يخفق
  فيه مثال لـ~\Ax{4}، وبالمثل لـ~\Ax{5}. ونستعمل النموذج نفسه لكلا
  الادعاءين. انظر النموذج التناظري الانعكاسي في
  \olref{fig:KTBnot45}. عندئذ
  $\mSat/{M}{\Box p \lif \Box \Box p}[w_1]$، فتخفق \Ax{4} في $w_1$.
  وبالمثل، $\mSat/{M}{\Diamond \lnot p \lif \Box \Diamond \lnot
    p}[w_2]$، فيخفق مثال \Ax{5} الذي فيه $!A = \lnot p$ في~$w_2$.
\end{proof}

\begin{thm}\ollabel{thm:KD5not4}
  $\Log{KD5} \neq \Log{KT4} = \Log{S4}$.
\end{thm}

\begin{proof}
  نعلم من \olref[frd][acc]{thm:soundschemas} أن جميع أمثلة \Ax{D}
  و~\Ax{5} صادقة في جميع النماذج التسلسلية الإقليدية. ولذلك يكفي أن
  نجد نموذجا تسلسليا إقليديا يشتمل على عالم يخفق فيه بعض أمثلة
  \Ax{4}. انظر نموذج \olref{fig:KD5not4}، ولاحظ أن
  $\mSat/{M}{\Box p \lif \Box\Box p}[w_1]$.
\end{proof}
