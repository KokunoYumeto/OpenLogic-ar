\olsection{القواعد المشتقة}

من الواضح أن العثور على !!{derivation}s وكتابتها صعب ومرهق ومتكرر.
فكثيرا ما نريد مثلا الانتقال من $!A \lif !B$ إلى
$\Box !A \lif \Box !B$، أي تطبيق القاعدة~\RK. ويتطلب ذلك تطبيقا لـ
\Nec، ثم تسجيل المثال المناسب لـ~\Ax{K}، ثم تطبيق~\MP. كما يتطلب
الانتقال من $!A \lif !B$ و$!B \lif !C$ إلى $!A \lif !C$ تسجيل
المثال (الطويل) للقضية الصادقة صدقا تحليليا
\[
(!A \lif !B) \lif ((!B \lif !C) \lif (!A \lif !C))
\]
وتطبيق \MP{} مرتين. وكثيرا ما نريد أن نستبدل بـ!!{formula} جزئية
صيغة نعلم أنها مكافئة لها، مثلا \iftag{prvDiamond}{$\Diamond
  !A$ بـ$\lnot\Box\lnot !A$}{$\Box !A$ بـ$\lnot\Diamond\lnot !A$}،
أو $\lnot\lnot !A$ بـ$!A$. ولذلك، بدلا من كتابة !!{derivation}
الفعلي، يكون من الأنسب أن نسجل فقط سبب كون الخطوات الوسيطة
!!{derivable}. ولهذا الغرض، لنجمع بعض الحقائق عن !!{derivability}.

\begin{prop}
  إذا كان $\Log{K} \Proves !A_1$، \dots، $\Log{K} \Proves !A_n$،
  وكانت $!B$ تلزم عن $!A_1$، \dots، $!A_n$ بمنطق القضايا، فعندئذ
  $\Log{K} \Proves !B$.
\end{prop}

\begin{proof}
  إذا كانت $!B$ تلزم عن $!A_1$، \dots،~$!A_n$ بمنطق القضايا، فإن
  \[
  !A_1 \lif (!A_2 \lif \cdots (!A_n \lif !B)\dots)
  \]
  مثال لقضية صادقة صدقا تحليليا. وبتطبيق \MP{} عدد $n$ من المرات
  نحصل على !!{derivation} لـ~$!B$.
\end{proof}

سنشير إلى استعمال هذه القضية بـ~\PL.

\begin{prop}
  إذا كان $\Log{K} \Proves !A_1 \lif (!A_2 \lif \cdots (!A_{n-1} \lif
  !A_n)\dots)$، فعندئذ $\Log{K} \Proves \Box !A_1 \lif (\Box !A_2 \lif
  \cdots (\Box !A_{n-1} \lif \Box !A_n)\dots)$.
\end{prop}

\begin{proof}
  بالاستقراء على $n$، تماما كما في برهان \olref[nor]{prop:rk}.
\end{proof}

سنشير إلى استعمال هذه القضية بـ~\RK. ولنوضح كيف تساعد هذه النتائج
على إثبات نتائج !!{derivability} بسهولة أكبر.

\begin{prop}
  $\Log{K} \Proves (\Box!A \land \Box!B) \lif \Box (!A \land !B)$
\end{prop}

\begin{proof}
  \begin{derivation}
    1. & $\Log{K} \Proves !A \lif (!B \lif (!A \land !B))$ & \Taut \\
    2. & $\Log{K} \Proves \Box!A \lif (\Box !B \lif \Box(!A \land !B))$ & \RK, 1\\
    3. & $\Log{K} \Proves (\Box!A \land \Box!B) \lif \Box(!A \land !B)$ & \PL, 2
  \end{derivation}
\end{proof}

\begin{prop}\ollabel{prop:rewriting}
  إذا كان $\Log{K} \Proves !A \liff !B$ و$\Log{K} \Proves
  \Subst{!C}{!A}{q}$، فعندئذ $\Log{K} \Proves \Subst{!C}{!B}{q}$.
\end{prop}

\begin{proof}
  تمرين.
\end{proof}

\begin{prob}
  أثبت \olref[nml][prf][der]{prop:rewriting} بأن تثبت، بالاستقراء على
  تعقيد~$!C$، أنه إذا كان $\Log{K} \Proves !A \liff !B$، فعندئذ
  $\Log{K} \Proves \Subst{!C}{!A}{q} \liff \Subst{!C}{!B}{q}$.
\end{prob}

تفيد هذه القضية بوجه خاص عندما نريد تحويل $\Diamond$ إلى $\Box$
(أو العكس)، أو حذف النفي المزدوج من داخل !!a{formula}. وفيما يأتي،
سنعلّم تطبيقات \olref{prop:rewriting} بعبارة «$!B$ بدلا من $!A$»
كلما أعدنا كتابة !!a{formula}~$!C(!A)$ على صورة~$!C(!B)$. وبعبارة
أخرى، تختصر «$!B$ بدلا من $!A$» ما يأتي:
  \begin{derivation}
    & $\Proves !C(!A)$ \\
    & $\Proves !A \liff !B$\\
    & $\Proves !C(!B)$ & بواسطة \olref{prop:rewriting}
  \end{derivation}
فمثلا:

\begin{prop}
  $\Log{K} \Proves \lnot\Box p \lif \Diamond \lnot p$
\end{prop}

\begin{proof}
  \iftag{prvDiamond}{% Diamond is primitive
  \begin{derivation}
    1. & $\Log{K} \Proves \Diamond \lnot p \liff \lnot\Box\lnot\lnot p$ &
    \Dual\\
    2. & $\Log{K} \Proves \lnot \Box \lnot\lnot p \lif \Diamond \lnot p$ & \PL, 1\\
    3. & $\Log{K} \Proves \lnot\Box p \lif \Diamond \lnot p$ & $p$ بدلا من $\lnot\lnot p$\\
  \end{derivation}
  }{% Diamond is defined
  \begin{derivation}
    1. & $\Log{K} \Proves \lnot \Box \lnot\lnot p \lif \Diamond \lnot p$ & \Taut\\
    2. & $\Log{K} \Proves \lnot\Box p \lif \Diamond \lnot p$ & $p$ بدلا من $\lnot\lnot p$\\
  \end{derivation}
  الصيغة في السطر~$1$ مثال لـ$\lnot q \lif \lnot q$، بإحلال
  $\Box\lnot\lnot p$ محل~$q$. والصيغتان $\Diamond\lnot p$ و
  $\lnot\Box\lnot\lnot p$ متطابقتان، لأن $\Diamond$ معرّف بأنه
  ~ $\lnot\Box\lnot$.}
\end{proof}

في !!{derivation} أعلاه، تختصر الخطوة الأخيرة «$p$ بدلا من
$\lnot\lnot p$» ما يأتي:
  \begin{derivation}
    & $\Log{K} \Proves \lnot \Box \lnot\lnot p \lif \Diamond \lnot
    p$\\
    & $\Log{K} \Proves \lnot\lnot p \liff p$ & \Taut\\
    & $\Log{K} \Proves \lnot\Box p \lif \Diamond \lnot p$ & بواسطة \olref{prop:rewriting}
  \end{derivation}
وتقوم هنا الأدوار التي لـ$!C(q)$ و$!A$ و$!B$ في
\olref{prop:rewriting}، على الترتيب، الصيغ $\lnot \Box q \lif
\Diamond \lnot p$ و$\lnot\lnot p$ و~$p$.

عندما تحتوي !!a{formula} على !!{formula} جزئية $\lnot\Diamond !A$،
يمكننا استبدال $\Box\lnot !A$ بها باستعمال \olref{prop:rewriting}،
لأن $\Log{K} \Proves \lnot\Diamond !A \liff \Box\lnot !A$. وسنشير
إلى هذا الاستبدال وأمثاله ببساطة بعبارة «$\Box\lnot$ بدلا من
$\lnot\Diamond$».

تسوغ القضية الآتية إثباتنا نتائج !!{derivability} على نحو مخططي.
فمثلا لا تثبت القضية السابقة أن $\Log{K} \Proves \lnot\Box p \lif
\Diamond \lnot p$ فحسب، بل تثبت أيضا أن
$\Log{K} \Proves \lnot\Box !A \lif \Diamond \lnot !A$ لكل~$!A$.

\begin{prop}
  إذا كانت $!A$ مثال إحلال لـ$!B$ وكان $\Log{K} \Proves !B$، فعندئذ
  $\Log{K} \Proves !A$.
\end{prop}

\begin{proof}
  من الممل، وإن كان روتينيا، التحقق (بالاستقراء على طول
  !!{derivation} لـ~$!B$) من أن تطبيق إحلال على !!{derivation} بأكمله
  ينتج أيضا !!{derivation} صحيحا. وعلى وجه التحديد، فأمثلة إحلال
  أمثلة القضايا الصادقة صدقا تحليليا هي نفسها أمثلة من ذلك القبيل،
  وأمثلة إحلال أمثلة~\iftag{prvDiamond}{\Dual{} و~}{}\Ax{K} هي نفسها
  أمثلة لـ~\iftag{prvDiamond}{\Dual{} و~}{}\Ax{K}، وتظل تطبيقات
  \MP{} و~\Nec{} صحيحة عند إحلال !!{formula}s محل
  !!{propositional variable}s في المقدمة (أو المقدمات) وفي النتيجة.
\end{proof}
