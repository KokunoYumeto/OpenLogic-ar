\olsection{سلامة القواعد الإضافية}

نقول إن قاعدة ما سليمة لفئة من النماذج إذا كان الفرع الناتج من تطبيقها
قابلًا للإرضاء في نموذج من تلك الفئة كلما كان الفرع الأصلي في
!!a{tableau} قابلًا للإرضاء في نموذج منها.

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج الانعكاسية.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{يُوسَّع الفرع بتطبيق T$\Box$ على
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{تمرين.}{وينتج من ذلك !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ جديدة على الفرع. افترض
    $\mSat{M}{\Gamma}[f]$، وبوجه خاص $\mSat{M}{\Box
      !B}[f(\sigma)]$. ولأن $R$ انعكاسية، نعلم أن
    $Rf(\sigma)f(\sigma)$. ومن ثم $\mSat{M}{!B}[f(\sigma)]$، أي إن
    $\mModel{M}, f$ يرضيان $\sFmla{\True}{!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{يُوسَّع الفرع بتطبيق T$\Diamond$ على
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{وينتج من ذلك !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$ جديدة على الفرع. افترض
    $\mSat{M}{\Gamma}[f]$، وبوجه خاص $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$. ولأن $R$ انعكاسية، نعلم أن
    $Rf(\sigma)f(\sigma)$. ومن ثم $\mSat/{M}{!B}[f(\sigma)]$، أي إن
    $\mModel{M}, f$ يرضيان $\sFmla{\False}{!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-T}.
\end{prob}
\tagendprob
