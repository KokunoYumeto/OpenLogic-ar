\olsection{منطق الإعلان العام}

تتيح لنا المنطقيات المعرفية الديناميكية تمثيل الطرق التي تتغير بها
معرفة العوامل مع مرور الزمن أو مع حصولها على معلومات جديدة. ويمثل
كثير من هذه المنطقيات تغيرات المعرفة باستعمال \emph{أحداث} أو
\emph{تحديثات} معلوماتية. وأبسط أنواع التحديث إعلان عام تُعلن فيه صيغة
ما إعلانًا صادقًا وتشهد جميع العوامل وقوعه معًا. ولتحقيق ذلك نوسّع
اللغة كما يأتي:

\begin{defn}
لتكن $G$ مجموعة من رموز العوامل. تحتوي اللغة الأساسية للمنطق المعرفي
متعدد العوامل ذي الإعلانات العامة ما يأتي:
\begin{enumerate}
  \tagitem{prvFalse}{الثابت القضي لل!!{falsity}~$\lfalse$.}{}
  \tagitem{prvTrue}{الثابت القضي لل!!{truth}~$\ltrue$.}{}
  \item مجموعة !!{denumerable} من ال!!{propositional variable}s:
    $\Obj p_0$، $\Obj p_1$، $\Obj p_2$، \dots
  \item الروابط القضوية: \startycommalist
  \iftag{prvNot}{\ycomma $\lnot$ (النفي)}{}%
  \iftag{prvAnd}{\ycomma $\land$ (العطف)}{}%
  \iftag{prvOr}{\ycomma $\lor$ (الفصل)}{}%
  \iftag{prvIf}{\ycomma $\lif$ (!!{conditional})}{}%
  \iftag{prvIff}{\ycomma $\liff$ (!!{biconditional})}{}%
  \item مؤثر المعرفة $\Knows_a$، حيث $a \in G$.
  \item مؤثر الإعلان العام $[!B]$، حيث $!B$ هي !!a{formula}.
\end{enumerate}
\end{defn}

يعمل مؤثر الإعلان العام عمل مؤثر الصندوق، ولذلك يُعطى تعريفنا
الاستقرائي للغة على النحو الآتي:

\begin{defn}
  تُعرَّف \emph{!!^{formula}s} اللغة المعرفية استقرائيًا كما يأتي:
  \begin{enumerate}
  \tagitem{prvFalse}{$\lfalse$ !!{formula} ذرية.}{}

  \tagitem{prvTrue}{$\ltrue$ !!{formula} ذرية.}{}

  \item كل متغير قضي $\Obj p_i$ هو !!{formula} (ذرية).

  \tagitem{prvNot}{إذا كانت $!A$ !!a{formula}، فإن $\lnot !A$
  !!a{formula}.}{}

  \tagitem{prvAnd}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \land
  !B)$ !!a{formula}.}{}

  \tagitem{prvOr}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \lor !B)$
  !!a{formula}.}{}

  \tagitem{prvIf}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \lif !B)$
  !!a{formula}.}{}

  \tagitem{prvIff}{إذا كانت $!A$ و$!B$ !!{formula}s، فإن $(!A \liff !B)$
  !!a{formula}.}{}

  \item إذا كانت $!A$ !!a{formula} وكان $a \in G$، فإن $\Knows_a !A$
  !!a{formula}.
  
  \item إذا كانت $!A$ و$!B$ !!{formula}s، فإن $[!A] !B$ !!a{formula}.

  \tagitem{limitClause}{ولا شيء غير ذلك !!a{formula}.}{}
\end{enumerate}
\end{defn}

القراءة المقصودة لل!!{formula} $[!A] !B$ هي: «بعد أن تُعلن $!A$
إعلانًا صادقًا، تصدق~$!B$.» وسيكون من المفيد أحيانًا أيضًا الحديث عن
المعرفة المشتركة في سياق الإعلانات العامة، ولذلك قد تشتمل اللغة كذلك
على مؤثر المعرفة المشتركة~$\CKnows_G !A$.
