\olsection{بديهيات الكمّيات وقواعدها}

\begin{defn}[بديهيات الكمّيات]
\emph{البديهيات} التي تحكم الكمّيات هي جميع حالات الصيغ الآتية:
\begin{align}
\ollabel{ax:q1} & \lforall[x][!B] \lif !B(t), \\
\ollabel{ax:q2} & !B(t) \lif \lexists[x][!B],
\end{align}
لكل حد مغلق~$t$.
\end{defn}

\begin{defn}[قواعد الكمّيات]
\begin{enumerate}
\item إذا ظهرت $!B \lif !A(a)$ بالفعل في ال!!{derivation} ولم يظهر
  $a$ في~$\Gamma$ ولا في~$!B$، فإن $!B \lif
  \lforall[x][!A(x)]$ خطوة استدلال صحيحة.
\item إذا ظهرت $!A(a) \lif !B$ بالفعل في ال!!{derivation} ولم يظهر
  $a$ في~$\Gamma$ ولا في~$!B$، فإن $\lexists[x][!A(x)] \lif
  !B$ خطوة استدلال صحيحة.
\end{enumerate}
\end{defn}
