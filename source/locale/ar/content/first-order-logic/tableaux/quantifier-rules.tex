\olsection{قواعد المكممات}

\subsection{قواعد~$\lforall$}

\begin{defish}
\AxiomC{\sFmla{\True}{\lforall[x][!A(x)]}}
\RightLabel{\TRule{\True}{\forall}}
\UnaryInfC{\sFmla{\True}{!A(t)}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\False}{\lforall[x][!A(x)]}}
\RightLabel{\TRule{\False}{\lforall}}
\UnaryInfC{\sFmla{\False}{!A(a)}}
\DisplayProof
\end{defish}

في~\TRule{\True}{\lforall}، يكون~$t$ حدًا مغلقًا (أي حدًا بلا
متغيرات). وفي~\TRule{\False}{\lforall}، يكون~$a$ !!a{constant} يجب ألا
يرد في أي موضع من الفرع فوق قاعدة~\TRule{\False}{\lforall}. ونسمي~$a$
\emph{المتغير المميَّز} لاستدلال~\TRule{\False}{\forall}.
\footnote{نستعمل مصطلح ``المتغير المميَّز'' حتى وإن كان~$a$ في القاعدة
أعلاه !!a{constant}، وذلك لأسباب تاريخية.}

\subsection{قواعد~$\lexists$}

\begin{defish}
\AxiomC{\sFmla{\True}{\lexists[x][!A(x)]}}
\RightLabel{\TRule{\True}{\lexists}}
\UnaryInfC{\sFmla{\True}{!A(a)}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\False}{\lexists[x][!A(x)]}}
\RightLabel{\TRule{\False}{\lexists}}
\UnaryInfC{\sFmla{\False}{!A(t)}}
\DisplayProof
\end{defish}

مرة أخرى، يكون~$t$ حدًا مغلقًا، ويكون~$a$ !!a{constant} لا يرد في الفرع
فوق قاعدة~\TRule{\True}{\lexists}. ونسمي~$a$ \emph{المتغير المميَّز}
لاستدلال~\TRule{\True}{\lexists}.

يسمى الشرط القاضي بألا يرد المتغير المميَّز في الفرع فوق استدلال
\TRule{\False}{\lforall} أو~\TRule{\True}{\lexists}
\emph{شرط المتغير المميَّز}.

\begin{explain}
تذكّر الاصطلاح القائل إننا عندما تكون~$!A$ !!a{formula} يكون فيها
ال!!{variable}~$x$ حرًا، نشير إلى ذلك بكتابة~$!A(x)$. وفي السياق نفسه،
تكون~$!A(t)$ اختصارًا لـ~$\Subst{!A}{t}{x}$. ولذلك يمكننا أيضًا كتابة
قاعدة~\TRule{\False}{\lexists} هكذا:
\begin{prooftree}
  \AxiomC{\sFmla{\False}{\lexists[x][!A]}}
  \RightLabel{\TRule{\False}{\lexists}}
  \UnaryInfC{\sFmla{\False}{\Subst{!A}{t}{x}}}
\end{prooftree}
لاحظ أن~$t$ قد يرد أصلًا في~$!A$؛ فمثلًا قد تكون~$!A$ هي
$\Atom{\Obj P}{t,x}$. ولذلك فإن استنتاج
$\sFmla{\False}{\Atom{\Obj P}{t,t}}$ من
$ \sFmla{\False}{\lexists[x][\Atom{\Obj P}{t,x}]}$ تطبيق صحيح لـ
\TRule{\False}{\lexists}. غير أن شرطي المتغير المميَّز في
\TRule{\False}{\lforall} و\TRule{\True}{\lexists} يقتضيان ألا يرد
ال!!{constant}~$a$ في~$!A$. ولذلك لا يمكنك أن تستنتج على نحو صحيح
$\sFmla{\False}{\Atom{\Obj P}{a,a}}$ من
$\sFmla{\False}{\lforall[x][\Atom{\Obj P}{a,x}]}$ باستعمال
$\TRule{\False}{\lforall}$.
\end{explain}

\begin{explain}
لا توجد قيود على الحد~$t$ في~\TRule{\True}{\lforall}
و\TRule{\False}{\lexists}. أما في قاعدتي~\TRule{\True}{\lexists}
و\TRule{\False}{\lforall}، فيقتضي شرط المتغير المميَّز ألا يرد
ال!!{constant}~$a$ في أي موضع من الفروع فوق الاستدلال المعني. وهذا
ضروري لضمان سلامة النسق. ومن دون هذا الشرط، يكون الآتي !!{tableau}
مغلقًا لـ~$\lexists[x][\formula{A}(x)] \lif
\lforall[x][\formula{A}(x)]$:
\begin{center}
\begin{tableau}{}
  [\sFmla{\False}{\lexists[x][\formula{A}(x)] \lif \lforall[x][\formula{A}(x)]}, just=\TAss
    [\sFmla{\True}{\lexists[x][\formula{A}(x)]},
      just={\TRule{\False}{\lif}[1]}
      [\sFmla{\False}{\lforall[x][\formula{A}(x)]},
        just={\TRule{\False}{\lif}[1]}
        [\sFmla{\True}{\formula{A}(a)},
          just={\TRule{\True}{\lexists}[2]}
          [\sFmla{\False}{\formula{A}(a)},
            just={\TRule{\False}{\lforall}[3]}, close]
        ]
      ]
    ]
  ]
\end{tableau}
\end{center}
لكن~$\lexists[x][\formula{A}(x)] \lif
\lforall[x][\formula{A}(x)]$ غير صالحة.
\end{explain}
