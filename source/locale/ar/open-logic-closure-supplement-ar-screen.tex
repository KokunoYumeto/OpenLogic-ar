        {rules-tN2.tex}{\ol_screen_use_compact_float:}
        {rules-tN3}{\ol_screen_use_compact_float:}
        {rules-tN3.tex}{\ol_screen_use_compact_float:}

        {rules-G2c.tex}{\ol_screen_use_tight_proof_float:\cs_set_eq:NN \OLScreenCompactFloatSize \OLScreenResidualFloatSize}

% The retained closure sequence ends with a heading-only Chapter 8 driver
% immediately followed by the Part II driver. Memoir normally clears before
% every part; suppress that one redundant clear. All other part clears retain
% exact memoir behaviour. This is the proven FA transition implementation.
\makeatletter
\newcommand*{\OLScreenDefaultPartClear}{%
  \if@openright
    \cleardoublepage
  \else
    \clearpage
  \fi
}
\newcommand*{\OLScreenInstallPartTransition}{%
  \renewcommand{\@setuppart}{%
    \ifnum\value{chapter}=8\relax
      \ifnum\value{part}=1\relax
        \par
      \else
        \OLScreenDefaultPartClear
      \fi
    \else
      \OLScreenDefaultPartClear
    \fi
    \thispagestyle{part}%
    \if@twocolumn
      \onecolumn
      \@tempswatrue
    \else
      \@tempswafalse
    \fi
    \beforepartskip
  }%
}
\makeatother
\AddToHook{begindocument/before}{%
  \OLScreenInstallPartTransition
  \let\sFmla\OLScreenSignedFormula
  \let\OLScreenSavedCenter\center
  \let\OLScreenSavedEndCenter\endcenter
  \renewenvironment{center}{%
    \OLScreenSavedCenter
    \ifnum\value{chapter}=48\relax
      \ifnum\value{section}=4\relax
        \let\OLScreenSavedCenterCR\\
        \let\\\OLScreenCompactCenterCR
      \fi
    \fi
  }{%
    \ifnum\value{chapter}=48\relax
      \ifnum\value{section}=4\relax
        \let\\\OLScreenSavedCenterCR
      \fi
    \fi
    \OLScreenSavedEndCenter
  }%
  \allowdisplaybreaks[4]%
  \renewenvironment{sidewaysfigure}{\begin{figure}}{\end{figure}}%
  \aliaspagestyle{part}{empty}%
  \renewcommand*{\beforepartskip}{\null\vspace*{8pt}}%
  \renewcommand*{\midpartskip}{\par\vskip4pt}%
  \renewcommand*{\afterpartskip}{\par\vskip8pt}%
  % Chapter 8 follows a long Chapter 7 tail whose residual space can hold its
  % heading but not the Part II introduction. Move only that transition as a
  % unit; retain the compact six-line rule for every other replayed QA unit.
  \renewcommand*{\clearforchapter}{%
    \par
    \ifnum\value{chapter}=7\relax
      \clearpage
    \else
      \Needspace{6\baselineskip}%
    \fi
  }%
  \setlength{\beforechapskip}{0pt}%
  \setlength{\midchapskip}{0.25\baselineskip}%
  \setlength{\afterchapskip}{0.55\baselineskip}%
  \hypersetup{pdfstartview=Fit,pdfpagelayout=OneColumn,pdfpagemode=UseOutlines}%
}
