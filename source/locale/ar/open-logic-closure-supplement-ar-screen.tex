% Screen-reading Arabic closure supplement. It resolves external references
% against the screen reader so page links and page-number references agree.

\PassOptionsToClass{14pt,oneside}{memoir}
\AddToHook{class/memoir/after}{%
  \setstocksize{11in}{8.5in}%
  \settrimmedsize{\stockheight}{\stockwidth}{*}%
  \settrims{0pt}{0pt}%
  \settypeblocksize{9.3in}{6.5in}{*}%
  \setlrmarginsandblock{1in}{1in}{*}%
  \setulmarginsandblock{0.85in}{0.85in}{*}%
  % The 14 pt Arabic running heads need both a taller box and real clearance
  % from the physical top edge; an 8 pt head drop clipped Scheherazade marks.
  \setheadfoot{22pt}{24pt}%
  \setheaderspaces{20pt}{*}{*}%
  \checkandfixthelayout
  \raggedbottom
  % Give mixed RTL prose and inline mathematics modest screen-only flexibility.
  \setlength{\emergencystretch}{1.5em}%
}

% In chapter 2, section 1, the eighth align* is the long interpolation proof.
% Set only that block at memoir's supported 11 pt footnotesize; at 14 pt it is
% wider than the physical page, while the surrounding Arabic remains 14 pt.
\newcount\OLScreenInterpolationAlign
\AddToHook{env/align*/before}{%
  \ifnum\value{chapter}=2\relax
    \ifnum\value{section}=1\relax
      \global\advance\OLScreenInterpolationAlign by 1\relax
      \ifnum\OLScreenInterpolationAlign=8\relax
        \footnotesize
      \fi
    \fi
  \fi
}
\AddToHook{env/align*/after}{%
  \ifnum\value{chapter}=2\relax
    \ifnum\value{section}=1\relax
      \ifnum\OLScreenInterpolationAlign=8\relax
        \normalsize
      \fi
    \fi
  \fi
}
