% Keep Polish names in the Latin font even inside Arabic prose. The pinned

% TeX's legacy short bar is a control-slot glyph absent from Unicode text
% fonts. Draw the equivalent mathematical rule instead.
\renewcommand*{\bar}[1]{\overline{#1}}

% A bar nested inside the upstream \Th macro is otherwise swallowed by
% \mathbf under LuaLaTeX.  Keep ordinary theory names unchanged, but render a
% complemented theory name as bold Latin text plus a real combining overline;
% this preserves both the printed distinction and Unicode text extraction.
\ExplSyntaxOn
\cs_new:Npn \ol_build_theory_bar:n #1
  { \mathord{\text{\foreignlanguage{english}{\bfseries #1\char"0305}}} }
\RenewDocumentCommand \Th { m }
  {
    \tl_if_head_eq_meaning:nNTF {#1} \bar
      { \exp_args:No \ol_build_theory_bar:n { \tl_tail:n {#1} } }
      { \mathbf{#1} }
  }
\ExplSyntaxOff
