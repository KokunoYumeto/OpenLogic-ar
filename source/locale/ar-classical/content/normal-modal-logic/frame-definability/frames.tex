\olsection{الأطر}

\begin{defn}
  \emph{الإطار} زوج~$\mModel{F} = \tuple{W,R}$ حيث~$W$ مجموعة عوالم
  غير خالية و$R$ علاقة ثنائية على~$W$. ويكون النموذج~$\mModel{M}$
  \emph{مبنيًا على} إطار~$\mModel{F} = \tuple{W,R}$ إذا وفقط إذا كان
  $\mModel{M} = \tuple{W, R, V}$ لتقييم ما~$V$.
\end{defn}

\begin{defn}
  إذا كان~$\mModel{F}$ إطارًا، نقول إن~$!A$ \emph{صالحة في
    $\mModel{F}$،} ونكتب~$\mModel{F} \Entails !A$، إذا كان
  $\mSat{M}{!A}$ لكل نموذج~$\mModel{M}$ مبني على~$\mModel{F}$.

  وإذا كان~$\mClass{F}$ صنفًا من الأطر، نقول إن~$!A$ \emph{صالحة في
    $\mClass{F}$،} ونكتب~$\mClass{F} \Entails !A$، إذا وفقط إذا كان
  $\mModel{F} \Entails !A$ لكل إطار~$\mModel{F} \in \mClass{F}$.
\end{defn}

وإنما تظهر فائدة الأطر لأن المطابقة بين المخططات وخصائص~$R$
تنعقد للأطر، \emph{لا للنماذج}. فصدق~\Ax{T} في كل نموذج انعكاسي
لا يستلزم أن كل نموذج تصدق فيه~\Ax{T} انعكاسي.
أما على مستوى الأطر فيصح الاتجاهان: \Ax{T} \emph{صالحة} في كل
\emph{إطار} انعكاسي، وكل إطار تصلح فيه~\Ax{T} انعكاسي.

\begin{rem}
ولا يخرج تعريف الصلاحية في صنف الأطر عن التعريف السابق في أصناف
النماذج؛ فإن~$\mClass{F} \Entails !A$ تكافئ
$\mClass{C} \Entails !A$ متى كان~$\mClass{C}$ صنف جميع النماذج
المبنية على إطار من~$\mClass{F}$.

ومن ثم، إذا كانت !!a{formula} أو مخطط صالحًا في \emph{جميع}
النماذج، لزم أن يكون صالحًا في أي صنف أطر~$\mClass{F}$.
\end{rem}
