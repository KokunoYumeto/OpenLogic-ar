\olsection{الإحلال المتزامن}

إذا أُخذت !!a{formula}~$!A$، وأُبدل كل ورود لـ!!a{propositional variable}
في~$!A$ بـ!!{formula} أخرى، كان الحاصل \emph{حالة} لها. وسيكثر
استعمال حالات ال!!{formula}s في بحث الصلاحية وفي بحث
!!{derivability}؛ فلا بد من ضبط معنى الإحلال الذي تتولد به.

\begin{defn}\ollabel{def:subst-inst}
  إذا كانت~$!A$ !!{formula} موجهية لا تخرج !!{propositional
    variable}s فيها عن~$p_1$، و\dots، و$p_n$، وكانت~$!D_1$، و\dots،
  و$!D_n$ !!{formula}s موجهية أيضًا، عرّفنا
  $\SSubst{!A}{\subst{!D_1}{p_1}, \dots, \subst{!D_n}{p_n}}$ بأنه
  الحاصل من وضع كل~$!D_i$ موضع~$p_i$ في~$!A$ دفعة واحدة،
  لا بإعادة الإحلال في الصيغ التي أُدخلت. ويضبط ذلك الاستقراء على~$!A$:
  \begin{enumerate}
    \tagitem{prvFalse}{\indcase{!A}{\lfalse}{%
        $\SSubst{\indfrm}{\subst{!D_1}{p_1}, \dots,
          \subst{!D_n}{p_n}}$ هي~$\lfalse$.}}{}
    \tagitem{prvTrue}{\indcase{!A}{\ltrue}{%
        $\SSubst{\indfrm}{\subst{!D_1}{p_1}, \dots,
          \subst{!D_n}{p_n}}$ هي~$\ltrue$.}}{}
    \item[] وللاختصار في عرض الحالات الآتية، نرمز بـ~$\sigma$ إلى قائمة
      الإحلال المتزامن
      \[
        \sigma =
        \bigl(\subst{!D_1}{p_1}, \dots, \subst{!D_n}{p_n}\bigr).
      \]
    \item \indcase{!A}{q}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma} &= q
        \end{aligned}
      \]
      بشرط~$q \not\ident p_i$ لكل~$i = 1$، و\dots، و$n$.}
    \item \indcase{!A}{p_i}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma} &= !D_i.
        \end{aligned}
      \]}
    \tagitem{prvNot}{\indcase{!A}{\lnot !B}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \lnot \SSubst{!B}{\sigma}.
        \end{aligned}
      \]}}{}
    \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \bigl(\SSubst{!B}{\sigma} \land
              \SSubst{!C}{\sigma}\bigr).
        \end{aligned}
      \]}}{}
    \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \bigl(\SSubst{!B}{\sigma} \lor
              \SSubst{!C}{\sigma}\bigr).
        \end{aligned}
      \]}}{}
    \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \bigl(\SSubst{!B}{\sigma} \lif
              \SSubst{!C}{\sigma}\bigr).
        \end{aligned}
      \]}}{}
    \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \bigl(\SSubst{!B}{\sigma} \liff
              \SSubst{!C}{\sigma}\bigr).
        \end{aligned}
      \]}}{}
    \item \indcase{!A}{\Box !B}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \Box \SSubst{!B}{\sigma}.
        \end{aligned}
      \]}
    \tagitem{prvDiamond}{\indcase{!A}{\Diamond !B}{%
      \[
        \begin{aligned}
          \SSubst{\indfrm}{\sigma}
            &= \Diamond \SSubst{!B}{\sigma}.
        \end{aligned}
      \]}}{}
  \end{enumerate}
  وتسمى ال!!{formula}~$\SSubst{!A}{\subst{!D_1}{p_1}, \dots,
    \subst{!D_n}{p_n}}$ \emph{حالة إحلال} لـ$!A$.
\end{defn}

\begin{ex}
  افترض أن~$!A$ هي~$p_1 \lif \Box(p_1 \land p_2)$، وأن~$!D_1$ هي
  $\Diamond(p_2 \lif p_3)$، وأن~$!D_2$ هي~$\lnot\Box p_1$. عندئذ
  تكون~$\SSubst{!A}{\subst{!D_1}{p_1}, \subst{!D_2}{p_2}}$ هي
  % تصحيح تنسيق (0411-S1): يفصل سطران من المثال بعلامة نهاية صف كاملة.
  \begin{align*}
    \Diamond(p_2 \lif p_3) &
    \lif \Box(\Diamond(p_2 \lif p_3) \land \lnot\Box p_1)
    \intertext{بينما تكون~$\SSubst{!A}{\subst{!D_2}{p_1}, \subst{!D_1}{p_2}}$ هي}
    \lnot\Box p_1 &
    \lif \Box(\lnot\Box p_1 \land \Diamond(p_2 \lif p_3))
    \intertext{لاحظ أن الإحلال المتزامن لا يساوي، بوجه عام، الإحلال
      المتكرر. فقارن مثلًا
      $\SSubst{!A}{\subst{!D_1}{p_1}, \subst{!D_2}{p_2}}$ أعلاه بـ
      $\Subst{(\Subst{!A}{!D_1}{p_1})}{!D_2}{p_2}$، وهو:}
    & \Subst{\Diamond(p_2 \lif p_3) \lif
      \Box(\Diamond(p_2 \lif p_3) \land p_2)}{\lnot\Box p_1}{p_2},
      \text{ أي}\\
    \Diamond(\lnot\Box p_1 \lif p_3) &
    \lif \Box(\Diamond(\lnot\Box p_1
    \lif p_3) \land \lnot\Box p_1)\\
    \intertext{وقارنه أيضًا بـ$\Subst{(\Subst{!A}{!D_2}{p_2})}{!D_1}{p_1}$:}
    & \Subst{p_1 \lif \Box(p_1 \land \lnot\Box p_1)}
      {\Diamond(p_2 \lif p_3)}{p_1}, \text{ أي}\\
    \Diamond(p_2 \lif p_3) &
    \lif \Box(\Diamond(p_2
    \lif p_3) \land \lnot\Box\Diamond(p_2 \lif p_3)).
  \end{align*}
\end{ex}
