\olsection{سلامة القواعد الإضافية}

سلامة القاعدة لفئة من النماذج معناها أنه متى أمكن إرضاء فرع من
!!a{tableau} في نموذج من تلك الفئة، أمكن إرضاء الفرع الناتج من إجراء
القاعدة عليه في نموذج من الفئة نفسها.
ونفترض فيما يأتي انغلاق مجموعة بادئات الفرع تحت السوابق، كما يقع في
الـ!!{tableau}s المولدة بالقواعد؛ فورود~$\sigma.n$ يضمن ورود~$\sigma$.
% توضيح برهاني (0466-I1): تحتاج حالتا B و4r إلى كون البادئة الأم مفسرة بالفعل.

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج الانعكاسية.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{أما تطبيق T$\Box$ على
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{تمرين.}{فنتيجته زيادة !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ في الفرع. ولنفرض
    $\mSat{M}{\Gamma}[f]$، فيكون خاصةً~$\mSat{M}{\Box
      !B}[f(\sigma)]$. ومن انعكاسية~$R$ يلزم
    $Rf(\sigma)f(\sigma)$، ومنه~$\mSat{M}{!B}[f(\sigma)]$.
    فإذن~$\mModel{M}, f$ يرضيان~$\sFmla{\True}{!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{وأما تطبيق T$\Diamond$ على
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{فيزيد في الفرع !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$. فإن كان
    $\mSat{M}{\Gamma}[f]$ كان خاصةً~$\mSat/{M}{\Diamond
      !B}[f(\sigma)]$. ولانعكاسية~$R$ يكون
    $Rf(\sigma)f(\sigma)$، فيلزم~$\mSat/{M}{!B}[f(\sigma)]$.
    وبذلك يرضي~$\mModel{M}, f$ الصيغة~$\sFmla{\False}{!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-T}.
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-D} 
  \iftag{prvBox}{\Ax{D}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{D}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج التسلسلية.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{في تطبيق D$\Box$ على
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{تمرين.}{تُزاد !!{signed
      formula}~$\sFmla{\True}{\Diamond !B}[\sigma]$ في الفرع.
    فإذا كان~$\mSat{M}{\Gamma}[f]$ صدق خاصةً
    $\mSat{M}{\Box !B}[f(\sigma)]$. وبما أن~$R$ تسلسلية، نجد~$w
    \in W$ يحقق~$Rf(\sigma)w$. فيكون~$\mSat{M}{!B}[w]$،
    ويشهد هذا العالم لـ~$\mSat{M}{\Diamond !B}[f(\sigma)]$.
    ومن ثم يرضي~$\mModel{M}, f$ الصيغة
    $\sFmla{\True}{\Diamond!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{وفي تطبيق D$\Diamond$ على
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{تُزاد !!{signed
      formula}~$\sFmla{\False}{\Box !B}[\sigma]$ في الفرع.
    ولنفرض~$\mSat{M}{\Gamma}[f]$، فيلزم خاصةً
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$. ولتسلسل~$R$ يوجد~$w
    \in W$ بحيث~$Rf(\sigma)w$. وعنده~$\mSat/{M}{!B}[w]$،
    فيشهد بكذب الضرورة:~$\mSat/{M}{\Box !B}[f(\sigma)]$.
    فالنموذج وتأويله~$\mModel{M}, f$ يرضيان
    $\sFmla{\False}{\Box!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-D}.
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-B} 
  \iftag{prvBox}{\Ax{B}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{B}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج المتناظرة.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{إذا أُجريت B$\Box$ على
  $\sFmla{\True}{\Box !B}[\sigma.n] \in \Gamma$:
  \iftag{probBox}{تمرين.}{نتجت !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$. ولنفترض
    $\mSat{M}{\Gamma}[f]$، ومنه خاصةً~$\mSat{M}{\Box
      !B}[f(\sigma.n)]$. أما~$f$ فتأويل لبادئات الفرع في~$\mModel{M}$،
    ولذلك~$Rf(\sigma)f(\sigma.n)$. وأما~$R$ فمتناظرة،
    ولذلك~$Rf(\sigma.n)f(\sigma)$. ومن
    $\mSat{M}{\Box !B}[f(\sigma.n)]$ نستنتج
    $\mSat{M}{!B}[f(\sigma)]$، فيرضي~$\mModel{M}, f$
    الصيغة~$\sFmla{\True}{!B}[\sigma]$.}}{}
\tagitem{prvDiamond}{وإذا أُجريت B$\Diamond$ على
  $\sFmla{\False}{\Diamond !B}[\sigma.n] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{نتجت !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$. فإذا كان
    $\mSat{M}{\Gamma}[f]$ كان خاصةً
    $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$. ولأن~$f$ تأويل لبادئات
    الفرع في~$\mModel{M}$، ثبت~$Rf(\sigma)f(\sigma.n)$.
    ومن تناظر~$R$ يثبت~$Rf(\sigma.n)f(\sigma)$.
    فمع~$\mSat/{M}{\Diamond !B}[f(\sigma.n)]$ يلزم
    $\mSat/{M}{!B}[f(\sigma)]$. وهكذا يرضي~$\mModel{M}, f$
    الصيغة~$\sFmla{\False}{!B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-B}.
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-4} 
  \iftag{prvBox}{\Ax{4}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{4}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج المتعدية.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{أما 4$\Box$ إذا أُجريت على
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$:
  \iftag{probBox}{تمرين.}{فإنها تزيد !!{signed
      formula}~$\sFmla{\True}{\Box!B}[\sigma.n]$ في الفرع.
    نأخذ~$\mSat{M}{\Gamma}[f]$، فيلزم خاصةً
    $\mSat{M}{\Box !B}[f(\sigma)]$. وكون~$f$ تأويلًا لبادئات
    الفرع في~$\mModel{M}$، مع كون~$\sigma.n$ مستعملة، يقتضي
    $Rf(\sigma)f(\sigma.n)$. خذ الآن عالمًا~$w$ كيفما اتفق يحقق
    $Rf(\sigma.n)w$. فتعدي~$R$ يوجب~$Rf(\sigma)w$.
    ولما كان~$\mSat{M}{\Box !B}[f(\sigma)]$، لزم~$\mSat{M}{!B}[w]$.
    فثبت~$\mSat{M}{\Box !B}[f(\sigma.n)]$، وبذلك يرضي~$\mModel{M}, f$
    الصيغة~$\sFmla{\True}{\Box !B}[\sigma.n]$.}}{}
\tagitem{prvDiamond}{وأما 4$\Diamond$ إذا أُجريت على
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{فإنها تزيد !!{signed
      formula}~$\sFmla{\False}{\Diamond!B}[\sigma.n]$ في الفرع.
    ولنفرض~$\mSat{M}{\Gamma}[f]$، فيكون خاصةً
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$. وكون~$f$ تأويلًا لبادئات
    الفرع في~$\mModel{M}$، مع استعمال~$\sigma.n$، يقتضي
    $Rf(\sigma)f(\sigma.n)$. فمهما أخذنا~$w$ بحيث~$Rf(\sigma.n)w$،
    أوجب تعدي~$R$ أن~$Rf(\sigma)w$. ومع
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$ يلزم~$\mSat/{M}{!B}[w]$.
    فحصل~$\mSat/{M}{\Diamond !B}[f(\sigma.n)]$، فيرضي
    $\mModel{M}, f$ الصيغة~$\sFmla{\False}{\Diamond !B}[\sigma.n]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-4}.
\end{prob}
\tagendprob

\begin{prop}
  \ollabel{prop:soundness-4r} 
  \iftag{prvBox}{\Ax{4r}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ و}%
  \iftag{prvDiamond}{\Ax{4r}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{سليمة}{سليمتان}
  للنماذج الإقليدية.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{إذا أُجريت 4r$\Box$ على
  $\sFmla{\True}{\Box !B}[\sigma.n] \in \Gamma$:
  \iftag{probBox}{تمرين.}{كانت الزيادة !!{signed
      formula}~$\sFmla{\True}{\Box!B}[\sigma]$.
    نأخذ~$\mSat{M}{\Gamma}[f]$، فيصدق خاصةً~$\mSat{M}{\Box
      !B}[f(\sigma.n)]$. ولما كان~$f$ تأويلًا لبادئات الفرع
    في~$\mModel{M}$، علمنا~$Rf(\sigma)f(\sigma.n)$.
    خذ عالمًا~$w$ يحقق~$Rf(\sigma)w$. فإقليدية~$R$ تستلزم
    $Rf(\sigma.n)w$، ومن~$\mSat{M}{\Box !B}[f(\sigma.n)]$
    ينتج~$\mSat{M}{!B}[w]$. فثبت~$\mSat{M}{\Box !B}[f(\sigma)]$،
    أي إن~$\mModel{M}, f$ يرضيان~$\sFmla{\True}{\Box !B}[\sigma]$.}}{}
\tagitem{prvDiamond}{وإذا أُجريت 4r$\Diamond$ على
  $\sFmla{\False}{\Diamond !B}[\sigma.n] \in \Gamma$:
  \iftag{probDiamond}{تمرين.}{كانت الزيادة !!{signed
      formula}~$\sFmla{\False}{\Diamond!B}[\sigma]$.
    فإذا كان~$\mSat{M}{\Gamma}[f]$ كان خاصةً
    $\mSat/{M}{\Diamond !B}[f(\sigma.n)]$. ولأن~$f$ تأويل لبادئات
    الفرع في~$\mModel{M}$، كان~$Rf(\sigma)f(\sigma.n)$.
    وأي عالم~$w$ يحقق~$Rf(\sigma)w$، توجب إقليدية~$R$ أن
    $Rf(\sigma.n)w$. فمع~$\mSat/{M}{\Diamond !B}[f(\sigma.n)]$
    يلزم~$\mSat/{M}{!B}[w]$. ومنه~$\mSat/{M}{\Diamond !B}[f(\sigma)]$،
    فيرضي~$\mModel{M}, f$ الصيغة
    $\sFmla{\False}{\Diamond !B}[\sigma]$.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  أكمل برهان \olref[nml][tab][msn]{prop:soundness-4r}.
\end{prob}
\tagendprob
