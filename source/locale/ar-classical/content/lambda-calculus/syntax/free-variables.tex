\olsection{المتغيرات الحرة}

موضوع حساب لامبدا هو الدوال، وبالتجريد تنشأ فيه الدالة.
فالمعنى الحدسي لـ$\lambd[x][M]$ أنه دالة يعيّن~$M$ قيمتها حين يسند وسيطها إلى~$x$.
غير أن وقائع~$x$ في~$M$ ليست سواءً: فإذا اشتمل~$M$ على تجريد آخر~$\lambd[x][N]$،
تعلقت وقائع~$x$ في~$N$ بالتجريد~$\lambd[x][N]$ دون~$\lambd[x][M]$.
وعليه فإن تجريد لامبدا~$\lambd[x]$ في~$\lambd[x][M]$ \emph{يربط}
من وقائع~$x$ في~$M$ ما لم يربطه تجريد آخر؛ وتلك هي الوقائع \emph{الحرة} لـ$x$ في~$M$.

\begin{defn}[المجال]
متى وقع~$\lambd[x][M]$ في حد~$N$، سميت الواقعة المناظرة لـ$M$
\emph{مجال}~$\lambd[x]$.
\end{defn}

\begin{defn}[الواقعة الحرة والمقيدة]
  واقعة المتغير~$x$ في الحد~$M$ \emph{حرة} إن لم تدخل في مجال
  $\lambd[x]$، و\emph{مقيدة} إن دخلت فيه. وتكون واقعة المتغير~$x$
  في~$\lambd[x][M]$ مقيدة بـ$\lambd[x]$ الابتدائية إذا وفقط إذا كانت
  واقعة~$x$ في~$M$ حرة.
\end{defn}
% إصلاح صياغة (0360-N1): صيغ شرط الربط تكافؤًا صريحًا.

\begin{ex}
  انظر إلى~$\lambd[x][x y]$: فإن~$x$ و$y$ كلاهما داخل مجال~$\lambd[x]$،
  ومن ثم يربط~$x$ التجريد~$\lambd[x]$. وأما~$y$ فحر، إذ لا يدخل في مجال
  $\lambd[y]$. وفي~$\lambd[x][x x]$ يربط واقعتي~$x$ التجريد~$\lambd[x]$،
  لأنهما حرتان في~$xx$. وأما في~$((\lambd[x][xx])x)$ فآخر وقائع~$x$ حر،
  لخروجه من مجال~$\lambd[x]$. وفي
  $\lambd[x][(\lambd[x][x])x]$ مجال~$\lambd[x]$ الأولى هو
  $(\lambd[x][x])x$، ومجال~$\lambd[x]$ الثانية هو الواقعة التي قبل الأخيرة
  من~$x$. فإذا نظرنا إلى~$(\lambd[x][x])x$ وجدنا آخر وقائع~$x$ حرًا،
  والذي قبله مقيدًا. ولهذا فإن الواقعة قبل الأخيرة من~$x$ في
  $\lambd[x][(\lambd[x][x])x]$ مقيدة بـ$\lambd[x]$ الثانية، وأما الأخيرة فمقيدة بـ$\lambd[x]$ الأولى.
\end{ex}

إذا تتبعنا وقائع المتغيرات في الحد~$P$، أمكن جمع متغيراته الحرة في مجموعة نرمز إليها بـ$\FV{P}$.
والتعريف التالي يصف هذه المجموعة وصفًا استقرائيًا مباشرًا:

\begin{defn}[المتغيرات الحرة لحد] \ollabel{def:fv}
  لمجموعة \emph{المتغيرات الحرة} في الحد التعريف الاستقرائي الآتي:
  \begin{enumerate}
    \item $\FV{x} = \{x\}$ \ollabel{def:fv1}
    \item $\FV{\lambd[x][N]} = \FV{N} \setminus \{x\}$    \ollabel{def:fv2}
    \item $\FV{PQ} = \FV{P} \cup \FV{Q}$ \ollabel{def:fv3}
  \end{enumerate}
\end{defn}

\begin{prob}
  \begin{enumerate}
  \item بيّن مجال~$\lambd[g]$ ومجال كل واحدة من واقعتي~$\lambd[x]$ في الحد
    $\lambd[g][(\lambd[x][g (x x)]) \lambd[x][g (x x)]]$.
  \item انظر في وقائع المتغيرات في
    $\lambd[g][(\lambd[x][g (x x)]) \lambd[x][g (x x)]]$: هل هي جميعًا مقيدة؟ وعيّن التجريد الذي يربط كل واقعة منها.
    \item احسب~$\FV{\lambd[x][(\lambd[y][(\lambd[z][x y]) z]) y]}$.
  \end{enumerate}
\end{prob}

\begin{explain}
المتغير الحر كالإحالة إلى خارج الحد، أي إلى \emph{البيئة}.
فالحد المشتمل على متغيرات حرة لم يكتمل تحديده؛ إذ يتوقف سلوكه على ما تعيّنه البيئة.
فالحد~$\lambd[x][f x]$، مثلًا، يأخذ وسيطًا~$x$ ويعيد حاصل تطبيق~$f$ عليه،
وفيه المتغير~$f$ حر. ولذلك تتوقف قيمة هذا الحد على بيئته،
ولا سيما على قيمة~$f$ فيها.

فإذا جرّدنا هذا الحد حصلنا على~$\lambd[f][\lambd[x][f
    x]]$. وعندئذ ينقطع اعتماده على متغير البيئة~$f$؛
إذ يصير دالًا على دالة ذات وسيطين تطبق أولهما على ثانيهما.
فلو تغيرت قيمة~$f$ في البيئة، لم يتغير سلوك الحد؛
لأنه يستعمل ما يدفع إليه وسيطًا، ولا يستعمل قيمة~$f$ في البيئة.
\end{explain}

\begin{defn}[الحد المغلق، المؤلف]
  الحد الخالي من المتغيرات الحرة يسمى \emph{حدًا مغلقًا}،
  ويسمى أيضًا \emph{مؤلفًا}.
\end{defn}

\begin{lem}\ollabel{lem:fv}
  \begin{enumerate}
    \item \ollabel{lem:fv-abs} بشرط~$y \neq x$، تتكافأ العبارتان~$y \in
      \FV{\lambd[x][N]}$ و$y \in \FV{N}$.
    \item \ollabel{lem:fv-app} شرط~$y \in \FV{PQ}$، وهو كاف أيضًا، أن يكون
      $y \in \FV{P}$ أو~$y \in \FV{Q}$.
    \end{enumerate}
\end{lem}

\begin{proof}
  يترك إثباتها تمرينًا.
\end{proof}

\begin{prob}
  أقم برهان~\olref[lam][syn][fv]{lem:fv}.
\end{prob}
