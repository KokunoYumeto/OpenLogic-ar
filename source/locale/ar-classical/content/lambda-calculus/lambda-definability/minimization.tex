\olsection{التصغير}

الدوال العودية العامة هي الحاصلة من الدوال الأساس~$\Zero$ و$\Succ$
و$\Proj{n}{i}$، بإعمال التركيب والعودية البدائية والتصغير النظامي.
وقد ثبت أمر الدوال الأساس والإغلاق تحت العمليتين الأوليين؛ فلكي نثبت
أن كل دالة عودية عامة !!{lambda definable}، بقي أن نثبت أن التصغير
النظامي لدالة~!!a{lambda definable} ينتج دالة !!{lambda definable}.

\begin{lem}
  \ollabel{lem:min} إذا كانت~$f(x_1, \dots, x_k, y)$ نظامية
  و!!{lambda definable}، فإن الدالة~$h$ المعرفة بـ
  \[
  h(x_1, \dots, x_k) = \umin{y}{f(x_1,\dots,x_k, y) = 0}
  \]
  تكون هي أيضًا !!{lambda definable}.
\end{lem}

\begin{proof}
  افترض أن حد~$\lambda$~$F$ يعرّف بلامبدا الدالة النظامية
  $f(\vec x, y)$. ونبني، لكي !!{lambda define}~$h$، حد بحث نستعمل
  معه مركّب نقطة ثابتة:
  \begin{align*}
    \fn{Search} & \ident \lambd[g][\lambd[f\,\vec{x}\,y][
        \fn{IsZero} (f\, \vec{x}\, y)\, y\, (g\, f\, \vec{x}\, (\fn{Succ}\, y))]]\\
    H & \ident \lambd[\vec x][(Y \, \fn{Search}) F\, \vec{x}\, \num{0}],
  \end{align*}
  حيث يجوز أن يكون~$Y$ أي مركّب للنقطة الثابتة. ومعنى~$\fn{Search}$
  أنها عملية بحث عودية تبدأ من~$y$: تختبر~$f\,\vec x\,y$؛ فإن كان
  صفرًا ردّت~$y$، وإن لم يكن نادت نفسها عند~$\fn{Succ}\,y$.
  فإذا بدأنا بالصفر، أعاد
  $(Y \, \fn{Search}) F \num{n_1}\dots\num{n_k}\,\num{0}$
  أصغر~$m$ يحقق~$f(n_1, \dots, n_k, m) = 0$.
  
  وتفصيل ذلك أن
  \begin{align*}
    (Y \, \fn{Search}) F \num{n_1}
    \dots\num{n_k}\, \num{m} & \red \num{m}
    \intertext{إذا كان~$f(n_1, \dots, n_k, m) = 0$، أو}
    & \red (Y \, \fn{Search}) F\, \num{n_1} \dots\num{n_k}\, \num{m+1}
    \intertext{في غير ذلك. وبما أن~$f$ نظامية، يوجد~$y$ يحقق
      $f(n_1, \dots, n_k, y) = 0$، ومن ثم}
    (Y \, \fn{Search}) F \num{n_1} \dots\num{n_k}\,\num{0}
    & \red \num{h(n_1, \dots, n_k)}.
    \end{align*}
  فالنظامية تضمن انتهاء هذا البحث، والحد الأخير يمثل الدالة المطلوبة.
\end{proof}


\begin{prop}
  كل دالة عودية عامة !!{lambda definable}.
\end{prop}

\begin{proof}
 فالدوال الأساس !!{lambda definable} بـ\olref[prf]{lem:basic}،
 وصنف الدوال !!{lambda definable} مغلق تحت التركيب والعودية البدائية والتصغير النظامي
 بـ\olref[prf]{lem:comp} و\olref[prf]{lem:prim} و\olref{lem:min}.
 وهذه هي عمليات بناء الدوال العودية العامة، فيثبت المطلوب.
\end{proof}
