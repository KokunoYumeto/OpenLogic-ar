\olsection{اختزال~$\beta$}

\begin{lem}\ollabel{lem:one-par}
  من~$M \bredone M'$ يلزم~$M \bredpar M'$.
\end{lem}
\begin{proof}
  نستقرئ اشتقاق~$M \bredone M'$ من أصغر علاقة متوافقة، فتقع أربع حالات:
  \begin{enumerate}
  \item في تقلص الجذر يكون~$M=(\lambd[x][N])Q$ و
    $M'=\Subst{N}{Q}{x}$. ولما كان~$N\bredpar N$ و$Q\bredpar Q$
    بموجب \olref[pb]{thm:refl}، أعطت
    \olref[pb]{defn:bredpar}\olref[pb]{defn:bredpar4} المطلوب.
  \item وإن وقع التقلص تحت تجريد، كان~$M=\lambd[x][P]$ و
    $M'=\lambd[x][P']$ مع~$P\bredone P'$. يفيد فرض الاستقراء
    $P\bredpar P'$، ثم تفيد \olref[pb]{defn:bredpar2} المطلوب.
  \item وإن وقع في الطرف الأيسر من تطبيق~$PQ$، جمعنا فرض
    الاستقراء لذلك الطرف إلى~$Q\bredpar Q$ بالانعكاس، ثم طبقنا
    \olref[pb]{defn:bredpar3}.
  \item وإن وقع في الطرف الأيمن، جمعنا~$P\bredpar P$ بالانعكاس
    إلى فرض الاستقراء للطرف الأيمن، ثم طبقنا القاعدة نفسها.
  \end{enumerate}
  % تصحيح مصدر (0370-I1): شمل الاستقراء حالات التوافق الثلاث فضلًا على تقلص الجذر.
\end{proof}

\begin{lem}\ollabel{lem:par-red}
  من~$M \bredpar M'$ يلزم~$M \bred M'$.
\end{lem}

\begin{proof}
  نستقرئ~!!{derivation} العبارة~$M \bredpar M'$، ونفصل بحسب آخر قاعدة:
  \begin{enumerate}
  \item إن كانت~\olref[pb]{defn:bredpar1}، لم يكن~$M$ و$M'$ إلا~$x$،
    ولدينا~$x \bred x$.
  \item وإن كانت~\olref[pb]{defn:bredpar2}، كان~$M$ هو~$\lambd[x][N]$
    و$M'$ هو~$\lambd[x][N']$، لبعض~$x$ و$N$ و$N'$، مع~$N \bredpar N'$.
    ومن فرض الاستقراء~$N \bred N'$؛ فتنتج~$\lambd[x][N] \bred \lambd[x][N']$
    بإجراء تقلصات~$\bredone$ نفسها التي أعطت~$N \bred N'$.
  \item وإن كانت~\olref[pb]{defn:bredpar3}، كان~$M$ هو~$PQ$ و$M'$ هو~$P'Q'$،
    لبعض~$P$ و$Q$ و$P'$ و$Q'$، مع~$P \bredpar P'$ و$Q \bredpar Q'$.
    ويفيد فرض الاستقراء~$P \bred P'$ و$Q \bred Q'$.
    فنحصل على~$PQ \bred P'Q'$ بإجراء الاختزال~$P \bred P'$
    ثم الاختزال~$Q \bred Q'$.
  \item وإن كانت~\olref[pb]{defn:bredpar4}، كان~$M$ هو~$(\lambd[x][N])Q$
    و$M'$ هو~$\Subst{N'}{Q'}{x}$، لبعض~$x$ و$N$ و$N'$ و$Q$ و$Q'$،
    مع~$N \bredpar N'$ و$Q \bredpar Q'$.
    ومن فرض الاستقراء~$Q \bred Q'$ و$N \bred N'$.
    فبلوغ~$(\lambd[x][N])Q \bred \Subst{N'}{Q'}{x}$
    يكون بإجراء~$N \bred N'$، ثم~$Q \bred Q'$،
    ثم تقليص~$(\lambd[x][N'])Q'$ إلى~$\Subst{N'}{Q'}{x}$.
  \end{enumerate}
\end{proof}

\begin{lem}\ollabel{lem:str}
  العلاقة~$\bred$ أصغر علاقة متعدية تشتمل على~$\bredpar$.
\end{lem}

\begin{proof}
  لتكن~$\xred$ أصغر العلاقات المتعدية التي تشتمل على~$\bredpar$.

  أما~$\bred \subseteq \xred$، فلنفرض~$M \bred M'$، أي
  $M \ident M_1 \bredone \dots \bredone M_k \ident M'$.
  وتفيد~\olref{lem:one-par} أن~$M \ident M_1 \bredpar \dots \bredpar M_k \ident M'$.
  ولما كانت~$\xred$ تشتمل على~$\bredpar$ وتتصف بالتعدي، نتج~$M \xred M'$.

  وأما~$\xred \subseteq \bred$، فلنفرض~$M \xred M'$، أي
  $M \ident M_1 \bredpar \dots \bredpar M_k \ident M'$.
  وتفيد~\olref{lem:par-red} أن~$M \ident M_1 \bred \dots \bred M_k \ident M'$.
  ومن تعدي~$\bred$ ينتج~$M \bred M'$.
\end{proof}

\begin{thm}\ollabel{thm:cr}
  العلاقة~$\bred$ ذات خاصية تشيرش--روسر.
\end{thm}

\begin{proof}
  تجمع~\olref[dap]{thm:str} و\olref[pb]{thm:cr} و\olref{lem:str}
  فتنتج المبرهنة مباشرة.
\end{proof}
