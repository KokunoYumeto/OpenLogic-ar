\olsection{قواعد القضايا لمنطقات مختارة}

العمدة في تعيين قواعد رابط ما في حساب تتابعيات ذي جوانب~$n$ هي
دالة الصدق المميزة لذلك الرابط وحدها. فإن اشترك منطقان أو أكثر في
هذه الدالة، اشتركا في قواعد التتابعيات ذات الجوانب~$n$ الخاصة
بالرابط، وإن اختلفا من وجوه أخرى.

\subsection{قواعد~$\lnot$}

أما~$\lnot$ في منطقي~\L ukasiewicz وكلايني وفي صورهما المختلفة،
فقواعده كما يأتي.

\begin{defish}
  \begin{center}
\AxiomC{$ \Gamma \nSequent \Pi \nSequent \Delta, !A $}
\RightLabel{\iR{\lnot}{\False}}
\UnaryInfC{$ \lnot !A, \Gamma \nSequent \Pi \nSequent \Delta$}
\DisplayProof
\\[2ex]
\AxiomC{$ \Gamma \nSequent !A, \Pi \nSequent \Delta $}
\RightLabel{\iR{\lnot}{\Undef}}
\UnaryInfC{$\Gamma \nSequent \lnot !A, \Pi \nSequent \Delta$}
\DisplayProof
\\[2ex]
\AxiomC{$!A, \Gamma \nSequent \Pi \nSequent \Delta$}
\RightLabel{\iR{\lnot}{\True}}
\UnaryInfC{$\Gamma \nSequent \Pi \nSequent \Delta,  \lnot !A$}
\DisplayProof
  \end{center}
\end{defish}

تنطبق القواعد الآتية لـ$\lnot$ على منطق غودل.

\begin{defish}
\AxiomC{$ \Gamma \nSequent !A, \Pi \nSequent \Delta, !A $}
\RightLabel{\iR{\lnot}{\False}[\LogGod]}
\UnaryInfC{$ \lnot !A, \Gamma \nSequent \Pi \nSequent \Delta$}
\DisplayProof
\hfill
\AxiomC{$!A, \Gamma \nSequent \Pi \nSequent \Delta$}
\RightLabel{\iR{\lnot}{\True}[\LogGod]}
\UnaryInfC{$\Gamma \nSequent \Pi \nSequent \Delta,  \lnot !A$}
\DisplayProof
\hfill
\end{defish}

وأما~$\lnot !A$ فلا سبيل إلى أن تكون قيمته~$\Undef$ في منطق غودل؛
ولذلك خلا الموضع الأوسط من قاعدة خاصة به.

\subsection{قواعد~$\land$}

هذه قواعد~$\land$ في منطق~\L ukasiewicz ومنطق كلايني القوي ومنطق
غودل.

\begin{defish}
\[
  \AxiomC{$\begin{gathered}
    !A, !B, \Gamma \\[-0.25ex]
    \nSequent \Pi \\[-0.25ex]
    \nSequent \Delta
  \end{gathered}$}
  \RightLabel{$\iR{\land}{\False}$}
  \UnaryInfC{$\begin{gathered}
    !A \land !B, \Gamma \\[-0.25ex]
    \nSequent \Pi \\[-0.25ex]
    \nSequent \Delta
  \end{gathered}$}
  \DisplayProof
\]
\[
  \AxiomC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent !A, \Pi \\[-0.25ex]
    \nSequent !A, \Delta
  \end{gathered}$}
  \AxiomC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent !B, \Pi \\[-0.25ex]
    \nSequent !B, \Delta
  \end{gathered}$}
  \AxiomC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent !A, !B, \Pi \\[-0.25ex]
    \nSequent \Delta
  \end{gathered}$}
  \RightLabel{$\iR\land\Undef$}
  \TrinaryInfC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent !A \land !B, \Pi \\[-0.25ex]
    \nSequent \Delta
  \end{gathered}$}
  \DisplayProof
\]
\[
  \AxiomC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent \Pi \\[-0.25ex]
    \nSequent \Delta, !A
  \end{gathered}$}
  \AxiomC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent \Pi \\[-0.25ex]
    \nSequent \Delta, !B
  \end{gathered}$}
  \RightLabel{$\iR\land\True$}
  \BinaryInfC{$\begin{gathered}
    \Gamma \\[-0.25ex]
    \nSequent \Pi \\[-0.25ex]
    \nSequent \Delta, !A \land !B
  \end{gathered}$}
  \DisplayProof
\]
\end{defish}
