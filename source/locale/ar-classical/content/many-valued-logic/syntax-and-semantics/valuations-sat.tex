\olsection{\usetoken{P}{valuation} والإشباع}

\begin{defn}[!!^{valuation}s]
إذا كانت~$V$ مجموعة قيم صدق، فـ\emph{!!{valuation}} اللغة
$\Lang{L}$ في~$V$ هو دالة~$\pAssign{v}$ تسند~!!a{element} من~$V$
إلى كل واحد من~!!{propositional variable}s اللغة؛ أي
$\pAssign{v} \colon \PVar \to V$.
\end{defn}

\begin{defn}\ollabel{defn:pValue}
  إذا كان~!!a{valuation}~$\pAssign{v}$ في مجموعة قيم الصدق~$V$
  لمنطق متعدد القيم~$\Log L$، عرفت دالة التقييم
  $\pValue{v} \colon \Frm[L] \to V$ بالاستقراء الآتي:
  \begin{enumerate}
    \item $\pValue{v}(\Obj p_n) = \pAssign{v}(\Obj p_n)$؛
    \item إذا كان~$\star$ رابطًا ذا~$0$ من المواضع، كان
    $\pValue{v}(\star) = \tf{\star}[\Log L]$؛
    \item إذا كان~$\star$ رابطًا ذا~$n$ من المواضع، و$n>0$، كان
    \[
      \pValue{v}(\star(!A_1, \dots, !A_n)) = \tf{\star}[\Log L]
      (\pValue{v}(!A_1), \dots, \pValue{v}(!A_n)).
    \]
  \end{enumerate}
\end{defn}

\begin{defn}[الإشباع]
\ollabel{defn:satisfaction} تكون~!!{formula}~$!A$ \emph{مشبعة}
  في~!!a{valuation}~$\pAssign{v}$، ونكتب
  $\pSat{v}{!A}[\Log L]$، إذا وفقط إذا كان
  $\pValue{v}(!A)[\Log L] \in V^+$؛ حيث~$V^+$ مجموعة قيم الصدق
  المميزة للمنطق~$\Log L$.
  
  ونرمز بـ~$\pSat/{v}{!A}[\Log L]$ إلى قولنا «ليس
  $\pSat{v}{!A}[\Log L]$.» وإذا كانت~$\Gamma$ مجموعة
  !!{formula}s، كان~$\pSat{v}{\Gamma}[\Log L]$ إذا وفقط إذا تحقق
  $\pSat{v}{!A}[\Log L]$ لكل~$!A \in \Gamma$.
\end{defn}
