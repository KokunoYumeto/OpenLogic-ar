\olsection{المفاهيم الدلالية}

نضبط الآن المعاني الدلالية التي سيجري عليها البحث:

\begin{defn}
\begin{enumerate}
\item !!^a{formula}~$!A$ \emph{قابلة للإرضاء} متى وُجد تقييم~$\pAssign{v}$
  بحيث $\pSat{v}{!A}$؛ وهي \emph{غير قابلة للإرضاء} متى لم يوجد تقييم~$\pAssign{v}$
  بحيث $\pSat{v}{!A}$.
\item !!^a{formula}~$!A$ \emph{تحصيل حاصل} متى كان
  $\pSat{v}{!A}$ لجميع !!{valuation}s~$\pAssign{v}$.
\item !!^a{formula}~$!A$ \emph{عارضة} متى أمكن إرضاؤها ولم تكن تحصيل حاصل.
\item لتكن $\Gamma$ مجموعة من !!{formula}s. قولنا $\Gamma \Entails !A$،
  وقراءته «$\Gamma$ تستلزم~$!A$»، معناه أن $\pSat{v}{!A}$ لكل
  !!{valuation}~$\pAssign{v}$ يكون فيه $\pSat{v}{\Gamma}$، وهذا شرط لازم وكاف.
\item ولتكن $\Gamma$ مجموعة من !!{formula}s. إن وُجد لـ~$\Gamma$
  !!a{valuation}~$\pAssign{v}$ بحيث
  $\pSat{v}{\Gamma}$، قلنا إنها \emph{قابلة للإرضاء}؛ وإلا كانت $\Gamma$
  \emph{غير قابلة للإرضاء}.
\end{enumerate}
\end{defn}

\begin{prob}
 انظر في كل واحدة من !!{formula}s الأربع الآتية، وبيّن هل هي
 (أ)~قابلةٌ للإرضاء، و(ب)~تحصيلُ حاصل، و(ج)~عارضةٌ.
\begin{enumerate}
  \item \( ( \Obj p_0 \lif ( \lnot \Obj p_1 \lif \lnot \Obj p_0 ) ) \).
  \item \( ( ( \Obj p_0 \land \lnot \Obj p_1 ) \lif ( \lnot \Obj p_0 \land \Obj p_2 )) \liff ( ( \Obj p_2 \lif \Obj p_0 ) \lif ( \Obj p_0 \lif \Obj p_1 )) \).
  \item \( ( \Obj p_0 \liff \Obj p_1 ) \lif ( \Obj p_2 \liff \lnot \Obj p_1 ) \).
  \item \( (( \Obj p_0 \liff ( \lnot \Obj p_1 \land \Obj p_2 )) \lor ( \Obj p_2 \lif ( \Obj p_0 \liff \Obj p_1 ))) \).
\end{enumerate}
\end{prob}

\begin{prop}
\ollabel{prop:semanticalfacts}
\begin{enumerate}
\item تكون $!A$ تحصيلَ حاصل إذا وفقط إذا كان
  $\emptyset \Entails !A$؛
\item إذا كان $\Gamma \Entails !A$ و$\Gamma \Entails !A \lif !B$، فإن
  $\Gamma \Entails !B$؛
\item إذا كانت $\Gamma$ قابلةً للإرضاء، فإن كل مجموعة جزئية منتهية
  من~$\Gamma$ قابلةٌ للإرضاء أيضًا؛
\item \ollabel{def:monotonicity} الرتابة: إذا كانت $\Gamma \subseteq \Delta$
  وكان $\Gamma \Entails !A$، فإن $\Delta \Entails !A$ أيضًا؛
\item \ollabel{def:Cut} التعدي: إذا كان $\Gamma \Entails !A$ و
  $\Delta \cup \{ !A\} \Entails !B$، فإن $\Gamma \cup \Delta \Entails
  !B$.
\end{enumerate}
\end{prop}

\begin{proof}
تمرين.
\end{proof}

أثبت \olref[pl][syn][sem]{prop:semanticalfacts}
\end{prob}

\begin{prop}\ollabel{prop:entails-unsat}
  $\Gamma \Entails !A$ إذا وفقط إذا كانت $\Gamma \cup \{\lnot !A\}$
  غير قابلة للإرضاء.
\end{prop}

\begin{proof}
تمرين.
\end{proof}

\begin{thm}[مبرهنة الاستنتاج الدلالي]
  \ollabel{thm:sem-deduction} $\Gamma \Entails !A \lif !B$ إذا وفقط
  إذا كان $\Gamma \cup \{!A\} \Entails !B$.
\end{thm}
