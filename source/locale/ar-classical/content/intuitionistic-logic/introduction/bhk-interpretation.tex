\olsection{تأويل براور--هايتنغ--كولموغوروف}

\begin{editorial}
  عرض براهين الصدق العام الحدسي بتأويل BHK في هذه المادة عسير المتابعة؛
  ويحتاج إلى مزيد من البيان.
\end{editorial}

للروابط الحدسية تأويل بنائي غير صوري يسمى عادة تأويل
براور--هايتنغ--كولموغوروف. وأصله مفهوم «البناء»، وهو قريب من البرهان
البنائي. غير أنا نعدل هنا عن لفظ «البرهان» في سياق BHK، لئلا يلتبس
بــ!!a{derivation} داخل نظام !!{derivation} صوري. ومن هذا الأصل يبين
التأويل معاني الروابط الحدسية.

\begin{enumerate}
\item نأخذ معنى البناء للعبارة الذرية معلومًا ابتداءً.
\item بناء~$!A_1 \land !A_2$ زوج~$\tuple{M_1, M_2}$، يكون~$M_1$
  بناءً لــ$!A_1$ ويكون~$M_2$ بناءً لــ$!A_2$.
\item بناء~$!A_1 \lor !A_2$ زوج~$\tuple{s, M}$: إما أن يكون~$s=1$
  ويكون~$M$ بناءً لــ$!A_1$، وإما أن يكون~$s=2$ ويكون~$M$ بناءً لــ$!A_2$.
\item بناء~$!A \lif !B$ دالة تنقل كل بناء لــ$!A$ إلى بناء لــ$!B$.
\item لا بناء لــ$\lfalse$، أي للمحال.
\item تحد~$\lnot !A$ بما يرادف~$!A \lif \lfalse$؛ فبناء~$\lnot !A$
  دالة تنقل بناءً لــ$!A$ إلى بناء لــ$\lfalse$.
\end{enumerate}

\begin{ex}
مثاله~$\lnot \lfalse$. بناؤها دالة تُخرج بناءً لــ$\lfalse$ متى أُدخل
إليها بناء لــ$\lfalse$. ودالة الهوية~$\Id{}$ من هذا القبيل: فإذا دخلها
بناء~$M$ لــ$\lfalse$، كان~$\Id{}(M) = M$ بناءً لــ$\lfalse$.
\end{ex}

فالمعنى العام لــ$\lnot !A$ أن «بناء~$!A$ محال».

\begin{ex}
نريد بناء~$!A \lif \lnot \lnot !A$ لكل قضية~$!A$، أي بناء
$!A \lif ((!A \lif \lfalse) \lif \lfalse)$. فلا بد من دالة~$f$ تأخذ
بناءً~$M$ لــ$!A$ وترد بناءً~$f(M)$ لــ
$(!A \lif \lfalse) \lif \lfalse$. ولكي تنشئ~$f$ بناء
$(!A \lif \lfalse) \lif \lfalse$ نحد دالة~$g$
تخرج بناءً لــ$\lfalse$ إذا دخلها بناء~$h$ لــ$!A \lif \lfalse$.
وتعريف~$g$ أن تطبق المدخل~$h$ على بناء~$!A$، أعني~$M$ الذي سبق أخذه.
فمخرج~$h$، وهو~$h(M)$، بناء لــ$\lfalse$؛ ولذلك يكون
$f(M)(h) = h(M)$ بناءً لــ$\lfalse$ متى كان~$M$ بناءً لــ$!A$.
\end{ex}

\begin{ex}
ولنبن~$\lnot(!A \land \lnot !A)$، أي
$(!A \land (!A \lif \lfalse)) \lif \lfalse$. المطلوب دالة~$f$ تخرج
بناءً لــ$\lfalse$ إذا دخلها بناء~$M$ لــ
$!A \land (!A \lif \lfalse)$. وبناء العطف~$!B_1 \land !B_2$ زوج
$\tuple{N_1, N_2}$، يكون~$N_1$ فيه بناءً لــ$!B_1$ و$N_2$ بناءً
لــ$!B_2$. فلنحد دالتين~$p_1$ و$p_2$ تستخرجان من بناء
$!B_1 \land !B_2$ بناءي~$!B_1$ و$!B_2$ على الترتيب:
\begin{align*}
  p_1(\tuple{N_1, N_2}) &= N_1\\
  p_2(\tuple{N_1, N_2}) &= N_2
\end{align*}
وعمل~$f$ أن تطبق أولًا~$p_1$ على مدخلها~$M$، فيخرج بناء لــ$!A$، ثم
تطبق~$p_2$ على~$M$، فيخرج بناء لــ$!A \lif \lfalse$. وهذا الأخير دالة
$p_2(M)$ تخرج بناء لــ$\lfalse$ إذا دخلها بناء لــ$!A$. فإذا طبقنا
$p_2(M)$ على~$p_1(M)$ خرج بناء لــ$\lfalse$؛ ومن ثم نضع
$f(M) = p_2(M)(p_1(M))$.
\end{ex}

\begin{ex}
ولنبن~$((!A \land !B) \lif !C) \lif (!A \lif
(!B \lif !C))$. فهذا البناء دالة~$f$ تنقل بناء~$g$ لــ
$(!A \land !B) \lif !C$ إلى بناء لــ$(!A \lif (!B \lif !C))$.
والبناء~$g$ نفسه دالة من بناءات~$!A \land !B$ إلى بناءات~$!C$؛ وأما
$f(g)$ فدالة~$h_g$ من بناءات~$!A$ إلى دوال تنقل بناءات~$!B$ إلى
بناءات~$!C$.

وقد تشعب الوصف؛ فلنرتبه. نطلب دالة معينة~$h_g$ تكون مخرج~$f$ عند
المدخل~$g$. يدخل~$h_g$ بناء~$M$ لــ$!A$، ويجب أن يكون مخرج~$h_g(M)$
دالة~$k_M$ تنقل بناءات~$N$ لــ$!B$ إلى بناءات~$!C$. فنضع
$k_{g,M}(N) = g(\tuple{M, N})$. وقد علم أن~$\tuple{M, N}$ بناء لــ
$!A \land !B$؛ فتكون~$k_{g,M}$ بناء لــ$!B \lif !C$، إذ تنقل بناءات~$N$
لــ$!B$ إلى بناءات~$!C$. ثم نضع~$h_g(M) = k_{g,M}$؛ فهذه دالة تنقل
بناءات~$M$ لــ$!A$ إلى بناءات~$k_{g,M}$ لــ$!B \lif !C$. وأخيرًا نضع
$f(g) = h_g$؛ فتحصل الدالة التي تنقل بناءات~$g$ لــ
$(!A \land !B) \lif !C$ إلى بناءات~$!A \lif (!B \lif !C)$. وقد طال
الطريق!{}
\end{ex}

تسمى~$!A \lor \lnot !A$ قانون الوسط المستبعد. وقد نبنيها لقضايا معينة
$!A$، مثل~$\lfalse \lor \lnot \lfalse$، ولا نبنيها على العموم؛ لأن
الفصل الحدسي يقتضي بناء أحد شقيه، مع أن قضايا، كحدسية غولدباخ، لم يتهيأ
الآن إثباتها ولا دحضها. ومع ذلك لا يدحض قانون الوسط المستبعد؛ بل تصدق
$\lnot \lnot (!A \lor \lnot !A)$.

\begin{ex}
لبرهنة $\lnot \lnot (!A \lor \lnot !A)$ نحتاج إلى دالة~$f$ تحوّل
بناءً لـ$\lnot(!A \lor \lnot !A)$، أي لـ
$(!A \lor (!A \lif \lfalse)) \lif \lfalse$، إلى بناء لـ~$\lfalse$.
وبعبارة أخرى، نحتاج إلى دالة~$f$ يكون $f(g)$ فيها بناء لـ~$\lfalse$
إذا كان $g$ بناءً لـ$\lnot(!A \lor \lnot !A)$.

افترض أن $g$ بناء لـ$\lnot(!A \lor \lnot !A)$؛ أي دالة تحوّل بناء
$!A \lor \lnot !A$ إلى بناء لـ~$\lfalse$. وبناء
$!A \lor \lnot !A$ زوج $\tuple{s, M}$ يكون فيه إما $s = 1$ و$M$
بناءً لـ$!A$، وإما $s = 2$ و$M$ بناءً لـ$\lnot !A$. ولتكن $h_1$
الدالة التي تربط بناء~$M_1$ لـ$!A$ ببناء لـ$!A \lor \lnot !A$:
فهي تربط $M_1$ بـ$\tuple{1, M_1}$. ولتكن $h_2$ الدالة التي تربط
بناء~$M_2$ لـ$\lnot !A$ ببناء لـ$!A \lor \lnot !A$: فهي تربط
$M_2$ بـ$\tuple{2, M_2}$.

لتكن $k$ هي $\comp{h_1}{g}$: إنها دالة تعيد، إذا أُعطيت بناءً لـ~$!A$،
بناءً لـ~$\lfalse$؛ أي إنها بناء لـ$!A \lif \lfalse$ أو لـ$\lnot !A$.
والآن لتكن $l$ هي $\comp{h_2}{g}$. إنها دالة تعطي، إذا أُعطيت بناءً
لـ$\lnot !A$، بناءً لـ~$\lfalse$. ولأن $k$ بناء لـ$\lnot !A$، فإن
$l(k)$ بناء لـ~$\lfalse$.

وما فعلناه في مجموعه هو وصف الطريقة التي نحول بها بناء~$g$ لـ
$\lnot(!A \lor \lnot !A)$ إلى بناء لـ~$\lfalse$؛ أي إن الدالة~$f$
التي تربط بناء~$g$ لـ$\lnot(!A \lor \lnot !A)$ ببناء~$l(k)$ لـ
$\lfalse$ هي بناء لـ$\lnot\lnot(!A \lor \lnot !A)$.
\end{ex}

فاستعمال تأويل BHK لإظهار الصدق العام الحدسي لل!!{formula}s لا يلبث أن
يطول ويشتبه. وللمنطق الحدسي، لحسن الحظ، أنظمة !!{derivation} أيسر،
وتأويلات دلالية أدق.
