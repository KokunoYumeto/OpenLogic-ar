\olsection{مبرهن التمام}

\begin{thm}\ollabel{thm:completeness}
  من~$\Gamma \Entails !A$ يلزم~$\Gamma \Proves !A$.
\end{thm}

\begin{proof}
  نسلك طريق النقيض المقابل. ليكن~$\Gamma \Proves/ !A$. تعطي
  \olref[lin]{lem:lindenbaum} مجموعة أولية
  $\Gamma^* \supseteq \Gamma$ تبقى معها~$\Gamma^* \Proves/ !A$. ولنأخذ
  النموذج القانوني~$\mModel{M(\Gamma^*)}$ للمجموعة~$\Gamma^*$ وفق
  \olref[mod]{defn:canonical-model}. فإن كل~$!B \in \Gamma$ يحقق
  $\Gamma^* \Proves !B$. ومن التعريف
  $\Gamma^*(\emptyseq)=\Gamma^*$. فتعطي لمّة الصدق
  (\olref[tru]{lem:truth})
  $\mSat{M(\Gamma^*)}{!B}[\emptyseq]$ لكل~$!B \in \Gamma$، مع
  $\mSat/{M(\Gamma^*)}{!A}[\emptyseq]$. وهذا شاهد على
  $\Gamma \Entails/ !A$.
\end{proof}

\begin{prob}
  أقم الدليل على أن اقتصار~$!A$ على !!{propositional variable}s و
  $\lor$ و$\land$ يقتضي~$\Entails/ !A$. ثم استخرج من ذلك امتناع تعريف~$\lif$
  في المنطق الحدسي بواسطة~$\lor$ و$\land$.
\end{prob}

\begin{prob}
  برهن، مستعينًا بمبرهن التمام، أن~$\Proves !A \lor !B$ لا تكون إلا مع
  $\Proves !A$ أو~$\Proves !B$. (تلميح: خذ
  $\mSat/{M_1}{!A}$ و$\mSat/{M_2}{!B}$، وأنشئ منهما نموذجًا
  $\mModel{M}$ يحقق~$\mSat/{M}{!A \lor !B}$.)
\end{prob}

\begin{prob}
  أقم الدليل على أن~$\mModel{M}$، إذا كان نموذجًا علائقيًا ذا ترتيب خطي،
  حقق~$\mSat{M}{(!A \lif !B)\lor(!B \lif !A)}$.
\end{prob}
