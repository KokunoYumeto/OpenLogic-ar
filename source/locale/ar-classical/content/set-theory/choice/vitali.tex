\olsection{ملحق: مفارقة فيتالي}

لا يستقيم لنا حكم راسخ في قبول بناء باناخ--تارسكي حتى ننظر في برهانه.
غير أن بسطه يفتقر إلى قدر من الجبر يجاوز ما يتسع له هذا الموضع. ونستطيع
مع ذلك أن نسوق إشارات وجيزة تنير بعض سبيل البرهان،\footnote{لمعالجة
أوفى بكثير، انظر~\cite{Weston2003} أو~\cite{Wagon2016}.} وذلك بالنظر
في النتيجة الآتية.

\begin{thm}[مفارقة فيتالي (في $\ZFC$)]\ollabel{vitaliparadox}
يمكن تفكيك أية دائرة قطعًا معدودة العدد، ثم تعاد هذه القطع، بالدوران
والإزاحة، فتؤلف دائرتين مماثلتين للدائرة الأولى.
\end{thm}

وبرهان مفارقة فيتالي أيسر كثيرًا من برهان باناخ--تارسكي. ونسبناها إلى
فيتالي لأنها ناشئة عن بنائه، في~\citeyear{Vitali1905}، مجموعة غير
قابلة للقياس. وأما جهة نظرية المجموعات في البرهانين فمتقاربة جدًا.
والفارق الأعظم أن تفكيك باناخ--تارسكي متناه، وتفكيك فيتالي غير متناه
لكنه معدود. وبعبارة~\citet{Weston2003}، فإن مفارقة فيتالي «ليست
قطعًا بمثل إدهاش مفارقة باناخ--تارسكي، لكنها تبين أن المفارقات
الهندسية قد تقع حتى في أوضاع `بسيطة'.»

ومفارقة فيتالي في شكل ذي بعدين، وهو الدائرة؛ فلنعمل على
المستوى~$\Real^2$. ولتكن~$\rotationsgroup$ مجموعة دورانات النقاط في
اتجاه عقارب الساعة، حول الأصل، بزوايا معاملاتُها نسبية:
$2\pi q$، حيث~$q \in \Rat \cap [0,1)$. ولـ~$\rotationsgroup$ الخواص
الجبرية الآتية، وسيتضح معناها من البرهان إن لم يتضح من العبارة.

\begin{lem}\ollabel{rotationsgroupabelian}
تكوّن~$\rotationsgroup$ {group} أبيلية تحت تركيب الدوال.
\end{lem}

\begin{proof}
لنكتب~$0_{\rotationsgroup}$ للدوران بزاوية~$0$. فهذا محايد في
$\rotationsgroup$، لأن
$\comp{0_{\rotationsgroup}}{\rho} = \comp{\rho}{0_{\rotationsgroup}} =
\rho$ لكل~$\rho \in \rotationsgroup$.

وهي مغلقة تحت التركيب؛ فإذا كان~$\rho$ و$\sigma$ دورانين بزاويتين
$2\pi q$ و$2\pi r$، كان تركيبهما دورانًا بزاوية
$2\pi((q+r) \bmod 1)$، وهذا عدد نسبي يقع في~$[0,1)$.

ولكل عنصر معكوس. فإذا كان~$\rho \in \rotationsgroup$ دورانه بزاوية
$2\pi q$، كان معكوسه الدوران بزاوية~$0$ إن كان~$q=0$، وبزاوية
$2\pi(1-q)$ إن كان~$q>0$؛ وفي الحالين
$\rho^{-1} \in \rotationsgroup$ و
$\rho \circ \rho^{-1} = 0_\rotationsgroup$.

والتركيب تجميعي:
$\comp{\rho}{(\comp{\sigma}{\tau})} =
\comp{(\comp{\rho}{\sigma})}{\tau}$ لكل
$\rho, \sigma, \tau \in \rotationsgroup$.

وهو تبادلي أيضًا:~$\comp{\rho}{\sigma} =
\comp{\sigma}{\rho}$ لكل~$\rho, \sigma \in \rotationsgroup$.
\end{proof}

بل يمكن شطر~$\rotationsgroup$ نصفين، ثم استرداد المجموعة كلها من أي
واحد منهما.

\begin{lem}\ollabel{disjointgroup}
يوجد تقسيم لـ$\rotationsgroup$ إلى مجموعتين منفصلتين،
$\rotationsgroup_{1}$ و$\rotationsgroup_{2}$، وكل منهما أساس لاستعادة
$\rotationsgroup$.
\end{lem}

\begin{proof}
اجعل~$\rotationsgroup_{1}$ مؤلفة من الدورانات ذات الزوايا~$2\pi q$ حيث
$q \in \Rat \cap [0,\nicefrac{1}{2})$، ولتكن
$\rotationsgroup_{2} = \rotationsgroup \setminus
\rotationsgroup_1$، أي الدورانات التي يكون فيها
$q \in \Rat \cap [\nicefrac{1}{2},1)$. ويدل الجبر الأولي على أن
تضعيف~$q$ بترديد~$1$ يرسل كل نصف منهما تقابليًا إلى
$\Rat \cap [0,1)$. فمن ثم، لكل~$i=1,2$:
\[
	\Setabs{\comp{\rho}{\rho}}{\rho \in \rotationsgroup_i} =
	\rotationsgroup.
\]
\end{proof}

وبهذه الحقيقة عن الزمر نثبت~\olref{vitaliparadox}. لتكن
$\onesphere$ دائرة الوحدة، أي النقاط التي بعدها من أصل المستوى وحدة
بعينها، وهي
$\Setabs{\tuple{r,s} \in \Real^2}{\sqrt{r^2+s^2}=1}$. ونقسم
$\onesphere$ باعتبار العلاقة الآتية عليها:
\[
	r \sim s \emph{ إذا وفقط إذا كان }(\exists \rho \in \rotationsgroup)\rho(r) = s.
\]
فترتبط نقطتان من~$\onesphere$ إذا وفقط إذا أمكن الانتقال من إحداهما
إلى الأخرى بدوران ذي معامل نسبي حول الأصل. وكما ينبغي أن يكون:

\begin{lem}
$\sim$ علاقة تكافؤ.
\end{lem}

\begin{proof}
ظاهر من~\olref{rotationsgroupabelian}.
\end{proof}

والآن نستعمل الاختيار فنحصل على مجموعة~$C$ فيها عنصر واحد بالضبط من
كل صنف تكافؤ لـ~$\onesphere$ بحسب~$\sim$. أي نأخذ دالة اختيار~$f$
على مجموعة أصناف التكافؤ،\footnote{لأن
$\rotationsgroup$ !!{enumerable}، تكون كل !!{element} من~$E$
!!{enumerable}. ولأن~$\onesphere$ !!{nonenumerable}، ينتج من
\olref{vitalicover} و
\olref[card-arithmetic][simp]{kappaunionkappasize} أن~$E$
!!{nonenumerable}. ومن ثم فهذا استخدام للاختيار \emph{غير} المعدود.}
\[
	E = \Setabs{\equivrep{r}{\sim}}{r \in \onesphere},
\]
ولنضع~$C = \ran{f}$. ولكل دوران~$\rho \in \rotationsgroup$ تتألف
$\funimage{\rho}{C}$ من النقاط التي يرسل إليها~$\rho$ كل نقطة في~$C$.
وتبين اللمتان
الآتيتان أن هذه المجموعات تغطي الدائرة كلها من غير تداخل.

\begin{lem}\ollabel{vitalicover}
$\onesphere = \bigcup_{\rho \in \rotationsgroup} \funimage{\rho}{C}$.
\end{lem}

\begin{proof}
خذ~$s \in \onesphere$؛ يوجد~$r \in C$ بحيث
$r \in \equivrep{s}{\sim}$، أي~$r \sim s$، أي
$\rho(r) = s$ لبعض~$\rho \in \rotationsgroup$.
\end{proof}

\begin{lem}\ollabel{vitalinooverlap}
إذا كان~$\rho_1 \neq \rho_2$، فإن
$\funimage{\rho_1}{C} \cap \funimage{\rho_2}{C} = \emptyset$.
\end{lem}

\begin{proof}
افترض~$s \in \funimage{\rho_{1}}{C} \cap
\funimage{\rho_{2}}{C}$. فيكون
$s = \rho_{1}(r_{1}) = \rho_{2}(r_{2})$ لبعض
$r_{1}, r_{2} \in C$. وعليه
$\rho^{-1}_2(\rho_1(r_1)) = r_2$، و
$\comp{\rho_1}{\rho^{-1}_2} \in \rotationsgroup$، ومن ثم
$r_{1} \sim r_{2}$. فيلزم~$r_1 = r_2$، لأن~$C$ لا تختار من كل صنف
تكافؤ بحسب~$\sim$ إلا عنصرًا واحدًا. ولذلك
$s = \rho_1(r_1) = \rho_2(r_1)$، ومنه~$\rho_1 = \rho_2$.
\end{proof}

ولنطبق الآن ما تقدم من حقائق الجبر على الدائرة.

\begin{lem}\ollabel{pseudobanachtarski}
يوجد تقسيم لـ$\onesphere$ إلى مجموعتين منفصلتين~$D_{1}$ و$D_{2}$،
بحيث يمكن تقسيم~$D_{1}$ إلى عدد معدود من المجموعات التي يمكن تدويرها
لتكوين نسخة من~$\onesphere$ (وكذلك~$D_{2}$).
\end{lem}

\begin{proof}
باستعمال~$\rotationsgroup_{1}$ و$\rotationsgroup_{2}$ من
\olref{disjointgroup}، ضع:
\begin{align*}
	D_{1} &= \bigcup_{\rho \in \rotationsgroup_1} \funimage{\rho}{C} &
	D_{2} &= \bigcup_{\rho \in \rotationsgroup_2} \funimage{\rho}{C}
\end{align*}
فهذا تقسيم لـ~$\onesphere$ بحسب~\olref{vitalicover}، و$D_1$ و$D_2$
منفصلتان بحسب~\olref{vitalinooverlap}. ومن نفس البناء ينقسم~$D_1$ إلى
مجموعات معدودة، هي~$\funimage{\rho}{C}$ لكل
$\rho \in \rotationsgroup_1$. ويمكن تدويرها حتى تؤلف نسخة من
$\onesphere$، لأن
$\onesphere = \bigcup_{\rho \in
\rotationsgroup}\funimage{\rho}{C} = \bigcup_{\rho \in
\rotationsgroup_1}\funimage{(\comp{\rho}{\rho})}{C}$ بموجب
\olref{disjointgroup} و\olref{vitalicover}. والحجة نفسها جارية على
$D_2$. \end{proof}\noindent
وتلزم مفارقة فيتالي من فورها. فنحن نولد نسختين من~$\onesphere$ من
$\onesphere$ نفسها: نقسمها قطعًا معدودة العدد، وهي مختلف مجموعات
$\funimage{\rho}{C}$، ثم نحركها حركات صلبة؛ فندور كل قطعة من~$D_1$،
ونزيح كل قطعة من~$D_2$ أولًا ثم ندورها.

وخلاصة مسلك البرهان أننا بدأنا بخصائص جبرية لزمرة الدورانات في
المستوى، وبها قسمنا~$\onesphere$ أصناف تكافؤ، ثم جاء وجه «المفارقة»
من استعمال الاختيار لانتقاء عنصر من كل صنف.

وهذا بعينه مسلك برهان باناخ--تارسكي؛ وإنما تزيد خصائص الجبر اللازمة له
تعقيدًا عظيمًا على التي احتجنا إليها في مفارقة فيتالي، وليس لتلك
الخصائص الجبرية اتصال بالاختيار. ولنوجزها.

نبدأ في برهان باناخ--تارسكي بنظير~\olref{disjointgroup}: كل
\emph{زمرة حرة} تنقسم أربع قطع يمكن، في التصور، «تحريكها» حتى نسترد
نسختين من الزمرة كلها.\footnote{ترجع حقيقة إمكان استخدام
\emph{أربع} قطع إلى~\cite{Robinson1947}. ولبرهان حديث، انظر
\citet[Theorem 5.2]{Wagon2016}. ونتبع~\citet[p.~3]{Weston2003} في
وصف ذلك بأنه «تحريك» قطع الزمرة.} ثم نبين أن دورانين مخصوصين حول
أصل~$\Real^3$ يولدان زمرة حرة من الدورانات~$F$.\footnote{انظر
\citet[Theorem 2.1]{Wagon2016}.} ولم نستعمل الاختيار إلى هنا.

ولنرمز إلى سطح كرة الوحدة في~$\Real^3$ بـ
\[
  S^2 = \Setabs{x\in\Real^3}{\lVert x\rVert=1}.
\]

لكن فعل~$F$ على سطح الكرة ليس حرًا؛ فلكل دوران غير محايد نقطتان
ثابتتان على محوره. لذلك نحذف أولًا مجموعة الاستثناءات المعدودة
\[
	D = \Setabs{x \in S^2}{(\exists \rho \in F \setminus \{1_F\})\rho(x)=x}.
\]
وعلى~$S^2 \setminus D$ يصير فعل~$F$ حرًا. ثم نقسم السطح
المنقوص إلى مدارات~$F$، ونستعمل الاختيار لانتقاء نقطة واحدة من كل
مدار. وننقل تقسيم~$F$ إلى السطح المنقوص كما في
\olref{pseudobanachtarski}، فنحصل على تفكيك مفارقي له إلى قطع متناهية
العدد.

ويبقى رد~$D$. والزمرة~$F$ معدودة لأنها متولدة بمولدين؛ ولكل دوران
غير محايد نقطتا تثبيت فقط على~$S^2$، فتكون~$D$ معدودة. اختر محورًا لا
تقع نقطتا تقاطعه مع~$S^2$ في~$D$. ولكل~$d,d'\in D$ ولكل~$k>0$، لا
يوجد إلا عدد منته من الزوايا~$\theta$ التي يحقق دوران ذلك المحور بها
$\sigma_\theta^k(d)=d'$. فاتحاد هذه الزوايا معدود؛ فنختار زاوية خارجه،
ولتكن~$\sigma$ الدوران بها. عندئذ تكون المجموعات
$\funimage{\sigma^n}{D}$، حيث~$n < \omega$، منفصلة اثنتين اثنتين.
وضع~$D^* = \bigcup_{n < \omega}\funimage{\sigma^n}{D}$. فعندئذ
$\funimage{\sigma}{D^*} = D^* \setminus D$. فيمتص تدوير~$D^*$
بـ~$\sigma$، مع إبقاء سائر السطح ثابتًا، فيعرّف تقابلًا من~$S^2$ إلى
$S^2 \setminus D$؛ فهو امتصاص لـ$D$ بتفكيك من قطعتين. ثم ننقل
التفكيك المفارقي للسطح المنقوص بهذا الامتصاص، ونرد~$D$، فننال نسختين
من سطح الكرة. وهذا يثبت~\citep{Hausdorff1914}:
% تصحيح مصدر عربي (0600-N1، 0600-N2): فُصل سطح R^3 عن دائرة R^2، وأُثبت وجود دوران الامتصاص.

\begin{thm}[مفارقة هاوسدورف (في $\ZFC$)]
يمكن تفكيك سطح أية كرة قطعًا متناهية العدد، ثم تعاد هذه القطع، بالدوران
والإزاحة، فتؤلف نسختين منفصلتين من تلك الكرة.
\end{thm}

ويحتاج تمام مبرهنة باناخ--تارسكي، حيث يدخل باطن الكرة لا سطحها وحده،
إلى حيلتين جبريتين أخريين. وليستا في الحقيقة إلا تمامًا للصنعة
الجبرية. ولذلك يقول ويستون:
\begin{quote}	
  [\ldots] النتيجة الخاصة بالزمر الحرة هي \emph{الخطوة الأساسية} في
  برهان مفارقة باناخ--تارسكي. ومن وجهة النظر هذه، ليست مفارقة
  باناخ--تارسكي عبارة عن~$\Real^3$ بقدر ما هي عبارة عن تعقيد زمرة
  [الإزاحات والدورانات في~$\Real^3$]. \cite[p.~16]{Weston2003}
\end{quote}
فإمكان التفكيك المتناهي في باناخ--تارسكي، أو التفكيك غير المتناهي
المعدود في فيتالي، مرده إلى خصائص معينة في نظرية الزمر بحسب العمل في
بعدين أو ثلاثة.

وهذه الخاتمة تفسد شيئًا من نكتة آخر~\olref[banach]{sec}. فإن اسم
«باناخ--تارسكي» مرسوم في بعدين، فينبغي تفكيكه قطعًا غير متناهية معدودة
حتى يعاد ترتيبها إلى «باناخ--تارسكي باناخ--تارسكي». ولا تصلح النكتة
إلا إذا كتبت في ثلاثة أبعاد؛ فلنترك ذلك تمرينًا للقارئ.

وبقي تعليق. قلنا في~\olref[banach]{sec} إن «قطع» الكرة لا تكون قابلة
للقياس، بل تكون بعثرات غير متناهية لا سبيل إلى تصويرها. وكذلك ما
اخترناه للحصول على~\olref{pseudobanachtarski}. وهذا محتاج إلى بيان.

فنوجز الخلفية مرة أخرى، وليس هذا إلا موجزًا، ومن أراد التفصيل فليرجع
إلى مدخل في نظرية القياس. قياس مجموعة~$X$ إسناد قيمة~$\mu(E) \in \Real$
إلى كل~$E$ في «جبر~$\sigma$» على~$X$. والذي يعنينا أن~$\mu$ تحقق
الإضافة المعدودة: قياس اتحاد معدود لمجموعات منفصلة هو مجموع قياسات
آحاده، أي
$\mu(\bigcup_{n < \omega} X_n) = \sum_{n < \omega}\mu(X_n)$ متى
كانت~$X_n$ منفصلة. والمجموعة غير القابلة للقياس هي التي لا يمكن إسناد
قياس مناسب إليها. فباستعمال~$\rotationsgroup$ المتقدمة نحصل على:

\begin{cor}[فيتالي]
ليكن~$\mu$ قياسًا بحيث~$\mu(\onesphere) = 1$، وبحيث
$\mu(X) = \mu(Y)$ إذا كانت~$X$ و$Y$ متطابقتين هندسيًا. عندئذ تكون
$\funimage{\rho}{C}$ غير قابلة للقياس لكل
$\rho \in \rotationsgroup$.
\end{cor}

\begin{proof}
افترض نقيض ذلك. فليكن
$\mu(\funimage{\sigma}{C}) = r$ لبعض
$\sigma \in \rotationsgroup$ وبعض~$r \in \Real$. ولكل
$\rho \in \rotationsgroup$ تكون~$\funimage{\rho}{C}$ و
$\funimage{\sigma}{C}$ متطابقتين هندسيًا، ومن ثم
$\mu(\funimage{\rho}{C}) = r$ لكل~$\rho \in \rotationsgroup$.
وبحسب~\olref{vitalicover} و\olref{vitalinooverlap}، تكون
$\onesphere =
\bigcup_{\rho \in \rotationsgroup}\funimage{\rho}{C}$ اتحادًا
معدودًا لمجموعات منفصلة اثنتين اثنتين. فتقضي الإضافة المعدودة بأن
$\mu(\onesphere) = 1$ هو مجموع قياسات مختلف
$\funimage{\rho}{C}$، أي:
\[
	1 = \mu(\onesphere) = \sum_{\rho \in \rotationsgroup}\mu(\funimage{\rho}{C}) = \sum_{\rho \in \rotationsgroup}r
\]
لكن~$r \geq 0$ لأن~$\mu$ قياس؛ فإذا كان~$r = 0$، كان
$\sum_{\rho \in \rotationsgroup}r = 0$، وإذا كان~$r > 0$، كان
$\sum_{\rho \in \rotationsgroup}r = \infty$؛ وهو تناقض.
\end{proof}
