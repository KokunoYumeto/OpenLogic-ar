\olsection{التأسيس}

لم يتم لنا المقصود \emph{كلّه}، وإنما قاربناه؛ وذلك أن~$\ZFminus$
لا تضمن أن تقع \emph{كل} مجموعة في بعض~$V_\alpha$، أي أن تكون كل
مجموعة قد نشأت في مرحلة من المراحل.

وقد يقال إن في وسعنا، من الوجه الرياضي المباشر، ألا \emph{نبالي}
بالمجموعات الخارجة عن الهرم؛ فإن وُجدت أغفلناها. غير أن تصورنا للمجموعة
إنما بُني على أن كل مجموعة تتكون في مرحلة ما (انظر~\stageshier{} في
\olref[z][story]{sec}). فوجب أن ننفي إمكان خروج مجموعة عن الهرم، ولا
سبيل إلى ذلك إلا بإضافة مسلّمة تجعل لكل مجموعة موضعًا فيه.

ولما كانت مجموعات~$V_\alpha$ هي مراحلنا، كان أقرب ما يلوح أن نتخذ
القول الآتي مسلّمة:

\begin{defish}
\emph{الانتظام.} $\forall A \exists \alpha\, A \subseteq V_\alpha$
\end{defish}

وهذا مسلك مستقيم لا غبار عليه. إلا أننا سنعدل عنه، لعلة يأتي بيانها في
القسم التالي، إلى المسلّمة الآتية:

\begin{axiom}[التأسيس]
$(\forall A \neq \emptyset)(\exists B \in A)A \cap B = \emptyset$.
\end{axiom}

ومع شيء من العمل يتبين، داخل~$\ZFminus$، أن التأسيس يوجب الانتظام.
ونبدأ بالتعريف الآتي:
\begin{defn}
لكل مجموعة~$A$، لتكن:
\begin{align*}
	\text{cl}_0(A) &= A,\\
	\text{cl}_{n+1}(A) &= \bigcup \text{cl}_n(A),\\
	\text{trcl}(A) &= \bigcup_{n < \omega} \text{cl}_{n}(A).
\end{align*}
ونسمي~$\text{trcl}(A)$ \emph{الإغلاق المتعدي} لـ$A$.
\end{defn}
\noindent
ووجه هذه التسمية ظاهر من القضية الآتية:
\begin{prop}\ollabel{subsetoftrcl}
$A \subseteq \trcl{A}$ و$\trcl{A}$ مجموعة متعدية.
\end{prop}

\begin{proof}
أما~$A = \text{cl}_0(A) \subseteq \text{trcl}(A)$ فظاهر. وإذا كان
$x \in b \in \trcl{A}$، كان~$b \in \text{cl}_n(A)$ لبعض~$n$؛ ومن ثم
$x \in \text{cl}_{n+1}(A) \subseteq \text{trcl}(A)$.
\end{proof}

\begin{lem}\ollabel{lem:TransitiveWellFounded}
إذا كانت~$A$ مجموعة متعدية، وجد~$\alpha$ يحقق
$A \subseteq V_\alpha$.
\end{lem}

\begin{proof}
نستحضر تعريف «$\supstrict(X)$» من
\olref[ordinals][opps]{defsupstrict}، ثم نعرّف مجموعتين:
\begin{align*}
	D  &= \Setabs{x \in A}{\forall \delta\ x \nsubseteq V_\delta}\\
	\alpha &= \supstrict\Setabs{\delta}{(\exists x \in A)
	(x \subseteq V_\delta \land (\forall \gamma \in \delta)x \nsubseteq V_\gamma)}
\end{align*}
ولنفرض أولًا~$D = \emptyset$. إذا كان~$x \in A$، وجد~$\delta$ بحيث
$x \subseteq V_\delta$؛ وبما أن الأعداد الترتيبية مرتبة ترتيبًا جيدًا،
جاز اختياره بحيث
$(\forall \gamma \in \delta)x \nsubseteq V_\gamma$. فيكون
$\delta \in \alpha$، ويلزم~$x \in V_\alpha$ من
\olref[spine][Valphabasic]{Valphabasicprops}. وإذن
$A \subseteq V_{\alpha}$، وهو المطلوب.

فلم يبق إلا إثبات~$D = \emptyset$. نفرض خلافه طلبًا للمحال. يقضي
التأسيس بوجود~$B \in D \subseteq A$ بحيث~$D \cap B = \emptyset$.
فإذا كان~$x \in B$ كان~$x \in A$ لتعدي~$A$؛ ولأن~$x \notin D$،
ينتج~$\exists \delta\ x \subseteq V_\delta$. ضع الآن
\[
	\beta = \supstrict\Setabs{\delta}{(\exists x \in B)(x \subseteq V_\delta \land (\forall \gamma < \delta)x \nsubseteq V_\gamma)}.
\]
فنستدل كما تقدم على~$B \subseteq V_\beta$، وهذا يناقض كون~$B \in D$.
\end{proof}

\begin{thm}\ollabel{zfentailsregularity}
يصدق الانتظام.
\end{thm}

\begin{proof}
خذ~$A$ كيفما اتفق. من~\olref{subsetoftrcl} نعلم أن
$A \subseteq \trcl{A}$ وأن الأخيرة متعدية؛ فتوجد~$\alpha$ بحيث
$A \subseteq \trcl{A} \subseteq V_\alpha$، بمقتضى
\olref{lem:TransitiveWellFounded}.
\end{proof}

فقد ثبت أن~$\ZFminus$ تقضي بـ
$\emph{التأسيس}\Rightarrow\emph{الانتظام}$. وسيأتي في
\olref[spine][rank]{zfminusregularityfoundation} أن~$\ZFminus$ تقضي
كذلك بـ$\emph{الانتظام}\Rightarrow\emph{التأسيس}$. فالتأسيس
والانتظام، مع افتراض~$\ZFminus$، متكافئان. وعلى هذا، بالنظر إلى
$\ZFminus$، يجوز تسويغ التأسيس بتكافئه مع الانتظام؛ أما الانتظام
نفسه فيسوّغه تصور~\stageshier{} مباشرة.
