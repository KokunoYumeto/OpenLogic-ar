\olsection{الأشجار}

\emph{الشجرة} صنف مخصوص من الترتيب الجزئي، عظيم النفع في فروع
المنطق كلها. فالأشجار المنتهية ترد في أولياته: إذ تفهم
!!{formula}s بتحليلها إلى شجر تركيبية، وتتخذ !!{derivation}s في
كثير من أنساق !!{derivation} صورة أشجار منتهية.
%
وأما غير المنتهية فترد منذ براهين مبرهنات الاكتمال للمنطق القضي
ومنطق الرتبة الأولى، ثم يستمر استعمالها في المنطق الرياضي كله.

وبين الشجرة في نظرية المجموعات والشجرة في نظرية الرسوم صلة وثيقة.
وهذه صورة شجرة منتهية:

أسفل العقد~$r$، وهي الجذر. وكل عقدة غير $r$ لها عقدة أصل واحدة
تحتها مباشرة. والعلاقة بين عقدة~$x$ وعقدة~$y$ هي أنه إذا أمكن بلوغ $y$ من~$x$ بسلوك
الحواف صعودًا، قلنا إن $x$ \emph{سلف} لـ~$y$.

وعلاقة السلف ترتيب جزئي صارم؛ ومنها نهتدي إلى التعريف بالمجموعات.
ولا بد له من اصطلاحين: \emph{العنصر الأصغر} لمجموعة~$A$ مرتبة
جزئيًا بـ~$\le$ هو !!a{element} $x \in A$ يحقق لكل $y \in A$
أن~$x \le y$. ونقول إن المجموعة \emph{مرتبة ترتيبًا جيدًا}
بـ~$\le$ إذا كان لكل جزئية غير خالية منها عنصر أصغر.

\begin{defn}[الشجرة]
\emph{الشجرة} زوج $T = \tuple{A, \le}$، فيه $A$ مجموعة و$\le$
ترتيب جزئي على~$A$، وله أصغر وحيد $r \in A$ يسمى \emph{الجذر}،
وتكون لكل $x \in A$ المجموعة $\Setabs{y}{y \le x}$ مرتبة ترتيبًا
جيدًا بـ~$\le$.
\end{defn}

\begin{defn}[اللواحق]
لتكن $T = \tuple{A, \le}$ شجرة. متى كان $x,y \in A$ و$x < y$،
وامتنع وجود $z \in A$ يحقق $x < z < y$، سمي $y$ \emph{لاحقًا} لـ~$x$.
\end{defn}

ولواحق $x \in A$ تسمى \emph{أبناءه} أيضًا. فإن كان $y$ لاحقًا
لـ~$x$، كان $x$ \emph{سابق}~$y$، ويسمى كذلك \emph{عقدته الأصل}.

\begin{prop}
في الشجرة $\tuple{A,\le}$ لا يكون لـ$x \in A$، إن لم يكن الجذر،
أكثر من سابق واحد.
\end{prop}

\begin{proof}
افرض $y_1 < x$ و$y_2 < x$ و$y_1 \neq y_2$. فيكون
$\{y_1, y_2\} \subseteq \Setabs{z}{z<x}$. ولأن
$\Setabs{z}{z<x}$ جيدة الترتيب بـ~$\le$، فلا بد لجزئيتها
$\{y_1, y_2\}$ من أصغر، وليس إلا $y_1$ أو~$y_2$.
إذن $y_1 \le y_2$ أو $y_2 \le y_1$؛ ومع $y_1 \neq y_2$
يصير ذلك $y_1 < y_2$ أو $y_2 < y_1$. وقد فرضنا
$y_1 < x$ و$y_2 < x$، فينتج $y_1 < y_2 < x$ أو $y_2 < y_1 < x$.
ففي كلتا الحالتين يمتنع أن يكون $y_1$ و$y_2$ كلاهما سابقًا لـ~$x$.
\end{proof}

\begin{defn}
الشجرة $T = \tuple{A, \le}$ \emph{غير منتهية} إن كانت $A$ غير
منتهية، و\emph{منتهية} فيما عدا ذلك. وإن كانت $T$ بحيث لكل $x \in A$ فيها
لواحق عددها منته، قلنا إن $T$ \emph{منتهية التفرع}.
\end{defn}

\begin{defn}[الفروع]
في شجرة $T = \tuple{A, \le}$، \emph{الفرع} من~$T$ سلسلة قصوى في~$T$:
أي $B \subseteq A$، لكل $x, y \in B$ فيها $x \le y$ أو $y \le x$،
ولكل $z \in A \setminus B$ يوجد $u \in B$ لا يصدق معه $z \le u$
ولا $u \le z$.
%
ورمزها $[T]$، أعني مجموعة جميع فروع $T$.
\end{defn}

\begin{ex}
المثال المشهور لمنتهي التفرع هو \emph{الشجرة الثنائية غير المنتهية}:
متتاليات منتهية من $0$ و~$1$، يرمز إليها بـ$\{0,1\}^*$ أو~$\Bin^*$،
وترتب بالامتداد $\sqsubseteq$، نحو $101 \sqsubseteq 101101$.
ويمكن أن يلحق بآخر كل سلسلة ثنائية $0$ أو $1$؛ لذلك عناصر الشجرة
غير متناهية، ولكل عنصر~$s$ لاحقان لا غير هما $s0$ و~$s1$.
وجذرها المتتالية الخالية $\emptyseq$.
\end{ex}

\begin{ex}
وأعم قليلًا من ذلك متتاليات الطبيعيّات المنتهية~$\Nat^*$ مع
الامتداد~$\sqsubseteq$؛ فهي شجرة أيضًا، ولكن تفرعها غير منته:
لكل $s \in \Nat^*$ لواحق~$sn$ لا تتناهى، واحد لكل $n \in \Nat$.
وكل مجموعة غير خالية $A \subseteq \Nat^*$ مغلقة تحت~$\sqsubseteq$ تعد
\emph{شجرة جزئية} من~$\Nat^*$؛ أي إن $A$ تحقق: $s \in A$ و$s' \sqsubseteq s$
يقتضيان $s' \in A$. وكل شجرة منتهية يمكن تمثيلها بشجرة جزئية
منتهية من~$\Nat^*$.
\end{ex}

\textbf{تنبيه تصحيحي على الأصل.} قُيّدت الشجرة الجزئية هنا بغير الخالية؛
إذ إن المجموعة الخالية مغلقة تحت السوابق، ولكن لا جذر لها، فلا تحقق تعريف
الشجرة المتقدم.

\begin{prop}[قضية \foreignlanguage{english}{K\H{o}nig} المساعدة]
كل شجرة $T = \tuple{A,\le}$ غير منتهية ومنتهية التفرع يكون لـ$T$ فرع غير منته.
\end{prop}

وتختص من قضية \foreignlanguage{english}{K\H{o}nig} المساعدة
صيغة كثيرة الاستعمال في نظرية قابلية الحساب، تسمى
\emph{الصيغة الضعيفة من قضية \foreignlanguage{english}{K\H{o}nig} المساعدة}:
كل شجرة جزئية غير منتهية من $\{0,1\}^*$ لها فرع غير منته.
