\olsection{الاتحادات والتقاطعات}

\begin{explain}
سبق في \olref[sfr][set][bas]{sec} تعريف المجموعات بالتجريد، أي بالصورة
$\Setabs{x}{\phi(x)}$. والخاصية~$\phi$ المستعملة فيه قد تذكر مجموعات
سبق تعريفها. فإذا كانت $A$ و~$B$ مجموعتين، جمعت
$\Setabs{x}{x \in A \lor x \in B}$ كل ما هو من !!{element}s $A$ أو~$B$؛
فهي تضم !!{element}s $A$ و~$B$ معًا. وفي \olref{fig:union} تصوير ذلك:
فالناحية المظللة تمثل !!{element}s المجموعتين $A$ و~$B$ مجتمعة.

\begin{figure}
  \olasset{assets/diagrams/union.tikz}
  \caption{الاتحاد $A \cup B$ يجمع !!{element}s $A$ و~$B$ في مجموعة واحدة.}
  \ollabel{fig:union}
\end{figure}

ولكثرة نفع هذه العملية وورودها---أعني جمع المجموعات---نضع لها اسمًا
اصطلاحيًا ورمزًا.
\end{explain}

\begin{defn}[الاتحاد]
\emph{اتحاد} $A$ و$B$، ورمزه $A \cup B$، مجموعة الأشياء التي هي من !!{element}s $A$ أو $B$
أو كلتيهما:
\[
A \cup B = \Setabs{x}{x \in A \lor x \in B}
\]
\end{defn}

\begin{ex}
لا أثر لتكرار !!{element}s؛ فإذا اشتركت مجموعتان في !!a{element}
لم يحتج اتحادهما إلى ذكر ذلك !!{element} إلا مرة واحدة، كما في
$\{ a, b, c\} \cup \{ a, 0, 1\} = \{a, b, c, 0, 1\}$.

واتحاد المجموعة بإحدى مجموعاتها الجزئية هو المجموعة الأصلية نفسها: $\{a,
b, c \} \cup \{a \} = \{a, b, c\}$.

وكذلك ضم الخالية إلى مجموعة يبقي المجموعة على حالها: $\{a,
b, c \} \cup \emptyset = \{a, b, c \}$.
\end{ex}

\begin{prob}
برهن على أن $A \subseteq B$ يستلزم $A \cup B = B$.
\end{prob}

\begin{explain}
وأما العملية «المزدوجة» للاتحاد فتجمع !!{element}s التي هي من
!!{element}s~$A$ ومن !!{element}s~$B$ معًا. واسم هذه العملية
\emph{التقاطع}، وصورتها في \olref{fig:intersection}.
\begin{figure}
  \olasset{assets/diagrams/intersection.tikz}
  \caption{التقاطع $A \cap B$ يجمع ما بين المجموعتين من !!{element}s مشتركة.}
  \ollabel{fig:intersection}
\end{figure}
\end{explain}

\begin{defn}[التقاطع]
\emph{تقاطع} $A$ و$B$، ورمزه $A \cap B$، مجموعة الأشياء التي هي من
!!{element}s $A$ و~$B$ معًا:
\[
A \cap B = \Setabs{x}{x \in A \land x \in B}
\]
فإن خلا التقاطع سميت المجموعتان \emph{متباينتين}؛ أي لا !!{element}s
مشتركة بينهما.
\end{defn}

\begin{ex}
متى انتفت !!{element}s المشتركة بين مجموعتين خلا تقاطعهما، نحو:
$\{ a, b, c\} \cap \{ 0, 1\} = \emptyset$.

ومتى وجدت !!{element}s مشتركة بين مجموعتين، كان تقاطعهما المجموعة
التي تضمها جميعًا ولا تضم غيرها، نحو:
$\{a, b, c \} \cap \{a, b, d \} = \{a, b\}$.

وتقاطع المجموعة بإحدى مجموعاتها الجزئية هو تلك المجموعة الجزئية:
$\{a, b, c\} \cap \{a, b\} = \{a, b\}$.

وأما التقاطع مع الخالية فخالٍ أبدًا: $\{a, b, c \}
\cap \emptyset = \emptyset$.
\end{ex}

\begin{prob}
أقم برهانًا دقيقًا على أن $A \subseteq B$ يستلزم $A \cap B = A$.
\end{prob}

\begin{explain}
وليس الاتحاد والتقاطع مقصورين على مجموعتين. ولتعميمهما وجه موجز:
نجمع أولًا المجموعات المراد اتحادها أو تقاطعها في مجموعة واحدة.
فالاتحاد يجمع كل ما ينتمي إلى !!{element} واحد على الأقل من تلك
المجموعة، وأما التقاطع فيجمع كل ما ينتمي إلى كل !!{element} منها.
\end{explain}

\begin{defn}
لتكن $A$ مجموعة من المجموعات. فاتحادها $\bigcup A$ يجمع
!!{element}s جميع !!{element}s~$A$:
\begin{align*}
\bigcup A & = \Setabs{x}{x \text{ ينتمي إلى !!a{element} من } A},
\text{ أي}\\
& = \Setabs{x}{\text{يوجد } B \in A
  \text{ بحيث } x \in B}
\end{align*}
\end{defn}

\begin{defn}
لتكن $A$ مجموعة غير خالية من المجموعات. فتقاطعها $\bigcap A$ يجمع ما تشترك
فيه جميع عناصر~$A$ من الأشياء:
\begin{align*}
\bigcap A & = \Setabs{x}{x \text{ ينتمي إلى كل !!{element} من } A},
\text{ أي}\\
 & = \Setabs{x}{\text{لكل } B \in A, x \in B}
\end{align*}
\end{defn}

\textbf{تنبيه تصحيحي على الأصل.} اشترطنا ألا تخلو عائلة التقاطع؛ لأن
التعريف المعروض لا يعيّن تقاطع العائلة الخالية من غير تعيين مجموعة كلية
تحصر عناصره. ولا يلحق هذا الشرط اتحاد العائلة.

\begin{ex}
لتكن $A = \{ \{ a, b \}, \{ a, d, e \}, \{ a, d \} \}$.
فيكون $\bigcup A = \{ a, b, d, e \}$ و$\bigcap A = \{ a \}$.
\end{ex}
\begin{prob}
	أثبت أنه إذا كانت $A$ مجموعة وكان $A \in B$، فإن
$A \subseteq \bigcup B$.
\end{prob}

وكذلك يكون التعريف لمتتالية مجموعات $A_1$، $A_2$، \dots
\begin{align*}
\bigcup_i A_i & = \Setabs{x}{x \text{ ينتمي إلى إحدى المجموعات } A_i}\\
\bigcap_i A_i & = \Setabs{x}{x \text{ ينتمي إلى كل } A_i}.
\end{align*}

وإن أعطيت \emph{مجموعة فهرسة} $I$، وجعلت مجموعة $A_i$ لكل $i \in I$،
جاز استعمال اختصار الاتحاد الآتي؛ وأما اختصار التقاطع فنشترط فيه أن تكون
مجموعة الفهرسة غير خالية:
\begin{align*}
	\bigcup_{i \in I} A_i & = \bigcup \Setabs{A_i }{i \in I}\\
	\bigcap_{i \in I} A_i & = \bigcap\Setabs{A_i}{i \in I}
\end{align*}

ويبقى أن ننظر فيما هو من !!{element}s~$A$ وليس في~$B$؛ وتمثيله
في \olref{difference}.

\begin{figure}
  \olasset{assets/diagrams/difference.tikz}
  \caption{الفرق $A \setminus B$ يجمع !!{element}s~$A$ التي ليست من !!{element}s~$B$.}
  \ollabel{difference}
\end{figure}

\begin{defn}[الفرق]
\emph{فرق المجموعتين}~$A \setminus B$ هو مجموعة جميع !!{element}s
$A$ التي ليست من !!{element}s~$B$، أي
\[
A\setminus B = \Setabs{x}{x\in A \text{ و } x \notin B}.
\]
\end{defn}

\begin{prob}
برهن على أن $A \subsetneq B$ يقتضي $B \setminus A \neq \emptyset$.
\end{prob}
