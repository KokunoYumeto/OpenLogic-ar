\olsection{تمثيل العلاقات}

نحدّ الآن قابلية التمثيل في حق \emph{العلاقات}، بعد أن حدّدناها في حق الدوال.

\begin{defn}
\ollabel{defn:representing-relations} لتكن
$R(x_0,\dots,x_k)$ علاقة على الأعداد الطبيعية.
نقول إنها {\em قابلة للتمثيل في $\Th{Q}$} إذا أمكن اختيار صيغة
$!A_R(x_0,\dots,x_k)$ تستوفي الشرطين الآتيين:
إذا صدقت $R(n_0,\dots,n_k)$ أثبتت $\Th{Q}$
$!A_R(\num{n_0},\dots,\num{n_k})$،
وإذا كذبت $R(n_0,\dots,n_k)$ أثبتت $\Th{Q}$
$\lnot !A_R(\num{n_0},\dots,\num{n_k})$.
\end{defn}

\begin{thm}
\ollabel{thm:representing-rels} قابلية العلاقة للتمثيل في
$\Th{Q}$ تكافئ قابليتها للحساب.
\end{thm}

\begin{proof}
أما من التمثيل إلى الحساب، فلتكن الصيغة $!A_R(x_0,\dots,x_k)$ ممثلةً
لـ$R(x_0,\dots,x_k)$. نحسب $R$ بالإجراء الآتي:
إذا أعطيت $n_0$، \dots،~$n_k$، أجرينا البحث عن برهان
$!A_R(\num{n_0},\dots,\num{n_k})$ وعن برهان
$\lnot !A_R(\num{n_0},\dots,\num{n_k})$ بالتوازي.
وبفرض التمثيل ينتهي أحد البحثين، ولا يجتمع البرهانان لاتساق النظرية.
فإن وُجد الأول كان الجواب «نعم»، وإن وُجد الآخر كان الجواب «لا».

وأما من الحساب إلى التمثيل، فنفرض أن $R(x_0,\dots,x_k)$ قابلة للحساب.
وهذا، بحسب التعريف، هو قابلية الدالة $\Char{R}(x_0,\dots,x_k)$ للحساب.
ومن \olref[int]{thm:representable-iff-comp} نجد لتمثيل $\Char{R}$
صيغةً $!A_{\Char{R}}(x_0,\dots,x_k,y)$.
فنجعل $!A_R(x_0,\dots,x_k)$ هي الصيغة
$!A_{\Char{R}}(x_0,\dots,x_k,\num{1})$.
وخذ الآن $n_0$، \dots،~$n_k$ كيفما كانت.
إذا صدقت $R(n_0,\dots,n_k)$، كانت
$\Char{R}(n_0,\dots,n_k)=1$؛ فتثبت $\Th{Q}$
$!A_{\Char{R}}(\num{n_0},\dots,\num{n_k},\num{1})$،
أي تثبت $\Th{Q}$ الصيغة $!A_R(\num{n_0},\dots,\num{n_k})$.
وأما إذا كذبت $R(n_0,\dots,n_k)$، كانت $\Char{R}(n_0,\dots,n_k)=0$،
فتثبت $\Th{Q}$:
\[
\lforall[y][(!A_{\Char{R}}(\num{n_0}, \dots, \num{n_k}, y) \lif y =
  \num{0})].
\]
ثم إن $\Th{Q}$ تثبت $\eq/[\num{0}][\num{1}]$؛
فلا بد أن تثبت $\Th{Q}$
$\lnot !A_{\Char{R}}(\num{n_0},\dots,\num{n_k},\num{1})$،
وهذا هو $\lnot !A_R(\num{n_0},\dots,\num{n_k})$ المطلوب.
\end{proof}

\begin{prob}
استخرج من قابلية $R$ للتمثيل في~$\Th{Q}$
قابليةَ $\Char{R}$ للتمثيل فيها.
\end{prob}
