\olsection{ال\printtoken{p}{structure} الجزئية}

إذا كان !!{domain}~!!a{structure}~$\Struct{M}$ داخلًا في
!!{domain}~$\Struct{M'}$ أخرى، لم يكف هذا وحده لجعل~$\Struct{M}$
«جزءًا» من~$\Struct{M'}$. فلا بد، مع الشرط $\Domain{M} \subseteq
\Domain{M'}$، من اتفاق $\Struct{M}$ و$\Struct{M'}$ في تأويل
رموز اللغة على الجزء المشترك~$\Domain{M}$ على الأقل.

\begin{defn}
\ollabel{defn:substructure}
لتكن !!{structure}s~$\Struct M$ و$\Struct M'$ للغة واحدة~$\Lang
L$. نقول إن $\Struct M$~\emph{!!{structure} جزئية} من~$\Struct M'$،
وإن $\Struct M'$~\emph{امتداد} لـ~$\Struct M$، وكتبنا $\Struct M
\substruct \Struct M'$، إذا وفقط إذا
\begin{enumerate}
% تصحيح مصطلحي (C185-01): رُدّ n-place إلى «ذو/ذات n مواضع» بدل «n-محلية» المبهمة؛ والاختيار مفتوح لمراجعة خبير في مصطلح arity.
\item $\Domain{M} \subseteq \Domain{M'}$؛
\item لكل ثابت~$c \in \Lang L$، $\Assign{c}{M} =
    \Assign{c}{M'}$؛
\item لكل !!{function}~$f \in \Lang L$ ذات $n$ مواضع،
  $\Assign{f}{M}(a_1, \dots, a_n) = \Assign{f}{M'}(a_1, \dots, a_n)$
  لجميع $a_1$، و\dots، $a_n \in \Domain{M}$؛
\item لكل !!{predicate}~$R \in \Lang L$ ذي $n$ مواضع، تكون $\langle
  a_1, \dots, a_n\rangle \in \Assign{R}{M}$ إذا وفقط إذا كانت
  $\langle a_1, \dots, a_n\rangle \in \Assign{R}{M'}$ لجميع $a_1$،
  و\dots، $a_n \in \Domain{M}$.
\end{enumerate}
\end{defn}

\begin{rem}
\ollabel{rem:substructure}
إن خلت اللغة من الثوابت ومن !!{function}s، كفت كل مجموعة جزئية غير
خالية $N \subseteq \Domain{M}$ لتعيين !!{structure} جزئية~$\Struct{N}$
من~$\Struct M$ ذات !!{domain}~$\Domain{N} = N$، وذلك بأن نضع
$\Assign{R}{N} = \Assign{R}{M} \cap N^n$.
\end{rem}
