\olsection{وصف المجالات: الـ\usetoken{S}{enumerable} منها و\usetoken{P}{domain} اللانهائية}

المجموعة~$M$ لانهائية بمعنى ديديكند إذا وفقط إذا وجدت دالة
!!a{injective}~$f\colon M \to M$ ليست !!{surjective}، أي
$\ran{f} \neq M$. ولنا في منطق الرتبة الأولى أن نأخذ
!!{function} أحادية الموضع~$f$، فنقرر أن الدالة~$\Assign{f}{M}$
المسندة إليها في !!a{structure}~$\Struct{M}$ هي !!{injective}
وأن~$\ran{f} \neq \Domain{M}$ بالصيغة:
\[
\lforall[x][\lforall[y][(\eq[f(x)][f(y)] \lif \eq[x][y])]] \land
\lexists[y][\lforall[x][\eq/[y][f(x)]]].
\]
فإن أرضت~$\Struct{M}$ هذه !!{sentence}، كانت
$\Assign{f}{M}: \Domain{M} \to \Domain{M}$ !!{injective}،
ولزم أن يكون~$\Domain{M}$ لانهائيًا. وإن كان~$\Domain{M}$
لانهائيًا، وجدت الدالة المطلوبة، وأمكن اختيارها تأويلًا
لـ$\Assign{f}{M}$، فترضى !!{sentence} في~$\Struct{M}$.
لكن هذا الطريق مشروط بأن يكون في اللغة الرمز غير المنطقي~$f$
المخصص لهذا الغرض. أما في الرتبة الثانية فنقول إن مثل هذه الدالة
\emph{موجود}، ولا نحتاج إلى~$f$؛ فتحصل !!{sentence} من منطق
الرتبة الثانية الخالص:
\[
\fn{Inf} \ident \lexists[u][(\lforall[x][\lforall[y][(\eq[u(x)][u(y)]
      \lif \eq[x][y])]] \land \lexists[y][\lforall[x][\eq/[y][u(x)]]])].
\]
وعندئذ~$\Sat{M}{\fn{Inf}}$ إذا وفقط إذا كان~$\Domain{M}$ لانهائيًا.
ثم نضع~$\fn{Fin} \ident \lnot \fn{Inf}$، فيكون
$\Sat{M}{\fn{Fin}}$ إذا وفقط إذا كان~$\Domain{M}$ منتهيًا.
ولا تبلغ !!{sentence} واحدة من الرتبة الأولى الخالصة هذا المبلغ
في التعبير عن لا نهائية !!{domain}، وإن بلغته مجموعة لانهائية
من الجمل. وأما التناهي فلا توجد مجموعة من !!{sentence}s
الرتبة الأولى الخالصة تُرضى في !!a{structure} إذا وفقط إذا
كان مجالها منتهيًا.

\begin{prop}
يكون~$\Sat{M}{\fn{Inf}}$ إذا وفقط إذا كان~$\Domain{M}$ لانهائيًا.
\end{prop}

\begin{proof}
معنى~$\Sat{M}{\fn{Inf}}$ أن يكون
  $\Sat{M}{\lforall[x][\lforall[y][(\eq[u(x)][u(y)] \lif \eq[x][y])]]
    \land \lexists[y][\lforall[x][\eq/[y][u(x)]]]}[s]$
  لبعض~$s$.
فمتى تحقق ذلك كانت~$s(u)$ دالة !!a{injective}، ووجد
$y \in \Domain{M}$ ليس في مدى~$s(u)$.
وعكسًا، إن وجدت دالة !!a{injective}~$f\colon \Domain{M} \to \Domain{M}$
تحقق~$\ran{f} \neq \Domain{M}$، أخذنا تعيينًا~$s$
يحقق~$s(u) = f$، فحصل المطلوب.
\end{proof}

والمجموعة~$M$ !!{enumerable} متى أمكن تعدادها على الصورة
\[
m_0, m_1, m_2, \dots
\]
بحيث ترد !!{element}s فيها بلا تكرار، ويجوز أن ينتهي التعداد.
ومتى كانت~$M$ غير خالية، تكافأ وجود هذا التعداد ووجود
!!a{element}~$z \in M$ ودالة~$f\colon M \to M$ تكون بها
  $z$، و$f(z)$، و$f(f(z))$، و\dots جميع !!{element}s~$M$.
  فإن أعطينا التعداد، أخذنا~$z = m_0$ و$f(m_k) = m_{k+1}$،
  أو~$f(m_k) = m_k$ إن كان~$m_k$ آخر !!{element} فيه؛
  فيحصل بذلك !!{element} والدالة المطلوبان؛ والمدار المنتهي يكرر
  آخر عنصر، غير أنه يمر بعناصر~$M$ كلها.
  وإن أعطينا مثل~$z$ و$f$، مر المدار~$z$، و$f(z)$،
  و$f(f(z))$، و\dots بكل عناصر~$M$، وربما كرر منها.
  فنُبقي أول ورود لكل عنصر ونطرح ما تكرر بعده؛ فإن كانت~$M$
  منتهية وقفنا بعد آخر ورود أول. وهكذا نحصل على تعداد بلا تكرار،
  فتكون~$M$ !!{enumerable}.
وبالتعبير عن وجود~$z$ و$f$ في الرتبة الثانية نحصل على
!!{sentence} تصدق في !!a{structure} إذا وفقط إذا كانت
!!{structure} !!{enumerable}:
\[
\fn{Count} \ident
\lexists[z][\lexists[u][\lforall[X][((X(z) \land
      \lforall[x][(X(x) \lif X(u(x)))]) \lif \lforall[x][X(x)])]]]
\]
% تصحيح مصدر (0329-I1): فُصل المدار المغطي، الذي قد يكرر، عن التعداد الخالي من التكرار المستخرج منه.

\begin{prop}
يكون~$\Sat{M}{\fn{Count}}$ إذا وفقط إذا كان~$\Domain{M}$ !!{enumerable}.
\end{prop}

\begin{proof}
لتكن~$\Domain{M}$ !!{enumerable}، ولنعددها
بـ$m_0$، و$m_1$، و\dots. نحذف المكرر من التعداد حتى لا يرد
$m_k$ مرتين. ونضع~$f(m_k) = m_{k+1}$،
و$f(m_k)=m_k$ إذا كان~$m_k$ آخر عناصر التعداد المنتهي.
ثم نضع~$s(z) = m_0$ و$s(u) = f$، وندعي أن
\[
\Sat{M}{\lforall[X][((X(z) \land \lforall[x][(X(x) \lif X(u(x)))])
    \lif \lforall[x][X(x)])]}[s]
\]
خذ~$S \subseteq \Domain{M}$ كيفما اتفق، ولنفترض
$\Sat{M}{(X(z) \land \lforall[x][(X(x) \lif
X(u(x)))])}[\Subst{s}{S}{X}]$.
ينتج حينئذ~$\Subst{s}{S}{X}(z) \in S$،
وأنه متى كان~$x \in S$ كان~$(\Subst{s}{S}{X}(u))(x) \in S$.
ولما كان~$\varAssign{\Subst{s}{S}{X}}{s}{X}$،
فهذا بعينه قولنا~$m_0 \in S$، وأن~$x \in S$
يستلزم~$f(x) \in S$. وعليه
$m_0 \in S$، و$m_1 = f(m_0) \in S$، و
$m_2 = f(f(m_0)) \in S$، وهكذا في سائر التعداد.
فيلزم~$S = \Domain{M}$، ويحصل
$\Sat{M}{\lforall[x][X(x)]}[\Subst{s}{S}{X}]$.
ولأن~$S \subseteq \Domain{M}$ لم تُخص بشرط، ثبت
$\Sat{M}{\fn{Count}}$.

وللعكس، نفرض~$\Sat{M}{\fn{Count}}$، أي
\[
\Sat{M}{\lforall[X][((X(z) \land \lforall[x][(X(x) \lif X(u(x)))])
    \lif \lforall[x][X(x)])]}[s]
\]
لبعض~$s$. ضع~$m = s(z)$ و$f = s(u)$، وخذ
$S = \{m, f(m), f(f(m)), \dots\}$. فالمجموعة~$S$
!!{enumerable} بهذا التعريف. ومن الفرض نحصل على
\[
\Sat{M}{(X(z) \land \lforall[x][(X(x) \lif X(u(x)))])
    \lif \lforall[x][X(x)]}[\Subst{s}{S}{X}]
\]
ويتحقق~$\Sat{M}{X(z)}[\Subst{s}{S}{X}]$، إذ
$S \ni m = \Subst{s}{S}{X}(z)$. ويتحقق كذلك
$\Sat{M}{\lforall[x][(X(x) \lif
X(u(x)))]}[\Subst{s}{S}{X}]$، لأن~$f(x) \in S$ كلما كان~$x \in S$.
فقد أُرضي المقدم والشرطية معًا، فوجب إرضاء التالي:
$\Sat{M}{\lforall[x][X(x)]}[\Subst{s}{S}{X}]$.
وهذا يقتضي~$S = \Domain{M}$، فتكون~$\Domain{M}$ !!{enumerable}،
كما أن~$S$ كذلك بحسب تعريفها.
\end{proof}

\begin{prob}
!!{sentence}~$\fn{Inf} \land \fn{Count}$ تصدق في المجالات
!!{denumerable}، ولا تصدق في غيرها. غيّر تعريف~$\fn{Count}$
لتحصل على !!{sentence} أخرى تقرر مباشرة أن المجال
!!{denumerable}، ثم أثبت أنها تؤدي هذا المعنى.
\end{prob}
