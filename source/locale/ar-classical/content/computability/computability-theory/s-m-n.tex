\olsection{مبرهنة $s$-$m$-$n$}

\begin{explain}
تسمى المبرهنة الآتية «مبرهنة $s$-$m$-$n$»، وسيظهر وجه
التسمية. وإنما الصعوبة في ضبط معناها؛ فإذا استبانت عبارتها
قرب مأخذها.
\end{explain}

\begin{thm}
\ollabel{thm:s-m-n}
  مهما أخذنا عددين طبيعيين $n$ و~$m$، وجدت دالة عودية بدائية~$s^m_n$
  تتحقق بها، لكل أعداد طبيعية $e$، $a_0$، …، $a_{m-1}$، $y_0$،
  …،~$y_{n-1}$، المساواة الجزئية الآتية:
  \[
  \cfind{s^m_n(e, a_0, \dots, a_{m-1})}[n](y_0, \dots, y_{n-1}) \simeq
  \cfind{e}[m+n](a_0, \dots, a_{m-1}, y_0, \dots, y_{n-1}).
\]
\end{thm}

% تصحيح ترجمة (C232-01): الأعيان المذكورة أعداد طبيعية، لا متتالية واحدة.
\textbf{تنبيه تصحيحي على الترجمة الأساس.} الأعيان $e$ و$a_i$ و$y_i$
أعداد طبيعية، لا متتالية واحدة؛ فأُظهر نوعها الصحيح.

\begin{explain}
ويقرب المعنى إذا جعلنا عمل $s^m_n$ في \emph{البرامج}. فالدالة $s^m_n$
تتلقى برنامجًا~$e$ لدالة ذات $m+n$ حجة، وتتلقى معه المداخل المثبتة $a_0$،
…،~$a_{m-1}$؛ ثم تعطي برنامجًا
$s^m_n(e,a_0,\dots,a_{m-1})$ لدالة ذات $n$ حجة، لا يبقى من حججها إلا ما لم يُثبّت.
\iftag{TMs}{فإذا كان $e$ وصفًا لآلة تورنغ، كانت
$s^m_n(e,a_0,\dots,a_{m-1})$ هي آلة تورنغ التي تسبق سلسلة الدخل
$y_0$، …،~$y_{n-1}$ بـ$a_0$، …،~$a_{m-1}$، ثم تشغل~$e$.
فكل $s^m_n$ حينئذ دالة عودية بدائية تستخرج رمز آلة تورنغ
المطلوبة.}{}
\end{explain}
