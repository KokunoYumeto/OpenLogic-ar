\olsection{السلامة}

\begin{explain}
نسق !!^a{derivation}، كالجداول الدلالية، \emph{سليم} متى امتنع عليه
!!{derive} ما لا يلزم في الحقيقة. فالسلامة ضمان لأنساق ال!!{derivation}
من تجاوز ما يسوّغه الاستدلال. ويتبيّن المراد بحسب الخاصية البرهانية
التي نبحث عنها؛ فمن ذلك:
\begin{enumerate}
\item كل~$!A$ !!{derivable} تكون \iftag{FOL}{صالحة}{تحصيل حاصل}؛
\item إذا كانت !!a{sentence} !!{derivable} من جمل أخرى، فهي أيضًا نتيجة لها؛
\item إذا كانت مجموعة من ال!!{sentence}s غير متسقة، فهي غير قابلة للإرضاء.
\end{enumerate}
هذه خصائص لا غنى عنها لنسق !!a{derivation}. فإن تخلّفت إحداها
كان نسق ال!!{derivation} معيبًا، إذ أمكنه !!{derive} ما لا حق له
في إثباته. فمن أجل ذلك كان إثبات سلامة نسق !!a{derivation}
من أوكد مطالب دراسته.

وجميع هذه الخصائص البرهانية مردّها إلى !!{tableau}s مغلقة
تختلف صورها. فإثبات~(1)--(3) يقتضي بيان خاصية دلالية مشتركة
لل!!{tableau}s المغلقة. نعرّف أولًا إرضاء
!!a{signed formula} في \iftag{FOL}{بنية}{تقييم}، ثم نثبت أن
ال!!{tableau} متى أغلق امتنع أن ترضي \iftag{FOL}{بنية}{تقييم}
جميع افتراضاته. وعندئذ تتبع~(1)--(3) هذه النتيجة.
\end{explain}

\begin{defn}
\iftag{FOL}{!!^a{structure}}{!!^a{valuation}}~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
\emph{ترضي}
!!a{signed formula} $\sFmla{\True}{!A}$ إذا وفقط إذا كان
$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!A}$، وتُرضي
$\sFmla{\False}{!A}$ إذا وفقط إذا كان
$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!A}$. وتُرضي
$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ مجموعة من ال!!{signed formula}s
~$\Gamma$ إذا وفقط إذا أرضت كل $\sFmla{S}{!A} \in \Gamma$. وتكون
$\Gamma$~\emph{قابلة للإرضاء} إذا وُجدت
\iftag{FOL}{!!a{structure}}{!!a{valuation}} ترضيها، و\emph{غير قابلة
للإرضاء} خلاف ذلك.
\end{defn}

\begin{thm}[السلامة]
  \ollabel{thm:tableau-soundness}
  إذا كان لــ$\Gamma$ !!{tableau} مغلق، فإن $\Gamma$ غير قابلة للإرضاء.
\end{thm}

\begin{proof}
نقول عن فرع من !!a{tableau} إنه قابل للإرضاء إذا وفقط إذا كانت
مجموعة ال!!{signed formula}s التي عليه قابلة للإرضاء.
ونقول عن !!a{tableau} إنه قابل للإرضاء متى كان له فرع واحد
قابل للإرضاء على الأقل.

نثبت أن مدّ !!{tableau} قابل للإرضاء بإحدى قواعد الاستدلال
يبقيه !!{tableau} قابلًا للإرضاء. وفي ذلك إثبات المبرهنة؛
فإن كل !!{tableau} مغلق يُبنى بإجراء القواعد على ال!!{tableau}
المؤلّف من افتراضات في~$\Gamma$ وحدها. فلو كانت~$\Gamma$ قابلة
للإرضاء، لكان كل !!{tableau} مبني لها قابلًا للإرضاء.
لكن ال!!{tableau} المغلق لا يقبل الإرضاء؛ إذ يجتمع في كل فرع
$\sFmla{\True}{!A}$ و$\sFmla{\False}{!A}$، ولا
\iftag{FOL}{بنية}{تقييم} ترضي~$!A$ ولا ترضيها معًا.

نفرض !!{tableau} قابلًا للإرضاء، أي !!a{tableau} له فرع قابل
للإرضاء على الأقل. وإجراء القاعدة إما إلحاق !!{signed formula}s
بفرع، وإما شقّه إلى فرعين. فإن كان في ال!!{tableau} فرع قابل
للإرضاء لم تتناوله هذه الخطوة، بقي كذلك في ال!!{tableau} الممدود؛
فيصح المطلوب. فليس علينا إلا النظر في إجراء قاعدة على فرع قابل
للإرضاء.

نسمّي~$\Gamma$ مجموعة ال!!{signed formula}s على هذا الفرع،
ونفرض أن~$\sFmla{S}{!A} \in \Gamma$ هي ال!!{signed formula}
التي تناولتها القاعدة. إن لم تشعّب القاعدة الفرع، وجب إثبات
قابلية إرضاء~$\Gamma$ مع النتائج المضافة. وإن شعّبته،
كفى أن يقبل الإرضاء أحد الفرعين الناشئين على الأقل.

أما الاستدلالات التي لا تشعّب الفرع، فتفصيلها كما يأتي:
\begin{enumerate}
\item إذا أجرينا~$\TRule{\True}{\lnot}$ على
  $\sFmla{\True}{\lnot !B} \in \Gamma$، صارت
  ال!!{signed formula}s على الفرع~$\Gamma \cup
  \{\sFmla{\False}{!B}\}$. نفرض
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$؛ فيلزم، على الخصوص،
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{\lnot !B}$.
  ومن تعريف النفي~$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B}$؛
  أي إن~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  ترضي~$\sFmla{\False}{!B}$، كما أردنا.
\item يُوسَّع الفرع بتطبيق $\TRule{\False}{\lnot}$ على
  $\sFmla{\False}{\lnot !B} \in \Gamma$: تمرين.
\item وإذا أجرينا~$\TRule{\True}{\land}$ على
  $\sFmla{\True}{!B \land !C} \in \Gamma$،
  أضفنا !!{signed formula}s اثنتين:~$\sFmla{\True}{!B}$
  و$\sFmla{\True}{!C}$. نفرض
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$، ومنه
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{!B \land !C}$.
  فيلزم~$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!B}$ و
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{!C}$؛
  فالبنية أو التقييم~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  ترضي~$\sFmla{\True}{!B}$ و$\sFmla{\True}{!C}$ جميعًا.
\item يُوسَّع الفرع بتطبيق $\TRule{\False}{\lor}$ على
  $\sFmla{\False}{!B \lor !C} \in \Gamma$: تمرين.
\item وإذا أجرينا~$\TRule{\False}{\lif}$ على
  $\sFmla{\False}{!B \lif !C} \in \Gamma$،
  أضفنا !!{signed formula}s اثنتين:~$\sFmla{\True}{!B}$
  و$\sFmla{\False}{!C}$. نفرض
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$، فيكون، على الخصوص،
  $\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B \lif !C}$.
  ولا يكون ذلك إلا مع~$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!B}$
  و$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!C}$؛
  فالبنية أو التقييم~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  ترضي~$\sFmla{\True}{!B}$ و$\sFmla{\False}{!C}$.


\iftag{FOL}{%
\item وإذا أجرينا~$\TRule{\True}{\lforall}$ على
  $\sFmla{\True}{\lforall[x][!A(x)]} \in \Gamma$،
  ألحقنا بالفرع !!{signed formula} هي~$\sFmla{\True}{!A(t)}$.
  نفرض~$\Sat{M}{\Gamma}$؛ فمنه
  $\Sat{M}{\lforall[x][!A(x)]}$. وبمقتضى
  \olref[syn][sem]{prop:quant-terms} نحصل على~$\Sat{M}{!A(t)}$؛
  أي إن~$\Struct{M}$ ترضي~$\sFmla{\True}{!A(t)}$.


\item وإذا أجرينا~$\TRule{\False}{\lforall}$ على
  $\sFmla{\False}{\lforall[x][!A(x)]} \in \Gamma$،
  ألحقنا !!{signed formula} هي~$\sFmla{\False}{!A(a)}$، حيث~$a$
  !!a{constant} لم يرد في~$\Gamma$. والمجموعة~$\Gamma$ قابلة
  للإرضاء، فنأخذ~$\Struct{M}$ تحقق~$\Sat{M}{\Gamma}$، ومنه
  $\Sat/{M}{\lforall[x][!A(x)]}$. ولإثبات قابلية إرضاء
  $\Gamma \cup \{\sFmla{\False}{!A(a)}\}$ نبني بنية
  $\Struct{M'}$ على الوجه الآتي.

  تقضي \olref[syn][ass]{prop:sat-quant} بأن
  $\Sat/{M}{\lforall[x][!A(x)]}$ يكافئ وجود~$s$ له
  $\Sat/{M}{!A(x)}[s]$. نأخذ~$\Struct{M'}$ موافقةً
  لـ~$\Struct{M}$ في كل شيء سوى~$\Assign{a}{M'} = s(x)$.
  وبمقتضى \olref[syn][ext]{cor:extensionality-sent}،
  لكل~$\sFmla{\True}{!C} \in \Gamma$ يكون~$\Sat{M'}{!C}$،
  ولكل~$\sFmla{\False}{!C} \in \Gamma$ يكون~$\Sat/{M'}{!C}$؛
  إذ لم يرد~$a$ في~$\Gamma$.

  ثم تعطينا \olref[syn][ext]{prop:extensionality}
  $\Sat/{M'}{!A(x)}[s]$، وتعطينا
  \olref[syn][ext]{prop:ext-formulas}~$\Sat/{M'}{!A(a)}[s]$.
  ولأن~$!A(a)$ !!a{sentence}، تنقلنا
  \olref[syn][ass]{prop:sentence-sat-true} إلى~$\Sat/{M'}{!A(a)}$؛
  أي إن~$\Struct{M'}$ ترضي~$\sFmla{\False}{!A(a)}$.
\item يُوسَّع الفرع بتطبيق $\TRule{\True}{\lexists}$ على
  $\sFmla{\True}{\lexists[x][!B(x)]} \in \Gamma$: تمرين.
\item يُوسَّع الفرع بتطبيق $\TRule{\False}{\lexists}$ على
  $\sFmla{\False}{\lexists[x][!B(x)]} \in \Gamma$: تمرين.}{}
\end{enumerate}
وأما الاستدلالات التي تشعّب الفرع، فنفصّلها الآن:
\begin{enumerate}
\item إذا أجرينا~$\TRule{\False}{\land}$ على
  $\sFmla{\False}{!B \land !C} \in \Gamma$،
  نشأ فرع أيسر يمر بـ~$\sFmla{\False}{!B}$،
  وفرع أيمن يمر بـ~$\sFmla{\False}{!C}$.
  نفرض~$\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$، فيلزم
  $\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B \land !C}$؛
  ومن ثم إما~$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B}$
  وإما~$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!C}$.
  أما في الأولى، فـ~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  ترضي~$\sFmla{\False}{!B}$؛ أي إن
  $\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ ترضي صيغ الفرع الأيسر.
  وأما في الثانية، فـ~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  ترضي~$\sFmla{\False}{!C}$؛ أي إن
  $\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ ترضي صيغ الفرع الأيمن.
\item يُوسَّع الفرع بتطبيق $\TRule{\True}{\lor}$ على
  $\sFmla{\True}{!B \lor !C} \in \Gamma$: تمرين.
\item يُوسَّع الفرع بتطبيق $\TRule{\True}{\lif}$ على
  $\sFmla{\True}{!B \lif !C} \in \Gamma$: تمرين.
\item وإذا أجرينا~\Cut، نشأ فرع فيه~$\sFmla{\True}{!B}$،
  وآخر فيه~$\sFmla{\False}{!B}$. ونحن نفرض
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$؛ ولا يخلو الأمر من
  $\iftag{FOL}{\Sat{M}}{\pSat{v}}{!B}$ أو
  $\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B}$.
  فـ~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ ترضي أحد الفرعين،
  الأيسر أو الأيمن.
\end{enumerate}
\end{proof}

\begin{cor}
\ollabel{cor:weak-soundness}
إذا كان $\Proves !A$، فإن $!A$ \iftag{FOL}{صالحة}{تحصيل حاصل}.
\end{cor}

\begin{cor}
\ollabel{cor:entailment-soundness}
إذا كان $\Gamma \Proves !A$، فإن $\Gamma \Entails !A$.
\end{cor}

\begin{proof}
  من~$\Gamma \Proves !A$ نأخذ~$!B_1$، \dots، $!B_n \in
  \Gamma$ بحيث يكون للمجموعة~$\{\sFmla{\False}{!A}, \sFmla{\True}{!B_1}, \dots,
  \sFmla{\True}{!B_n}\}$ !!{tableau} مغلق. فتقتضي
  \olref{thm:tableau-soundness} أن كل
  \iftag{FOL}{!!{structure}}{!!{valuation}}~$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$
  إما أن تكذّب بعض~$!B_i$ وإما أن تصدّق~$!A$.
  فإذا كان~$\iftag{FOL}{\Sat{M}}{\pSat{v}}{\Gamma}$ انتفى الأول،
  فلزم~$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!A}$.
\end{proof}

\begin{cor}
\ollabel{cor:consistency-soundness}
إذا كانت $\Gamma$ قابلة للإرضاء، فهي متسقة.
\end{cor}

\begin{proof}
نأخذ المقابل بالنقيض. فإذا كانت~$\Gamma$ غير متسقة، وجد~$!B_1$،
\dots، $!B_n \in \Gamma$ ومعها !!{tableau} مغلق لـ
$\{\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n}\}$.
وبمبرهنة \olref{thm:tableau-soundness} لا توجد
$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ تحقق
$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!B_i}$ لكل~$i=1$، \dots،~$n$.
فيمتنع إرضاء~$\Gamma$، وهو المطلوب.
\end{proof}
