\olsection{\usetoken{S}{derivability} والمكمّمات}

\begin{explain}
  ومن مقدمات برهان الاكتمال أن تثبت بقواعد الجداول الدلالية هذه
  النتائج في علاقة~$\Proves$؛ فلذلك نبيّنها هنا.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} إذا كان $c$ ثابتًا لا يظهر في
$\Gamma$ ولا في~$!A(x)$، وكان $\Gamma \Proves !A(c)$، فإن $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
من~$\Gamma \Proves !A(c)$ نأخذ~$!B_1$، \dots، $!B_n
\in \Gamma$ ومعها !!{tableau} مغلق للمجموعة
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{وعلينا أن نثبت أن هناك أيضًا !!{tableau} مغلقًا لــ}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
نأخذ ال!!{tableau} المغلق، ونجعل افتراضه الأول
$\sFmla{\False}{\lforall[x][!A(x)]}$، ثم نلحق
$\sFmla{\False}{!A(c)}$ بالافتراضات، كما في الرسم:
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n}, 
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
ولا يزول الإغلاق بهذا التغيير؛ فإن ال!!{sentence}s التي كانت
افتراضات ما زالت جميعًا حاضرة في أعلى ال!!{tableau}.
وأما السطر المضاف فمأخوذ بوجه صحيح من~$\TRule{\False}{\lforall}$:
إذ ال!!{constant}~$c$ لا يرد في~$!B_1$، \dots، $!B_n$
ولا في~$\lforall[x][!A(x)]$، فلا يرد فوق السطر المضاف في
ال!!{tableau} الجديد.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{نأخذ !!{tableau} للمجموعة
  $\sFmla{\False}{\lexists[x][!A(x)]}, \sFmla{\True}{!A(t)}$،
  ويكفي لإغلاقه إجراء قاعدة كذب الوجود مرة واحدة:
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{ونأخذ !!{tableau} للمجموعة
  $\sFmla{\False}{\formula{A}(t)}, \sFmla{\True}{\lforall[x][!A(x)]}$،
  ويغلق بإجراء قاعدة صدق الكل مرة واحدة:
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
