\olsection{\usetoken{P}{tableau} مع \usetoken{S}{identity}}

إذا دخلت ال!!{identity} في !!^{tableau}s لزم إلحاق قواعد بها.
فقواعد~$\eq$ هي الآتية، على أن تكون~$t$ و$t_1$ و$t_2$ حدودًا مغلقة:

\begin{defish}
\AxiomC{}
\RightLabel{$\eq$}
\UnaryInfC{\sFmla{\True}{\eq[t][t]}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\True}{\eq[t_1][t_2]}}
\noLine
\UnaryInfC{\sFmla{\True}{!A(t_1)}}
\RightLabel{$\TRule{\True}{\eq}$}
\UnaryInfC{\sFmla{\True}{!A(t_2)}}
\DisplayProof
\hfill
\AxiomC{\sFmla{\True}{\eq[t_1][t_2]}}
\noLine
\UnaryInfC{\sFmla{\False}{!A(t_1)}}
\RightLabel{$\TRule{\False}{\eq}$}
\UnaryInfC{\sFmla{\False}{!A(t_2)}}
\DisplayProof
\end{defish}
وتنفرد~$\TRule{\True}{\eq}$ و$\TRule{\False}{\eq}$ عن القواعد
المتقدمة بأنهما تحتاجان إلى !!{formula}s موقّعتين \emph{اثنتين}
سبقتا على الفرع:~$\sFmla{\True}{\eq[t_1][t_2]}$ و
$\sFmla{S}{!A(t_1)}$.

\begin{ex}
إذا كان $s$ و$t$ حدين مغلقين، فإن $\eq[s][t], !A(s)
\Proves !A(t)$:
\begin{oltableau}
  [\sFmla{\False}{\formula{A}(t)}, just = \TAss
    [\sFmla{\True}{\eq[s][t]}, just = \TAss
      [\sFmla{\True}{\formula{A}(s)}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just={\TRule{\True}{\eq}[2, 3]}, close]
      ]
    ]
  ]
\end{oltableau}
وهذا هو المعروف بمبدأ إحلال المتطابقات، أو قانون لايبنتس.

وبال!!^{tableau}s نثبت تناظر~$\eq$؛ أي
$\eq[s_1][s_2]
\Proves \eq[s_2][s_1]$، كما يأتي:
\begin{oltableau}
  [\sFmla{\False}{\eq[s_2][s_1]}, just = \TAss
    [\sFmla{\True}{\eq[s_1][s_2]}, just = \TAss
      [\sFmla{\True}{\eq[s_1][s_1]}, just = {$\eq$}
        [\sFmla{\True}{\eq[s_2][s_1]}, just = {\TRule{\True}{\eq}[2, 3]}, close]
      ]
    ]
  ]
\end{oltableau}
أما السطر~$2$ ففيه ال!!{formula} الأولى المطلوبة،
وهي~$\sFmla{\True}{\eq[s_1][s_2]}$، لقاعدة~$\TRule{\True}{\eq}$.
وأما السطر~$3$ ففيه الثانية، وصورتها~$\sFmla{\True}{!A(s_1)}$.
وبيان المطابقة أن نأخذ~$!A(x)$ هي~$\eq[x][s_1]$؛
فيكون~$!A(s_1)$ هو~$\eq[s_1][s_1]$،
ويكون~$!A(s_2)$ هو~$\eq[s_2][s_1]$.

ونثبت كذلك تعدّي~$\eq$؛ أي~$\eq[s_1][s_2],
\eq[s_2][s_3] \Proves \eq[s_1][s_3]$:
\begin{oltableau}
  [\sFmla{\False}{\eq[s_1][s_3]}, just = \TAss
    [\sFmla{\True}{\eq[s_1][s_2]}, just = \TAss
      [\sFmla{\True}{\eq[s_2][s_3]}, just = \TAss
        [\sFmla{\True}{\eq[s_1][s_3]}, just = {\TRule{\True}{\eq}[3, 2]}, close]
      ]
    ]
  ]
\end{oltableau}
في هذا ال!!{tableau} تؤخذ ال!!{formula} الأولى لقاعدة
$\TRule{\True}{\eq}$ من السطر~$3$، وهي~$\sFmla{\True}{\eq[s_2][s_3]}$؛
فيقوم~$s_2$ مقام~$t_1$، و$s_3$ مقام~$t_2$.
وأما الثانية، ذات الصورة~$\sFmla{\True}{!A(s_2)}$، ففي السطر~$2$.
خذ~$!A(x)$ هي~$\eq[s_1][x]$؛ فيصير~$!A(s_2)$
$\eq[s_1][s_2]$، وهو السطر~$2$، ويصير~$!A(s_3)$
ال!!{formula}~$\eq[s_1][s_3]$ التي انتهينا إليها.
\end{ex}

\begin{prob}
أنشئ !!{tableau}s مغلقة لما يأتي:
\begin{enumerate}
\item $\sFmla{\False}{\lforall[x][\lforall[y][((x = y \land !A(x))
      \lif !A(y))]]}$
\item $\sFmla{\False}{\lexists[x][(!A(x) \land
    \lforall[y][(!A(y) \lif y = x)])]}$,\\
  $\sFmla{\True}{\lexists[x][!A(x)] \land
    \lforall[y][\lforall[z][((!A(y) \land !A(z)) \lif y = z)]]}$
\end{enumerate}
\end{prob}
